# powerbi_client.py
import requests
from azure.identity import ClientSecretCredential

class PowerBIClient:
    def __init__(self, tenant_id, client_id, client_secret):
        self.credential = ClientSecretCredential(tenant_id, client_id, client_secret)
        self.base_url = "https://api.powerbi.com/v1.0/myorg"

    def get_token(self):
        token = self.credential.get_token("https://analysis.windows.net/powerbi/api/.default")
        return token.token

    def execute_dax(self, dataset_id, dax_query):
        """Send a DAX query to Power BI and get results"""
        url = f"{self.base_url}/datasets/{dataset_id}/executeQueries"
        headers = {
            "Authorization": f"Bearer {self.get_token()}",
            "Content-Type": "application/json"
        }
        body = {
            "queries": [{"query": dax_query}],
            "serializerSettings": {"includeNulls": True}
        }
        response = requests.post(url, headers=headers, json=body)
        return response.json()

# main.py
def run_agent(user_question):
    messages = [
        {"role": "system", "content": SYSTEM_PROMPT},
        {"role": "user", "content": user_question}
    ]

    while True:
        response = llm_client.chat.completions.create(
            model="gpt-4o",  # or claude via Anthropic API
            messages=messages,
            tools=tools,
            tool_choice="auto"
        )

        msg = response.choices[0].message

        # If the model wants to call a tool
        if msg.tool_calls:
            messages.append(msg)
            for tool_call in msg.tool_calls:
                name = tool_call.function.name
                args = json.loads(tool_call.function.arguments)

                if name == "schema_lookup":
                    result = chroma_collection.query(query_texts=[args["query"]], n_results=3)
                    tool_result = str(result["documents"])
                elif name == "execute_dax":
                    tool_result = str(pbi_client.execute_dax(DATASET_ID, args["dax_query"]))
                elif name == "read_excel":
                    df = pd.read_excel(args["file_path"], sheet_name=args.get("sheet_name"))
                    tool_result = df.head(50).to_string()

                messages.append({
                    "role": "tool",
                    "tool_call_id": tool_call.id,
                    "content": tool_result
                })
        else:
            # Final answer
            return msg.content

What You MUST Know (4 Key Areas)
1. The Data Model (Most Important — spend 2-3 days here)
This is the foundation your agent reasons over. You need to understand:

Tables and relationships — how your fact tables (e.g., Sales) connect to dimension tables (e.g., Products, Dates, Regions) via keys. Think of it like a database schema.
Star schema concept — Power BI models are typically one central fact table surrounded by dimension tables. If you understand SQL joins, this will click fast.
Columns vs Measures — columns are raw data fields, measures are calculated aggregations (like "Total Revenue" or "Avg Order Value"). Your agent needs to know the difference because DAX queries use them differently.
How to learn this fast: Open your .pbix file in Power BI Desktop, go to the Model View (the icon on the left that looks like a diagram). Spend an hour just reading the table names, columns, and the lines connecting them. That diagram IS your agent's brain map.

2. DAX (The Query Language — spend 3-5 days here)
DAX is how your agent will actually pull data from Power BI. You don't need to master it — you need to know the patterns so you can write a good system prompt and debug when the LLM generates bad DAX.

The 20% of DAX that covers 80% of use cases:

dax
// Basic aggregation — "what's total revenue?"
EVALUATE
SUMMARIZECOLUMNS(
    Sales[Region],
    "Total Revenue", SUM(Sales[Revenue])
)

// Filtering — "what's revenue for North region?"
EVALUATE
CALCULATETABLE(
    SUMMARIZECOLUMNS(
        Sales[Region],
        "Total Revenue", SUM(Sales[Revenue])
    ),
    Sales[Region] = "North"
)

// Time intelligence — "compare this year vs last year"
EVALUATE
SUMMARIZECOLUMNS(
    'Date'[Year],
    "Revenue", SUM(Sales[Revenue])
)

// Top N — "top 5 products by revenue"
EVALUATE
TOPN(5,
    SUMMARIZECOLUMNS(
        Products[ProductName],
        "Revenue", SUM(Sales[Revenue])
    ),
    [Revenue], DESC
)
Key DAX functions to know: EVALUATE, SUMMARIZECOLUMNS, CALCULATETABLE, FILTER, SUM, AVERAGE, COUNT, TOPN, DATEADD, SAMEPERIODLASTYEAR, DIVIDE

Important: The Power BI REST API only accepts EVALUATE style DAX queries (returns a table). It does NOT accept standalone measure definitions. This is a common gotcha — make sure your system prompt tells the LLM this.

3. Power BI Service Basics (spend 1 day here)
You need to know just enough to set up the API pipeline:

How to publish a .pbix from Desktop to the Power BI Service (it's literally "Publish" button → select a workspace)
Workspaces — where reports and datasets live. You'll need the workspace ID and dataset ID for API calls
Semantic model (dataset) — once published, the data model becomes a "semantic model" in the service. This is what your API queries against.
To find your IDs: Go to Power BI Service → open your workspace → click on the dataset → the URL will look like: app.powerbi.com/groups/{workspaceId}/datasets/{datasetId} — those GUIDs are what your code needs.

4. Power BI REST API (spend 1-2 days here)
You really only need 3 API endpoints:

Execute Queries — POST /datasets/{id}/executeQueries — this is your main one, sends DAX and gets data back
Get Tables — GET /datasets/{id}/tables — helps your agent discover what tables exist
Get Measures — GET /datasets/{id}/measures — helps your agent know what pre-built measures are available
That's it. You don't need the 50+ other endpoints.
