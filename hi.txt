"""One scene in, one clip file out.

This is where the "Veo only does 8 seconds" constraint stops being a problem and
becomes a loop: each scene of the plan is generated on its own, written to
``storage/projects/<id>/clips/``, measured, and recorded back on the project.
The renderer later accumulates whatever is ready.
"""

from __future__ import annotations

import asyncio
from pathlib import Path

from ..config import settings
from ..domain.formats import veo_aspect_for
from ..domain.models import Clip, Project, Scene, StoryOption, now_iso
from ..errors import BadRequest, Conflict
from ..logging_setup import get_logger
from ..storage import clips_dir, ensure_dir, persist_media, resolve_in_storage, work_dir
from . import veo
from .veo import InlineAsset
from .ffmpeg import extract_poster, probe
from .imagegen import asset_from_image, generate_keyframe, livery_assets
from .jobs import JobContext, map_with_concurrency
from .store import require_project, update_project

log = get_logger("clips")


def selected_story(project: Project) -> StoryOption:
    story = next((s for s in project.stories if s.id == project.selected_story_id), None)
    if story is None:
        raise BadRequest("No story is selected for this project yet.")
    if not story.scenes:
        raise BadRequest("The selected story has no scenes yet. Generate the storyboard first.")
    return story


def _find_scene(project: Project, scene_id: str) -> tuple[StoryOption, Scene]:
    story = selected_story(project)
    scene = next((s for s in story.scenes if s.id == scene_id), None)
    if scene is None:
        raise BadRequest(f"Scene {scene_id} is not part of the selected story.")
    return story, scene


def sync_clips(project: Project) -> list[Clip]:
    """Derive the clip rows from the selected story, keeping what is generated."""
    story = next((s for s in project.stories if s.id == project.selected_story_id), None)
    if story is None:
        return []

    existing = {clip.scene_id: clip for clip in project.clips}
    clips: list[Clip] = []

    for scene in story.scenes:
        previous = existing.get(scene.id)
        # A scene whose length changed invalidates the clip generated from the
        # old version — the renderer would otherwise pad or cut the wrong shot.
        stale = previous is not None and previous.seconds != scene.seconds

        if previous is not None and not stale:
            refreshed = previous.model_copy(update={"order": scene.order, "seconds": scene.seconds})
            clips.append(refreshed)
        else:
            clips.append(
                Clip(sceneId=scene.id, order=scene.order, status="idle", seconds=scene.seconds)
            )

    return clips


async def _patch_clip(project_id: str, scene_id: str, **patch) -> None:
    def mutate(project: Project) -> Project:
        project.clips = [
            clip.model_copy(update=patch) if clip.scene_id == scene_id else clip
            for clip in project.clips
        ]
        return project

    await update_project(project_id, mutate)


async def generate_scene_clip(
    project: Project,
    scene_id: str,
    *,
    force: bool = False,
    on_progress=None,
    seed_frame: InlineAsset | None = None,
) -> Clip:
    """Generate one scene's clip.

    ``seed_frame`` is the previous clip's final frame, supplied by the chained
    path in ``generate_all_clips``. It takes precedence over a generated
    keyframe: a real frame of the film we are already making is a stronger
    continuity anchor than a fresh still of what that frame should have been.
    """
    story, scene = _find_scene(project, scene_id)
    current = next((clip for clip in project.clips if clip.scene_id == scene_id), None)

    if current and current.status == "generating" and not force:
        raise Conflict(f"Scene {scene.order} is already generating.")
    if current and current.status == "ready" and not force:
        return current

    aspect = veo_aspect_for(project.brief.ratio)
    destination = Path(clips_dir(project.id)) / f"scene-{scene.order:02d}.mp4"
    poster = destination.with_suffix(".jpg")

    await _patch_clip(
        project.id,
        scene_id,
        status="generating",
        progress=0.0,
        error=None,
        started_at=now_iso(),
        attempts=(current.attempts if current else 0) + 1,
    )

    async def report(fraction: float, message: str) -> None:
        if on_progress is not None:
            result = on_progress(fraction, message)
            if asyncio.iscoroutine(result):
                await result
        await _patch_clip(project.id, scene_id, progress=fraction)

    try:
        # Lock continuity with a first frame: the previous clip's tail if we are
        # chaining, otherwise an optional generated still.
        first_frame = seed_frame
        if first_frame is None and settings.veo_use_keyframes:
            await report(0.02, "Rendering the opening frame…")
            first_frame = await generate_keyframe(scene, story, aspect)

        # A keyframe already carries the livery in its pixels, so references are
        # the fallback for when one was not produced — keyframes switched off, no
        # image key, or a blocked generation. Without either, the aircraft is
        # whatever Veo invents, which is the failure this reference set exists to
        # prevent, so it is worth saying per scene.
        references = None if first_frame else list(livery_assets())
        if references:
            log.warning(
                "Scene %d has no keyframe; falling back to Veo reference images for the "
                "IndiGo livery. Set VEO_USE_KEYFRAMES=true and GEMINI_IMAGE_API_KEY for the "
                "stronger path, which fixes the livery in the frame itself.",
                scene.order,
            )

        generated = await veo.generate_clip(
            destination,
            prompt=scene.prompt,
            seconds=scene.seconds,
            aspect_ratio=aspect,
            negative_prompt=scene.negative_prompt,
            model=project.model or None,
            first_frame=first_frame,
            reference_images=references or None,
            on_progress=report,
        )

        info = await probe(generated.file)

        try:
            await extract_poster(generated.file, poster, min(0.5, max(0.0, info.seconds / 2)))
            poster_file, poster_url = await asyncio.to_thread(
                persist_media, poster, project.id, "posters"
            )
        except Exception as error:  # noqa: BLE001 — a missing poster is cosmetic
            log.warning("Poster extraction failed for scene %d: %s", scene.order, error)
            poster_file = None
            poster_url = None

        if info.seconds < scene.seconds - 0.5:
            log.warning(
                "Scene %d came back at %.2fs for a %ds request; the renderer will hold the last "
                "frame to fill the slot.",
                scene.order, info.seconds, scene.seconds,
            )

        durable_file, durable_url = await asyncio.to_thread(
            persist_media, generated.file, project.id, "clips"
        )
        await _patch_clip(
            project.id,
            scene_id,
            status="ready",
            progress=1.0,
            actual_seconds=info.seconds,
            file=durable_file,
            url=durable_url,
            poster_file=poster_file,
            poster_url=poster_url,
            operation=generated.operation,
            error=None,
            completed_at=now_iso(),
        )

        log.info("Scene %d ready (%.2fs).", scene.order, info.seconds)
        refreshed = await require_project(project.id)
        return next(clip for clip in refreshed.clips if clip.scene_id == scene_id)

    except asyncio.CancelledError:
        await _patch_clip(project.id, scene_id, status="idle", progress=0.0, error=None)
        raise
    except Exception as error:
        await _patch_clip(
            project.id,
            scene_id,
            status="failed",
            progress=0.0,
            error=str(error),
            completed_at=now_iso(),
        )
        raise


async def generate_all_clips(
    project: Project, context: JobContext, *, force: bool = False
) -> dict[str, int]:
    """Generate every scene that is not already done, ``VEO_CONCURRENCY`` at a time.

    One clip failing does not stop the others — the UI shows which one broke and
    offers a retry, which is much better than losing eleven good generations.
    """
    story = selected_story(project)

    by_scene = {clip.scene_id: clip for clip in project.clips}
    pending = [
        scene
        for scene in story.scenes
        if force or (by_scene.get(scene.id) is None or by_scene[scene.id].status != "ready")
    ]

    if not pending:
        return {"ready": len(story.scenes), "failed": 0}

    log.info(
        "Generating %d clip(s) for project %s, %d at a time.",
        len(pending), project.id, settings.veo_concurrency,
    )

    per_scene: dict[str, float] = {}

    def publish() -> None:
        done = sum(per_scene.values())
        context.progress(
            done / len(pending), f"{int(done)} of {len(pending)} clips generated"
        )

    async def run(scene: Scene, seed_frame: InlineAsset | None = None) -> Clip:
        attempt = 0
        while True:
            attempt += 1
            try:

                async def track(fraction: float, _message: str) -> None:
                    per_scene[scene.id] = fraction
                    publish()

                return await generate_scene_clip(
                    project, scene.id, force=True, on_progress=track, seed_frame=seed_frame
                )
            except asyncio.CancelledError:
                raise
            except Exception as error:
                if attempt >= settings.veo_max_attempts:
                    raise
                log.warning(
                    "Scene %d failed on attempt %d; retrying. %s", scene.order, attempt, error
                )

    if settings.veo_chain_keyframes:
        return await _generate_chained(project, pending, run_scene=run, publish=publish)

    results = await map_with_concurrency(pending, settings.veo_concurrency, run)
    failed = sum(1 for result in results if isinstance(result, BaseException))

    return {"ready": len(pending) - failed, "failed": failed}


async def _tail_frame(clip: Clip, work: Path) -> InlineAsset | None:
    """The last frame of a finished clip, as a seed for the next one.

    Sampled a beat before the true end: the final frame of a generated clip is
    the one most likely to carry compression mush or the start of a fade, and
    seeding the next shot with a smeared frame propagates the smear.
    """
    if not clip.file:
        return None
    try:
        source = resolve_in_storage(clip.file)
        info = await probe(source)
        at = max(0.0, (info.seconds or 0.0) - 0.15)
        frame = work / f"tail-{clip.scene_id}.jpg"
        await extract_poster(source, frame, at)
        return asset_from_image(frame)
    except Exception as error:  # noqa: BLE001 — chaining is an optimisation
        log.warning("Could not take a tail frame from %s: %s", clip.scene_id, error)
        return None


async def _generate_chained(
    project: Project,
    pending: list[Scene],
    *,
    run_scene,
    publish,
) -> dict[str, int]:
    """Generate clips in story order, each seeded with the previous tail frame.

    Serial by necessity — clip N+1 cannot start until N has produced a frame.
    That is the cost of the feature, and it is why it is off by default.

    A failure breaks the chain but not the run: the next scene simply starts
    from its own keyframe, which is exactly the behaviour without chaining.
    """
    ordered = sorted(pending, key=lambda scene: scene.order)
    work = Path(work_dir(project.id)) / "chain"
    ensure_dir(work)

    log.info("Chaining %d clip(s) for project %s in story order.", len(ordered), project.id)

    ready = 0
    failed = 0
    seed: InlineAsset | None = None

    for scene in ordered:
        try:
            clip = await run_scene(scene, seed_frame=seed)
        except asyncio.CancelledError:
            raise
        except Exception as error:  # noqa: BLE001 — one bad clip must not stop the film
            log.warning("Scene %d failed; the chain restarts from scene %d. %s",
                        scene.order, scene.order + 1, error)
            failed += 1
            seed = None
            continue

        ready += 1
        seed = await _tail_frame(clip, work)
        publish()

    return {"ready": ready, "failed": failed}
