class BriefBody(BaseModel):
    brief: str = Field(min_length=8)
    seconds: int = Field(ge=4)
    style: StylePreset = "cinematic"
    ratio: AspectRatio = "16:9"
    pacing: Pacing | None = None
    references: list[str] = Field(default_factory=list)
    avoid: list[str] = Field(default_factory=list)
    language: str = "English"
    transition: TransitionStyle = "hardcut"
    quality: Literal["standard", "draft"] = "standard"

    def to_brief(self) -> ProjectBrief:
        return ProjectBrief(
            brief=self.brief.strip(),
            seconds=self.seconds,
            style=self.style,
            ratio=self.ratio,
            pacing=self.pacing or default_pacing(self.style),
            references=self.references,
            avoid=self.avoid,
            language=self.language,
        )


class BriefPatch(BaseModel):
    brief: str | None = None
    seconds: int | None = Field(default=None, ge=4)
    style: StylePreset | None = None
    ratio: AspectRatio | None = None
    pacing: Pacing | None = None
    references: list[str] | None = None
    avoid: list[str] | None = None
    language: str | None = None
    transition: TransitionStyle | None = None


class SelectStoryBody(BaseModel):
    storyId: str


class ScenePatch(BaseModel):
    prompt: str | None = None
    title: str | None = None
    summary: str | None = None
    negativePrompt: str | None = None
    transitionOut: TransitionStyle | None = None


class RewriteBody(BaseModel):
    instruction: str = ""


class TrackPatch(BaseModel):
    selected: bool | None = None
    gain: float | None = Field(default=None, ge=0, le=2)
    prompt: str | None = None
    label: str | None = None

@router.get("/plan")
async def get_plan(
    seconds: float = Query(default=30),
    style: str = Query(default="cinematic"),
    pacing: Pacing | None = Query(default=None),
) -> dict[str, Any]:
    resolved = pacing or default_pacing(style)
    plan = plan_duration(seconds, resolved)

    return {
        "plan": plan.model_dump(by_alias=True, mode="json"),
        "limits": {
            "maxClipSeconds": settings.veo_max_clip_seconds,
            "allowedClipSeconds": settings.allowed_clip_seconds,
            "maxVideoSeconds": settings.max_video_seconds,
            "maxSegments": settings.max_segments,
        },
    }

def _apply_analysis(analysis: BriefAnalysis) -> Callable[[Project], Project]:
    """Distribute a brief analysis across the project.

    Each part lands where the system that needs it will look: intent on the
    brief for the prompt layer, supers and end card on the project for the
    compositor, music on the brief for the composer.
    """

    def mutate(current: Project) -> Project:
        current.brief.intent = analysis.intent
        current.brief.unsupported = analysis.unsupported
        current.brief.music_direction = analysis.music
        # dict.fromkeys de-duplicates while keeping the user's ordering first.
        current.brief.avoid = list(dict.fromkeys([*current.brief.avoid, *analysis.avoid]))
        current.supers = analysis.supers
        current.end_card = analysis.end_card
        return current

    return mutate


@router.post("/projects", status_code=201)
async def post_project(body: BriefBody) -> dict[str, Any]:
    brief = body.to_brief()
    plan = plan_duration(brief.seconds, brief.pacing)

    project = Project(
        id=short_id("vid"),
        ownerId=current_owner_id(),
        brief=brief,
        plan=plan,
        transition=body.transition,
        model=settings.veo_fast_model if body.quality == "draft" else settings.veo_model,
    )
    await create_project(project)

    # Kick the treatments off immediately — the UI shows a spinner on the cards.
    async def work(context: JobContext) -> dict[str, int]:
        # Split the brief before anything reads it. The scene writer must never
        # see on-screen copy, or it will ask Veo to spell.
        context.progress(0.05, "Reading the brief…")
        analysis = await analyse_brief(brief.brief, seconds=brief.seconds, ratio=brief.ratio)
        analysed = await update_project(project.id, _apply_analysis(analysis))

        context.progress(0.2, "Writing director treatments…")
        stories = await generate_treatments(analysed.brief, plan)

        def mutate(current: Project) -> Project:
            current.stories = stories
            current.selected_story_id = stories[0].id if stories else None
            return current

        await update_project(project.id, mutate)
        return {"stories": len(stories)}

    job = start_job(project.id, "storyboard", work, message="Reading the brief…")
    return {"project": project.dump(), "jobId": job.id}

@router.patch("/projects/{project_id}")
async def patch_project(project_id: str, body: BriefPatch) -> dict[str, Any]:
    """Changing the brief re-plans the segments and invalidates the storyboard."""
    existing = await require_project(project_id)
    changes = body.model_dump(exclude_none=True)

    merged = existing.brief.model_copy(
        update={key: value for key, value in changes.items() if key != "transition"}
    )
    if body.style and not body.pacing:
        merged.pacing = default_pacing(body.style)

    duration_changed = (
        merged.seconds != existing.brief.seconds or merged.pacing != existing.brief.pacing
    )
    plan = plan_duration(merged.seconds, merged.pacing) if duration_changed else existing.plan

    # New brief text means the old split no longer describes it. A stale
    # `intent` is worse than none: the prompt layer would keep quoting the
    # previous brief while the supers came from the new one.
    text_changed = merged.brief != existing.brief.brief
    if text_changed:
        merged.intent = ""
        merged.unsupported = []
        merged.music_direction = None

    def mutate(current: Project) -> Project:
        current.brief = merged
        current.plan = plan
        if body.transition:
            current.transition = body.transition
        if text_changed:
            current.supers = []
            current.end_card = EndCard(enabled=False)
        if duration_changed or text_changed:
            # A new segment count or a new brief makes the scene list meaningless.
            for story in current.stories:
                story.scenes = []
            current.clips = []
        return current

    project = await update_project(project_id, mutate)

    if text_changed:
        analysis = await analyse_brief(merged.brief, seconds=merged.seconds, ratio=merged.ratio)
        project = await update_project(project_id, _apply_analysis(analysis))

    return {"project": project.dump()}

@router.post("/projects/{project_id}/stories/{story_id}/regenerate", status_code=202)
async def post_regenerate_story(project_id: str, story_id: str) -> dict[str, str]:
    project = await require_project(project_id)
    if not any(story.id == story_id for story in project.stories):
        raise NotFound(f"Story {story_id} not found.")

    async def work(context: JobContext) -> dict[str, str]:
        context.progress(0.2, "Finding a different direction…")
        replacement = await regenerate_treatment(
            project.brief, project.plan, project.stories, story_id
        )

        def mutate(current: Project) -> Project:
            current.stories = [
                replacement if story.id == story_id else story for story in current.stories
            ]
            if current.selected_story_id == story_id:
                current.clips = []
            return current

        await update_project(project_id, mutate)
        return {"storyId": story_id}

    return {"jobId": start_job(project_id, "storyboard", work, target_id=story_id).id}


@router.post("/projects/{project_id}/stories/{story_id}/scenes", status_code=202)
async def post_scenes(project_id: str, story_id: str) -> dict[str, str]:
    """Write the scene-by-scene prompts for one story."""
    project = await require_project(project_id)
    story = next((candidate for candidate in project.stories if candidate.id == story_id), None)
    if story is None:
        raise NotFound(f"Story {story_id} not found.")

    async def work(context: JobContext) -> dict[str, Any]:
        written, repair_passes = await generate_scenes(
            project.brief,
            project.plan,
            story,
            lambda fraction, message: context.progress(fraction, message),
        )

        def mutate(current: Project) -> Project:
            current.stories = [
                written if candidate.id == story_id else candidate for candidate in current.stories
            ]
            current.clips = sync_clips(current)
            return current

        await update_project(project_id, mutate)
        return {
            "storyId": story_id,
            "scenes": len(written.scenes),
            "audit": written.audit.model_dump(by_alias=True, mode="json"),
            "repairPasses": repair_passes,
        }

    job = start_job(
        project_id,
        "storyboard",
        work,
        target_id=story_id,
        message=f"Writing {len(project.plan.segments)} scene prompts…",
    )
    return {"jobId": job.id}

@router.patch("/projects/{project_id}/scenes/{scene_id}")
async def patch_scene(project_id: str, scene_id: str, body: ScenePatch) -> dict[str, Any]:
    """Direct edit of a scene prompt from the storyboard drawer."""
    changes = body.model_dump(exclude_none=True)
    field_map = {
        "prompt": "prompt",
        "title": "title",
        "summary": "summary",
        "negativePrompt": "negative_prompt",
        "transitionOut": "transition_out",
    }
    update = {field_map[key]: value for key, value in changes.items()}

    def mutate(current: Project) -> Project:
        for story in current.stories:
            story.scenes = [
                scene.model_copy(update=update) if scene.id == scene_id else scene
                for scene in story.scenes
            ]
        if "prompt" in update:
            # An edited prompt means the existing clip no longer matches the scene.
            current.clips = [
                clip.model_copy(update={"status": "idle", "progress": 0.0, "error": None})
                if clip.scene_id == scene_id
                else clip
                for clip in current.clips
            ]
        return current

    project = await update_project(project_id, mutate)
    return {"project": project.dump()}


@router.post("/projects/{project_id}/scenes/{scene_id}/rewrite", status_code=202)
async def post_rewrite_scene(project_id: str, scene_id: str, body: RewriteBody) -> dict[str, str]:
    project = await require_project(project_id)
    story = selected_story(project)
    scene = next((candidate for candidate in story.scenes if candidate.id == scene_id), None)
    if scene is None:
        raise NotFound(f"Scene {scene_id} not found in the selected story.")

    async def work(context: JobContext) -> dict[str, str]:
        context.progress(0.3, "Rewriting the scene…")
        rewritten = await rewrite_scene(project.brief, story, scene, body.instruction)

        def mutate(current: Project) -> Project:
            for candidate in current.stories:
                if candidate.id != story.id:
                    continue
                candidate.scenes = [
                    rewritten if existing.id == scene_id else existing
                    for existing in candidate.scenes
                ]
            current.clips = [
                clip.model_copy(update={"status": "idle", "progress": 0.0, "error": None})
                if clip.scene_id == scene_id
                else clip
                for clip in current.clips
            ]
            return current

        await update_project(project_id, mutate)
        return {"sceneId": scene_id}

    return {"jobId": start_job(project_id, "storyboard", work, target_id=scene_id).id}

@router.post("/projects/{project_id}/audio/suggest", status_code=202)
async def post_audio_suggest(project_id: str) -> dict[str, str]:
    project = await require_project(project_id)
    story = selected_story(project)

    async def work(context: JobContext) -> dict[str, int]:
        context.progress(0.3, "Writing music direction…")
        tracks = await suggest_tracks(project.brief, story, project.plan.target_seconds)

        def mutate(current: Project) -> Project:
            # Uploaded tracks survive a re-suggest; generated ideas are replaced.
            current.audio = [
                track for track in current.audio if track.source == "uploaded"
            ] + tracks
            return current

        await update_project(project_id, mutate)
        return {"tracks": len(tracks)}

    return {"jobId": start_job(project_id, "audio", work).id}

@router.patch("/projects/{project_id}/audio/{track_id}")
async def patch_audio(project_id: str, track_id: str, body: TrackPatch) -> dict[str, Any]:
    changes = body.model_dump(exclude_none=True)

    def mutate(current: Project) -> Project:
        target = next((track for track in current.audio if track.id == track_id), None)
        if target is None:
            return current

        updated = []
        for track in current.audio:
            if track.id == track_id:
                patch = dict(changes)
                # Editing the prompt invalidates the rendered audio.
                if (
                    "prompt" in patch
                    and patch["prompt"] != track.prompt
                    and track.source == "generated"
                ):
                    patch |= {"status": "idle", "file": None, "url": None}
                updated.append(track.model_copy(update=patch))
            elif changes.get("selected") is True and track.kind == target.kind:
                # Selecting a track deselects the others of the same kind.
                updated.append(track.model_copy(update={"selected": False}))
            else:
                updated.append(track)

        current.audio = updated
        return current

    project = await update_project(project_id, mutate)
    return {"project": project.dump()}


@router.post("/projects/{project_id}/audio/upload", status_code=201)
async def post_audio_upload(
    project_id: str,
    file: UploadFile = File(...),
    kind: str = Form(default="music"),
) -> dict[str, Any]:
    project = await require_project(project_id)
    payload = await file.read()

    if len(payload) > 40 * 1024 * 1024:
        raise BadRequest("Audio uploads are capped at 40MB.")
    if not payload:
        raise BadRequest("The uploaded file is empty.")

    track = await save_uploaded_track(
        project,
        file.filename or "track.mp3",
        payload,
        "voiceover" if kind == "voiceover" else "music",
    )
    return {"track": track.model_dump(by_alias=True, mode="json")}

@router.get("/projects/{project_id}/events")
async def stream_events(project_id: str, request: Request) -> StreamingResponse:
    """Live project state over SSE.

    Every event carries full state, so a dropped connection recovers simply by
    reconnecting — there is no incremental log to replay.
    """
    project = await require_project(project_id)
    queue = subscribe(project_id)

    async def generator():
        try:
            yield _sse("project", {"project": project.dump()})
            for job in list_jobs(project_id):
                yield _sse("job", {"job": job.dump()})

            while True:
                if await request.is_disconnected():
                    break
                try:
                    event = await asyncio.wait_for(queue.get(), timeout=20.0)
                except asyncio.TimeoutError:
                    # Heartbeat, so proxies keep the connection open through the
                    # quiet stretches of a long render.
                    yield ": ping\n\n"
                    continue

                kind = event.pop("type")
                yield _sse(kind, event)
        finally:
            unsubscribe(project_id, queue)

    return StreamingResponse(
        generator(),
        media_type="text/event-stream",
        headers={
            "Cache-Control": "no-cache, no-transform",
            "Connection": "keep-alive",
            "X-Accel-Buffering": "no",
        },
    )


def _sse(event: str, data: dict[str, Any]) -> str:
    return f"event: {event}\ndata: {json.dumps(data)}\n\n"
