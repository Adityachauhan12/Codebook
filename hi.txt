"""FastAPI service behind the React console."""
from __future__ import annotations

import logging
import threading
from pathlib import Path
from typing import Any

from fastapi import FastAPI, HTTPException
from fastapi.middleware.cors import CORSMiddleware
from fastapi.responses import PlainTextResponse
from pydantic import BaseModel

from .config import settings
from .knowledge.backends import probe_all
from .pipeline import STAGES, Pipeline, RunStore, discover_repos

logging.basicConfig(
    level=logging.INFO,
    format="%(asctime)s  %(levelname)-7s %(name)s  %(message)s",
)

app = FastAPI(title="Agentic Codebase Analyzer", version="1.0.0")

app.add_middleware(
    CORSMiddleware,
    allow_origins=["http://localhost:5173", "http://127.0.0.1:5173"],
    allow_methods=["*"],
    allow_headers=["*"],
)

store = RunStore()
pipeline = Pipeline(store)

# run_id -> thread, so a second start request on the same run is rejected
_active: dict[str, threading.Thread] = {}
_active_lock = threading.Lock()


class StartRunRequest(BaseModel):
    repo_path: str
    run_id: str | None = None


@app.get("/api/health")
def health() -> dict[str, Any]:
    ok, detail = settings.ready()
    backends = {
        name: {
            "available": status.available,
            "detail": status.detail,
            "executable": status.executable,
            "version": status.version,
        }
        for name, status in probe_all().items()
    }
    provider_detail = settings.llm_provider
    if settings.is_azure:
        provider_detail = (
            f"Azure OpenAI · {settings.azure_deployment} · "
            f"{'Responses' if settings.azure_use_responses_api else 'Chat'} API "
            f"· {settings.azure_api_version}"
        )

    return {
        "ok": ok,
        "detail": detail,
        "api_key_configured": not settings.key_is_placeholder,
        "dry_run": settings.dry_run,
        "provider": settings.llm_provider,
        "provider_detail": provider_detail,
        "azure_endpoint": settings.azure_endpoint if settings.is_azure else None,
        "knowledge_backend": settings.knowledge_backend,
        "knowledge_model": settings.knowledge_model,
        "deep_model": settings.deep_model,
        "worker_model": settings.worker_model,
        "backends": backends,
        "target_dir": str(settings.target_dir),
        "reports_dir": str(settings.reports_dir),
        "stages": [{"key": k, "label": l} for k, l in STAGES],
    }


@app.get("/api/repos")
def repos() -> dict[str, Any]:
    return {"repos": discover_repos()}


@app.get("/api/runs")
def runs() -> dict[str, Any]:
    return {"runs": store.list_runs()}


@app.post("/api/runs")
def start_run(request: StartRunRequest) -> dict[str, Any]:
    repo_path = Path(request.repo_path)
    if not repo_path.exists() or not repo_path.is_dir():
        raise HTTPException(404, f"Repository not found: {request.repo_path}")

    ok, detail = settings.ready()
    if not ok:
        raise HTTPException(400, detail)

    run_id = request.run_id or pipeline.new_run_id()

    with _active_lock:
        if run_id in _active and _active[run_id].is_alive():
            raise HTTPException(409, f"Run {run_id} is already in progress")

        thread = threading.Thread(
            target=pipeline.execute,
            kwargs={"repo_path": repo_path, "run_id": run_id},
            daemon=True,
            name=f"run-{run_id}",
        )
        _active[run_id] = thread
        thread.start()

    return {"run_id": run_id, "repo_name": repo_path.name, "status": "running"}


@app.get("/api/runs/{repo_name}/{run_id}")
def get_run(repo_name: str, run_id: str) -> dict[str, Any]:
    record = store.load(repo_name, run_id)
    if record is None:
        raise HTTPException(404, "Run not found")
    return record.to_dict()


@app.get("/api/runs/{repo_name}/{run_id}/artifacts/{name}", response_class=PlainTextResponse)
def get_artifact(repo_name: str, run_id: str, name: str) -> str:
    if "/" in name or "\\" in name or ".." in name:
        raise HTTPException(400, "Invalid artifact name")
    content = store.read_artifact(repo_name, run_id, name)
    if content is None:
        raise HTTPException(404, f"Artifact not found: {name}")
    return content


@app.get("/api/portfolio")
def portfolio() -> dict[str, Any]:
    """Cross-repository view - the reason this is a platform and not a script."""
    summary: list[dict[str, Any]] = []
    for run in store.list_runs():
        if run.get("status") != "completed":
            continue
        conv = store.read_artifact(run["repo_name"], run["run_id"], "convertibility.json")
        readiness, candidates = None, 0
        if conv:
            import json

            try:
                data = json.loads(conv)
                readiness = (data.get("overall") or {}).get("agentic_readiness_score")
                candidates = len(data.get("candidates") or [])
            except json.JSONDecodeError:
                pass
        stats = run.get("stats") or {}
        summary.append({
            "repo_name": run["repo_name"],
            "run_id": run["run_id"],
            "started_at": run.get("started_at"),
            "stacks": stats.get("stacks") or [],
            "languages": list((stats.get("languages") or {}).keys())[:6],
            "source_files": stats.get("total_source_files"),
            "lines": stats.get("total_lines"),
            "readiness": readiness,
            "candidates": candidates,
        })
    return {"portfolio": summary}
