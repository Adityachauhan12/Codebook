"""Audio has three sources, in descending order of importance:

  1. Veo's native audio, baked into every clip. Room tone, texture, body sound,
     dialogue. This is the bed the film actually sits on.
  2. A music track, either generated (when a provider is configured) or
     uploaded. Mixed underneath at ``MUSIC_GAIN``.
  3. A voiceover, synthesised from a script by the Azure TTS deployment.

Only (1) is guaranteed. Everything here degrades to "upload your own" rather
than blocking the render.
"""

from __future__ import annotations

from pathlib import Path
from urllib.parse import quote

import httpx
from pydantic import BaseModel, Field

from ..config import settings
from ..domain.models import AudioKind, AudioTrack, Project, ProjectBrief, StoryOption
from ..errors import BadRequest, Unavailable, UpstreamError
from ..ids import short_id
from ..logging_setup import get_logger
from ..skills.registry import compose_skill
from ..storage import audio_dir, ensure_dir, media_url, to_storage_relative
from .ffmpeg import probe
from .llm import chat_json
from .store import require_project, update_project

log = get_logger("audio")


class _MusicIdea(BaseModel):
    label: str = ""
    prompt: str = ""


class _Voiceover(BaseModel):
    label: str = "Voiceover"
    script: str = ""


class _Suggestions(BaseModel):
    music: list[_MusicIdea] = Field(default_factory=list)
    voiceover: _Voiceover | None = None


async def suggest_tracks(
    brief: ProjectBrief, story: StoryOption, target_seconds: int
) -> list[AudioTrack]:
    """Write music direction for the cut.

    The skill's sound vocabulary is loaded so the briefs come back as texture and
    instrumentation rather than adjectives.
    """
    max_words = int(target_seconds * 2.5)

    user = "\n".join(
        [
            "## The film",
            "",
            f"Brief: {brief.creative()}",
            *(
                [f"The brief's own music direction, to honour: {brief.music_direction}"]
                if brief.music_direction
                else []
            ),
            f"Length: {target_seconds}s. Style preset: {brief.style}. Ratio: {brief.ratio}.",
            f"Story: {story.title} — {story.treatment}",
            f"Emotion: {story.anchors.emotion}. Motif: {story.anchors.motif}. "
            f"Ends on: {story.anchors.final_image}",
            "",
            "The clips already carry native audio — room tone, body sound, and the sound motif. "
            "What you are writing is the score that sits UNDER that, plus an optional voiceover.",
            "",
            "## Task",
            "",
            "Write 5 distinct music directions. Each one names instrumentation, tempo in BPM, "
            "texture, and where it lifts or drops against the beat map. No adjective soup — "
            "'warm cinematic build' is not a direction.",
            "",
            f"Then write one optional voiceover script that fits {target_seconds} seconds at "
            f"roughly 2.5 words per second (about {max_words} words maximum). Leave it empty if "
            "the film is stronger without narration.",
            "",
            "## Output",
            "",
            "Return JSON only:",
            '{"music":[{"label":"","prompt":""}],"voiceover":{"label":"","script":""}}',
        ]
    )

    raw = await chat_json(
        [
            {"role": "system", "content": compose_skill("audio")},
            {"role": "user", "content": user},
        ],
        label="audio-directions",
        fast=True,
        temperature=0.9,
        max_output_tokens=2000,
    )

    try:
        parsed = _Suggestions.model_validate(raw)
    except Exception as error:  # noqa: BLE001
        raise UpstreamError("The model returned audio directions we could not read.") from error

    tracks = [
        AudioTrack(
            id=short_id("track"),
            kind="music",
            label=idea.label,
            prompt=idea.prompt,
            selected=index == 0,
            status="idle",
            gain=settings.music_gain,
        )
        for index, idea in enumerate(parsed.music[:5])
    ]

    if parsed.voiceover and parsed.voiceover.script.strip():
        tracks.append(
            AudioTrack(
                id=short_id("track"),
                kind="voiceover",
                label=parsed.voiceover.label or "Voiceover",
                prompt=parsed.voiceover.script.strip(),
                selected=False,
                status="idle",
                gain=settings.voiceover_gain,
            )
        )

    return tracks

# Music providers
# ---------------------------------------------------------------------------


async def _music_elevenlabs(prompt: str, seconds: int) -> tuple[bytes, str]:
    if not settings.elevenlabs_api_key:
        raise Unavailable("ELEVENLABS_API_KEY is not set.")

    length_ms = round(min(max(seconds, 10), 300) * 1000)

    async with httpx.AsyncClient(timeout=300.0) as client:
        response = await client.post(
            "https://api.elevenlabs.io/v1/music",
            headers={
                "xi-api-key": settings.elevenlabs_api_key,
                "content-type": "application/json",
            },
            json={
                "prompt": prompt,
                "music_length_ms": length_ms,
                "model_id": settings.elevenlabs_music_model,
            },
        )

    if response.status_code >= 400:
        raise UpstreamError(
            f"ElevenLabs music generation failed ({response.status_code}).",
            details=response.text[:400],
        )

    return response.content, "mp3"


async def _music_lyria(prompt: str) -> tuple[bytes, str]:
    if not (settings.google_cloud_project and settings.google_access_token):
        raise Unavailable(
            "Vertex Lyria needs GOOGLE_CLOUD_PROJECT and GOOGLE_ACCESS_TOKEN "
            "(run `gcloud auth print-access-token`)."
        )

    location = settings.google_cloud_location
    url = (
        f"https://{location}-aiplatform.googleapis.com/v1/projects/"
        f"{settings.google_cloud_project}/locations/{location}/publishers/google/models/"
        f"{settings.vertex_lyria_model}:predict"
    )

    async with httpx.AsyncClient(timeout=300.0) as client:
        response = await client.post(
            url,
            headers={
                "authorization": f"Bearer {settings.google_access_token}",
                "content-type": "application/json",
            },
            json={"instances": [{"prompt": prompt}], "parameters": {}},
        )

    if response.status_code >= 400:
        raise UpstreamError(
            f"Lyria music generation failed ({response.status_code}).", details=response.text[:400]
        )

    predictions = response.json().get("predictions") or []
    encoded = None
    if predictions:
        encoded = predictions[0].get("bytesBase64Encoded") or predictions[0].get("audioContent")
    if not encoded:
        raise UpstreamError("Lyria returned no audio.")

    import base64

    return base64.b64decode(encoded), "wav"

# Voiceover
# ---------------------------------------------------------------------------


async def _synthesise_voiceover(script: str) -> tuple[bytes, str]:
    deployment = settings.azure_openai_tts_deployment_name
    if not deployment:
        raise Unavailable(
            "Voiceover needs AZURE_OPENAI_TTS_DEPLOYMENT_NAME. Veo still generates in-scene "
            "dialogue without it."
        )
    if not settings.azure_configured:
        raise Unavailable("Azure OpenAI is not configured.")

    assert settings.azure_openai_base_url and settings.azure_openai_key
    url = (
        f"{settings.azure_openai_base_url.rstrip('/')}/openai/deployments/{quote(deployment)}"
        f"/audio/speech?api-version={quote(settings.azure_openai_version)}"
    )

    async with httpx.AsyncClient(timeout=300.0) as client:
        response = await client.post(
            url,
            headers={"api-key": settings.azure_openai_key, "content-type": "application/json"},
            json={
                "model": deployment,
                "input": script,
                "voice": settings.azure_openai_tts_voice,
                "response_format": "mp3",
            },
        )

    if response.status_code >= 400:
        raise UpstreamError(
            f"Voiceover synthesis failed ({response.status_code}).", details=response.text[:400]
        )

    return response.content, "mp3"

# Orchestration

async def _patch_track(project_id: str, track_id: str, **patch) -> None:
    def mutate(project: Project) -> Project:
        project.audio = [
            track.model_copy(update=patch) if track.id == track_id else track
            for track in project.audio
        ]
        return project

    await update_project(project_id, mutate)


async def generate_track(project: Project, track_id: str) -> AudioTrack:
    track = next((candidate for candidate in project.audio if candidate.id == track_id), None)
    if track is None:
        raise BadRequest(f"Audio track {track_id} not found.")

    await _patch_track(project.id, track_id, status="generating", error=None)

    try:
        seconds = project.plan.target_seconds

        if track.kind == "voiceover":
            payload, extension = await _synthesise_voiceover(track.prompt)
        elif settings.music_provider == "elevenlabs":
            payload, extension = await _music_elevenlabs(track.prompt, seconds)
        elif settings.music_provider == "vertex-lyria":
            payload, extension = await _music_lyria(track.prompt)
        else:
            raise Unavailable(
                "No music provider is configured. Set MUSIC_PROVIDER to `elevenlabs` or "
                "`vertex-lyria`, or upload a track instead."
            )

        destination = Path(audio_dir(project.id)) / f"{track_id}.{extension}"
        ensure_dir(destination.parent)
        destination.write_bytes(payload)

        try:
            info = await probe(destination)
            duration = info.seconds
        except Exception:  # noqa: BLE001 — duration is informational
            duration = None

        relative = to_storage_relative(destination)
        await _patch_track(
            project.id,
            track_id,
            status="ready",
            file=relative,
            url=media_url(relative),
            seconds=duration,
            source="generated",
            error=None,
        )

        log.info("Generated %s track %s.", track.kind, track_id)
        refreshed = await require_project(project.id)
        return next(candidate for candidate in refreshed.audio if candidate.id == track_id)

    except Exception as error:
        await _patch_track(project.id, track_id, status="failed", error=str(error))
        raise


async def save_uploaded_track(
    project: Project, filename: str, payload: bytes, kind: AudioKind = "music"
) -> AudioTrack:
    suffix = Path(filename).suffix.lower() or ".mp3"
    suffix = "".join(char for char in suffix if char.isalnum() or char == ".")

    track_id = short_id("track")
    destination = Path(audio_dir(project.id)) / f"{track_id}{suffix}"
    ensure_dir(destination.parent)
    destination.write_bytes(payload)

    try:
        info = await probe(destination)
        duration = info.seconds
    except Exception:  # noqa: BLE001
        duration = None

    relative = to_storage_relative(destination)
    track = AudioTrack(
        id=track_id,
        kind=kind,
        label=Path(filename).stem,
        prompt="Uploaded track",
        selected=True,
        status="ready",
        gain=settings.voiceover_gain if kind == "voiceover" else settings.music_gain,
        seconds=duration,
        file=relative,
        url=media_url(relative),
        source="uploaded",
    )

    def mutate(current: Project) -> Project:
        # An uploaded track of the same kind takes over the selection.
        current.audio = [
            existing.model_copy(update={"selected": False})
            if existing.kind == kind
            else existing
            for existing in current.audio
        ]
        current.audio.append(track)
        return current

    await update_project(project.id, mutate)
    return track
