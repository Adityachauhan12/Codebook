"""The accumulation step.

Takes the N independently generated Veo clips and produces one file of exactly
``plan.target_seconds``. Three phases, weighted for the progress bar roughly by
how long they actually take.
"""

from __future__ import annotations

import shutil
from pathlib import Path

from ..brand.assets import font_path
from ..brand.supers import build_lockups, build_supers
from ..config import settings
from ..domain.formats import output_size_for
from ..domain.models import Project, RenderResult, now_iso
from ..errors import BadRequest
from ..logging_setup import get_logger
from ..storage import (
    ensure_dir,
    media_url,
    output_dir,
    resolve_in_storage,
    to_storage_relative,
    work_dir,
)
from .clips import selected_story
from .ffmpeg import (
    MixTrack,
    Segment,
    burn_brand,
    concat_segments,
    mix_and_finalise,
    normalise_segment,
    probe,
)
from .jobs import JobContext
from .store import update_project

log = get_logger("render")

PHASE_NORMALISE = 0.50
PHASE_CONCAT = 0.18
PHASE_BRAND = 0.12
PHASE_MIX = 0.20


async def _set_render(project_id: str, **patch) -> Project:
    def mutate(project: Project) -> Project:
        project.render = project.render.model_copy(update=patch)
        return project

    return await update_project(project_id, mutate)


async def render_project(project: Project, context: JobContext) -> dict:
    story = selected_story(project)
    plan = project.plan

    # Order matters: the timeline is the story order, not the order clips finished.
    by_scene = {clip.scene_id: clip for clip in project.clips}
    ordered = [(scene, by_scene.get(scene.id)) for scene in story.scenes]

    missing = [scene for scene, clip in ordered if clip is None or clip.status != "ready" or not clip.file]
    if missing:
        listed = ", ".join(str(scene.order) for scene in missing)
        raise BadRequest(
            f"{len(missing)} of {len(ordered)} clips are not ready yet: scene {listed}."
        )

    work = Path(work_dir(project.id))
    out = Path(output_dir(project.id))
    ensure_dir(work)
    ensure_dir(out)

    segments_dir = work / "segments"
    # A re-render must not concatenate last run's leftovers.
    shutil.rmtree(segments_dir, ignore_errors=True)
    ensure_dir(segments_dir)

    width, height = output_size_for(project.brief.ratio, settings.veo_resolution)
    started_at = now_iso()

    await _set_render(
        project.id,
        status="rendering",
        progress=0.0,
        stage="Preparing clips…",
        started_at=started_at,
        error=None,
    )

    try:
        # --- Phase 1: conform every clip to identical parameters -------------
        normalised: list[Segment] = []
        total = len(ordered)

        for index, (scene, clip) in enumerate(ordered):
            assert clip is not None and clip.file
            source = resolve_in_storage(clip.file)
            target = segments_dir / f"seg-{index:03d}.mp4"
            stage = f"Conforming clip {index + 1} of {total}"

            await _set_render(project.id, stage=stage, progress=(index / total) * PHASE_NORMALISE)

            async def on_progress(fraction: float, _message: str, i: int = index) -> None:
                context.progress(((i + fraction) / total) * PHASE_NORMALISE, stage)

            await normalise_segment(
                source,
                target,
                width=width,
                height=height,
                seconds=scene.seconds,
                fps=settings.render_fps,
                on_progress=on_progress,
            )

            normalised.append(Segment(file=target, seconds=scene.seconds))

        # --- Phase 2: accumulate ---------------------------------------------
        joined_path = work / "joined.mp4"
        await _set_render(
            project.id,
            stage=f"Accumulating {len(normalised)} clips into one timeline…",
            progress=PHASE_NORMALISE,
        )

        async def concat_progress(fraction: float, message: str) -> None:
            context.progress(PHASE_NORMALISE + fraction * PHASE_CONCAT, message)

        joined_seconds = await concat_segments(
            normalised,
            joined_path,
            transition=project.transition,
            crossfade_seconds=0.4,
            work_dir=work,
            on_progress=concat_progress,
        )

        # Hard cuts land on the plan exactly; a crossfade eats (n-1) x fade seconds.
        final_seconds = round(min(plan.target_seconds, joined_seconds), 3)
        if final_seconds < plan.target_seconds - 0.05:
            log.info(
                "Crossfade overlap shortened the film to %.2fs against a %ds target.",
                final_seconds, plan.target_seconds,
            )

        # --- Phase 2b: burn supers and the brand mark ------------------------
        # Veo returns a clean plate by policy; everything a viewer reads is
        # drawn here, from a font file, so it is spelled the way it was written.
        supers = build_supers(
            project.supers, project.end_card, seconds=final_seconds, height=height
        )
        lockups = build_lockups(project.end_card, seconds=final_seconds, width=width)

        if supers or lockups:
            branded_path = work / "branded.mp4"
            await _set_render(
                project.id,
                stage=f"Compositing {len(supers)} super(s) and the brand mark…",
                progress=PHASE_NORMALISE + PHASE_CONCAT,
            )

            async def brand_progress(fraction: float, message: str) -> None:
                context.progress(
                    PHASE_NORMALISE + PHASE_CONCAT + fraction * PHASE_BRAND, message
                )

            await burn_brand(
                joined_path,
                branded_path,
                seconds=final_seconds,
                supers=supers,
                lockups=lockups,
                font=font_path(),
                work_dir=work / "supers",
                on_progress=brand_progress,
            )
            mix_source = branded_path
        else:
            mix_source = joined_path

        # --- Phase 3: mix and trim -------------------------------------------
        tracks = [
            MixTrack(
                file=resolve_in_storage(track.file),
                gain=track.gain,
                # Music loops to cover the runtime; voiceover plays once from the top.
                loop=track.kind == "music",
                fade_in=1.0 if track.kind == "music" else 0.0,
                fade_out=2.0 if track.kind == "music" else 0.0,
            )
            for track in project.audio
            if track.selected and track.status == "ready" and track.file
        ]

        final_path = out / "final.mp4"
        done_before_mix = PHASE_NORMALISE + PHASE_CONCAT + PHASE_BRAND
        await _set_render(
            project.id,
            stage="Mixing audio…" if tracks else "Finishing…",
            progress=done_before_mix,
        )

        async def mix_progress(fraction: float, message: str) -> None:
            context.progress(done_before_mix + fraction * PHASE_MIX, message)

        await mix_and_finalise(
            mix_source,
            final_path,
            seconds=final_seconds,
            tracks=tracks,
            fade_out_seconds=0.6,
            on_progress=mix_progress,
        )

        info = await probe(final_path)
        size = final_path.stat().st_size
        relative = to_storage_relative(final_path)

        result = RenderResult(
            status="ready",
            progress=1.0,
            stage="Done.",
            file=relative,
            url=media_url(relative),
            seconds=info.seconds,
            bytes=size,
            startedAt=started_at,
            completedAt=now_iso(),
        )

        await _set_render(project.id, **result.model_dump())
        log.info(
            "Rendered project %s: %.2fs, %.1fMB from %d clips.",
            project.id, info.seconds, size / 1_048_576, total,
        )

        # The conformed intermediates are large and reproducible. Keep the
        # joined file — a re-mix with different music skips phases 1 and 2 —
        # and drop the rest.
        shutil.rmtree(segments_dir, ignore_errors=True)

        return result.model_dump(by_alias=True, mode="json")

    except Exception as error:
        await _set_render(
            project.id, status="failed", stage=None, error=str(error), completed_at=now_iso()
        )
        raise
