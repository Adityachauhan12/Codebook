"""LLM access layer.

One job: given a system prompt, a user prompt and a JSON schema, return a dict
that validates against that schema - reliably, with retries, caching and a
no-API-key offline mode.

The vendor lives in providers.py. Everything above this layer (the three
agents) only calls `LLMClient.json()` and `LLMClient.text()`, so switching
between Azure OpenAI and Anthropic changes one line in .env and nothing else.
"""
from __future__ import annotations

import hashlib
import json
import logging
import re
import threading
from dataclasses import dataclass, field
from pathlib import Path
from typing import Any

from ..config import settings
from .providers import Provider, ProviderError, get_provider

log = logging.getLogger(__name__)

LLMError = ProviderError  # agents catch one error type regardless of vendor


@dataclass
class Usage:
    """Running token tally for a whole pipeline run."""

    input_tokens: int = 0
    output_tokens: int = 0
    cache_read_tokens: int = 0
    calls: int = 0
    cached_calls: int = 0
    _lock: threading.Lock = field(default_factory=threading.Lock, repr=False)

    def add(self, usage: dict[str, int]) -> None:
        with self._lock:
            self.calls += 1
            self.input_tokens += usage.get("input_tokens", 0)
            self.output_tokens += usage.get("output_tokens", 0)
            self.cache_read_tokens += usage.get("cache_read_tokens", 0)

    def hit_cache(self) -> None:
        with self._lock:
            self.cached_calls += 1

    def as_dict(self) -> dict[str, int]:
        with self._lock:
            return {
                "calls": self.calls,
                "cached_calls": self.cached_calls,
                "input_tokens": self.input_tokens,
                "output_tokens": self.output_tokens,
                "cache_read_tokens": self.cache_read_tokens,
            }


# --------------------------------------------------------------------------
# JSON recovery - structured outputs make this rare, but never assume
# --------------------------------------------------------------------------

_FENCE = re.compile(r"```(?:json)?\s*(.*?)```", re.DOTALL)


def extract_json(text: str) -> dict[str, Any]:
    """Pull the first JSON object out of a model response."""
    text = (text or "").strip()
    if not text:
        raise LLMError("model returned an empty response")

    for candidate in (text, *(m.group(1).strip() for m in _FENCE.finditer(text))):
        try:
            parsed = json.loads(candidate)
            if isinstance(parsed, dict):
                return parsed
        except json.JSONDecodeError:
            pass

    # Last resort: brace matching from the first '{'.
    start = text.find("{")
    if start != -1:
        depth, in_string, escaped = 0, False, False
        for index, char in enumerate(text[start:], start):
            if in_string:
                if escaped:
                    escaped = False
                elif char == "\\":
                    escaped = True
                elif char == '"':
                    in_string = False
                continue
            if char == '"':
                in_string = True
            elif char == "{":
                depth += 1
            elif char == "}":
                depth -= 1
                if depth == 0:
                    try:
                        return json.loads(text[start : index + 1])
                    except json.JSONDecodeError:
                        break
    raise LLMError(f"could not parse JSON from model response: {text[:400]}")


# --------------------------------------------------------------------------
# Disk cache
# --------------------------------------------------------------------------


class DiskCache:
    """Content-addressed cache so re-running a repo costs nothing for unchanged parts."""

    def __init__(self, directory: Path, enabled: bool = True):
        self.dir = directory
        self.enabled = enabled
        if self.enabled:
            self.dir.mkdir(parents=True, exist_ok=True)

    @staticmethod
    def key(*parts: str) -> str:
        digest = hashlib.sha256()
        for part in parts:
            digest.update(part.encode("utf-8", "replace"))
            digest.update(b"\x00")
        return digest.hexdigest()[:40]

    def get(self, key: str) -> dict[str, Any] | None:
        if not self.enabled:
            return None
        path = self.dir / f"{key}.json"
        if not path.exists():
            return None
        try:
            return json.loads(path.read_text(encoding="utf-8"))
        except (OSError, json.JSONDecodeError):
            return None

    def put(self, key: str, value: dict[str, Any]) -> None:
        if not self.enabled:
            return
        try:
            (self.dir / f"{key}.json").write_text(
                json.dumps(value, ensure_ascii=False), encoding="utf-8"
            )
        except OSError as exc:  # cache failures must never break a run
            log.warning("could not write cache entry: %s", exc)


# --------------------------------------------------------------------------
# Client
# --------------------------------------------------------------------------


class LLMClient:
    def __init__(self, usage: Usage | None = None, provider: Provider | None = None):
        self.usage = usage or Usage()
        self.cache = DiskCache(settings.cache_dir, settings.llm_cache)
        self.provider = provider or get_provider()

    def json(
        self,
        *,
        system: str,
        prompt: str,
        schema: dict[str, Any],
        deep: bool = False,
        max_tokens: int = 32000,
        label: str = "call",
    ) -> dict[str, Any]:
        """Ask for a response that conforms to `schema` and return it as a dict."""
        model = settings.deep_model if deep else settings.worker_model
        effort = settings.deep_effort if deep else settings.worker_effort
        schema_text = json.dumps(schema, sort_keys=True)

        cache_key = DiskCache.key(
            self.provider.name, model, effort, system, prompt, schema_text, "json"
        )
        hit = self.cache.get(cache_key)
        if hit is not None:
            self.usage.hit_cache()
            log.info("[%s] cache hit", label)
            return hit

        text, usage = self.provider.complete(
            system=system, prompt=prompt, schema=schema, model=model,
            effort=effort, max_tokens=max_tokens, label=label,
        )
        self.usage.add(usage)

        result = extract_json(text)
        self.cache.put(cache_key, result)
        return result

    def text(
        self,
        *,
        system: str,
        prompt: str,
        deep: bool = False,
        max_tokens: int = 32000,
        label: str = "call",
    ) -> str:
        model = settings.deep_model if deep else settings.worker_model
        effort = settings.deep_effort if deep else settings.worker_effort

        cache_key = DiskCache.key(self.provider.name, model, effort, system, prompt, "text")
        hit = self.cache.get(cache_key)
        if hit is not None:
            self.usage.hit_cache()
            return hit.get("text", "")

        text, usage = self.provider.complete(
            system=system, prompt=prompt, schema=None, model=model,
            effort=effort, max_tokens=max_tokens, label=label,
        )
        self.usage.add(usage)
        self.cache.put(cache_key, {"text": text})
        return text
