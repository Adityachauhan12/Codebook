import type { Brief, CampaignRecord, ChatSSEEvent, SSEEvent } from '../types'

// Empty by default: the backend serves this bundle from its own origin
// (Dockerfile copies frontend/dist into backend/static), so relative URLs
// hit the right API in production, staging, or behind any custom domain.
//
// VITE_API_BASE exists only for local dev, where Vite (5173) and the API
// (8000) are different origins. Keep it in .env.development, NOT .env --
// Vite loads .env in every mode, so a localhost value there gets compiled
// into the production bundle and the deployed site calls the user's machine.
const API_BASE = import.meta.env.VITE_API_BASE ?? ''

async function readSSEStream<T>(
  response: Response,
  onEvent: (event: T) => void,
) {
  if (!response.ok || !response.body) {
    throw new Error(`Request failed (${response.status})`)
  }

  const reader = response.body.getReader()
  const decoder = new TextDecoder()
  let buffer = ''

  const flush = (raw: string) => {
    const payload = raw
      .split(/\r?\n/)
      .map((l) => l.trim())
      .filter((l) => l.startsWith('data:'))
      .map((l) => l.slice(5).trim())
      .join('')
    if (!payload) return
    onEvent(JSON.parse(payload) as T)
  }

  while (true) {
    const { value, done } = await reader.read()
    if (done) break
    buffer += decoder.decode(value, { stream: true })
    let idx = buffer.indexOf('\n\n')
    while (idx >= 0) {
      flush(buffer.slice(0, idx))
      buffer = buffer.slice(idx + 2)
      idx = buffer.indexOf('\n\n')
    }
  }
  if (buffer.trim()) flush(buffer)
}

export async function sendChatMessage(
  sessionId: string,
  message: string,
  options: {
    signal?: AbortSignal
    onEvent: (event: ChatSSEEvent) => void
  },
) {
  const response = await fetch(`${API_BASE}/chat/message`, {
    method: 'POST',
    headers: { 'Content-Type': 'application/json', Accept: 'text/event-stream' },
    body: JSON.stringify({ session_id: sessionId, message }),
    signal: options.signal,
  })
  await readSSEStream<ChatSSEEvent>(response, options.onEvent)
}

export async function streamCampaign(
  brief: Brief,
  options: {
    signal?: AbortSignal
    onEvent: (event: SSEEvent) => void
  },
) {
  const response = await fetch(`${API_BASE}/campaign/run`, {
    method: 'POST',
    headers: { 'Content-Type': 'application/json', Accept: 'text/event-stream' },
    body: JSON.stringify(brief),
    signal: options.signal,
  })
  await readSSEStream<SSEEvent>(response, options.onEvent)
}

export async function refineCampaign(
  sessionId: string,
  message: string,
  currentBrief: Brief,
  options: {
    signal?: AbortSignal
    onEvent: (event: SSEEvent) => void
  },
) {
  const response = await fetch(`${API_BASE}/campaign/refine`, {
    method: 'POST',
    headers: { 'Content-Type': 'application/json', Accept: 'text/event-stream' },
    body: JSON.stringify({ session_id: sessionId, message, current_brief: currentBrief }),
    signal: options.signal,
  })
  await readSSEStream<SSEEvent>(response, options.onEvent)
}

/* -------------------------------------------------------------------------- */
/* Campaign history — server-backed, so every machine sees the same list.      */
/* The backend has always exposed these; the app previously read only          */
/* localStorage, which is why each browser showed a different set.             */
/* -------------------------------------------------------------------------- */

export async function fetchCampaigns(signal?: AbortSignal): Promise<CampaignRecord[]> {
  const response = await fetch(`${API_BASE}/campaigns`, {
    headers: { Accept: 'application/json' },
    signal,
  })
  if (!response.ok) throw new Error(`Could not load campaigns (${response.status})`)
  const data = await response.json()
  return Array.isArray(data) ? (data as CampaignRecord[]) : []
}

export async function fetchCampaign(
  campaignId: string,
  signal?: AbortSignal,
): Promise<CampaignRecord | null> {
  const response = await fetch(`${API_BASE}/campaigns/${encodeURIComponent(campaignId)}`, {
    headers: { Accept: 'application/json' },
    signal,
  })
  if (response.status === 404) return null
  if (!response.ok) throw new Error(`Could not load campaign (${response.status})`)
  return (await response.json()) as CampaignRecord
}

export async function saveCampaign(campaign: CampaignRecord): Promise<void> {
  const response = await fetch(`${API_BASE}/campaigns/save`, {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({ campaign }),
  })
  if (!response.ok) throw new Error(`Could not save campaign (${response.status})`)
}

export async function deleteCampaign(campaignId: string): Promise<void> {
  const response = await fetch(`${API_BASE}/campaigns/${encodeURIComponent(campaignId)}`, {
    method: 'DELETE',
  })
  if (!response.ok && response.status !== 404) {
    throw new Error(`Could not delete campaign (${response.status})`)
  }
}

import { Link, NavLink } from 'react-router-dom'
import { BarChart3, CalendarDays, LayoutDashboard, PanelLeftClose, PanelLeftOpen } from 'lucide-react'

type SidebarProps = {
  collapsed: boolean
  onToggle: () => void
}

const items = [
  { to: '/', label: 'Dashboard', icon: LayoutDashboard },
  { to: '/history', label: 'History', icon: CalendarDays },
  { to: '/campaigns', label: 'Campaigns', icon: BarChart3 },
]

export function Sidebar({ collapsed, onToggle }: SidebarProps) {
  return (
    <aside className={`sidebar ${collapsed ? 'sidebar--collapsed' : ''}`}>
      <div className="sidebar__brand">
        <Link to="/" className="sidebar__brand-link">
          {!collapsed && (
            <span className="sidebar__badge">M</span>
          )}
          {!collapsed && (
            <span>
              <strong>Marketing OS</strong>
              <small>Autonomous campaign studio</small>
            </span>
          )}
        </Link>
        <button className="icon-button" type="button" onClick={onToggle} aria-label="Toggle sidebar">
          {collapsed ? <PanelLeftOpen size={18} /> : <PanelLeftClose size={18} />}
        </button>
      </div>

      <nav className="sidebar__nav" aria-label="Primary">
        {items.map((item) => {
          const Icon = item.icon
          return (
            <NavLink key={item.to} to={item.to} className={({ isActive }) => `nav-item ${isActive ? 'is-active' : ''}`}>
              <Icon size={18} />
              {!collapsed && <span>{item.label}</span>}
            </NavLink>
          )
        })}
      </nav>

      <div className="sidebar__foot">
        {!collapsed && <p>Live SSE campaign pipeline</p>}
      </div>
    </aside>
  )
}
