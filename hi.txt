"""Brand compliance, enforced in code.

Same reasoning as ``app/skills/audit.py``: a model asked whether it followed the
brand book will say yes. These rules run against the structured storyboard the
model actually returned, and anything at ``error`` severity is fed back as a
repair brief before a single second of Veo time is spent.

The rules here encode ``policy.INDIGO_VIDEO_BRAND_POLICY``. When that document
changes, this file changes with it — a guardrail nobody checks is decoration.
"""

from __future__ import annotations

import re

from ..domain.models import AuditIssue, StoryOption

# ---------------------------------------------------------------------------
# Brand name
# ---------------------------------------------------------------------------

_INDIGO_TOKEN = re.compile(r"\bindigo\b", re.IGNORECASE)

#: "indigo" is also a legitimate colour word, and the approved palette language
#: ("deep indigo navy-blue") depends on it. Colour usage is allowed; brand usage
#: must be the exact IndiGo casing.
_COLOUR_BEFORE = re.compile(
    r"\b(?:deep|dark|pale|light|rich|soft|muted|bright|colour|color|shade|tint|tone|hue)\s*[-,]?\s*$",
    re.IGNORECASE,
)
_COLOUR_AFTER = re.compile(
    r"^\s*[-,]?\s*(?:blue|navy|violet|dye|tone|hue|tint|shade|grade|palette|wash|cast)\b",
    re.IGNORECASE,
)

#: Name forms the brand book explicitly forbids.
_BANNED_NAME_FORMS = re.compile(
    r"\b(?:air\s+indigo|indigo\s+airways|indigo\s+airlines|goindigo(?!\.in\b)|"
    r"indigoing|indigoes|indigone|indiwent)\b",
    re.IGNORECASE,
)

#: The URL must carry both capitals.
_URL_TOKEN = re.compile(r"\bgo\s?indigo\s?\.\s?in\b", re.IGNORECASE)


def _brand_name_issues(text: str) -> list[str]:
    """Return the misspelt brand-name occurrences in ``text``."""
    problems: list[str] = []

    for match in _INDIGO_TOKEN.finditer(text):
        token = match.group(0)
        if token == "IndiGo":
            continue
        before = text[max(0, match.start() - 24) : match.start()]
        after = text[match.end() : match.end() + 24]
        if _COLOUR_BEFORE.search(before) or _COLOUR_AFTER.match(after):
            continue  # the colour, not the brand
        problems.append(token)

    return problems


# ---------------------------------------------------------------------------
# Frame content
# ---------------------------------------------------------------------------

#: Asking Veo for any of these produces a mangled mark or unreadable type.
#: Policy section 2 — all marks and type are composited after the render.
_MARK_OR_TYPE = re.compile(
    r"\b(?:logo|logotype|wordmark|brand\s?mark|arrow[- ]plane|20[- ]dot|twenty[- ]dot|"
    # "supers" plural only — "Super 35" and "super wide" are legitimate craft terms.
    r"subtitle|caption|supers\b|lower[- ]third|end\s?card|title\s?card|"
    # The original pattern matched "on-screen text" only, so a scene reading
    # "Final on-screen message ... as clean brand-safe typography" sailed
    # through and Veo drew the end card itself. Cover the synonyms.
    r"on[- ]screen\s+(?:text|message|copy|type|typography|graphic|line)|"
    r"typograph(?:y|ic)|typeface|"
    # "<copy> appears/reads/remains" is how a writer smuggles in a super
    # without ever using the word "text".
    r"(?:text|message|copy|headline|tagline|wording|lettering)\s+"
    r"(?:appears|appear|reads|read|remains|remain|sits|fades|holds)|"
    r"text\s+overlay|overlay\s+text|tail\s+letter|fuselage\s+letter|livery\s+letter|"
    r"signage\s+read|readable\s+(?:text|signage|lettering))\b",
    re.IGNORECASE,
)

#: Banned dominant palettes. Policy section 3.
_BANNED_COLOURS = re.compile(
    r"\b(?:red|crimson|scarlet|maroon|purple|violet|magenta|fuchsia)\b", re.IGNORECASE
)

#: A prompt that correctly restates the palette *names* the banned colours in
#: order to forbid them ("no red, no purple") — the guardrails ask for exactly
#: that. Negation is therefore scoped to the clause, so "no red, crimson,
#: magenta" clears the whole list while "crimson sunset" in its own clause does
#: not. Getting this wrong would flag the very compliance we asked for.
_NEGATION_CUE = re.compile(
    r"\b(?:no|not|never|avoid|avoiding|without|banned?|ban|excluding|exclude|free\s+of|"
    r"absent|zero|minus|prohibited|forbidden)\b",
    re.IGNORECASE,
)
_CLAUSE_SPLIT = re.compile(r"[.;\n]")


#: How far back a negation can reach. Long enough to cover "banned: red,
#: crimson, purple, violet", short enough that an unrelated "not" earlier in a
#: sentence does not silently excuse a colour genuinely in frame.
_NEGATION_WINDOW = 60


def _unnegated(pattern: re.Pattern[str], text: str) -> list[str]:
    """Matches of ``pattern`` that are asserted, not forbidden.

    Compliant prompts name the banned things in order to ban them — the
    continuity block says "no logo, wordmark or on-screen text in frame", and
    that block is restated in every scene. Matching naively would flag the exact
    compliance the guardrails demand, on every single scene. A negation only
    covers what follows it, so "crimson and gold, no blue" is still crimson.
    """
    found: list[str] = []
    for clause in _CLAUSE_SPLIT.split(text):
        for match in pattern.finditer(clause):
            lookback = clause[max(0, match.start() - _NEGATION_WINDOW) : match.start()]
            if not _NEGATION_CUE.search(lookback):
                found.append(match.group(0).lower())
    return found


def _banned_colour_hits(text: str) -> set[str]:
    return set(_unnegated(_BANNED_COLOURS, text))

#: The brand's own blues. The first cut of this accepted any word containing
#: "blue", and real output sailed through on "cobalt blue" — a bright greenish
#: blue that is not Pantone 2738 C. Checking for *a* blue is not checking for
#: *the* blue.
#: Named ``_REQUIRED_BLUE`` because that is what the two call sites ask for. It
#: was previously defined as ``_BRAND_BLUE`` and referenced as ``_REQUIRED_BLUE``
#: — a half-applied rename that left `brand_issues()` raising ``NameError`` on
#: every story with scenes, which took the whole brand audit down with it.
_REQUIRED_BLUE = re.compile(
    r"\b(?:indigo|navy|sky\s+blue|pale\s+blue|powder\s+blue)\b", re.IGNORECASE
)

#: Blues that are recognisably not IndiGo's.
_OFF_BRAND_BLUE = re.compile(
    r"\b(?:cobalt|azure|cyan|turquoise|teal|cerulean|electric\s+blue|royal\s+blue|"
    r"baby\s+blue|periwinkle|sapphire)\b",
    re.IGNORECASE,
)

#: The brand name itself, correctly spelled. Naming the airline in a prompt is
#: what makes Veo draw the wordmark on cabin walls and seat-backs — the exact
#: thing the composite-in-post design exists to prevent. Describe the cabin by
#: its colour instead.
_BRAND_NAME_IN_PROMPT = re.compile(r"\bIndiGo\b")

#: A gold-drenched grade. The brand wants clean, bright, generous daylight; a
#: sunrise-soaked frame reads as warm nostalgia, not IndiGo, however blue the
#: palette line claims to be. Lighting beats palette in a render.
_WARM_GRADE = re.compile(
    r"\b(?:golden\s+hour|sunset|sunrise|amber|golden|gold(?:en)?\s+light|honey\s+light|"
    r"warm\s+gold|orange\s+glow|burnt\s+orange|sodium|firelight|candlelit)\b",
    re.IGNORECASE,
)

#: Camera behaviour that contradicts "effortless". Policy section 4.
_FRANTIC_CAMERA = re.compile(
    r"\b(?:whip\s?pan|snap\s?zoom|crash\s?zoom|speed\s?ramp|shaky|frantic|chaotic|"
    r"jerky|violent\s+(?:shake|movement))\b",
    re.IGNORECASE,
)

# ---------------------------------------------------------------------------
# Safety. Policy section 9 — zero tolerance.
# ---------------------------------------------------------------------------

_SAFETY_TERMS = re.compile(
    r"\b(?:crash(?:es|ing|ed)?|collision|colliding|mid[- ]air|emergency\s+landing|"
    r"crash[- ]land|aircraft\s+(?:on\s+fire|in\s+distress)|uncontrolled\s+descent|"
    r"nose[- ]dive|nosedive|runway\s+(?:incident|overrun)|oxygen\s+mask|evacuation\s+slide|"
    r"emergency\s+slide|evacuat(?:e|ing|ion)|severe\s+turbulence|explosion|exploding|"
    r"detonat|terroris|sabotage|hijack|wreckage|debris\s+field|casualt|fatalit)\b",
    re.IGNORECASE,
)

# ---------------------------------------------------------------------------
# Tone. Policy sections 1 and 6.
# ---------------------------------------------------------------------------

_CHEAP_TONE = re.compile(
    r"\b(?:cheap(?:est)?|dirt[- ]cheap|bargain|discount|rock[- ]bottom|slash(?:ed|ing)?\s+(?:price|fare)|"
    r"unbeatable\s+(?:price|fare|deal)|lowest\s+(?:price|fare)\s+ever)\b",
    re.IGNORECASE,
)

#: "routes"/"sectors" are trade words; the brand book says "flights".
_TRADE_WORDS = re.compile(r"\b(?:routes|sectors)\b", re.IGNORECASE)

_ALL_CAPS_RUN = re.compile(r"\b[A-Z]{4,}\b")

#: Acronyms and codes that are legitimately upper case.
_CAPS_ALLOWLIST = {
    "FPV", "SFX", "LED", "HDR", "IMAX", "JSON", "MMKG", "USB", "IFE", "ATC", "VFX",
}

#: Distress. Policy section 3b — nobody in an IndiGo film is frightened. The
#: craft beat map pushes hard toward tension, so this is the backstop that
#: catches it when the rebranded beat labels do not.
_DISTRESS = re.compile(
    r"\b(?:frightened|afraid|fear(?:ful)?|scared|terrified|panic(?:ked|king)?|anxious|anxiety|"
    r"dread|nervous|distress(?:ed)?|desperate|despair|helpless|overwhelmed|trembl(?:e|ing)|"
    r"shak(?:y|ing)\s+hands|tears?\s+of\s+(?:fear|grief)|sob(?:bing)?|weep(?:ing)?|grief|"
    r"clench(?:ed|ing)?\s+jaw|white[- ]knuckle|brac(?:e|ing)\s+for|flinch|cower|"
    r"stress(?:ed|ful)?|worried|worry|uneasy|apprehensive)\b",
    re.IGNORECASE,
)

#: Friction blamed on the journey itself. Policy section 3b — the obstacle is
#: never IndiGo and never flying.
_JOURNEY_FRICTION = re.compile(
    r"\b(?:delay(?:ed|s)?|cancelled\s+flight|long\s+quel?ue|queue\s+snak|lost\s+(?:bag|luggage)|"
    r"missed\s+(?:the\s+)?(?:flight|connection)|cramped|crowded\s+(?:cabin|gate)|turbulence|"
    r"stuffy|delayed\s+boarding|rush(?:ing|ed)\s+through\s+security)\b",
    re.IGNORECASE,
)

#: Things Veo *can* render that read as IndiGo. Policy section 2b — with the
#: mark banned, a scene carrying none of these has no brand in it at all.
_BRAND_SURFACE = re.compile(
    r"\b(?:cabin|aisle|seat[- ]?back|seat\s+fabric|headrest|tray\s+table|overhead\s+bin|"
    r"cabin\s+window|porthole|crew|attendant|steward(?:ess)?|pilot|uniform|lanyard|"
    r"boarding\s+pass|bag\s+tag|baggage\s+tag|aircraft|airplane|aeroplane|plane|jet|"
    r"fuselage|wing(?:tip)?|tail|livery|jet\s?bridge|air\s?bridge|tarmac|runway|"
    r"gate|terminal|departure\s+board|trolley)\b",
    re.IGNORECASE,
)

#: A hero shot of hardware with nobody in it. Policy section 5 — a warning, not
#: a block: aviation settings are legitimate in a brand film, the cliché is not.
_AVIATION_SUBJECT = re.compile(
    r"\b(?:aircraft|airplane|aeroplane|plane|jet|airbus|a320|a321|runway|tarmac|"
    r"airport|terminal|apron|taxiway)\b",
    re.IGNORECASE,
)
_HUMAN_PRESENCE = re.compile(
    r"\b(?:passenger|traveller|traveler|child|kid|boy|girl|man|woman|mother|father|"
    r"crew|attendant|pilot|hand|face|eyes|shoulder|family|couple|person|people|"
    r"grandmother|grandfather|daughter|son|figure|silhouette\s+of\s+a)\b",
    re.IGNORECASE,
)


def _all_caps_hits(text: str) -> list[str]:
    return [
        word
        for word in _ALL_CAPS_RUN.findall(text)
        if word not in _CAPS_ALLOWLIST and not word.isdigit()
    ]


def brand_issues(story: StoryOption) -> list[AuditIssue]:
    """Audit one story option against the IndiGo video guardrails."""
    issues: list[AuditIssue] = []

    def push(rule: str, severity: str, message: str) -> None:
        issues.append(AuditIssue(rule=rule, severity=severity, message=message))  # type: ignore[arg-type]

    def at(scene, rule: str, severity: str, message: str) -> None:
        issues.append(
            AuditIssue(
                sceneId=scene.id,
                sceneOrder=scene.order,
                rule=rule,
                severity=severity,  # type: ignore[arg-type]
                message=message,
            )
        )

    # --- Story level --------------------------------------------------------
    story_text = " ".join(
        [story.title, story.summary, story.treatment, story.angle, *story.continuity.rules]
    )

    misspelt = _brand_name_issues(story_text)
    if misspelt:
        push(
            "brand/name",
            "error",
            f"The brand name is written as {', '.join(sorted(set(misspelt)))}. "
            "It is always IndiGo — capital I, capital G.",
        )

    if _BANNED_NAME_FORMS.search(story_text):
        push(
            "brand/name-form",
            "error",
            "Forbidden name form. Never Air IndiGo, IndiGo Airways, IndiGo Airlines, or "
            "goindigo, and never the name as a verb.",
        )

    for url in _URL_TOKEN.findall(story_text):
        if url != "goindigo.in":
            push(
                "brand/url",
                "warning",
                f'The website is written as "{url}". It is goindigo.in — one word, lowercase, '
                "no space before the dot.",
            )
            break

    palette = story.continuity.palette
    if palette:
        if not _REQUIRED_BLUE.search(palette):
            push(
                "brand/palette-blue",
                "error",
                "The locked palette does not name a blue. Deep indigo navy-blue "
                "(Pantone 2738 C) must dominate, with pale sky blue (Pantone 291 C) secondary.",
            )
        banned = _banned_colour_hits(palette)
        if banned:
            push(
                "brand/palette-banned",
                "error",
                f"The locked palette contains prohibited colours: {', '.join(sorted(banned))}. "
                "Red and purple palettes are banned.",
            )

    if not any(
        _MARK_OR_TYPE.search(rule) or "text" in rule.lower() for rule in story.continuity.rules
    ):
        push(
            "brand/no-mark-rule",
            "warning",
            "The continuity block does not state that no logo, wordmark, or on-screen text "
            "appears in frame. Lock it there so every scene inherits it.",
        )

    caps = _all_caps_hits(f"{story.title} {story.summary}")
    if caps:
        push(
            "brand/sentence-case",
            "warning",
            f"ALL CAPS in story copy: {', '.join(sorted(set(caps))[:4])}. "
            "IndiGo is always sentence case.",
        )

    if _CHEAP_TONE.search(story_text):
        push(
            "brand/tone",
            "error",
            "Discount-led language. Low-cost must never read as cheap — sell effortlessness, "
            "not price.",
        )

    if _TRADE_WORDS.search(story_text):
        push("brand/vocabulary", "warning", 'Use "flights", never "routes" or "sectors".')

    # --- Scene level --------------------------------------------------------
    for scene in story.scenes:
        prompt = scene.prompt or ""
        card = scene.shot_card
        haystack = "\n".join(
            [
                prompt,
                scene.summary,
                *(value for value in card.model_dump().values() if isinstance(value, str)),
            ]
        )

        marks = _unnegated(_MARK_OR_TYPE, haystack)
        if marks:
            at(
                scene,
                "brand/generated-mark",
                "error",
                f'This scene asks the model to render "{marks[0]}". Veo cannot render a '
                "wordmark or type legibly — it will come back misspelt and off-colour. Remove it; "
                "the mark is composited from the master file after the render.",
            )

        scene_misspelt = _brand_name_issues(haystack)
        if scene_misspelt:
            at(
                scene,
                "brand/name",
                "error",
                f"Brand name written as {', '.join(sorted(set(scene_misspelt)))}. It is always "
                "IndiGo. Better still: describe the colour as deep indigo navy-blue and keep the "
                "brand name out of the prompt entirely.",
            )

        # Even correctly spelled, the name in a prompt is what makes Veo paint a
        # wordmark on the fuselage and the seat-backs — and it comes back as
        # "IndlGo" with a mangled device beside it. The mark is composited from
        # the master file; the prompt describes colour and form only.
        named = _BRAND_NAME_IN_PROMPT.findall(prompt)
        if named:
            at(
                scene,
                "brand/name-in-prompt",
                "error",
                f"This prompt names IndiGo {len(named)} time(s). Veo renders a named brand as "
                "drawn lettering, and it cannot spell — that is where the broken fuselage "
                "wordmark comes from. Describe it instead: 'deep indigo navy-blue tail and "
                "engine accents, pearl white fuselage'. The mark is composited after the render.",
            )

        off_blue = _unnegated(_OFF_BRAND_BLUE, haystack)
        if off_blue:
            at(
                scene,
                "brand/off-brand-blue",
                "warning",
                f"Off-brand blue in frame: {', '.join(sorted(set(off_blue)))}. Checking for *a* "
                "blue is not checking for *the* blue — use deep indigo navy-blue, with pale sky "
                "blue secondary.",
            )

        # Lighting beats palette in a render: a frame can name indigo in its
        # palette line and still come back entirely amber if the key is a
        # sunrise. Warn only when nothing pulls the other way.
        warm = _unnegated(_WARM_GRADE, haystack)
        if warm and not _REQUIRED_BLUE.search(f"{card.lighting} {card.palette}"):
            at(
                scene,
                "brand/warm-grade",
                "warning",
                f"The grade is carried by warm light ({', '.join(sorted(set(warm))[:3])}) with no "
                "brand blue in the lighting or palette line. Lighting beats palette in a render — "
                "this frame will come back gold, which reads as nostalgia rather than IndiGo.",
            )

        if not _REQUIRED_BLUE.search(f"{card.palette} {prompt}"):
            at(
                scene,
                "brand/palette-blue",
                "error",
                "No blue in this scene. Blue must dominate every frame, not just the average of "
                "the film — Veo has no memory between clips, so the palette must be restated here.",
            )

        banned = _banned_colour_hits(f"{card.palette}. {prompt}")
        if banned:
            at(
                scene,
                "brand/palette-banned",
                "error",
                f"Prohibited colours in frame: {', '.join(sorted(banned))}. "
                "Red and purple palettes are banned.",
            )

        if _SAFETY_TERMS.search(haystack):
            at(
                scene,
                "brand/safety",
                "error",
                "This scene depicts an aviation incident, emergency, or destruction. Absolutely "
                "prohibited — rewrite the beat entirely rather than softening it.",
            )

        if _FRANTIC_CAMERA.search(haystack):
            at(
                scene,
                "brand/camera-calm",
                "warning",
                "Frantic camera work contradicts effortless. Prefer a calm, motivated move — "
                "slow push-in, gentle track, or locked-off.",
            )

        if _CHEAP_TONE.search(haystack):
            at(scene, "brand/tone", "error", "Discount-led language. Never frame IndiGo as cheap.")

        distress = _unnegated(_DISTRESS, haystack)
        if distress:
            at(
                scene,
                "brand/distress",
                "error",
                f'This scene plays as distress ("{distress[0]}"). Nobody in an IndiGo film is '
                "frightened or stressed — an airline cannot advertise anxiety. Express the same "
                "beat as longing, anticipation, or relief instead.",
            )

        friction = _unnegated(_JOURNEY_FRICTION, haystack)
        if friction:
            at(
                scene,
                "brand/journey-friction",
                "error",
                f'This scene blames the journey ("{friction[0]}"). The obstacle is never IndiGo '
                "and never flying — flying is what resolves the story. Distance and time apart "
                "are fair game; delays, queues, and cramped cabins are not.",
            )

        if not _BRAND_SURFACE.search(haystack):
            at(
                scene,
                "brand/no-brand-surface",
                "error",
                "Nothing in this scene reads as IndiGo. With the logo and all type banned from "
                "the plate, brand has to be carried by what Veo can render — cabin interior, "
                "crew uniform, aircraft form, or a blue brand object. Otherwise this is stock "
                "footage with a blue grade.",
            )

        caps = _all_caps_hits(prompt)
        if caps:
            at(
                scene,
                "brand/sentence-case",
                "warning",
                f"ALL CAPS in the prompt: {', '.join(sorted(set(caps))[:4])}.",
            )

        if _AVIATION_SUBJECT.search(haystack) and not _HUMAN_PRESENCE.search(haystack):
            at(
                scene,
                "brand/aviation-cliche",
                "warning",
                "An aircraft, runway, or terminal with no human presence is a stock aviation "
                "shot, not a scene. Put a person and a feeling in the frame.",
            )

    return issues
