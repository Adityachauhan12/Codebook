"""Turn a project's copy into positioned, branded compositing instructions.

``domain.models`` holds *what* a super says and roughly where it sits;
``services.ffmpeg`` knows how to draw one. This is the layer between: it owns
type scale, safe areas, and the brand's own end-card composition.

Sizes are expressed as a fraction of frame height so a 1080p render and a 720p
render look identical rather than one of them looking like a ransom note.
"""

from __future__ import annotations

from ..domain.models import EndCard, SuperPosition, SuperSpec
from ..services.ffmpeg import Lockup, Super
from . import active_brand
from .assets import icon, wordmark

__all__ = ["build_supers", "build_lockups"]

#: Broadcast-style safe area. Copy inside this margin survives the crop that
#: social platforms apply to a 16:9 master.
_MARGIN = 0.062

#: Type scale, as a fraction of frame height.
_BODY = 0.041
_HEADLINE = 0.055
_SUBLINE = 0.030


#: Multiplier from type size to baseline-to-baseline distance.
_LINE_HEIGHT = 1.34


def _x(position: SuperPosition, height: int) -> str:
    """The ffmpeg x expression. ``tw`` is the measured text width."""
    if position in {"upper-centre", "lower-centre", "centre"}:
        return "(W-tw)/2"
    return str(int(height * _MARGIN))


def _y(position: SuperPosition, *, line: int, lines: int, size: int, height: int) -> str:
    """The ffmpeg y expression for one line of a stacked block.

    Line height is computed from the type size rather than read back from
    ffmpeg's ``th``, because ``th`` measures whatever that single line happens
    to contain — a line of all-caps and a line with descenders would other‑wise
    sit at different pitches inside the same super.
    """
    margin = int(height * _MARGIN)
    step = int(size * _LINE_HEIGHT)

    if position in {"upper-left", "upper-centre"}:
        return str(margin + line * step)
    if position in {"lower-left", "lower-centre"}:
        return f"H-{margin + (lines - line) * step}"
    return f"(H-{lines * step})/2+{line * step}"


def _stack(
    text: str,
    *,
    position: SuperPosition,
    start: float,
    end: float,
    size: int,
    height: int,
    fade: float = 0.5,
    y_offset: int = 0,
) -> list[Super]:
    """One ``Super`` per line.

    drawtext will not render a multi-line block cleanly: with ``expansion=none``
    it breaks the line *and* draws a tofu glyph where the newline was. Splitting
    here is what stops a box appearing mid-super, and it buys exact control over
    leading as a side effect.
    """
    lines = [line for line in text.splitlines() if line.strip()]
    if not lines:
        return []

    drawn: list[Super] = []
    for index, line in enumerate(lines):
        y = _y(position, line=index, lines=len(lines), size=size, height=height)
        if y_offset:
            y = f"{y}+{y_offset}"
        drawn.append(
            Super(
                text=line,
                start=start,
                end=end,
                x=_x(position, height),
                y=y,
                size=size,
                fade=fade,
            )
        )
    return drawn


def build_supers(
    items: list[SuperSpec],
    end_card: EndCard,
    *,
    seconds: float,
    height: int,
) -> list[Super]:
    """Every line of burnt-in copy for the film, in timeline order."""
    supers: list[Super] = []

    for item in items:
        supers += _stack(
            item.text,
            position=item.position,
            start=max(0.0, item.start),
            end=min(seconds, item.end),
            size=int(height * _BODY),
            height=height,
        )

    if not end_card.enabled:
        return supers

    # The card resolves over the tail of the film. The mark sits optically
    # above centre and the copy below it, so neither fights the other.
    start = max(0.0, seconds - end_card.seconds)

    if end_card.headline:
        supers += _stack(
            end_card.headline,
            position="centre",
            start=start + 0.6,
            end=seconds,
            size=int(height * _HEADLINE),
            height=height,
            fade=0.7,
            y_offset=int(height * 0.085),
        )
    if end_card.subline:
        supers += _stack(
            end_card.subline,
            position="centre",
            start=start + 1.0,
            end=seconds,
            size=int(height * _SUBLINE),
            height=height,
            fade=0.7,
            y_offset=int(height * 0.165),
        )

    return supers


def build_lockups(end_card: EndCard, *, seconds: float, width: int) -> list[Lockup]:
    """The brand mark for the end card, or nothing when brand is off.

    White cut only: the card resolves over sky, cloud or runway, all of which
    are mid-to-bright but never light enough for the blue cut to hold.
    """
    if not end_card.enabled or end_card.mark == "none" or active_brand() != "indigo":
        return []

    file = wordmark(on_dark=True) if end_card.mark == "wordmark" else icon(on_dark=True)
    # The lockup reads at about a quarter of frame width; smaller and the
    # 20-dot device closes up, larger and it stops feeling like a sign-off.
    target = int(width * (0.26 if end_card.mark == "wordmark" else 0.10))

    return [
        Lockup(
            file=file,
            start=max(0.0, seconds - end_card.seconds),
            width=target,
            x="(W-w)/2",
            y="(H-h)/2-" + str(int(width * 0.035)),
            fade=0.8,
        )
    ]
