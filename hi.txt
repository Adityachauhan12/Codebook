"""Loader for the vendored video-prompting skill.

The prompting craft lives in markdown, not in string literals scattered through
the services. ``app/skills/video_prompting`` is a vendored copy of the ``video``
skill from github.com/smixs/visual-skills (CC BY 4.0, Serge Shima — see NOTICE),
plus ``PIPELINE.md``, our own addendum binding the skill to the Veo-segment
architecture.

Files are read once and cached. Loading all twelve references into every call
would be ~90k characters of mostly irrelevant model syntax, so each task
declares exactly what it needs — the same routing the skill's own SKILL.md
prescribes.
"""

from __future__ import annotations

from functools import lru_cache
from pathlib import Path
from typing import Literal

from ..brand import active_brand, brand_system_block
from ..logging_setup import get_logger

log = get_logger("skills")

SKILL_DIR = Path(__file__).resolve().parent / "video_prompting"

SKILL_ATTRIBUTION = (
    "Prompting craft adapted from Visual Skills by Serge Shima "
    "(github.com/smixs/visual-skills), CC BY 4.0."
)

SkillTask = Literal["storyboard", "repair", "scene", "audio"]

#: Task -> reference files, mirroring the router in the skill's SKILL.md.
TASK_REFERENCES: dict[str, list[str]] = {
    # Step 1 dramaturgy, step 2 universal rules, step 3 the one model file, then
    # the task-shaped modules: genres for the style preset, vocabulary so the
    # writer has precise camera/light/sound terms, role-modes for storyboard mode.
    "storyboard": [
        "dramaturgy",
        "universal-rules",
        "veo",
        "patterns-and-genres",
        "camera-lighting-vocabulary",
        "role-modes",
    ],
    # A repair pass already has the structure; it needs the rules it broke and
    # the catalogue of known failure modes.
    "repair": ["universal-rules", "fixes-and-skeletons", "veo", "camera-lighting-vocabulary"],
    # Rewriting one scene in place.
    "scene": ["universal-rules", "veo", "camera-lighting-vocabulary", "fixes-and-skeletons"],
    # Sound design ideas for the music bed.
    "audio": ["camera-lighting-vocabulary", "patterns-and-genres"],
}


@lru_cache(maxsize=64)
def _read(relative: str) -> str:
    path = SKILL_DIR / relative
    try:
        return path.read_text(encoding="utf-8")
    except OSError:
        log.error("Missing skill file %s — the prompt will be weaker without it.", relative)
        return ""


def _reference(name: str) -> str:
    return _read(f"references/{name}.md")


@lru_cache(maxsize=8)
def compose_skill(task: SkillTask) -> str:
    """Build the system prompt for a task.

    The role, the vendored craft files it routes to, and the pipeline addendum
    that pins everything to Veo segments.
    """
    names = TASK_REFERENCES[task]
    parts: list[str] = [_read("SKILL.md").strip()]

    parts.append(
        "\n".join(
            [
                "---",
                "",
                "# Reference files loaded for this task",
                "",
                "The mandatory reading order in the router above has already been resolved for you.",
                f"Task: **{task}**. The files below are the ones it routes to — their contents "
                "follow in full.",
                "Do not ask for other files; work with these.",
            ]
        )
    )

    for name in names:
        body = _reference(name).strip()
        if body:
            parts.append(f"---\n\n<!-- references/{name}.md -->\n\n{body}")

    parts.append("---\n\n" + _read("PIPELINE.md").strip())

    # Brand goes last so it is the most recent instruction in the system prompt,
    # and it says outright that it overrides the craft references above.
    brand = brand_system_block()
    if brand:
        parts.append(brand)

    system = "\n\n".join(part for part in parts if part)
    log.debug(
        "Composed skill for %s: %d references, brand=%s, %d chars",
        task, len(names), active_brand() or "none", len(system),
    )
    return system


def skill_info() -> dict[str, object]:
    """Cheap accessor for the health endpoint."""
    return {
        "loaded": bool(_read("SKILL.md")),
        "attribution": SKILL_ATTRIBUTION,
        "source": "https://github.com/smixs/visual-skills",
        "license": "CC BY 4.0",
        "references": {
            task: [f"{name}.md" for name in names] for task, names in TASK_REFERENCES.items()
        },
    }

"""The dramaturgy and three-detail checks, enforced in code.

The skill says its two checks — the six-point dramaturgy check and the
three-detail audit — are the step writers skip, and that skipping them silently
degrades everything downstream. An LLM asked to self-check will cheerfully claim
it did.

So the check runs here, against the structured scene the model returned.
Anything at ``error`` severity is fed back to the model as a repair brief;
warnings are surfaced in the UI but do not block.
"""

from __future__ import annotations

import re

from ..brand import brand_issues

#: Words that mark an absent detail. Section 7 of PIPELINE.md.
BANNED_PHRASES = [
    "cinematic masterpiece",
    "professional",
    "high quality",
    "highest quality",
    "masterpiece",
    "stunning",
    "epic",
    "amazing",
    "beautiful lighting",
    "beautiful light",
    "dynamic camera",
    "intense moment",
    "powerful scene",
    "breathtaking",
    "award-winning",
    "award winning",
    "hyperrealistic",
    "ultra realistic",
    "ultra-realistic",
    "photorealistic masterpiece",
    "8k",
    "4k",
    "best quality",
    "highly detailed",
]

#: "cinematic" alone is banned, but "cinematic naturalism" is a named grade the
#: skill's own worked examples use. Allow the compound, reject the adjective.
ALLOWED_CINEMATIC_COMPOUNDS = ("cinematic naturalism", "cinematic widescreen")

TELL_NOT_SHOW = re.compile(
    r"\b(?:he|she|they|the (?:man|woman|hero|subject|character)|it)\s+"
    r"(?:is|are|was|were|looks?|seems?|feels?)\s+(?:very\s+|really\s+|so\s+)?"
    r"(?:sad|angry|afraid|scared|happy|excited|nervous|anxious|joyful|frustrated|"
    r"lonely|proud|confident|surprised)\b",
    re.IGNORECASE,
)

LENS_PATTERN = re.compile(
    r"\b(?:\d{2,3}\s?mm|macro|anamorphic|telephoto|wide-angle|wide angle)\b", re.IGNORECASE
)

CAMERA_MOVE_PATTERN = re.compile(
    r"\b(?:static camera|locked[- ]off|slow push[- ]in|fast push[- ]in|push[- ]in|pull[- ]back|"
    r"dolly[- ](?:in|out)|tracking|lateral track|handheld|whip pan|snap zoom|rack focus|"
    r"tilt (?:up|down)|orbit|gimbal|crane|aerial|dolly[- ]zoom|pan (?:left|right)|drift|glide|"
    r"arc|steadicam|FPV|top[- ]down)\b",
    re.IGNORECASE,
)

LIGHT_PATTERN = re.compile(
    r"\b(?:key(?:\s?light)?|rim(?:\s?light)?|backlight|back light|fill|bounce|practical|"
    r"underlight|top light|side light|spill|silhouette|motivated)\b",
    re.IGNORECASE,
)

LIGHT_DIRECTION_PATTERN = re.compile(
    r"\b(?:frame[- ](?:left|right)|from (?:left|right|behind|above|below|"
    r"the (?:window|door|screen|street))|overhead|off[- ]screen (?:left|right)|"
    r"camera[- ](?:left|right)|three[- ]quarter)\b",
    re.IGNORECASE,
)

COLOUR_PATTERN = re.compile(
    r"\b(?:red|orange|amber|gold(?:en)?|yellow|green|teal|cyan|blue|indigo|violet|magenta|pink|"
    r"crimson|scarlet|sepia|ochre|beige|cream|ivory|white|grey|gray|black|brown|tan|rust|bronze|"
    r"copper|silver|charcoal|slate|desaturated|monochrome|warm|cool|cold)\b",
    re.IGNORECASE,
)

AUDIO_LINE = re.compile(r"^\s*audio\s*:", re.IGNORECASE | re.MULTILINE)

FINAL_IMAGE_PATTERN = re.compile(
    r"\b(?:ends? on|final frame|final image|closes? on|last frame|holds? on|freezes? on|out on)\b",
    re.IGNORECASE,
)

DIALOGUE_PATTERN = re.compile(
    r"(?:says|whispers|shouts|mutters|asks|replies|answers)\s*[:,]\s*[\"“]([^\"”]+)",
    re.IGNORECASE,
)

#: Words per second of natural speech; Veo caps spoken audio at ~8s.
WORDS_PER_SECOND = 2.5


def _contains_banned(text: str) -> list[str]:
    lower = text.lower()
    hits = [phrase for phrase in BANNED_PHRASES if phrase in lower]

    # "cinematic" on its own, but not as part of an allowed compound.
    for match in re.finditer(r"cinematic", lower):
        index = match.start()
        allowed = any(lower.startswith(compound, index) for compound in ALLOWED_CINEMATIC_COMPOUNDS)
        if not allowed and "cinematic" not in hits:
            hits.append("cinematic")
            break

    return hits


def _filled(value: str | None, min_length: int = 12) -> bool:
    return bool(value and len(value.strip()) >= min_length)


def _significant_token(phrase: str) -> str | None:
    """Longest word in a phrase, lowercased — a cheap proxy for "was this restated"."""
    words = [word for word in re.sub(r"[^a-z0-9\s-]", " ", phrase.lower()).split() if len(word) > 4]
    return max(words, key=len) if words else None


def _audit_scene(scene: Scene, story: StoryOption) -> list[AuditIssue]:
    issues: list[AuditIssue] = []

    def at(rule: str, severity: str, message: str) -> None:
        issues.append(
            AuditIssue(
                sceneId=scene.id,
                sceneOrder=scene.order,
                rule=rule,
                severity=severity,  # type: ignore[arg-type]
                message=message,
            )
        )

    prompt = scene.prompt or ""
    card = scene.shot_card
    haystack = "\n".join(
        [prompt, *(str(value) for value in card.model_dump().values() if isinstance(value, str))]
    )

    if len(prompt.strip()) < 120:
        at("prompt-substance", "error", "Prompt is too thin to direct a shot.")

    # --- Details Law: the three-detail check --------------------------------
    if not _filled(card.environmental_pressure):
        at(
            "three-detail/environment",
            "error",
            "No environmental pressure. Name a physical fact of the space that carries the emotion.",
        )
    if not _filled(card.micro_action):
        at(
            "three-detail/micro-action",
            "error",
            "No physical micro-action. Translate the emotion onto the body — jaw, hands, breath, eyes.",
        )
    if not _filled(card.sound_anchor, 8):
        at(
            "three-detail/motif",
            "error",
            "No sound anchor or visual motif tying this shot to the spine of the piece.",
        )

    # --- Banned vocabulary --------------------------------------------------
    banned = _contains_banned(haystack)
    if banned:
        at(
            "banned-vocabulary",
            "error",
            f"Placeholder words instead of detail: {', '.join(banned)}. "
            "Replace each with a concrete physical fact.",
        )

    if TELL_NOT_SHOW.search(haystack):
        at("show-dont-tell", "error", "An emotion is named without a body. Render it as an action.")

    # --- Camera and light ---------------------------------------------------
    if not LENS_PATTERN.search(haystack):
        at("lens", "error", "No lens stated. Add a focal length (24/35/50/85/100mm macro).")
    if not CAMERA_MOVE_PATTERN.search(haystack):
        at(
            "camera-move",
            "error",
            "No camera direction. Name one primary move, or state 'static camera'.",
        )
    if not _filled(card.movement_reason, 6):
        at("motivated-camera", "warning", "Camera move has no stated reason. Answer 'what changed?'")
    if not LIGHT_PATTERN.search(haystack):
        at("light-source", "error", "No named light source or role (key, rim, fill, practical).")
    if not LIGHT_DIRECTION_PATTERN.search(haystack):
        at("light-direction", "warning", "Light has no direction. Say where it comes from.")
    if not COLOUR_PATTERN.search(f"{card.palette} {prompt}"):
        at("palette", "warning", "Palette is not defined with concrete colours.")

    # --- Composition and gaze -----------------------------------------------
    if not _filled(card.eye_trace, 6):
        at("controlled-gaze", "warning", "No eye-trace. Say where the viewer looks in the first 0.3s.")

    # --- Veo specifics ------------------------------------------------------
    if not AUDIO_LINE.search(prompt):
        at(
            "veo/audio-line",
            "error",
            "No `Audio:` line. Veo will not generate sound unless it is labelled explicitly.",
        )
    if not FINAL_IMAGE_PATTERN.search(prompt):
        at(
            "final-image",
            "error",
            "No explicit final frame. The model needs an emotional destination.",
        )
    if not re.search(rf"duration\s*:\s*{scene.seconds}\b", prompt, re.IGNORECASE):
        at(
            "duration-discipline",
            "warning",
            f'Prompt does not state "Duration: {scene.seconds} seconds", which is what this '
            "segment will be generated at.",
        )

    # Dialogue budget — Veo speeds up delivery when a line does not fit.
    for match in DIALOGUE_PATTERN.finditer(prompt):
        line = match.group(1).strip()
        words = len([word for word in line.split() if word])
        budget = int(min(scene.seconds, 8) * WORDS_PER_SECOND)
        if words > budget:
            at(
                "dialogue-budget",
                "error",
                f'"{line[:48]}…" is {words} words but this {scene.seconds}s shot only holds '
                f"about {budget}. Cut the line.",
            )

    # --- Continuity ---------------------------------------------------------
    # Two different bars, because they mean different things. Palette and
    # lighting are the film's grade — every scene shares them, no exceptions.
    # Characters and wardrobe are a roster — a multi-character story naturally
    # has scenes that feature only some of them (a ground-crew scene has no
    # reason to mention the flight attendant), so the bar is "at least one
    # locked identity is present", not "the whole cast is present every time".
    continuity = story.continuity
    lower_prompt = prompt.lower()

    universal_tokens = [
        token
        for token in (_significant_token(v) for v in (continuity.palette, continuity.lighting_constant))
        if token
    ]
    missing_universal = [token for token in universal_tokens if token not in lower_prompt]
    if missing_universal:
        at(
            "continuity-block",
            "error",
            "The locked palette or lighting constant is not restated in this prompt. These are "
            "the two things that must be identical in every scene — Veo has no memory between "
            "generations.",
        )

    roster_tokens = [
        token
        for token in (
            _significant_token(v) for v in (*continuity.characters, *continuity.wardrobe)
        )
        if token
    ]
    if roster_tokens and not any(token in lower_prompt for token in roster_tokens):
        at(
            "continuity-block",
            "error",
            "None of the story's locked characters or wardrobe are referenced in this prompt. "
            "Every scene should feature — and describe consistently with — at least one "
            "identity from the continuity block, not a new unlocked person.",
        )

    if not _filled(scene.continuity_note, 10):
        at(
            "cut-match",
            "warning",
            "No note on how this scene joins the previous one. State the matching pose, light and "
            "lens height.",
        )

    return issues


def _audit_story(story: StoryOption) -> list[AuditIssue]:
    issues: list[AuditIssue] = []

    def push(rule: str, severity: str, message: str) -> None:
        issues.append(AuditIssue(rule=rule, severity=severity, message=message))  # type: ignore[arg-type]

    # --- The five anchors ---------------------------------------------------
    anchors = story.anchors
    missing = [
        name
        for name, value in (
            ("emotion", anchors.emotion),
            ("motif", anchors.motif),
            ("object", anchors.object),
            ("break", anchors.break_),
            ("finalImage", anchors.final_image),
        )
        if not _filled(value, 3)
    ]
    if missing:
        push("five-anchors", "error", f"Missing anchors: {', '.join(missing)}. Commit to all five.")

    # --- Continuity block ---------------------------------------------------
    continuity = story.continuity
    if not continuity.characters:
        push("continuity/characters", "warning", "No character identity locked for this story.")
    if not _filled(continuity.palette, 8):
        push("continuity/palette", "error", "No palette defined with concrete colours.")
    if not _filled(continuity.lighting_constant, 8):
        push(
            "continuity/lighting",
            "error",
            "No lighting constant. Name the dominant source and direction.",
        )

    # --- The motif must actually recur --------------------------------------
    motif_token = _significant_token(anchors.motif)
    if motif_token and len(story.scenes) > 2:
        recurrences = sum(
            1
            for scene in story.scenes
            if motif_token in f"{scene.prompt} {scene.shot_card.sound_anchor}".lower()
        )
        if recurrences < 2:
            push(
                "motif-recurrence",
                "warning",
                f'The motif "{anchors.motif}" appears in {recurrences} scene(s). '
                "A motif that does not recur is decoration.",
            )

    # --- The arc must be complete -------------------------------------------
    beats = {scene.beat for scene in story.scenes}
    for required in ("crack", "impact"):
        if required not in beats:
            push(
                "beat-map",
                "warning",
                f'No "{required}" beat in the cut. Never skip the Crack or the Impact.',
            )

    # --- Every shot must do a job -------------------------------------------
    functions = [scene.shot_function for scene in story.scenes]
    if len(functions) > 3 and len(set(functions)) < 3:
        push(
            "three-jobs",
            "warning",
            "Nearly every shot has the same function. Vary establish / power / pressure / detail / "
            "reaction / shift / impact.",
        )

    return issues


def audit_story_option(story: StoryOption) -> AuditReport:
    issues = _audit_story(story)
    for scene in story.scenes:
        issues.extend(_audit_scene(scene, story))

    # Craft and brand are separate axes, but they share one report and one repair
    # loop — a beautifully directed film in the wrong blue is still a reshoot.
    issues.extend(brand_issues(story))

    errors = sum(1 for issue in issues if issue.severity == "error")
    warnings = len(issues) - errors

    # A clean story scores 100; each error costs 6 points, each warning 1.5.
    score = max(0, round(100 - errors * 6 - warnings * 1.5))

    return AuditReport(passed=errors == 0, score=score, issues=issues)


def build_repair_brief(story: StoryOption, report: AuditReport) -> str:
    """Turn failed checks into a brief the model can act on, grouped by scene."""
    errors = [issue for issue in report.issues if issue.severity == "error"]
    if not errors:
        return ""

    story_level = [issue for issue in errors if issue.scene_order is None]
    by_scene: dict[int, list[AuditIssue]] = {}
    for issue in errors:
        if issue.scene_order is None:
            continue
        by_scene.setdefault(issue.scene_order, []).append(issue)

    lines = [
        "The automated dramaturgy, three-detail and brand audit rejected this storyboard.",
        "Fix every item below and return the corrected storyboard in the same JSON schema.",
        "Rules prefixed `brand/` are brand-book violations and are not negotiable — never "
        "satisfy one by removing the scene's substance.",
        "Keep the scene count, the per-scene `seconds`, the beats and the shot functions exactly "
        "as they are — only the writing changes.",
        "",
    ]

    if story_level:
        lines.append("## Story level")
        lines.extend(f"- [{issue.rule}] {issue.message}" for issue in story_level)
        lines.append("")

    scenes_by_order = {scene.order: scene for scene in story.scenes}
    for order in sorted(by_scene):
        scene = scenes_by_order.get(order)
        suffix = f" — {scene.title} ({scene.seconds}s, {scene.beat})" if scene else ""
        lines.append(f"## Scene {order}{suffix}")
        lines.extend(f"- [{issue.rule}] {issue.message}" for issue in by_scene[order])
        lines.append("")

    return "\n".join(lines)

"""Two-pass storyboarding.

  Pass 1 — treatments. Four genuinely different directions in one call, each
           committing to the five anchors and a continuity block. Cheap, and it
           is all the UI needs to render the story cards.
  Pass 2 — scenes. Run per story, on demand, because writing 4 x 12 shot cards
           up front burns tokens on three directions the user will discard.

Every pass-2 result goes through the programmatic audit; failures come back as a
repair brief rather than shipping mush to Veo.
"""

def _brief_block(brief: ProjectBrief, plan: DurationPlan) -> str:
    lines = [
        "## The brief",
        "",
        f"Idea: {brief.brief.strip()}",
        f"Deliverable: {plan.target_seconds} seconds, {brief.ratio}, "
        f'style preset "{brief.style}", pacing "{brief.pacing}".',
        f"Response language for titles/summaries/treatments: {brief.language}. "
        "Prompts stay in English.",
    ]

    if brief.references:
        lines += ["", "Reference material the user attached:"]
        lines += [f"- {item}" for item in brief.references]
    if brief.avoid:
        lines += ["", "Must not appear:"]
        lines += [f"- {item}" for item in brief.avoid]

    lines += [
        "",
        "## The delivery constraint",
        "",
        f"Veo renders 4, 6 or 8 second clips. This film is {len(plan.segments)} separate "
        f"generations accumulated by ffmpeg into one {plan.target_seconds}s file.",
        *[f"- {note}" for note in plan.notes],
    ]

    brand = brand_brief_block()
    if brand:
        lines += ["", brand]

    return "\n".join(lines)


def _segment_table(plan: DurationPlan) -> str:
    rows = [
        f"| {segment.index + 1} | {segment.starts_at}s–{segment.starts_at + segment.seconds}s "
        f"| {segment.seconds}s | {segment.beat} | {segment.shot_function} "
        f"| {beat_label(segment.beat)} |"
        for segment in plan.segments
    ]
    return "\n".join(
        [
            "| Scene | Timecode | Seconds | Beat | Shot function | What this beat is for |",
            "|---|---|---|---|---|---|",
            *rows,
        ]
    )


def _veo_skeleton(seconds: int) -> str:
    """The `prompt` field's shape. Sound and duration are NOT part of it.

    Both are 100% mechanical — duration is already fixed by the plan, and the
    `Audio:`/`Duration:` lines are boilerplate the writer kept dropping under
    narrative pressure across a long freeform string. So the split moved them
    to their own JSON field / plan value and `_materialise_scenes` appends both
    deterministically. That guarantees every scene carries them regardless of
    what the model does — see the audit rules `veo/audio-line` and
    `duration-discipline`, which exist to catch exactly this failure mode.
    """
    return "\n".join(
        [
            "```text",
            "<Camera framing, lens, one primary move>. <Character identity block> "
            "<physical action> in <environment>.",

def _scenes_task(brief: ProjectBrief, plan: DurationPlan, story: StoryOption) -> str:
    first_seconds = plan.segments[0].seconds if plan.segments else 8
    return "\n".join(
        [
            _brief_block(brief, plan),
            "",
            _story_context(story),
            "",
            "## The shot plan — fixed, do not change it",
            "",
            _segment_table(plan),
            "",
            f"Write exactly {len(plan.segments)} scenes, one per row, in order. The seconds, beat "
            "and shot function of each row are already decided by the renderer — write to them. "
            "Do not merge, split, add or drop a scene.",
            "",
            "## How each prompt must be written",
            "",
            "Open with camera and lens: the model drops camera direction that arrives late. Then "
            "the identity block, then the action.",
            "Every scene carries, without exception: one environmental pressure, one physical "
            "micro-action on a body, one sound anchor or visual motif, a stated focal length, "
            "exactly one primary camera move with a reason, a named light source with its "
            "direction, named colours, the restated continuity block, and an explicit final frame.",
            "",
            "Skeleton to fill, per scene (this example is sized for one scene — use that scene's "
            "own seconds):",
            _veo_skeleton(first_seconds),
            "",
            "Budget the action to the seconds: a 4s shot holds one action and a reaction, an 8s "
            "shot holds two or three with a pause.",
            "Dialogue is optional and must fit the clip — roughly 2.5 words per second, never "
            "more than 8 seconds of speech.",
            "Do not ask Veo for music; a score is mixed in afterwards. Ask it for room tone, "
            "texture and the motif.",
            "",
            "`continuityNote` states how this scene joins the previous one — the matching pose, "
            "light and lens height that make the cut work. Scene 1's note describes the opening "
            "frame instead.",
            "",
            "## Output",
            "",
            "Return JSON only, this exact shape:",
            "```json",
            json.dumps(_SCENE_SHAPE, indent=2),
            "```",
        ]
    )


#: Strips a model-written Audio:/SFX:/Says:/Duration: line if one slipped into
#: `prompt` anyway, so composing our own copy can never double them up.
_BOILERPLATE_LINE = re.compile(r"^\s*(?:audio|sfx|says|duration)\s*:.*$\n?", re.IGNORECASE | re.MULTILINE)

_DEFAULT_AUDIO = "room tone and ambient texture consistent with the scene, no music."


def _compose_negative(written: str) -> str:
    """The model's negative prompt, with the brand's own bans forced back in.

    The model writes a per-scene negative, so brand terms cannot live in the
    fallback alone — a scene where the model wrote anything at all would ship
    without them. The brand bans are the reason the plate stays clean enough to
    composite a real logo onto, so they are appended unconditionally.
    """
    parts = [written.strip() or _CRAFT_NEGATIVE]
    brand = brand_negative_prompt()
    if brand:
        existing = parts[0].lower()
        parts += [term for term in (t.strip() for t in brand.split(",")) if term.lower() not in existing]
    return ", ".join(part for part in parts if part)


def _compose_prompt(base_prompt: str, audio: str, seconds: int) -> str:
    """Assemble the final Veo prompt from the model's shot description plus the
    two boilerplate lines it kept dropping — `Audio:` and `Duration:` — added
    here deterministically instead of trusted to survive a long freeform
    string. See the comment on `_veo_skeleton` for why.
    """
    body = _BOILERPLATE_LINE.sub("", base_prompt).strip()
    sound = audio.strip() or _DEFAULT_AUDIO
    return f"{body}\n\nAudio: {sound}\n\nDuration: {seconds} seconds."


def _materialise_scenes(
    plan: DurationPlan, story: StoryOption, written: list[_WrittenScene]
) -> list[Scene]:
    by_order = {scene.order: scene for scene in written}
    scenes: list[Scene] = []

    for index, segment in enumerate(plan.segments):
        order = index + 1
        # Fall back to positional matching when the model mislabels `order`.
        entry = by_order.get(order) or (written[index] if index < len(written) else None)
        if entry is None:
            raise UpstreamError(
                f"The storyboard model returned {len(written)} scenes but the plan needs "
                f"{len(plan.segments)}."
            )

        previous = story.scenes[index] if index < len(story.scenes) else None
        transition = entry.transition_out if entry.transition_out in ("hardcut", "crossfade") else "hardcut"

        scenes.append(
            Scene(
                id=previous.id if previous else short_id("scene"),
                order=order,
                title=entry.title or f"Scene {order}",
                beat=segment.beat,
                shotFunction=segment.shot_function,
                seconds=segment.seconds,
                startsAt=segment.starts_at,
                summary=entry.summary,
                prompt=_compose_prompt(entry.prompt, entry.audio, segment.seconds),
                shotCard=entry.shot_card,
                negativePrompt=_compose_negative(entry.negative_prompt),
                transitionOut=transition,  # type: ignore[arg-type]
                continuityNote=entry.continuity_note,
            )
        )

    return scenes


async def _run_scene_pass(
    messages: list[ChatMessage], story: StoryOption, plan: DurationPlan, label: str
) -> StoryOption:
    raw = await chat_json(messages, label=label, temperature=0.85)

    try:
        parsed = _ScenesResponse.model_validate(raw)
    except ValidationError as error:
        raise UpstreamError(
            "The storyboard model returned a scene payload we could not read.",
            details=error.errors()[:5],
        ) from error

    if not parsed.scenes:
        raise UpstreamError("The storyboard model returned no scenes.")

    candidate = story.model_copy(deep=True)
    candidate.scenes = _materialise_scenes(plan, story, parsed.scenes)
    return candidate


def _to_model_shape(story: StoryOption) -> list[dict[str, Any]]:
    """Feed the previous draft back to the model in the shape it produced it."""
    return [
        {
            "order": scene.order,
            "title": scene.title,
            "summary": scene.summary,
            "continuityNote": scene.continuity_note,
            "transitionOut": scene.transition_out,
            "shotCard": scene.shot_card.model_dump(by_alias=True),
            "prompt": scene.prompt,
            "negativePrompt": scene.negative_prompt,
        }
        for scene in story.scenes
    ]


async def _emit(callback: ProgressFn | None, fraction: float, message: str) -> None:
    if callback is None:
        return
    result = callback(fraction, message)
    if asyncio.iscoroutine(result):
        await result


async def generate_scenes(
    brief: ProjectBrief,
    plan: DurationPlan,
    story: StoryOption,
    on_progress: ProgressFn | None = None,
) -> tuple[StoryOption, int]:
    """Write the scene prompts, audit them, repair what fails.

    Returns ``(story_with_scenes, repair_passes_used)``.
    """
    if not plan.segments:
        raise BadRequest("The duration plan has no segments.")

    task = _scenes_task(brief, plan, story)
    messages: list[ChatMessage] = [
        {"role": "system", "content": compose_skill("storyboard")},
        {"role": "user", "content": task},
    ]

    await _emit(on_progress, 0.1, f"Writing {len(plan.segments)} scene prompts…")

    candidate = await _run_scene_pass(messages, story, plan, "scenes")
    report = audit_story_option(candidate)
    repair_passes = 0

    while not report.passed and repair_passes < MAX_REPAIR_PASSES:
        repair_passes += 1
        errors = sum(1 for issue in report.issues if issue.severity == "error")
        log.info(
            'Story "%s": audit found %d blocking issue(s), running repair pass %d/%d',
            story.title, errors, repair_passes, MAX_REPAIR_PASSES,
        )
        await _emit(
            on_progress,
            min(0.9, 0.4 + repair_passes * 0.2),
            f"Audit found {errors} issue{'' if errors == 1 else 's'} — rewriting…",
        )

        repair_messages: list[ChatMessage] = [
            {"role": "system", "content": compose_skill("repair")},
            {"role": "user", "content": task},
            {
                "role": "assistant",
                "content": json.dumps({"scenes": _to_model_shape(candidate)}),
            },
            {"role": "user", "content": build_repair_brief(candidate, report)},
        ]

        try:
            candidate = await _run_scene_pass(
                repair_messages, story, plan, f"scenes-repair-{repair_passes}"
            )
        except UpstreamError as error:
            log.warning("Repair pass failed; keeping the previous draft. %s", error)
            break

        report = audit_story_option(candidate)

    await _emit(
        on_progress, 1.0, "Storyboard ready." if report.passed else "Storyboard ready with warnings."
    )

    if not report.passed:
        errors = sum(1 for issue in report.issues if issue.severity == "error")
        log.warning(
            'Story "%s" ships with %d unresolved issue(s), score %d.',
            story.title, errors, report.score,
        )

    candidate.audit = report
    return candidate, repair_passes

# Single-scene rewrite
# ---------------------------------------------------------------------------


async def rewrite_scene(
    brief: ProjectBrief, story: StoryOption, scene: Scene, instruction: str
) -> Scene:
    neighbours = [other for other in story.scenes if abs(other.order - scene.order) == 1]

    lines = [
        _story_context(story),
        "",
        "## The scene to rewrite",
        "",
        f"Scene {scene.order} of {len(story.scenes)} — {scene.seconds}s, "
        f'beat "{scene.beat}", shot function "{scene.shot_function}".',
        f'Delivery ratio {brief.ratio}, style preset "{brief.style}".',
        "",
        "Current prompt:",
        "```text",
        scene.prompt,
        "```",
        "",
    ]

    if neighbours:
        lines += [
            "Neighbouring scenes, so the cut still matches:",
            *[
                f"- Scene {other.order}: {other.summary} ({other.continuity_note})"
                for other in neighbours
            ],
            "",
        ]

    lines += [
        "## What the user asked for",
        "",
        instruction.strip() or "Make it stronger without changing what happens.",
        "",
        "Keep the duration, the beat and the shot function. Keep the continuity block intact — "
        "the other scenes still depend on it. All the per-scene hard requirements still apply.",
        "",
        "Write sound direction in the separate `audio` field, not inside `prompt`. Do not write "
        "an `Audio:` or `Duration:` line in `prompt` — both are appended automatically.",
        "",
        "## Output",
        "",
        'Return JSON only: {"scenes": [ <one scene object, same shape as the storyboard schema, '
        f"order {scene.order}> ]}}",
    ]

    raw = await chat_json(
        [
            {"role": "system", "content": compose_skill("scene")},
            {"role": "user", "content": "\n".join(lines)},
        ],
        label="scene-rewrite",
        temperature=0.9,
    )

    parsed = _ScenesResponse.model_validate(raw)
    if not parsed.scenes:
        raise UpstreamError("The model did not return a rewritten scene.")

    written = parsed.scenes[0]
    transition = (
        written.transition_out
        if written.transition_out in ("hardcut", "crossfade")
        else scene.transition_out
    )

    updated = scene.model_copy(deep=True)
    updated.title = written.title or scene.title
    updated.summary = written.summary or scene.summary
    updated.prompt = (
        _compose_prompt(written.prompt, written.audio, scene.seconds)
        if written.prompt.strip()
        else scene.prompt
    )
    updated.shot_card = written.shot_card
    updated.negative_prompt = _compose_negative(written.negative_prompt or scene.negative_prompt)
    updated.transition_out = transition  # type: ignore[assignment]
    updated.continuity_note = written.continuity_note or scene.continuity_note
    return updated

    @property
    def storage_path(self) -> Path:
        path = Path(self.storage_dir)
        return path if path.is_absolute() else (SERVER_ROOT / path).resolve()

    @property
    def google_key(self) -> str | None:
        """The Google key, under either accepted name."""
        return self.gemini_image_api_key or self.gemini_api_key

    @property
    def azure_configured(self) -> bool:
        return bool(
            self.azure_openai_base_url
            and self.azure_openai_key
            and self.azure_openai_deployment_name
        )

    @property
    def allowed_clip_seconds(self) -> list[int]:
        """Durations Veo accepts for a single generation, capped by config."""
        allowed = [s for s in (4, 6, 8) if s <= self.veo_max_clip_seconds]
        return allowed or [4]
