/**
 * Holds the project and keeps it current over SSE.
 *
 * Every server event carries the whole project, so there is no merge logic
 * here — generation runs for minutes on the backend and this just reflects
 * whatever the latest snapshot says.
 */
function useProjectStream(projectId: string | null) {
  const [project, setProject] = useState<Project | null>(null);
  const [jobs, setJobs] = useState<Record<string, Job>>({});
  const [connected, setConnected] = useState(false);

  useEffect(() => {
    if (!projectId) {
      setProject(null);
      setJobs({});
      return;
    }

    let cancelled = false;

    // Seed from a plain fetch so the UI has state even if SSE is slow to open.
    videoApi
      .getProject(projectId)
      .then(({ project: initial, jobs: initialJobs }) => {
        if (cancelled) return;
        setProject(initial);
        setJobs(Object.fromEntries(initialJobs.map((job) => [job.id, job])));
      })
      .catch(() => undefined);

    const stream = streamProject(projectId, {
      onProject: (next) => {
        setConnected(true);
        setProject(next);
      },
      onJob: (job) => setJobs((current) => ({ ...current, [job.id]: job })),
      onError: () => setConnected(false),
    });

    return () => {
      cancelled = true;
      stream.close();
    };
  }, [projectId]);

  const activeJobs = useMemo(
    () => Object.values(jobs).filter((job) => job.status === "running" || job.status === "queued"),
    [jobs],
  );

  return { project, jobs, activeJobs, connected };
}

function jobFor(jobs: Record<string, Job>, kind: Job["kind"], targetId?: string): Job | undefined {
  return Object.values(jobs)
    .filter(
      (job) =>
        job.kind === kind &&
        (targetId === undefined || job.targetId === targetId) &&
        (job.status === "running" || job.status === "queued"),
    )
    .sort((a, b) => b.createdAt.localeCompare(a.createdAt))[0];
}

function VideoPage() {
  const [step, setStep] = useState<Step>("brief");
  const [projectId, setProjectId] = useState<string | null>(null);
  const [health, setHealth] = useState<Health | null>(null);
  const [error, setError] = useState<string | null>(null);

  const { project, jobs, activeJobs } = useProjectStream(projectId);

  useEffect(() => {
    videoApi
      .health()
      .then(setHealth)
      .catch(() => setHealth(null));
  }, []);

  const selectedStory = useMemo(
    () => project?.stories.find((story) => story.id === project.selectedStoryId) ?? null,
    [project],
  );

  const run = useCallback(async <T,>(action: () => Promise<T>): Promise<T | null> => {
    setError(null);
    try {
      return await action();
    } catch (caught) {
      setError(caught instanceof Error ? caught.message : "Something went wrong.");
      return null;
    }
  }, []);

  const clipsCount = selectedStory?.scenes.length ?? project?.plan.segments.length ?? 0;

  return (
    <AppShell>
      <div className="max-w-[1400px] mx-auto px-4 sm:px-6 py-6">
        <Header
          step={step}
          setStep={setStep}
          clipsCount={clipsCount}
          hasProject={Boolean(project)}
        />

        {error && (
          <div className="mb-4 flex items-start gap-3 rounded-xl border border-destructive/40 bg-destructive/5 p-4 text-sm">
            <AlertTriangle className="size-4 shrink-0 text-destructive mt-0.5" />
            <div className="flex-1">
              <div className="font-medium text-destructive">That didn't work</div>
              <div className="text-muted-foreground mt-0.5">{error}</div>
            </div>
            <button
              onClick={() => setError(null)}
              className="text-muted-foreground hover:text-foreground"
            >
              Dismiss
            </button>
          </div>
        )}

        {step === "brief" && (
          <BriefStep
            health={health}
            onSubmit={async (payload) => {
              const created = await run(() => videoApi.createProject(payload));
              if (created) {
                setProjectId(created.project.id);
                setStep("storyboard");
              }
            }}
          />
        )}

        {step === "storyboard" && project && (
          <StoryboardStep
            project={project}
            jobs={jobs}
            onSelectStory={(storyId) => run(() => videoApi.selectStory(project.id, storyId))}
            onRegenerateStory={(storyId) =>
              run(() => videoApi.regenerateStory(project.id, storyId))
            }
            onGenerateScenes={(storyId) => run(() => videoApi.generateScenes(project.id, storyId))}
            onUpdateScene={(sceneId, prompt) =>
              run(() => videoApi.updateScene(project.id, sceneId, { prompt }))
            }
            onRewriteScene={(sceneId, instruction) =>
              run(() => videoApi.rewriteScene(project.id, sceneId, instruction))
            }
            onBack={() => setStep("brief")}
            onNext={async () => {
              await run(() => videoApi.generateAllClips(project.id));
              setStep("clips");
            }}
          />
        )}

        {step === "clips" && project && (
          <ClipsStep
            project={project}
            jobs={jobs}
            onGenerate={(sceneId) => run(() => videoApi.generateClip(project.id, sceneId))}
            onGenerateAll={() => run(() => videoApi.generateAllClips(project.id, true))}
            onBack={() => setStep("storyboard")}
            onNext={async () => {
              if (project.audio.length === 0) await run(() => videoApi.suggestAudio(project.id));
              setStep("audio");
            }}
          />
        )}

        {step === "audio" && project && (
          <AudioStep
            project={project}
            jobs={jobs}
            health={health}
            onSuggest={() => run(() => videoApi.suggestAudio(project.id))}
            onGenerate={(trackId) => run(() => videoApi.generateAudio(project.id, trackId))}
            onSelect={(trackId) =>
              run(() => videoApi.updateAudio(project.id, trackId, { selected: true }))
            }
            onUpdatePrompt={(trackId, prompt) =>
              run(() => videoApi.updateAudio(project.id, trackId, { prompt }))
            }
            onUpload={(file, kind) => run(() => videoApi.uploadAudio(project.id, file, kind))}
            onBack={() => setStep("clips")}
            onNext={async () => {
              await run(() => videoApi.render(project.id));
              setStep("final");
            }}
          />
        )}

        {step === "final" && project && (
          <FinalStep
            project={project}
            onRerender={() => run(() => videoApi.render(project.id))}
            onBack={() => setStep("audio")}
            onRestart={() => {
              setProjectId(null);
              setStep("brief");
            }}
          />
        )}

        {step !== "brief" && !project && (
          <div className="rounded-2xl border border-border bg-surface p-10 text-center">
            <Loader2 className="size-6 animate-spin text-primary mx-auto mb-3" />
            <div className="text-sm text-muted-foreground">Loading your project…</div>
          </div>
        )}

        {activeJobs.length > 0 && <JobTicker jobs={activeJobs} />}
      </div>
    </AppShell>
  );
}

/* ---------------- Header / Stepper ---------------- */

function Header({
  step,
  setStep,
  clipsCount,
  hasProject,
}: {
  step: Step;
  setStep: (s: Step) => void;
  clipsCount: number;
  hasProject: boolean;
}) {
  const steps: { id: Step; label: string; icon: typeof Film }[] = [
    { id: "brief", label: "Brief", icon: Pencil },
    { id: "storyboard", label: "Storyboard", icon: Layers },
    { id: "clips", label: "Clips", icon: Film },
    { id: "audio", label: "Audio", icon: Music },
    { id: "final", label: "Final video", icon: Sparkles },
  ];
  const currentIndex = steps.findIndex((s) => s.id === step);

  return (
    <div className="mb-6">
      <div className="flex items-center justify-between mb-4">
        <div>
          <div className="flex items-center gap-2 text-sm text-muted-foreground mb-1">
            <Film className="size-4" />
            AI Video Studio
          </div>
          <h1 className="text-2xl font-bold tracking-tight">Turn your idea into a full video</h1>
          <p className="text-sm text-muted-foreground mt-1">
            Veo renders short clips, so we plan your film as scenes, generate each one, then
            accumulate them into a single video.
          </p>
        </div>
      </div>

      <div className="flex items-center gap-1 sm:gap-2 p-1.5 rounded-xl border border-border bg-surface overflow-x-auto">
        {steps.map((s, i) => {
          const Icon = s.icon;
          const active = s.id === step;
          const done = i < currentIndex;
          const clickable = hasProject ? i <= currentIndex : i === 0;
          return (
            <button
              key={s.id}
              disabled={!clickable}
              onClick={() => clickable && setStep(s.id)}
              className={`flex items-center gap-2 px-3 py-2 rounded-lg text-sm whitespace-nowrap transition-colors ${
                active
                  ? "bg-primary text-primary-foreground font-medium"
                  : done
                    ? "text-foreground hover:bg-accent"
                    : "text-muted-foreground"
              } ${!clickable ? "opacity-50 cursor-not-allowed" : ""}`}
            >
              <div
                className={`size-5 grid place-items-center rounded-full text-[10px] font-semibold ${
                  active
                    ? "bg-primary-foreground/20"
                    : done
                      ? "bg-primary/10 text-primary"
                      : "bg-accent"
                }`}
              >
                {done ? <Check className="size-3" /> : i + 1}
              </div>
              <Icon className="size-4" />
              <span className="hidden sm:inline">{s.label}</span>
              {s.id === "clips" && clipsCount > 0 && (
                <span className="hidden sm:inline text-[10px] opacity-70">({clipsCount})</span>
              )}
            </button>
          );
        })}
      </div>
    </div>
  );
}

/* ---------------- Capability banner ---------------- */

function CapabilityNotice({ health }: { health: Health | null }) {
  if (!health) {
    return (
      <div className="rounded-xl border border-amber-500/40 bg-amber-500/5 p-4 text-sm">
        <div className="flex items-center gap-2 font-medium">
          <AlertTriangle className="size-4 text-amber-600" />
          Backend not reachable
        </div>
        <p className="text-muted-foreground mt-1">
          Start it with{" "}
          <code className="text-xs">cd server && uvicorn app.main:app --reload --port 8787</code>.
        </p>
      </div>
    );
  }

  const blocked = health.capabilities.filter(
    (capability: Capability) =>
      !capability.ready && ["storyboard", "clips"].includes(capability.id),
  );
  if (blocked.length === 0 && health.ffmpeg.available) return null;

  return (
    <div className="rounded-xl border border-amber-500/40 bg-amber-500/5 p-4 text-sm">
      <div className="flex items-center gap-2 font-medium">
        <AlertTriangle className="size-4 text-amber-600" />
        Some steps are not configured yet
      </div>
      <ul className="mt-2 space-y-1 text-muted-foreground">
        {blocked.map((capability) => (
          <li key={capability.id}>• {capability.detail}</li>
        ))}
        {!health.ffmpeg.available && (
          <li>
            • ffmpeg was not found — clips can be generated but not accumulated into a final video.
          </li>
        )}
      </ul>
    </div>
  );
}

function BriefStep({

  onSubmit,

    brief: string;
    seconds: number;
    style: StylePreset;
    ratio: AspectRatio;
    pacing?: Pacing;
    references?: string[];
    transition?: "hardcut" | "crossfade";

  const [brief, setBrief] = useState("");
  const [seconds, setSeconds] = useState(30);
  const [style, setStyle] = useState<StylePreset>("cinematic");
  const [ratio, setRatio] = useState<AspectRatio>("16:9");
  const [pacing, setPacing] = useState<Pacing | undefined>(undefined);
  const [transition, setTransition] = useState<"hardcut" | "crossfade">("hardcut");
  const [attachments, setAttachments] = useState<AttachmentItem[]>([]);
  const [plan, setPlan] = useState<DurationPlan | null>(null);
  const [submitting, setSubmitting] = useState(false);

  const maxSeconds = health?.limits.maxVideoSeconds ?? 180;

  useEffect(() => {
    return () => attachments.forEach((item) => URL.revokeObjectURL(item.previewUrl));
    // Only on unmount — revoking on every change would kill live previews.
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, []);

    let cancelled = false;

      .plan(seconds, style, pacing)
      .then(({ plan: next }) => !cancelled && setPlan(next))
      .catch(() => !cancelled && setPlan(null));

      cancelled = true;

            onChange={(e) => setBrief(e.target.value)}

      <aside className="lg:sticky lg:top-20 self-start rounded-2xl border border-border bg-surface p-5">
        <div className="text-sm font-medium mb-3">What happens next</div>
        <ol className="space-y-3 text-sm">
          {[
            "We write four director treatments and you pick a direction.",
            `The story is broken into ${plan?.segments.length ?? "several"} scenes sized to what Veo can render.`,
            "Every scene prompt is audited against the dramaturgy rules before it ships.",
            "Each clip is generated, then accumulated with audio into one video.",
          ].map((text, index) => (
            <li key={index} className="flex gap-3">
              <div className="size-6 rounded-full bg-primary/10 text-primary text-xs grid place-items-center font-semibold shrink-0">
                {index + 1}
              </div>
              <span className="text-muted-foreground">{text}</span>
            </li>
          ))}
        </ol>

        <button
          onClick={async () => {
            setSubmitting(true);
            await onSubmit({
              brief,
              seconds,
              style,
              ratio,
              pacing,
              transition,
              references: attachments.map((item) => item.file.name),
            });
            setSubmitting(false);
          }}
          disabled={!brief.trim() || submitting}
          className="mt-5 cursor-pointer w-full inline-flex items-center justify-center gap-2 rounded-xl bg-primary text-primary-foreground py-3 text-sm font-medium hover:opacity-90 disabled:opacity-50 disabled:cursor-not-allowed"
        >
          {submitting ? <Loader2 className="size-4 animate-spin" /> : <Wand2 className="size-4" />}
          {submitting ? "Planning…" : "Plan my storyboard"}
          {!submitting && <ArrowRight className="size-4" />}
        </button>
      </aside>
    </div>
  );
}

/**
 * Shows how the requested duration decomposes into Veo generations.
 *
 * This is the thing users get wrong about AI video, so it is stated up front
 * rather than discovered at render time.
 */
function PlanSummary({ plan, health }: { plan: DurationPlan | null; health: Health | null }) {
  if (!plan) {
    return (
      <div className="mt-3 text-xs text-muted-foreground">Working out the clip breakdown…</div>
    );
  }

  const counts = plan.segments.reduce<Record<number, number>>((acc, segment) => {
    acc[segment.seconds] = (acc[segment.seconds] ?? 0) + 1;
    return acc;
  }, {});

  const breakdown = Object.entries(counts)
    .sort((a, b) => Number(b[0]) - Number(a[0]))
    .map(([secs, count]) => `${count} × ${secs}s`)
    .join(" + ");

  return (
    <div className="mt-4 rounded-xl border border-border bg-background p-3">
      <div className="flex items-start gap-2 text-xs">
        <Scissors className="size-3.5 mt-0.5 text-primary shrink-0" />
        <div className="flex-1">
          <div className="font-medium text-foreground">
            {plan.segments.length} clips · {breakdown}
          </div>
          <div className="text-muted-foreground mt-1">
            {health?.veo.model ?? "Veo"} renders at most {health?.veo.maxClipSeconds ?? 8}s per
            generation, so your {plan.targetSeconds}s film is generated as {plan.segments.length}{" "}
            separate clips and accumulated into one file.
          </div>
          {!plan.exact && (
            <div className="text-muted-foreground mt-1">
              The plan overshoots by {plan.trimSeconds}s and the tail is trimmed to land on exactly{" "}
              {plan.targetSeconds}s.
            </div>
          )}
        </div>
      </div>

      <div className="mt-3 flex gap-0.5 h-6">
        {plan.segments.map((segment) => (
          <div
            key={segment.index}
            title={`Scene ${segment.index + 1} · ${segment.seconds}s · ${BEAT_LABELS[segment.beat]}`}
            className="rounded-sm bg-primary/25 hover:bg-primary/50 transition-colors grid place-items-center text-[9px] text-primary font-semibold"
            style={{ flexGrow: segment.seconds }}
          >
            {segment.seconds}
          </div>
        ))}
      </div>
    </div>
  );
}

                  }}

                >

/** Surfaces the backend's dramaturgy audit so a weak storyboard is visible. */
function AuditBadge({ story }: { story: StoryOption }) {
  const errors = story.audit.issues.filter((issue) => issue.severity === "error").length;
  const warnings = story.audit.issues.length - errors;

  if (story.audit.passed && warnings === 0) {
    return (
      <span className="inline-flex items-center gap-1.5 rounded-lg bg-emerald-500/10 px-2.5 py-1.5 text-xs font-medium text-emerald-600">
        <ShieldCheck className="size-3.5" />
        Audit clean
      </span>
    );
  }

  return (
    <span
      title={story.audit.issues.map((issue) => `${issue.rule}: ${issue.message}`).join("\n")}
      className={`inline-flex items-center gap-1.5 rounded-lg px-2.5 py-1.5 text-xs font-medium ${
        errors > 0 ? "bg-destructive/10 text-destructive" : "bg-amber-500/10 text-amber-600"
      }`}
    >
      <AlertTriangle className="size-3.5" />
      {errors > 0
        ? `${errors} issue${errors === 1 ? "" : "s"}`
        : `${warnings} note${warnings === 1 ? "" : "s"}`}
      <span className="opacity-60">· {story.audit.score}/100</span>
    </span>
  );
}

function SceneCard({
  scene,
  issues,
  onUpdate,
  onRewrite,
}: {
  scene: Scene;
  issues: { rule: string; severity: string; message: string }[];
  onUpdate: (prompt: string) => void;
  onRewrite: (instruction: string) => void;
}) {
  const [draft, setDraft] = useState(scene.prompt);
  const [instruction, setInstruction] = useState("");
  const [showRewrite, setShowRewrite] = useState(false);

  useEffect(() => setDraft(scene.prompt), [scene.prompt]);
  const dirty = draft !== scene.prompt;

  return (
    <div className="rounded-xl border border-border bg-background p-3">
      <div className="flex items-center justify-between mb-2 gap-2">
        <div className="flex items-center gap-2 min-w-0">
          <div className="size-7 shrink-0 rounded-lg bg-primary/10 text-primary text-xs font-semibold grid place-items-center">
            {scene.order}
          </div>
          <div className="min-w-0">
            <div className="text-sm font-medium truncate">{scene.title}</div>
            <div className="text-[11px] text-muted-foreground">
              {scene.seconds}s · {BEAT_LABELS[scene.beat]} · {scene.shotFunction} ·{" "}
              {formatSeconds(scene.startsAt)}
            </div>
          </div>
        </div>
      </div>

      {issues.length > 0 && (
        <div className="mb-2 rounded-lg bg-amber-500/5 border border-amber-500/30 p-2 text-[11px] space-y-0.5">
          {issues.slice(0, 3).map((issue, index) => (
            <div key={index} className="text-muted-foreground">
              <span className="font-medium text-amber-600">{issue.rule}</span> — {issue.message}
            </div>
          ))}
        </div>
      )}

      <textarea
        value={draft}
        onChange={(e) => setDraft(e.target.value)}
        className="w-full min-h-[160px] rounded-lg border border-border bg-surface p-3 text-xs font-mono leading-relaxed resize-y focus:outline-none focus:ring-2 focus:ring-primary"
      />

      <div className="mt-2 flex items-center gap-2 flex-wrap">
        <button
          onClick={() => onUpdate(draft)}
          disabled={!dirty}
          className="rounded-lg bg-primary px-3 py-1.5 text-xs font-medium text-primary-foreground disabled:opacity-40"
        >
          Save prompt
        </button>
        <button
          onClick={() => setShowRewrite((value) => !value)}
          className="rounded-lg border border-border bg-surface px-3 py-1.5 text-xs hover:bg-accent"
        >
          Ask for a rewrite
        </button>
        {scene.continuityNote && (
          <span className="text-[11px] text-muted-foreground truncate flex-1">
            Cut: {scene.continuityNote}
          </span>
        )}
      </div>

      {showRewrite && (
        <div className="mt-2 flex gap-2">
          <input
            value={instruction}
            onChange={(e) => setInstruction(e.target.value)}
            placeholder="e.g. make it a macro insert on the hands instead"
            className="flex-1 rounded-lg border border-border bg-surface px-3 py-1.5 text-xs focus:outline-none focus:ring-2 focus:ring-primary"
          />
          <button
            onClick={() => {
              onRewrite(instruction);
              setInstruction("");
              setShowRewrite(false);
            }}
            className="rounded-lg bg-accent px-3 py-1.5 text-xs font-medium hover:bg-accent/70"
          >
            Rewrite
          </button>
        </div>
      )}
    </div>
  );
}

/* ---------------- Step 3: Clips ---------------- */

function ClipsStep({
  project,
  jobs,
  onGenerate,
  onGenerateAll,
  onBack,
  onNext,
}: {
  project: Project;
  jobs: Record<string, Job>;
  onGenerate: (sceneId: string) => void;
  onGenerateAll: () => void;
  onBack: () => void;
  onNext: () => void;
}) {
  const story = project.stories.find((candidate) => candidate.id === project.selectedStoryId);
  const scenes = story?.scenes ?? [];
  const byScene = useMemo(
    () => Object.fromEntries(project.clips.map((clip) => [clip.sceneId, clip])),
    [project.clips],
  );

  const ready = project.clips.filter((clip) => clip.status === "ready").length;
  const failed = project.clips.filter((clip) => clip.status === "failed").length;
  const batch = jobFor(jobs, "clips");
  const canProceed = scenes.length > 0 && ready === scenes.length;

  const aspect = useMemo(() => {
    const [w, h] = project.brief.ratio.split(":").map(Number);
    return `${w} / ${h}`;
  }, [project.brief.ratio]);

  return (
    <div>
      <div className="flex items-center justify-between mb-4 flex-wrap gap-3">
        <div>
          <h2 className="text-lg font-semibold">Generate your clips</h2>
          <p className="text-sm text-muted-foreground">
            {ready} of {scenes.length} clips ready
            {failed > 0 && <span className="text-destructive"> · {failed} failed</span>}
            {batch && <span> · {batch.message}</span>}
          </p>
        </div>
        <button
          onClick={onGenerateAll}
          disabled={Boolean(batch)}
          className="inline-flex items-center gap-2 rounded-xl bg-primary text-primary-foreground px-4 py-2 text-sm font-medium hover:opacity-90 disabled:opacity-50"
        >
          {batch ? <Loader2 className="size-4 animate-spin" /> : <Wand2 className="size-4" />}
          {batch ? "Generating…" : "Regenerate all"}
        </button>
      </div>

      <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 gap-4">
        {scenes.map((scene) => (
          <ClipCard
            key={scene.id}
            scene={scene}
            clip={byScene[scene.id]}
            aspect={aspect}
            onGenerate={() => onGenerate(scene.id)}
          />
        ))}
      </div>

      <FooterNav
        onBack={onBack}
        onNext={onNext}
        nextLabel={canProceed ? "Add audio" : `${scenes.length - ready} clip(s) still to generate`}
        nextIcon={<Music className="size-4" />}
        disabled={!canProceed}
      />
    </div>
  );
}

function ClipCard({

  clip,
  aspect,
  onGenerate,
}: {
  scene: Scene;
  clip: Clip | undefined;
  aspect: string;

          {scene.summary}
        </div>
        <div className="mt-3 flex items-center gap-2">
          {status === "generating" ? (
            <div className="flex-1 h-1.5 rounded-full bg-accent overflow-hidden">
              <div
                className="h-full bg-primary transition-all"
                style={{ width: `${Math.max(3, (clip?.progress ?? 0) * 100)}%` }}
              />
            </div>
          ) : (

              {clip.actualSeconds.toFixed(1)}s

          )}
        </div>

}

            className="inline-flex items-center gap-2 rounded-xl bg-accent px-3 py-2 text-sm font-medium hover:bg-accent/70 disabled:opacity-50"
          >
            {suggesting ? (
              <Loader2 className="size-4 animate-spin" />
            ) : (
              <RefreshCw className="size-4" />
            )}
            New directions
          </button>
        </div>
      </div>

      {!musicReady && (
        <div className="mb-4 rounded-xl border border-border bg-surface p-4 text-sm text-muted-foreground">
          <div className="flex items-center gap-2 font-medium text-foreground">
            <Info className="size-4" />
            No music generator configured
          </div>
          <p className="mt-1">
            These directions are still written for you — upload a track that matches one, or set{" "}
            <code className="text-xs">MUSIC_PROVIDER</code> in{" "}
            <code className="text-xs">server/.env</code>. You can also render with Veo's native
            audio alone.
          </p>
        </div>
      )}

      {project.audio.length === 0 && suggesting && (
        <div className="rounded-2xl border border-border bg-surface p-10 text-center">
          <Loader2 className="size-6 animate-spin text-primary mx-auto mb-3" />
          <div className="text-sm">Writing music direction…</div>
        </div>
      )}

      <div className="grid grid-cols-1 md:grid-cols-2 gap-3">
        {project.audio.map((track) => (
          <AudioCard
            key={track.id}
            track={track}
            canGenerate={track.kind === "voiceover" ? true : Boolean(musicReady)}
            onGenerate={() => onGenerate(track.id)}
            onSelect={() => onSelect(track.id)}
            onUpdatePrompt={(prompt) => onUpdatePrompt(track.id, prompt)}
          />
        ))}
      </div>

      <FooterNav
        onBack={onBack}
        onNext={onNext}
        nextLabel={`Accumulate into the final video${selectedCount > 0 ? ` (${selectedCount} track)` : " (no music)"}`}
        nextIcon={<Sparkles className="size-4" />}
      />
    </div>
  );
}

function AudioCard({
  track,
  canGenerate,
  onGenerate,
  onSelect,
  onUpdatePrompt,
}: {
  track: AudioTrack;
  canGenerate: boolean;
  onGenerate: () => void;
  onSelect: () => void;
  onUpdatePrompt: (prompt: string) => void;
}) {
  const [draft, setDraft] = useState(track.prompt);
  useEffect(() => setDraft(track.prompt), [track.prompt]);

  return (
    <div
      className={`rounded-2xl border p-4 transition-colors ${
        track.selected ? "border-primary bg-primary/5" : "border-border bg-surface"
      }`}
    >
      <div className="flex items-start gap-3">
        <button
          onClick={onSelect}
          className={`size-5 rounded-md border-2 shrink-0 mt-0.5 grid place-items-center ${
            track.selected ? "bg-primary border-primary text-primary-foreground" : "border-border"
          }`}
          aria-label={`Select ${track.label}`}
        >
          {track.selected && <Check className="size-3" />}
        </button>

        <div className="flex-1 min-w-0">
          <div className="flex items-center justify-between gap-2">
            <div className="text-sm font-medium truncate">{track.label}</div>
            <div className="text-xs text-muted-foreground capitalize shrink-0">
              {track.kind}
              {track.seconds ? ` · ${formatSeconds(track.seconds)}` : ""}
            </div>
          </div>

          <textarea
            value={draft}
            onChange={(e) => setDraft(e.target.value)}
            onBlur={() => draft !== track.prompt && onUpdatePrompt(draft)}
            className="mt-2 w-full min-h-[72px] rounded-lg border border-border bg-background p-2.5 text-xs resize-none focus:outline-none focus:ring-2 focus:ring-primary"
          />

          {track.status === "ready" && track.url && (
            <audio controls src={track.url} className="mt-2 w-full h-8" />
          )}
          {track.status === "failed" && (
            <div className="mt-2 text-[11px] text-destructive line-clamp-2">{track.error}</div>
          )}

          <div className="mt-2 flex items-center gap-2">
            <button
              onClick={onGenerate}
              disabled={track.status === "generating" || !canGenerate}
              title={canGenerate ? undefined : "No provider configured for this track type"}
              className="inline-flex items-center gap-1.5 rounded-lg bg-primary text-primary-foreground text-xs font-medium px-3 py-1.5 disabled:opacity-40"
            >
              {track.status === "generating" ? (
                <>
                  <Loader2 className="size-3 animate-spin" /> Generating…
                </>
              ) : track.status === "ready" ? (
                <>
                  <RefreshCw className="size-3" /> Regenerate
                </>
              ) : (
                <>
                  <Wand2 className="size-3" /> Generate
                </>
              )}
            </button>
            {track.source === "uploaded" && (
              <span className="text-[11px] text-muted-foreground">Uploaded</span>
            )}
          </div>
        </div>
      </div>
    </div>
  );
}

/* ---------------- Step 5: Final ---------------- */

function FinalStep({
  project,
  onRerender,
  onBack,
  onRestart,
}: {
  project: Project;
  onRerender: () => void;
  onBack: () => void;
  onRestart: () => void;
}) {
  const { render } = project;
  const story = project.stories.find((candidate) => candidate.id === project.selectedStoryId);
  const scenes = story?.scenes ?? [];

  const aspect = useMemo(() => {
    const [w, h] = project.brief.ratio.split(":").map(Number);
    return `${w} / ${h}`;
  }, [project.brief.ratio]);

  return (
    <div className="grid grid-cols-1 lg:grid-cols-3 gap-6">
      <div className="lg:col-span-2">
        <div className="rounded-2xl border border-border bg-surface p-4">
          <div
            className="w-full rounded-xl overflow-hidden bg-black relative"
            style={{ aspectRatio: aspect }}
          >
            {render.status === "ready" && render.url ? (
              <video
                src={render.url}
                controls
                autoPlay
                playsInline
                className="absolute inset-0 h-full w-full"
              />
            ) : render.status === "failed" ? (
              <div className="absolute inset-0 grid place-items-center text-white p-6">
                <div className="text-center">
                  <AlertTriangle className="size-8 mx-auto mb-3 text-destructive" />
                  <div className="text-sm font-medium">Render failed</div>
                  <div className="text-xs text-white/70 mt-1 max-w-md">{render.error}</div>
                </div>
              </div>
            ) : (
              <div className="absolute inset-0 grid place-items-center text-white">
                <div className="text-center w-full max-w-sm px-6">
                  <Loader2 className="size-8 animate-spin mx-auto mb-3" />
                  <div className="text-sm font-medium">
                    {render.stage ?? "Accumulating clips & mixing audio…"}
                  </div>
                  <div className="text-xs text-white/70 mt-1">
                    {scenes.length} scenes → {project.plan.targetSeconds}s
                  </div>
                  <div className="mt-3 h-1 rounded-full bg-white/20 overflow-hidden">
                    <div
                      className="h-full bg-white transition-all"
                      style={{ width: `${Math.max(3, render.progress * 100)}%` }}
                    />
                  </div>
                </div>
              </div>
            )}
          </div>

          <div className="mt-4 flex items-center gap-2 flex-wrap">
            {render.status === "ready" && render.url && (
              <a
                href={render.url}
                download={`video-${project.id}.mp4`}
                className="inline-flex items-center gap-2 rounded-xl bg-primary text-primary-foreground px-4 py-2 text-sm font-medium hover:opacity-90"
              >
                <Download className="size-4" />
                Download MP4
              </a>
            )}
            <button
              onClick={onRerender}
              disabled={render.status === "rendering" || render.status === "queued"}
              className="inline-flex items-center gap-2 rounded-xl border border-border bg-background px-4 py-2 text-sm font-medium hover:bg-accent disabled:opacity-50"
            >
              <RefreshCw className="size-4" />
              Re-render
            </button>
            <button
              onClick={onRestart}
              className="inline-flex items-center gap-2 rounded-xl border border-border bg-background px-4 py-2 text-sm font-medium hover:bg-accent"
            >
              <Sparkles className="size-4" />
              Start a new video
            </button>
          </div>

          {render.status === "ready" && (
            <div className="mt-3 text-xs text-muted-foreground">
              {render.seconds?.toFixed(1)}s · {render.bytes ? formatBytes(render.bytes) : "—"} ·{" "}
              {project.brief.ratio} · accumulated from {scenes.length} Veo clips
            </div>
          )}
        </div>
      </div>

      <aside className="space-y-4">
        <div className="rounded-2xl border border-border bg-surface p-4">
          <div className="text-sm font-medium mb-3 flex items-center gap-2">
            <Layers className="size-4 text-primary" />
            Timeline
          </div>
          <div className="space-y-2 max-h-[420px] overflow-auto">
            {scenes.map((scene) => (
              <div key={scene.id} className="flex items-center gap-2 text-xs">
                <div className="size-6 shrink-0 rounded bg-primary/10 text-primary grid place-items-center font-semibold">
                  {scene.order}
                </div>
                <div className="flex-1 min-w-0">
                  <div className="truncate">{scene.title}</div>
                  <div className="text-[10px] text-muted-foreground">{BEAT_LABELS[scene.beat]}</div>
                </div>
                <div className="text-muted-foreground shrink-0">{scene.seconds}s</div>
              </div>
            ))}
          </div>
        </div>

        <div className="rounded-2xl border border-border bg-surface p-4">
          <div className="text-sm font-medium mb-3 flex items-center gap-2">
            <Music className="size-4 text-primary" />
            Audio mix
          </div>
          {project.audio.filter((track) => track.selected && track.status === "ready").length ===
          0 ? (
            <div className="text-xs text-muted-foreground">
              Veo's native clip audio only — no music bed selected.
            </div>
          ) : (
            <div className="space-y-2">
              {project.audio
                .filter((track) => track.selected && track.status === "ready")
                .map((track) => (
                  <div key={track.id} className="flex items-center gap-2 text-xs">
                    <Check className="size-3.5 text-primary shrink-0" />
                    <div className="flex-1 truncate">{track.label}</div>
                    <div className="text-muted-foreground capitalize">{track.kind}</div>
                  </div>
                ))}
            </div>
          )}
        </div>

        <button
          onClick={onBack}
          className="w-full inline-flex items-center justify-center gap-2 rounded-xl border border-border bg-background py-2.5 text-sm hover:bg-accent"
        >
          <ArrowLeft className="size-4" />
          Back to audio
        </button>
      </aside>
    </div>
  );
}

/* ---------------- Shared footer ---------------- */

  onBack,
  onNext,
  nextLabel,
  nextIcon,
  disabled,
}: {
  onBack: () => void;
  onNext: () => void;
  nextLabel: string;
  nextIcon?: React.ReactNode;
  disabled?: boolean;
}) {
  return (
    <div className="mt-6 flex items-center justify-between gap-3">
      <button
        onClick={onBack}
        className="inline-flex items-center cursor-pointer gap-2 rounded-xl border border-border bg-background px-4 py-2.5 text-sm hover:bg-accent"
      >
        <ArrowLeft className="size-4" />
        Back
      </button>
      <button
        onClick={onNext}
        disabled={disabled}
        className="inline-flex cursor-pointer items-center gap-2 rounded-xl bg-primary text-primary-foreground px-5 py-2.5 text-sm font-medium hover:opacity-90 disabled:opacity-50 disabled:cursor-not-allowed"
      >
        {nextIcon}
        {nextLabel}
        <ArrowRight className="size-4" />
      </button>
    </div>
  );
}
