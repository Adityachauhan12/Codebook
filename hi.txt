"""Shared vocabulary for the AI Video pipeline.

The frontend mirrors these in ``src/lib/video-api-types.ts`` — keep the two in
sync when changing anything here.
"""

from __future__ import annotations

from datetime import datetime, timezone
from typing import Any, Literal

from pydantic import BaseModel, Field

AspectRatio = Literal["16:9", "9:16", "1:1", "4:5"]
#: Veo only renders 16:9 and 9:16. Everything else is cropped at render time.
VeoAspectRatio = Literal["16:9", "9:16"]
StylePreset = Literal["cinematic", "product", "docu", "animated", "ugc"]
Pacing = Literal["slow", "standard", "fast"]
TransitionStyle = Literal["hardcut", "crossfade"]

#: Beat map from the dramaturgy reference (Layer 1).
Beat = Literal["hook", "context", "pressure", "crack", "acceleration", "impact", "aftermath"]

#: Shot function taxonomy (Layer 2).
ShotFunction = Literal[
    "establish", "power", "pressure", "detail", "reaction", "shift", "impact", "aftermath", "exit"
]


def now_iso() -> str:
    return datetime.now(timezone.utc).isoformat()


# ---------------------------------------------------------------------------
# Duration planning
# ---------------------------------------------------------------------------


class PlannedSegment(BaseModel):
    index: int
    #: Seconds requested from Veo for this segment. Always 4, 6 or 8.
    seconds: int
    #: Where this segment starts on the finished timeline.
    starts_at: int = Field(alias="startsAt")
    beat: Beat
    shot_function: ShotFunction = Field(alias="shotFunction")

    model_config = {"populate_by_name": True}


class DurationPlan(BaseModel):
    #: What the user asked for.
    target_seconds: int = Field(alias="targetSeconds")
    #: Sum of every segment's requested seconds. May exceed the target.
    planned_seconds: int = Field(alias="plannedSeconds")
    #: planned_seconds - target_seconds. Trimmed off the tail during render.
    trim_seconds: int = Field(alias="trimSeconds")
    #: True when no trim was needed.
    exact: bool
    strategy: Literal["stitch"] = "stitch"
    segments: list[PlannedSegment]
    notes: list[str] = Field(default_factory=list)

    model_config = {"populate_by_name": True}


# ---------------------------------------------------------------------------
# Storyboard
# ---------------------------------------------------------------------------


class StoryAnchors(BaseModel):
    """The five anchors. One per piece, no more."""

    emotion: str = ""
    motif: str = ""
    object: str = ""
    break_: str = Field(default="", alias="break")
    final_image: str = Field(default="", alias="finalImage")

    model_config = {"populate_by_name": True}


class ContinuityBlock(BaseModel):
    """Repeated verbatim inside every scene prompt — Veo has no memory."""

    characters: list[str] = Field(default_factory=list)
    wardrobe: list[str] = Field(default_factory=list)
    location: str = ""
    palette: str = ""
    lighting_constant: str = Field(default="", alias="lightingConstant")
    camera_language: str = Field(default="", alias="cameraLanguage")
    props: list[str] = Field(default_factory=list)
    rules: list[str] = Field(default_factory=list)

    model_config = {"populate_by_name": True}


class ShotCard(BaseModel):
    """Compressed form of the 14-field shot card."""

    frame: str = ""
    composition: str = ""
    camera: str = ""
    movement_reason: str = Field(default="", alias="movementReason")
    action: str = ""
    eye_trace: str = Field(default="", alias="eyeTrace")
    environmental_pressure: str = Field(default="", alias="environmentalPressure")
    micro_action: str = Field(default="", alias="microAction")
    sound_anchor: str = Field(default="", alias="soundAnchor")
    lighting: str = ""
    palette: str = ""
    cut_type: str = Field(default="", alias="cutType")

    model_config = {"populate_by_name": True}


class Scene(BaseModel):
    id: str
    order: int
    title: str = ""
    beat: Beat
    shot_function: ShotFunction = Field(alias="shotFunction")
    #: Seconds this scene occupies in the finished cut.
    seconds: int
    starts_at: int = Field(alias="startsAt")
    summary: str = ""
    #: The full, ready-to-send Veo prompt.
    prompt: str = ""
    shot_card: ShotCard = Field(default_factory=ShotCard, alias="shotCard")
    negative_prompt: str = Field(default="", alias="negativePrompt")
    transition_out: TransitionStyle = Field(default="hardcut", alias="transitionOut")
    #: How this scene joins the previous one.
    continuity_note: str = Field(default="", alias="continuityNote")

    model_config = {"populate_by_name": True}


class AuditIssue(BaseModel):
    scene_id: str | None = Field(default=None, alias="sceneId")
    scene_order: int | None = Field(default=None, alias="sceneOrder")
    rule: str
    severity: Literal["error", "warning"]
    message: str

    model_config = {"populate_by_name": True}


class AuditReport(BaseModel):
    passed: bool = True
    score: int = 100
    issues: list[AuditIssue] = Field(default_factory=list)
    checked_at: str = Field(default_factory=now_iso, alias="checkedAt")

    model_config = {"populate_by_name": True}


class StoryOption(BaseModel):
    id: str
    title: str
    #: One-line pitch for the story card.
    summary: str = ""
    #: The prose treatment shown under "Brief story".
    treatment: str = ""
    angle: str = ""
    anchors: StoryAnchors = Field(default_factory=StoryAnchors)
    continuity: ContinuityBlock = Field(default_factory=ContinuityBlock)
    scenes: list[Scene] = Field(default_factory=list)
    audit: AuditReport = Field(default_factory=AuditReport)

    model_config = {"populate_by_name": True}

ClipStatus = Literal["idle", "queued", "generating", "ready", "failed"]
AudioKind = Literal["music", "voiceover", "sfx"]
RenderStatus = Literal["idle", "queued", "rendering", "ready", "failed"]


class Clip(BaseModel):
    scene_id: str = Field(alias="sceneId")
    order: int
    status: ClipStatus = "idle"
    progress: float = 0.0
    #: Requested seconds.
    seconds: int
    #: Actual duration of the downloaded file, measured with ffprobe.
    actual_seconds: float | None = Field(default=None, alias="actualSeconds")
    #: Storage-relative path to the mp4.
    file: str | None = None
    url: str | None = None
    poster_url: str | None = Field(default=None, alias="posterUrl")
    #: Veo operation name for the in-flight generation.
    operation: str | None = None
    attempts: int = 0
    error: str | None = None
    started_at: str | None = Field(default=None, alias="startedAt")
    completed_at: str | None = Field(default=None, alias="completedAt")

    model_config = {"populate_by_name": True}


class AudioTrack(BaseModel):
    id: str
    kind: AudioKind = "music"
    label: str = ""
    #: Prompt for a generator, or the script for a voiceover.
    prompt: str = ""
    selected: bool = False
    status: Literal["idle", "generating", "ready", "failed"] = "idle"
    #: Mix gain applied under the clip audio.
    gain: float = 0.28
    seconds: float | None = None
    file: str | None = None
    url: str | None = None
    source: Literal["generated", "uploaded"] | None = None
    error: str | None = None

    model_config = {"populate_by_name": True}


class RenderResult(BaseModel):
    status: RenderStatus = "idle"
    progress: float = 0.0
    stage: str | None = None
    file: str | None = None
    url: str | None = None
    seconds: float | None = None
    bytes: int | None = None
    error: str | None = None
    started_at: str | None = Field(default=None, alias="startedAt")
    completed_at: str | None = Field(default=None, alias="completedAt")

    model_config = {"populate_by_name": True}


#: Where a super sits in frame. Semantic rather than pixel coordinates, so the
#: render layer owns safe-area maths and the same project renders correctly at
#: any output size or ratio.
SuperPosition = Literal["upper-left", "upper-centre", "lower-left", "lower-centre", "centre"]


class SuperSpec(BaseModel):
    """One line of on-screen copy, burnt in after the render.

    Veo is never asked to draw text — it cannot spell, and it misspelt four of
    five cards the one time it was asked. Copy lives here instead, gets stripped
    out of the brief before the scene writer sees it, and is composited by
    ffmpeg from a font file.
    """

    text: str
    start: float
    end: float
    position: SuperPosition = "upper-left"

    model_config = {"populate_by_name": True}


class EndCard(BaseModel):
    """The closing brand frame: lockup, plus optional headline and subline."""

    enabled: bool = True
    headline: str | None = None
    subline: str | None = None
    mark: Literal["wordmark", "icon", "none"] = "wordmark"
    #: How long before the end of the film the card starts resolving.
    seconds: float = 4.0

    model_config = {"populate_by_name": True}


class BriefAnalysis(BaseModel):
    """A raw brief decomposed into the things that go to different places.

    A brief is not one instruction, it is four kinds of instruction wearing a
    trench coat: what the film is *about*, what a viewer should *read*, what the
    deliverable *is*, and what must not appear. Feeding the lot to the scene
    writer is how "On-Screen Text: A new horizon awaits." became a request for
    Veo to spell, and how "4K HDR" became a silent no-op.

    Only ``intent`` ever reaches a prompt.
    """

    #: The creative brief with copy, deliverable spec and prohibitions removed.
    intent: str
    #: Copy the viewer reads. Composited after the render, never prompted.
    supers: list[SuperSpec] = Field(default_factory=list)
    end_card: EndCard = Field(default_factory=lambda: EndCard(enabled=False), alias="endCard")
    #: Things the brief says must not appear, for the negative prompt.
    avoid: list[str] = Field(default_factory=list)
    #: Asks the pipeline cannot honour, surfaced to the user rather than
    #: silently dropped — "4K HDR" when Veo tops out at 1080p SDR, say.
    unsupported: list[str] = Field(default_factory=list)
    #: Music and sound direction. Goes to the audio service, not to Veo.
    music: str | None = None

    model_config = {"populate_by_name": True}


class ProjectBrief(BaseModel):
    brief: str
    seconds: int
    style: StylePreset = "cinematic"
    ratio: AspectRatio = "16:9"
    pacing: Pacing = "standard"
    #: Free-text notes pulled from user attachments.
    references: list[str] = Field(default_factory=list)
    #: Words/brands/visuals the user wants excluded.
    avoid: list[str] = Field(default_factory=list)
    language: str = "English"
    #: ``brief`` with copy, deliverable spec and prohibitions stripped out. This
    #: is what the scene writer sees; ``brief`` is kept verbatim for display and
    #: re-analysis. Empty until the brief has been analysed.
    intent: str = ""
    #: Asks the pipeline cannot honour, recorded so the UI can say so.
    unsupported: list[str] = Field(default_factory=list)
    #: Music and sound direction lifted out of the brief, for the composer. It
    #: is stripped from ``intent`` so it cannot distract the scene writer.
    music_direction: str | None = Field(default=None, alias="musicDirection")

    def creative(self) -> str:
        """What the prompt layer is allowed to read."""
        return (self.intent or self.brief).strip()

    model_config = {"populate_by_name": True}


class Project(BaseModel):
    id: str
    created_at: str = Field(default_factory=now_iso, alias="createdAt")
    updated_at: str = Field(default_factory=now_iso, alias="updatedAt")
    brief: ProjectBrief
    plan: DurationPlan
    stories: list[StoryOption] = Field(default_factory=list)
    selected_story_id: str | None = Field(default=None, alias="selectedStoryId")
    clips: list[Clip] = Field(default_factory=list)
    audio: list[AudioTrack] = Field(default_factory=list)
    render: RenderResult = Field(default_factory=RenderResult)
    #: On-screen copy, burnt in at render time. Extracted from the brief so the
    #: scene writer never sees it and never asks Veo to draw it.
    supers: list[SuperSpec] = Field(default_factory=list)
    end_card: EndCard = Field(default_factory=EndCard, alias="endCard")
    transition: TransitionStyle = "hardcut"
    #: Veo model actually used, recorded for reproducibility.
    model: str = ""

    model_config = {"populate_by_name": True}

    def dump(self) -> dict[str, Any]:
        """camelCase JSON for the browser."""
        return self.model_dump(by_alias=True, mode="json")

# Jobs

JobKind = Literal["storyboard", "clip", "clips", "audio", "render"]
JobStatus = Literal["queued", "running", "succeeded", "failed", "cancelled"]


class Job(BaseModel):
    id: str
    project_id: str = Field(alias="projectId")
    kind: JobKind
    status: JobStatus = "queued"
    progress: float = 0.0
    message: str = ""
    #: Scene id for clip jobs, track id for audio jobs.
    target_id: str | None = Field(default=None, alias="targetId")
    error: str | None = None
    created_at: str = Field(default_factory=now_iso, alias="createdAt")
    updated_at: str = Field(default_factory=now_iso, alias="updatedAt")
    result: Any = None

    model_config = {"populate_by_name": True}

    def dump(self) -> dict[str, Any]:
        return self.model_dump(by_alias=True, mode="json")
