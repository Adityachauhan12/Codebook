export class ApiError extends Error {
  readonly status: number;
  readonly code: string;
  readonly details: unknown;

  constructor(message: string, status: number, code: string, details?: unknown) {
    super(message);
    this.name = "ApiError";
    this.status = status;
    this.code = code;
    this.details = details;
  }
}

async function request<T>(path: string, init?: RequestInit): Promise<T> {
  const response = await fetch(`${BASE}${path}`, {
    ...init,
    headers: {
      ...(init?.body instanceof FormData ? {} : { "content-type": "application/json" }),
      ...init?.headers,
    },
  });

  if (!response.ok) {
    let message = `Request failed (${response.status})`;
    let code = "error";
    let details: unknown;
    try {
      const payload = await response.json();
      message = payload.message ?? message;
      code = payload.error ?? code;
      details = payload.details;
    } catch {
      // A non-JSON error body (a proxy 502, say) keeps the generic message.
    }
    throw new ApiError(message, response.status, code, details);
  }

  if (response.status === 204) return undefined as T;
  return (await response.json()) as T;
}

export const videoApi = {
  health: () => request<Health>("/api/health"),

  /**
   * How a duration breaks into Veo-sized clips. Needs no API key, so the brief
   * screen can show the real clip count while the user is still typing.
   */
  plan: (seconds: number, style: string, pacing?: Pacing) =>
    request<{
      plan: DurationPlan;
      limits: {
        maxClipSeconds: number;
        allowedClipSeconds: number[];
        maxVideoSeconds: number;
        maxSegments: number;
      };
    }>(
      `/api/video/plan?seconds=${seconds}&style=${encodeURIComponent(style)}` +
        (pacing ? `&pacing=${pacing}` : ""),
    ),

  createProject: (body: {
    brief: string;
    seconds: number;
    style: StylePreset;
    ratio: AspectRatio;
    pacing?: Pacing;
    references?: string[];
    avoid?: string[];
    language?: string;
    transition?: TransitionStyle;
    quality?: "standard" | "draft";
  }) =>
    request<{ project: Project; jobId: string }>("/api/video/projects", {
      method: "POST",
      body: JSON.stringify(body),
    }),

  getProject: (id: string) =>
    request<{ project: Project; jobs: Job[] }>(`/api/video/projects/${id}`),

  deleteProject: (id: string) =>
    request<{ ok: boolean }>(`/api/video/projects/${id}`, { method: "DELETE" }),

  regenerateStory: (projectId: string, storyId: string) =>
    request<{ jobId: string }>(`/api/video/projects/${projectId}/stories/${storyId}/regenerate`, {
      method: "POST",
    }),

  generateScenes: (projectId: string, storyId: string) =>
    request<{ jobId: string }>(`/api/video/projects/${projectId}/stories/${storyId}/scenes`, {
      method: "POST",
    }),

  selectStory: (projectId: string, storyId: string) =>
    request<{ project: Project }>(`/api/video/projects/${projectId}/select-story`, {
      method: "POST",
      body: JSON.stringify({ storyId }),
    }),

  updateScene: (
    projectId: string,
    sceneId: string,
    patch: Partial<
      Pick<Scene, "prompt" | "title" | "summary" | "negativePrompt" | "transitionOut">
    >,
  ) =>
    request<{ project: Project }>(`/api/video/projects/${projectId}/scenes/${sceneId}`, {
      method: "PATCH",
      body: JSON.stringify(patch),
    }),

  rewriteScene: (projectId: string, sceneId: string, instruction: string) =>
    request<{ jobId: string }>(`/api/video/projects/${projectId}/scenes/${sceneId}/rewrite`, {
      method: "POST",
      body: JSON.stringify({ instruction }),
    }),

  generateClip: (projectId: string, sceneId: string) =>
    request<{ jobId: string }>(`/api/video/projects/${projectId}/clips/${sceneId}/generate`, {
      method: "POST",
    }),

  generateAllClips: (projectId: string, force = false) =>
    request<{ jobId: string }>(
      `/api/video/projects/${projectId}/clips/generate-all${force ? "?force=true" : ""}`,
      { method: "POST" },
    ),

  suggestAudio: (projectId: string) =>
    request<{ jobId: string }>(`/api/video/projects/${projectId}/audio/suggest`, {
      method: "POST",
    }),

  generateAudio: (projectId: string, trackId: string) =>
    request<{ jobId: string }>(`/api/video/projects/${projectId}/audio/${trackId}/generate`, {
      method: "POST",
    }),

  updateAudio: (
    projectId: string,
    trackId: string,
    patch: { selected?: boolean; gain?: number; prompt?: string; label?: string },
  ) =>
    request<{ project: Project }>(`/api/video/projects/${projectId}/audio/${trackId}`, {
      method: "PATCH",
      body: JSON.stringify(patch),
    }),

  uploadAudio: (projectId: string, file: File, kind: "music" | "voiceover" = "music") => {
    const form = new FormData();
    form.append("file", file);
    form.append("kind", kind);
    return request<{ track: AudioTrack }>(`/api/video/projects/${projectId}/audio/upload`, {
      method: "POST",
      body: form,
    });
  },

  render: (projectId: string) =>
    request<{ jobId: string }>(`/api/video/projects/${projectId}/render`, { method: "POST" }),

  getJob: (jobId: string) => request<{ job: Job }>(`/api/video/jobs/${jobId}`),

  cancelJob: (jobId: string) =>
    request<{ cancelled: boolean }>(`/api/video/jobs/${jobId}/cancel`, { method: "POST" }),
};

export type ProjectStream = {
  close: () => void;
};

/**
 * Subscribes to a project's SSE feed.
 *
 * Every `project` event carries complete state, so there is nothing to
 * reconcile — the caller just replaces what it has. EventSource reconnects on
 * its own, and the first event after a reconnect resyncs everything.
 */
export function streamProject(
  projectId: string,
  handlers: {
    onProject?: (project: Project) => void;
    onJob?: (job: Job) => void;
    onError?: (error: Event) => void;
  },
): ProjectStream {
  const source = new EventSource(`${BASE}/api/video/projects/${projectId}/events`);

  source.addEventListener("project", (event) => {
    try {
      handlers.onProject?.(JSON.parse((event as MessageEvent).data).project);
    } catch {
      // A malformed frame is not worth tearing the stream down over.
    }
  });

  source.addEventListener("job", (event) => {
    try {
      handlers.onJob?.(JSON.parse((event as MessageEvent).data).job);
    } catch {
      /* ignore */
    }
  });

  source.onerror = (event) => handlers.onError?.(event);

  return { close: () => source.close() };
}

export function formatSeconds(total: number): string {
  const seconds = Math.max(0, Math.round(total));
  return `${Math.floor(seconds / 60)}:${String(seconds % 60).padStart(2, "0")}`;
}

export function formatBytes(bytes: number): string {
  if (bytes < 1024) return `${bytes} B`;
  if (bytes < 1024 * 1024) return `${(bytes / 1024).toFixed(0)} KB`;
  return `${(bytes / 1_048_576).toFixed(1)} MB`;
}
