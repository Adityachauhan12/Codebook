import React, { useState, useEffect, useRef } from 'react';
import { StyleSheet, View, ScrollView, KeyboardAvoidingView, Platform, Modal, Dimensions } from 'react-native';
import { Provider as PaperProvider, Text, Button, TextInput, Card, Menu } from 'react-native-paper';
import { SafeAreaProvider, SafeAreaView } from 'react-native-safe-area-context';
import { Video } from 'expo-av';
import DateTimePicker from '@react-native-community/datetimepicker';
import axios from 'axios';

const { width, height } = Dimensions.get('window');

const API_BASE = 'https://pomologically-overminute-linette.ngrok-free.dev';
const indigoBlue = '#001b94';
const indigoOrange = '#f14658';

export default function App() {
  return (
    <SafeAreaProvider>
      <PaperProvider>
        <AvatarBookingApp />
      </PaperProvider>
    </SafeAreaProvider>
  );
}

function AvatarBookingApp() {
  const [language, setLanguage] = useState('en');
  const [uid, setUid] = useState('');
  const [chats, setChats] = useState([]);
  const [inputText, setInputText] = useState('');
  const [isLoading, setIsLoading] = useState(false);
  const [showAvatarFlow, setShowAvatarFlow] = useState(false);
  const [currentStep, setCurrentStep] = useState(null);
  const [bookingData, setBookingData] = useState({});
  const [menuVisible, setMenuVisible] = useState(false);
  const [fullscreenVideo, setFullscreenVideo] = useState(null);
  const [showStepInput, setShowStepInput] = useState(false);
  const [pendingStepData, setPendingStepData] = useState(null);
  const chatEndRef = useRef(null);
  const videoRef = useRef(null);

  const getWelcomeMessage = (lang) => {
    return lang === 'hi' 
      ? 'नमस्ते! 6Eskai में आपका स्वागत है। मैं चरणबद्ध अवतार मार्गदर्शन के साथ उड़ानें बुक करने में आपकी सहायता कर सकता हूं।'
      : 'Hello! Welcome to 6Eskai. I can help you book flights with step-by-step avatar guidance.';
  };

  useEffect(() => {
    const newUid = 'uid_' + Math.random().toString(36).substr(2, 9);
    setUid(newUid);
    setChats([{
      type: 'text',
      text: getWelcomeMessage(language),
      sender: 'bot',
      timestamp: new Date().toISOString()
    }]);
  }, []);

  const startAvatarFlow = async () => {
    try {
      setIsLoading(true);
      setShowAvatarFlow(true);

      const response = await fetch(`${API_BASE}/chat`, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({
          query: language === 'hi' ? 'अवतार के साथ उड़ान बुक करें' : 'book flight with avatar',
          uid: uid,
          language: language
        })
      });

      const data = await response.json();
      if (data.type === 'avatar_flow') {
        setCurrentStep(data.step);
        setPendingStepData(data);
        setFullscreenVideo(data.avatar_video);
        setShowStepInput(false);
      }
    } catch (error) {
      console.error('Error:', error);
      setChats(prev => [...prev, {
        type: 'text',
        text: 'Error connecting to server. Please check backend is running.',
        sender: 'bot',
        timestamp: new Date().toISOString()
      }]);
    } finally {
      setIsLoading(false);
    }
  };

  const processStep = async (step, userInput) => {
    try {
      setIsLoading(true);
      setShowStepInput(false);

      // Format user input for display
      let userInputText = '';
      if (typeof userInput === 'string') {
        userInputText = userInput;
      } else if (typeof userInput === 'object') {
        if (userInput.origin) userInputText = `From: ${userInput.origin}`;
        else if (userInput.destination) userInputText = `To: ${userInput.destination}`;
        else if (userInput.date) userInputText = `Date: ${userInput.date}`;
        else if (userInput.adults !== undefined) {
          userInputText = `${userInput.adults} Adult(s)`;
          if (userInput.children > 0) userInputText += `, ${userInput.children} Child(ren)`;
          if (userInput.infants > 0) userInputText += `, ${userInput.infants} Infant(s)`;
        }
        else if (userInput.passengers) userInputText = `${userInput.passengers.length} passenger(s) details provided`;
        else if (userInput.email) userInputText = `Email: ${userInput.email}, Phone: ${userInput.phone}`;
        else if (userInput.selected_flight) userInputText = `Selected: ${userInput.selected_flight}`;
        else if (userInput.confirmed) userInputText = 'Booking confirmed';
        else if (userInput.continue) userInputText = null; // Don't show "continue: true"
        else userInputText = JSON.stringify(userInput);
      }

      if (userInputText) {
        setChats(prev => [...prev, {
          type: 'text',
          text: userInputText,
          sender: 'user',
          timestamp: new Date().toISOString()
        }]);
      }

      const response = await fetch(`${API_BASE}/avatar-step`, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({
          uid: uid,
          step: step,
          user_input: typeof userInput === 'string' ? { response: userInput } : userInput,
          language: language
        })
      });

      const data = await response.json();
      setCurrentStep(data.step);
      setBookingData(prev => ({ ...prev, [step]: userInput }));
      setPendingStepData(data);

      // Handle complete step - stop the flow
      if (data.step === 'complete' || data.type === 'complete') {
        setChats(prev => [...prev, {
          type: 'text',
          text: '✅ Booking complete! Thank you for using 6Eskai.',
          sender: 'bot',
          timestamp: new Date().toISOString()
        }]);
        setShowAvatarFlow(false);
        setCurrentStep(null);
        setShowStepInput(false);
        setIsLoading(false);
        return;
      }

      if (data.avatar_video) {
        setFullscreenVideo(data.avatar_video);
      } else {
        handleVideoEnd(data);
      }
    } catch (error) {
      console.error('Error:', error);
    } finally {
      setIsLoading(false);
    }
  };

  const handleVideoEnd = (stepData) => {
    setFullscreenVideo(null);
    
    // Don't add message for complete step
    if (stepData.step !== 'complete') {
      setChats(prev => [...prev, {
        type: stepData.type || 'avatar_step',
        text: stepData.message,
        sender: 'bot',
        step: stepData.step,
        flights: stepData.flights,
        timestamp: new Date().toISOString()
      }]);
    }
    
    // Auto-progress for steps that don't need input (except review_booking and payment)
    if (!requiresInput(stepData.step) && 
        stepData.step !== 'flight_search' && 
        stepData.step !== 'review_booking' && 
        stepData.step !== 'payment' &&
        stepData.step !== 'complete') {
      setTimeout(() => {
        processStep(stepData.step, { continue: true });
      }, 500);
    } else if (stepData.step === 'complete') {
      // End the flow completely
      setTimeout(() => {
        setChats(prev => [...prev, {
          type: 'text',
          text: '✅ Booking complete! Thank you for using 6Eskai.',
          sender: 'bot',
          timestamp: new Date().toISOString()
        }]);
        setShowAvatarFlow(false);
        setCurrentStep(null);
        setShowStepInput(false);
      }, 1000);
    } else {
      setShowStepInput(true);
    }
  };

  const renderChatMessage = (chat, index) => {
    if (chat.sender === 'user') {
      return (
        <View key={index} style={styles.userMessageContainer}>
          <Card style={styles.userMessage}>
            <Card.Content>
              <Text style={styles.userMessageText}>{chat.text}</Text>
            </Card.Content>
          </Card>
        </View>
      );
    }

    return (
      <View key={index} style={styles.botMessageContainer}>
        <Card style={styles.botMessage}>
          <Card.Content>
            <Text style={styles.botMessageText}>{chat.text}</Text>

            {chat.flights && index === chats.length - 1 && showStepInput && (
              <View style={styles.flightsContainer}>
                <Text style={styles.flightsTitle}>Available Flights:</Text>
                {chat.flights.map((flight, idx) => (
                  <Card key={idx} style={styles.flightCard} onPress={() => processStep('flight_selection', { selected_flight: flight.flight_number })}>
                    <Card.Content>
                      <Text style={styles.flightNumber}>{flight.flight_number}</Text>
                      <Text style={styles.flightDetails}>{flight.departure} → {flight.arrival} ({flight.duration})</Text>
                      <Text style={styles.flightPrice}>{flight.price}</Text>
                    </Card.Content>
                  </Card>
                ))}
              </View>
            )}

            {chat.step === 'review_booking' && index === chats.length - 1 && showStepInput && currentStep === 'review_booking' && (
              <ReviewBooking bookingData={bookingData} onNext={(data) => processStep('review_booking', data)} language={language} />
            )}

            {chat.step && index === chats.length - 1 && showStepInput && chat.step === currentStep && (
              <View style={styles.stepInputContainer}>
                {renderStepInput(chat.step)}
              </View>
            )}
          </Card.Content>
        </Card>
      </View>
    );
  };

  const requiresInput = (step) => {
    return ['origin_selection', 'destination_selection', 'date_selection', 'passenger_selection', 'passenger_details', 'contact_details', 'flight_selection', 'review_booking'].includes(step);
  };

  const renderStepInput = (step) => {
    if (!requiresInput(step)) return null;
    
    switch (step) {
      case 'origin_selection':
        return <OriginSelection onNext={(data) => processStep('origin_selection', data)} language={language} />;
      case 'destination_selection':
        return <DestinationSelection onNext={(data) => processStep('destination_selection', data)} language={language} />;
      case 'date_selection':
        return <DateSelection onNext={(data) => processStep('date_selection', data)} language={language} />;
      case 'passenger_selection':
        return <PassengerSelection onNext={(data) => processStep('passenger_selection', data)} language={language} />;
      case 'passenger_details':
        return <PassengerDetails onNext={(data) => processStep('passenger_details', data)} passengerCount={bookingData.passenger_selection} />;
      case 'contact_details':
        return <ContactDetails onNext={(data) => processStep('contact_details', data)} language={language} />;
      default:
        return null;
    }
  };

  return (
    <SafeAreaView style={styles.container}>
      <KeyboardAvoidingView 
        behavior={Platform.OS === 'ios' ? 'padding' : 'height'}
        style={styles.container}
      >
        <View style={styles.header}>
          <View>
            <Text style={styles.headerTitle}>6Eskai</Text>
            <Text style={styles.headerSubtitle}>AI Flight Assistant</Text>
          </View>
          <Menu
            visible={menuVisible}
            onDismiss={() => setMenuVisible(false)}
            anchor={
              <Button onPress={() => setMenuVisible(true)} textColor="white">
                {language === 'en' ? 'EN' : 'हि'}
              </Button>
            }
          >
            <Menu.Item onPress={() => { setLanguage('en'); setMenuVisible(false); }} title="English" />
            <Menu.Item onPress={() => { setLanguage('hi'); setMenuVisible(false); }} title="हिंदी" />
          </Menu>
        </View>

        <ScrollView 
          style={styles.chatArea}
          ref={chatEndRef}
          onContentSizeChange={() => chatEndRef.current?.scrollToEnd({ animated: true })}
        >
          {chats.map((chat, idx) => renderChatMessage(chat, idx))}
          {isLoading && <Text style={styles.loadingText}>Bot is typing...</Text>}
        </ScrollView>

        {!showAvatarFlow && chats.length <= 2 && (
          <View style={styles.quickActions}>
            <Button mode="outlined" onPress={startAvatarFlow} style={styles.quickButton}>
              📱 {language === 'hi' ? 'अवतार' : 'Avatar'}
            </Button>
          </View>
        )}

        <View style={styles.footer}>
          <Text style={styles.footerText}>Powered by IndiGo AI</Text>
        </View>

        <Modal visible={!!fullscreenVideo} animationType="fade" onRequestClose={() => {}}>
          <View style={styles.fullscreenContainer}>
            <Video
              ref={videoRef}
              source={{ uri: fullscreenVideo }}
              style={styles.fullscreenVideo}
              resizeMode="contain"
              shouldPlay
              onPlaybackStatusUpdate={(status) => {
                if (status.didJustFinish) {
                  handleVideoEnd(pendingStepData);
                }
              }}
              onError={(e) => {
                console.log('Video error:', e);
                handleVideoEnd(pendingStepData);
              }}
            />
          </View>
        </Modal>
      </KeyboardAvoidingView>
    </SafeAreaView>
  );
}

// Step Components
function OriginSelection({ onNext, language }) {
  const [city, setCity] = useState('');
  return (
    <View>
      <TextInput mode="outlined" placeholder={language === 'hi' ? 'शहर दर्ज करें' : 'Enter city'} value={city} onChangeText={setCity} style={styles.input} />
      <Button mode="contained" onPress={() => onNext({ origin: city })} disabled={!city.trim()} style={styles.button} buttonColor={indigoBlue}>
        {language === 'hi' ? 'अगला' : 'Next'}
      </Button>
    </View>
  );
}

function DestinationSelection({ onNext, language }) {
  const [city, setCity] = useState('');
  return (
    <View>
      <TextInput mode="outlined" placeholder={language === 'hi' ? 'शहर दर्ज करें' : 'Enter city'} value={city} onChangeText={setCity} style={styles.input} />
      <Button mode="contained" onPress={() => onNext({ destination: city })} disabled={!city.trim()} style={styles.button} buttonColor={indigoBlue}>
        {language === 'hi' ? 'अगला' : 'Next'}
      </Button>
    </View>
  );
}

function DateSelection({ onNext, language }) {
  const [date, setDate] = useState(new Date());
  const [show, setShow] = useState(false);
  const [dateString, setDateString] = useState('');
  
  const onChange = (event, selectedDate) => {
    setShow(Platform.OS === 'ios');
    if (selectedDate) {
      setDate(selectedDate);
      const formatted = selectedDate.toISOString().split('T')[0];
      setDateString(formatted);
    }
  };
  
  return (
    <View>
      <Button 
        mode="outlined" 
        onPress={() => setShow(true)} 
        style={styles.input}
        icon="calendar"
      >
        {dateString || (language === 'hi' ? 'तारीख चुनें' : 'Select Date')}
      </Button>
      {show && (
        <DateTimePicker
          value={date}
          mode="date"
          display="default"
          onChange={onChange}
          minimumDate={new Date()}
        />
      )}
      <Button 
        mode="contained" 
        onPress={() => onNext({ date: dateString })} 
        disabled={!dateString} 
        style={styles.button} 
        buttonColor={indigoBlue}
      >
        {language === 'hi' ? 'अगला' : 'Next'}
      </Button>
    </View>
  );
}

function PassengerSelection({ onNext, language }) {
  const [adults, setAdults] = useState('1');
  const [children, setChildren] = useState('0');
  const [infants, setInfants] = useState('0');
  return (
    <View>
      <TextInput mode="outlined" label={language === 'hi' ? 'वयस्क' : 'Adults'} value={adults} onChangeText={setAdults} keyboardType="numeric" style={styles.input} />
      <TextInput mode="outlined" label={language === 'hi' ? 'बच्चे' : 'Children'} value={children} onChangeText={setChildren} keyboardType="numeric" style={styles.input} />
      <TextInput mode="outlined" label={language === 'hi' ? 'शिशु' : 'Infants'} value={infants} onChangeText={setInfants} keyboardType="numeric" style={styles.input} />
      <Button mode="contained" onPress={() => onNext({ adults: parseInt(adults), children: parseInt(children), infants: parseInt(infants) })} style={styles.button} buttonColor={indigoBlue}>
        {language === 'hi' ? 'उड़ानें खोजें' : 'Search Flights'}
      </Button>
    </View>
  );
}

function PassengerDetails({ onNext, passengerCount }) {
  const totalPassengers = passengerCount ? (passengerCount.adults + passengerCount.children + passengerCount.infants) : 1;
  const [passengers, setPassengers] = useState(Array.from({ length: totalPassengers }, (_, i) => {
    let type = 'Adult';
    if (passengerCount) {
      if (i >= passengerCount.adults + passengerCount.children) type = 'Infant';
      else if (i >= passengerCount.adults) type = 'Child';
    }
    return { name: '', age: '', gender: 'male', type };
  }));

  const updatePassenger = (index, field, value) => {
    const updated = [...passengers];
    updated[index][field] = value;
    setPassengers(updated);
  };

  const isValid = passengers.every(p => p.name.trim() && p.age.trim());

  return (
    <View>
      <Text style={styles.sectionTitle}>Passenger Details</Text>
      {passengers.map((passenger, index) => (
        <View key={index} style={styles.passengerForm}>
          <Text>Passenger {index + 1} ({passenger.type})</Text>
          <TextInput mode="outlined" placeholder="Full Name" value={passenger.name} onChangeText={(val) => updatePassenger(index, 'name', val)} style={styles.input} />
          <TextInput mode="outlined" placeholder="Age" value={passenger.age} onChangeText={(val) => updatePassenger(index, 'age', val)} keyboardType="numeric" style={styles.input} />
        </View>
      ))}
      <Button mode="contained" onPress={() => onNext({ passengers })} disabled={!isValid} style={styles.button} buttonColor={indigoBlue}>Continue</Button>
    </View>
  );
}

function ContactDetails({ onNext, language }) {
  const [email, setEmail] = useState('');
  const [phone, setPhone] = useState('');
  return (
    <View>
      <Text style={styles.sectionTitle}>{language === 'hi' ? 'संपर्क जानकारी' : 'Contact Information'}</Text>
      <TextInput mode="outlined" placeholder="Email" value={email} onChangeText={setEmail} keyboardType="email-address" style={styles.input} />
      <TextInput mode="outlined" placeholder="Phone" value={phone} onChangeText={setPhone} keyboardType="phone-pad" style={styles.input} />
      <Button mode="contained" onPress={() => onNext({ email, phone })} disabled={!email.trim() || !phone.trim()} style={styles.button} buttonColor={indigoBlue}>
        {language === 'hi' ? 'जारी रखें' : 'Continue'}
      </Button>
    </View>
  );
}

function ReviewBooking({ bookingData, onNext, language }) {
  return (
    <Card style={styles.reviewCard}>
      <Card.Content>
        <Text style={styles.reviewTitle}>{language === 'hi' ? 'बुकिंग समीक्षा' : 'Booking Summary'}</Text>
        {bookingData.origin_selection && (
          <Text style={styles.reviewText}>From: {bookingData.origin_selection.origin}</Text>
        )}
        {bookingData.destination_selection && (
          <Text style={styles.reviewText}>To: {bookingData.destination_selection.destination}</Text>
        )}
        {bookingData.date_selection && (
          <Text style={styles.reviewText}>Date: {bookingData.date_selection.date}</Text>
        )}
        {bookingData.passenger_selection && (
          <Text style={styles.reviewText}>
            Passengers: {bookingData.passenger_selection.adults} Adult(s)
            {bookingData.passenger_selection.children > 0 && `, ${bookingData.passenger_selection.children} Child(ren)`}
            {bookingData.passenger_selection.infants > 0 && `, ${bookingData.passenger_selection.infants} Infant(s)`}
          </Text>
        )}
        {bookingData.flight_selection && (
          <Text style={styles.reviewText}>Flight: {bookingData.flight_selection.selected_flight}</Text>
        )}
        {bookingData.contact_details && (
          <View>
            <Text style={styles.reviewText}>Email: {bookingData.contact_details.email}</Text>
            <Text style={styles.reviewText}>Phone: {bookingData.contact_details.phone}</Text>
          </View>
        )}
        <Button mode="contained" onPress={() => onNext({ confirmed: true })} style={[styles.button, {marginTop: 16}]} buttonColor={indigoBlue}>
          {language === 'hi' ? 'पुष्टि करें और भुगतान करें' : 'Confirm & Pay'}
        </Button>
      </Card.Content>
    </Card>
  );
}
