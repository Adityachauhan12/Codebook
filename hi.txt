import React, { useState, useEffect, useRef } from 'react';
import {
  StyleSheet, View, ScrollView, KeyboardAvoidingView,
  Platform, Modal, Dimensions, ActivityIndicator, Keyboard, TouchableOpacity
} from 'react-native';
import { Provider as PaperProvider, Text, Button, TextInput, Card, Menu } from 'react-native-paper';
import { SafeAreaProvider, SafeAreaView } from 'react-native-safe-area-context';
import { Video, ResizeMode } from 'expo-av';
import DateTimePicker from '@react-native-community/datetimepicker';

const { width, height } = Dimensions.get('window');
const API_BASE = 'https://pomologically-overminute-linette.ngrok-free.dev';
const indigoBlue = '#001b94';
const indigoOrange = '#f14658';
const indigoYellow = '#ffc107';
const indigoWhite = '#ffffff';

export default function App() {
  return (
    <SafeAreaProvider>
      <PaperProvider>
        <AvatarBookingApp />
      </PaperProvider>
    </SafeAreaProvider>
  );
}

function AvatarBookingApp() {
  const [language, setLanguage]       = useState('en');
  const [uid, setUid]                 = useState('');
  const [chats, setChats]             = useState([]);
  const [isLoading, setIsLoading]     = useState(false);
  const [showAvatarFlow, setShowAvatarFlow] = useState(false);
  const [showCheckinFlow, setShowCheckinFlow] = useState(false);
  const [currentStep, setCurrentStep] = useState(null);
  const [bookingData, setBookingData] = useState({});
  const [checkinData, setCheckinData] = useState({});
  const [menuVisible, setMenuVisible] = useState(false);
  const [fullscreenVideo, setFullscreenVideo] = useState(null);
  const [videoFinished, setVideoFinished]     = useState(false);
  const [showStepInput, setShowStepInput]     = useState(false);
  const [pendingStepData, setPendingStepData] = useState(null);
  const [showVideoLoader, setShowVideoLoader] = useState(false);
  const [flowChatStartIndex, setFlowChatStartIndex] = useState(0);
  const [showHomeScreen, setShowHomeScreen] = useState(true);

  // ✅ FIX 1: Track keyboard height
  const [keyboardOffset, setKeyboardOffset]   = useState(0);

  const chatEndRef = useRef(null);
  const videoRef   = useRef(null);

  const getWelcomeMessage = (lang) =>
    lang === 'hi'
      ? 'नमस्ते! 6Eskai में आपका स्वागत है।'
      : 'Hello! Welcome to 6Eskai.';

  const resetApp = () => {
    const newUid = 'uid_' + Math.random().toString(36).substr(2, 9);
    setUid(newUid);
    setChats([{ type: 'text', text: getWelcomeMessage(language), sender: 'bot', timestamp: new Date().toISOString() }]);
    setShowHomeScreen(true);
    setShowAvatarFlow(false);
    setShowCheckinFlow(false);
    setFullscreenVideo(null);
    setVideoFinished(false);
    setShowStepInput(false);
    setPendingStepData(null);
    setShowVideoLoader(false);
    setCurrentStep(null);
    setBookingData({});
    setCheckinData({});
    setIsLoading(false);
    setFlowChatStartIndex(0);
    setKeyboardOffset(0);
  };

  // Init UID + welcome chat
  useEffect(() => {
    const newUid = 'uid_' + Math.random().toString(36).substr(2, 9);
    setUid(newUid);
    setChats([{ type: 'text', text: getWelcomeMessage(language), sender: 'bot', timestamp: new Date().toISOString() }]);
  }, []);

  // ✅ FIX 1: Keyboard listener — shifts panel above keyboard
  useEffect(() => {
    const showEvt = Platform.OS === 'ios' ? 'keyboardWillShow' : 'keyboardDidShow';
    const hideEvt = Platform.OS === 'ios' ? 'keyboardWillHide' : 'keyboardDidHide';
    const sub1 = Keyboard.addListener(showEvt, (e) => setKeyboardOffset(e.endCoordinates.height));
    const sub2 = Keyboard.addListener(hideEvt, () => setKeyboardOffset(0));
    return () => { sub1.remove(); sub2.remove(); };
  }, []);

  // ✅ FIX 3: Seek to 0 on new video URL
  useEffect(() => {
    if (fullscreenVideo && videoRef.current) {
      videoRef.current.setPositionAsync(0).catch(() => {});
    }
  }, [fullscreenVideo]);

  const mountNextVideo = (videoUrl) => {
    setShowVideoLoader(true);
    setFullscreenVideo(videoUrl);
    setVideoFinished(false);
    setShowStepInput(false);
    // No timeout here — Video component handles all 3 fallback layers
  };

  const startCheckinFlow = async () => {
    try {
      setShowHomeScreen(false);
      setShowVideoLoader(true);   // ✅ Show loader IMMEDIATELY — no blank screen
      setIsLoading(true);
      setShowCheckinFlow(true);
      setFlowChatStartIndex(chats.length);

      const response = await fetch(`${API_BASE}/chat`, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({
          query: language === 'hi' ? 'चेक इन' : 'check in',
          uid,
          language
        })
      });
      const data = await response.json();
      if (data.type === 'avatar_checkin_flow') {
        setCurrentStep(data.step);
        setPendingStepData(data);
        if (data.message) {
          setChats(prev => [...prev, {
            type: 'text', text: data.message,
            sender: 'bot', timestamp: new Date().toISOString()
          }]);
        }
        mountNextVideo(data.avatar_video);
      }
    } catch (error) {
      console.error('Error:', error);
    } finally {
      setIsLoading(false);
    }
  };

  const processCheckinStep = async (step, userInput) => {
    try {
      setIsLoading(true);
      setShowStepInput(false);
      
      // Add user response to chat
      if (userInput && typeof userInput === 'object') {
        let userText = '';
        if (userInput.pnr) userText = `PNR: ${userInput.pnr}`;
        else if (userInput.lastname) userText = `Last Name: ${userInput.lastname}`;
        else if (userInput.mobile) userText = `Mobile: ${userInput.mobile}`;
        else if (userInput.email) userText = `Email: ${userInput.email}`;
        else if (userInput.consent !== undefined) userText = userInput.consent ? 'Yes, I consent' : 'No';
        else if (userInput.disclaimer_accepted) userText = 'I understand and accept';
        
        if (userText) {
          setChats(prev => [...prev, {
            type: 'text', text: userText,
            sender: 'user', timestamp: new Date().toISOString()
          }]);
        }
      }
      
      const response = await fetch(`${API_BASE}/checkin-step`, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({
          uid,
          step,
          user_input: typeof userInput === 'string' ? { response: userInput } : userInput,
          language
        })
      });
      const data = await response.json();
      setCurrentStep(data.step);
      setCheckinData(prev => ({ ...prev, [step]: userInput }));
      setPendingStepData(data);

      if (data.type === 'checkin_complete') {
        const successMsg = language === 'hi' 
          ? '✅ चेक-इन सफल! आपको ईमेल के माध्यम से बोर्डिंग पास प्राप्त होगा।'
          : '✅ Check-in successful! You\'ll receive your boarding pass via email.';
        const errorMsg = language === 'hi'
          ? '❌ चेक-इन विफल। कृपया अपने विवरण सत्यापित करें।'
          : '❌ Check-in failed. Please verify your details.';
        
        setChats(prev => [...prev, {
          type: 'text',
          text: data.success ? successMsg : errorMsg,
          sender: 'bot',
          timestamp: new Date().toISOString()
        }]);
        setTimeout(() => {
          setShowCheckinFlow(false);
          setFullscreenVideo(null);
          setVideoFinished(false);
          setShowVideoLoader(false);
          setShowStepInput(false);
          setTimeout(() => chatEndRef.current?.scrollToEnd({ animated: true }), 300);
        }, 800);
        return;
      }

      if (data.message) {
        setChats(prev => [...prev, {
          type: 'text', text: data.message,
          sender: 'bot', timestamp: new Date().toISOString()
        }]);
      }

      if (data.avatar_video) {
        mountNextVideo(data.avatar_video);
      } else {
        handleVideoEnd(data);
      }
    } catch (error) {
      console.error('Error:', error);
    } finally {
      setIsLoading(false);
    }
  };

  const startAvatarFlow = async () => {
    try {
      setShowHomeScreen(false);
      setShowVideoLoader(true);   // ✅ Show loader IMMEDIATELY — no blank screen
      setIsLoading(true);
      setShowAvatarFlow(true);
      // ✅ Change 2: Record where flow messages start (exclude welcome msg)
      setFlowChatStartIndex(chats.length);

      const response = await fetch(`${API_BASE}/chat`, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({
          query: language === 'hi' ? 'अवतार के साथ उड़ान बुक करें' : 'book flight with avatar',
          uid,
          language
        })
      });
      const data = await response.json();
      if (data.type === 'avatar_flow') {
        setCurrentStep(data.step);
        setPendingStepData(data);
        // ✅ Add first step message to chats so it shows on overlay
        if (data.message) {
          setChats(prev => [...prev, {
            type: 'text', text: data.message,
            sender: 'bot', timestamp: new Date().toISOString()
          }]);
        }
        mountNextVideo(data.avatar_video);
      }
    } catch (error) {
      console.error('Error:', error);
    } finally {
      setIsLoading(false);
    }
  };

  const processStep = async (step, userInput) => {
    try {
      setIsLoading(true);
      setShowStepInput(false);
      
      // Add user response to chat
      if (userInput && typeof userInput === 'object' && !userInput.continue) {
        let userText = '';
        if (userInput.origin) userText = `From: ${userInput.origin}`;
        else if (userInput.destination) userText = `To: ${userInput.destination}`;
        else if (userInput.date) userText = `Date: ${userInput.date}`;
        else if (userInput.adults !== undefined) userText = `Passengers: ${userInput.adults} Adult(s), ${userInput.children || 0} Child(ren), ${userInput.infants || 0} Infant(s)`;
        else if (userInput.selected_flight) userText = `Flight: ${userInput.selected_flight}`;
        else if (userInput.email) userText = `Email: ${userInput.email}, Phone: ${userInput.phone}`;
        else if (userInput.confirmed) userText = 'Booking confirmed';
        
        if (userText) {
          setChats(prev => [...prev, {
            type: 'text', text: userText,
            sender: 'user', timestamp: new Date().toISOString()
          }]);
        }
      }
      
      const response = await fetch(`${API_BASE}/avatar-step`, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({
          uid,
          step,
          user_input: typeof userInput === 'string' ? { response: userInput } : userInput,
          language
        })
      });
      const data = await response.json();
      setCurrentStep(data.step);
      setBookingData(prev => ({ ...prev, [step]: userInput }));
      setPendingStepData(data);

      // Check if booking is complete (payment step or complete step)
      if (data.step === 'complete' || data.type === 'complete' || data.step === 'payment') {
        const successMsg = language === 'hi'
          ? '✅ आपकी फ्लाइट सफलतापूर्वक बुक हो गई है! IndiGo चुनने के लिए धन्यवाद। सुरक्षित यात्रा! ✈️'
          : '✅ Your flight has been booked successfully! Thank you for choosing IndiGo. Have a great journey! ✈️';
        
        setChats(prev => [...prev, {
          type: 'text',
          text: successMsg,
          sender: 'bot',
          timestamp: new Date().toISOString()
        }]);

        // Return to chatbot immediately
        setTimeout(() => {
          setShowAvatarFlow(false);
          setFullscreenVideo(null);
          setVideoFinished(false);
          setShowVideoLoader(false);
          setShowStepInput(false);
          setTimeout(() => chatEndRef.current?.scrollToEnd({ animated: true }), 300);
        }, 1500);
        return;
      }

      // ✅ Change 3: Add each step's bot message to chats for the left overlay
      if (data.message) {
        setChats(prev => [...prev, {
          type: 'text', text: data.message,
          sender: 'bot', timestamp: new Date().toISOString()
        }]);
      }

      if (data.avatar_video) {
        mountNextVideo(data.avatar_video);
      } else {
        handleVideoEnd(data);
      }
    } catch (error) {
      console.error('Error:', error);
    } finally {
      setIsLoading(false);
    }
  };

  const handleVideoEnd = (stepData) => {
    if (!stepData) return;
    setVideoFinished(true);
    
    // Check-in flow handling
    if (showCheckinFlow) {
      // Give user time to read disclaimer_explanation (3 seconds)
      if (stepData.step === 'disclaimer_explanation') {
        setTimeout(() => {
          setShowStepInput(true);
        }, 3000);
        return;
      }
      
      if (!requiresCheckinInput(stepData.step) && stepData.step !== 'processing_checkin' && stepData.step !== 'complete') {
        setTimeout(() => {
          setVideoFinished(false);
          processCheckinStep(stepData.step, { continue: true });
        }, 300);
      } else if (stepData.step === 'complete') {
        setTimeout(() => {
          setFullscreenVideo(null);
          setVideoFinished(false);
          setShowCheckinFlow(false);
          setShowVideoLoader(false);
        }, 1000);
      } else {
        setShowStepInput(true);
      }
      return;
    }
    
    // Booking flow handling
    if (
      !requiresInput(stepData.step) &&
      stepData.step !== 'flight_search' &&
      stepData.step !== 'review_booking' &&
      stepData.step !== 'payment' &&
      stepData.step !== 'complete'
    ) {
      setTimeout(() => {
        setVideoFinished(false);
        processStep(stepData.step, { continue: true });
      }, 300);
    } else if (stepData.step === 'complete') {
      setTimeout(() => {
        setFullscreenVideo(null);
        setVideoFinished(false);
        setShowAvatarFlow(false);
        setShowVideoLoader(false);
      }, 1000);
    } else {
      setShowStepInput(true);
    }
  };

  const requiresInput = (step) =>
    ['origin_selection','destination_selection','date_selection',
     'passenger_selection','passenger_details','contact_details',
     'flight_selection','review_booking'].includes(step);

  const requiresCheckinInput = (step) =>
    ['pnr_collection','lastname_collection','mobile_collection',
     'email_collection','disclaimer_explanation','seat_consent'].includes(step);

  // ✅ Fix 1: Chat overlay with proper bubble styling
  const renderChatOverlay = () => {
    const flowChats = chats.slice(flowChatStartIndex);
    if (!flowChats.length) return null;
    const recent = flowChats.slice(-4); // Last 4 messages

    return (
      <View style={styles.chatOverlay} pointerEvents="none">
        {recent.map((chat, idx) => {
          // Oldest = 0.2 opacity, newest = 1.0
          const opacity = 0.2 + (idx / Math.max(recent.length - 1, 1)) * 0.8;
          const isUser = chat.sender === 'user';
          return (
            <View
              key={idx}
              style={[
                styles.chatOverlayBubble,
                isUser ? styles.chatOverlayUserBubble : styles.chatOverlayBotBubble,
                { opacity }
              ]}
            >
              <Text
                style={[
                  styles.chatOverlayText,
                  isUser ? styles.chatOverlayUserText : styles.chatOverlayBotText
                ]}
                numberOfLines={3}
              >
                {chat.text}
              </Text>
            </View>
          );
        })}
      </View>
    );
  };

  // ✅ Change 6: Faded history in panel — only flow messages
  const renderFadedHistoryInPanel = () => {
    // ✅ Same — exclude welcome message
    const flowChats = chats.slice(flowChatStartIndex);
    if (flowChats.length < 2) return null; // Don't show if only 1 message
    const history = flowChats.slice(-4, -0); // All but most recent (shown in stepMessageBox)

    return (
      <View style={styles.fadedHistory} pointerEvents="none">
        {history.map((chat, idx) => {
          const opacity = 0.06 + (idx / Math.max(history.length - 1, 1)) * 0.22;
          return (
            <View key={idx} style={[styles.fadedHistoryBubble, {
              alignSelf: chat.sender === 'user' ? 'flex-end' : 'flex-start',
              opacity,
            }]}>
              <Text style={[styles.fadedHistoryText, {
                color: chat.sender === 'user' ? indigoBlue : '#333'
              }]} numberOfLines={1}>
                {chat.sender === 'bot' ? '🤖 ' : '👤 '}{chat.text}
              </Text>
            </View>
          );
        })}
      </View>
    );
  };

  const renderStepInputInModal = (step) => {
    // Check-in inputs
    if (showCheckinFlow) {
      switch (step) {
        case 'pnr_collection':
          return <PNRInput onNext={(d) => processCheckinStep('pnr_collection', d)} language={language} />;
        case 'lastname_collection':
          return <LastNameInput onNext={(d) => processCheckinStep('lastname_collection', d)} language={language} />;
        case 'mobile_collection':
          return <MobileInput onNext={(d) => processCheckinStep('mobile_collection', d)} language={language} />;
        case 'email_collection':
          return <EmailInput onNext={(d) => processCheckinStep('email_collection', d)} language={language} />;
        case 'disclaimer_explanation':
          return <DisclaimerAcceptance onNext={(d) => processCheckinStep('disclaimer_explanation', d)} language={language} />;
        case 'seat_consent':
          return <SeatConsent onNext={(d) => processCheckinStep('seat_consent', d)} language={language} />;
        default: return null;
      }
    }
    
    // Booking inputs
    if (!requiresInput(step)) return null;
    switch (step) {
      case 'origin_selection':
        return <OriginSelection onNext={(d) => processStep('origin_selection', d)} language={language} />;
      case 'destination_selection':
        return <DestinationSelection onNext={(d) => processStep('destination_selection', d)} language={language} />;
      case 'date_selection':
        return <DateSelection onNext={(d) => processStep('date_selection', d)} language={language} />;
      case 'passenger_selection':
        return <PassengerSelection onNext={(d) => processStep('passenger_selection', d)} language={language} />;
      case 'passenger_details':
        return <PassengerDetails onNext={(d) => processStep('passenger_details', d)} passengerCount={bookingData.passenger_selection} />;
      case 'contact_details':
        return <ContactDetails onNext={(d) => processStep('contact_details', d)} language={language} />;
      default: return null;
    }
  };

  const renderChatMessage = (chat, index) => {
    if (chat.sender === 'user') {
      return (
        <View key={index} style={styles.userMessageContainer}>
          <Card style={styles.userMessage}>
            <Card.Content><Text style={styles.userMessageText}>{chat.text}</Text></Card.Content>
          </Card>
        </View>
      );
    }
    return (
      <View key={index} style={styles.botMessageContainer}>
        <Card style={styles.botMessage}>
          <Card.Content><Text style={styles.botMessageText}>{chat.text}</Text></Card.Content>
        </Card>
      </View>
    );
  };

  return (
    <SafeAreaView style={styles.container}>
      <KeyboardAvoidingView behavior={Platform.OS === 'ios' ? 'padding' : 'height'} style={styles.container}>

        {/* Header */}
        <View style={styles.header}>
          <View>
            <Text style={styles.headerTitle}>6Eskai</Text>
            <Text style={styles.headerSubtitle}>AI Flight Assistant</Text>
          </View>
          <View style={{ flexDirection: 'row', alignItems: 'center', gap: 8 }}>
            {/* Language toggle */}
            <Menu
              visible={menuVisible}
              onDismiss={() => setMenuVisible(false)}
              anchor={
                <Button
                  onPress={() => setMenuVisible(true)}
                  textColor="white"
                  mode="outlined"
                  style={styles.headerButton}
                  compact
                >
                  {language === 'en' ? '🌐 EN' : '🌐 HI'}
                </Button>
              }
            >
              <Menu.Item onPress={() => { setLanguage('en'); setMenuVisible(false); }} title="English" />
              <Menu.Item onPress={() => { setLanguage('hi'); setMenuVisible(false); }} title="हिंदी" />
            </Menu>
            {/* Home/Reset button */}
            <Button
              onPress={resetApp}
              textColor="white"
              icon="home"
              mode="outlined"
              style={styles.headerButton}
              compact
            >
              Home
            </Button>
          </View>
        </View>

        {/* ── IndiGo Themed Home Screen ── */}
        {showHomeScreen && (
          <View style={styles.homeScreen}>
            {/* Blue hero section */}
            <View style={styles.homeHero}>
              {/* IndiGo Logo */}
              <View style={styles.homeLogoRow}>
                <View style={styles.homeLogoCircle}>
                  <Text style={styles.homeLogoNumber}>6</Text>
                  <Text style={styles.homeLogoE}>E</Text>
                </View>
                <View style={styles.homeLogoTextWrap}>
                  <Text style={styles.homeLogoIndigo}>indigo</Text>
                  <View style={styles.homeLogoDivider} />
                </View>
              </View>
              <Text style={styles.homeTagline}>Your AI Travel Companion</Text>
              <Text style={styles.homeSubTagline}>Lean. Clean. Flying Machine.</Text>

              {/* Decorative wave strip */}
              <View style={styles.homeWaveStrip}>
                {[...Array(18)].map((_, i) => (
                  <View key={i} style={[styles.homeWaveDot, { opacity: 0.12 + (i % 3) * 0.12 }]} />
                ))}
              </View>
            </View>

            {/* Action cards */}
            <View style={styles.homeCardsArea}>
              <Text style={styles.homeCardsLabel}>What would you like to do?</Text>

              {/* Flight Booking card */}
              <TouchableOpacity style={styles.homeActionCard} onPress={startAvatarFlow} activeOpacity={0.75}>
                <View style={styles.homeActionLeft}>
                  <View style={[styles.homeActionIcon, { backgroundColor: '#e8f0ff' }]}>
                    <Text style={styles.homeActionEmoji}>✈️</Text>
                  </View>
                  <View>
                    <Text style={styles.homeActionTitle}>Flight Booking</Text>
                    <Text style={styles.homeActionSub}>
                      {language === 'hi' ? 'अवतार के साथ बुक करें' : 'Book with Avatar Guidance'}
                    </Text>
                  </View>
                </View>
                <View style={styles.homeActionArrow}>
                  <Text style={styles.homeActionArrowText}>›</Text>
                </View>
              </TouchableOpacity>

              {/* Web Check-in card */}
              <TouchableOpacity style={styles.homeActionCard} onPress={startCheckinFlow} activeOpacity={0.75}>
                <View style={styles.homeActionLeft}>
                  <View style={[styles.homeActionIcon, { backgroundColor: '#fff4e6' }]}>
                    <Text style={styles.homeActionEmoji}>🎫</Text>
                  </View>
                  <View>
                    <Text style={styles.homeActionTitle}>Web Check-in</Text>
                    <Text style={styles.homeActionSub}>
                      {language === 'hi' ? 'ऑनलाइन चेक-इन करें' : 'Check-in Online'}
                    </Text>
                  </View>
                </View>
                <View style={styles.homeActionArrow}>
                  <Text style={styles.homeActionArrowText}>›</Text>
                </View>
              </TouchableOpacity>
            </View>
          </View>
        )}

        {/* Chat area — hidden during avatar flow */}
        {!showAvatarFlow && !showCheckinFlow && !showHomeScreen && (
          <ScrollView
            style={styles.chatArea}
            ref={chatEndRef}
            onContentSizeChange={() => chatEndRef.current?.scrollToEnd({ animated: true })}
          >
            {chats.map((chat, idx) => renderChatMessage(chat, idx))}
            {isLoading && <Text style={styles.loadingText}>Bot is typing...</Text>}
          </ScrollView>
        )}

        {!showAvatarFlow && !showCheckinFlow && chats.length <= 2 && !showHomeScreen && (
          <View style={styles.quickActions}>
            <Button mode="outlined" onPress={startAvatarFlow} style={styles.quickButton}>
              📱 {language === 'hi' ? 'अवतार बुकिंग शुरू करें' : 'Start Avatar Booking'}
            </Button>
            <Button mode="outlined" onPress={startCheckinFlow} style={styles.quickButton}>
              ✈️ {language === 'hi' ? 'चेक इन' : 'Web Check-in'}
            </Button>
          </View>
        )}

        <View style={styles.footer}>
          <Text style={styles.footerText}>Powered by IndiGo AI</Text>
        </View>

        {/* ── Fullscreen Modal ── */}
        <Modal
          visible={!!fullscreenVideo || showVideoLoader}
          animationType="fade"
          onRequestClose={() => {}}
          statusBarTranslucent
        >
          {/* ✅ Change 7: IndiGo blue not '#000' — any micro-flash matches loader color */}
          <View style={{ flex: 1, backgroundColor: indigoBlue }}>

            {/* ✅ FIX 3: Video ALWAYS renders and plays underneath — no opacity toggling */}
            {fullscreenVideo && (
              <Video
                ref={videoRef}
                source={{ uri: fullscreenVideo }}
                style={styles.fullscreenVideo}
                resizeMode={ResizeMode.COVER}
                shouldPlay={!videoFinished}
                isLooping={false}
                onLoad={async () => {
                  try { await videoRef.current?.setPositionAsync(0); } catch {}
                  // Layer 3: hard timeout — if nothing else fires within 3s, force hide
                  setTimeout(() => setShowVideoLoader(false), 3000);
                }}
                onReadyForDisplay={() => {
                  // Layer 1: first frame decoded (most reliable on iOS)
                  setShowVideoLoader(false);
                }}
                onPlaybackStatusUpdate={(status) => {
                  // Layer 2: playback started (most reliable on Android)
                  if (status.isPlaying) {
                    setShowVideoLoader(false);
                  }
                  if (status.didJustFinish) {
                    handleVideoEnd(pendingStepData);
                  }
                }}
                onError={(e) => {
                  console.log('Video error:', e);
                  setShowVideoLoader(false);
                  handleVideoEnd(pendingStepData);
                }}
              />
            )}

            {/* ✅ FIX 3: Loader as ABSOLUTE OVERLAY on top of video — no gap/black flash */}
            {showVideoLoader && (
              <View style={styles.loaderOverlay}>
                {/* Top branding strip */}
                <View style={styles.loaderTopStrip} />

                {/* Logo */}
                <View style={styles.loaderLogoWrap}>
                  <View style={styles.loaderGlowRing} />
                  <View style={styles.loaderLogo}>
                    <Text style={styles.loaderLogoNum}>6</Text>
                    <Text style={styles.loaderLogoLetter}>E</Text>
                  </View>
                </View>

                {/* Brand name */}
                <Text style={styles.loaderBrandName}>indigo</Text>
                <View style={styles.loaderBrandUnderline} />

                <Text style={styles.loaderSubtitle}>
                  {language === 'hi' ? 'अगला चरण लोड हो रहा है...' : 'Preparing next step...'}
                </Text>

                {/* Spinner */}
                <ActivityIndicator size="large" color={indigoOrange} style={{ marginTop: 32 }} />

                {/* Orange progress dots */}
                <View style={styles.loaderDotRow}>
                  <View style={[styles.dot, { backgroundColor: indigoOrange, opacity: 1 }]} />
                  <View style={[styles.dot, { backgroundColor: indigoOrange, opacity: 0.5 }]} />
                  <View style={[styles.dot, { backgroundColor: indigoOrange, opacity: 0.2 }]} />
                </View>

                {/* Bottom strip */}
                <View style={styles.loaderBottomStrip} />
              </View>
            )}

            {/* Chat overlay on left while video plays */}
            {!showVideoLoader && !videoFinished && renderChatOverlay()}

            {/* ✅ Floating Home button — always visible during video/loader */}
            <View style={styles.floatingHomeBtn} pointerEvents="box-none">
              <Button
                onPress={resetApp}
                textColor="white"
                icon="home"
                mode="contained"
                buttonColor="rgba(0,0,0,0.45)"
                style={styles.floatingHomeBtnInner}
                labelStyle={{ fontSize: 12 }}
                compact
              >
                Home
              </Button>
            </View>

            {/* ✅ FIX 1 + FIX 2: Step input panel — keyboard-aware + faded history */}
            {!showVideoLoader && videoFinished && pendingStepData && (
              <View style={[styles.stepInputPanel, { bottom: keyboardOffset }]}>
                <ScrollView
                  keyboardShouldPersistTaps="handled"
                  showsVerticalScrollIndicator={false}
                >
                  {/* ✅ FIX 2: Faded previous messages at top of panel */}
                  {renderFadedHistoryInPanel()}

                  {/* Step message */}
                  <View style={styles.stepMessageBox}>
                    <Text style={styles.stepMessage}>{pendingStepData.message}</Text>
                  </View>

                  {/* Flights */}
                  {pendingStepData.flights && showStepInput && (
                    <View>
                      <Text style={styles.sectionLabel}>Available Flights</Text>
                      {pendingStepData.flights.map((flight, idx) => (
                        <Card
                          key={idx}
                          style={styles.flightCard}
                          onPress={() => processStep('flight_selection', { selected_flight: flight.flight_number })}
                        >
                          <Card.Content>
                            <Text style={{ fontWeight: 'bold' }}>{flight.flight_number}</Text>
                            <Text style={{ fontSize: 12 }}>{flight.departure} → {flight.arrival} ({flight.duration})</Text>
                            <Text style={{ color: indigoOrange, fontWeight: 'bold' }}>{flight.price}</Text>
                          </Card.Content>
                        </Card>
                      ))}
                    </View>
                  )}

                  {/* Review booking */}
                  {pendingStepData.step === 'review_booking' && showStepInput && (
                    <ReviewBooking
                      bookingData={bookingData}
                      onNext={(data) => processStep('review_booking', data)}
                      language={language}
                    />
                  )}

                  {/* Dynamic form inputs */}
                  {showStepInput && pendingStepData.step === currentStep && (
                    <View style={{ marginTop: 16 }}>
                      {renderStepInputInModal(pendingStepData.step)}
                    </View>
                  )}

                </ScrollView>
              </View>
            )}

          </View>
        </Modal>

      </KeyboardAvoidingView>
    </SafeAreaView>
  );
}

function PassengerDetails({ onNext, passengerCount }) {
  const total = passengerCount ? passengerCount.adults + passengerCount.children + passengerCount.infants : 1;
  const [passengers, setPassengers] = useState(
    Array.from({ length: total }, (_, i) => {
      let type = 'Adult';
      if (passengerCount) {
        if (i >= passengerCount.adults + passengerCount.children) type = 'Infant';
        else if (i >= passengerCount.adults) type = 'Child';
      }
      return { name: '', age: '', gender: 'male', type };
    })
  );
  const update = (i, field, val) => {
    const updated = [...passengers]; updated[i][field] = val; setPassengers(updated);
  };
  const isValid = passengers.every(p => p.name.trim() && p.age.trim());
  return (
    <View>
      <Text style={styles.sectionTitle}>Passenger Details</Text>
      {passengers.map((p, i) => (
        <View key={i} style={styles.passengerForm}>
          <Text style={{ marginBottom: 6, color: indigoBlue, fontWeight: '600' }}>Passenger {i + 1} ({p.type})</Text>
          <TextInput mode="outlined" placeholder="Full Name" value={p.name} onChangeText={v => update(i, 'name', v)} style={styles.input} />
          <TextInput mode="outlined" placeholder="Age" value={p.age} onChangeText={v => update(i, 'age', v)} keyboardType="numeric" style={styles.input} />
        </View>
      ))}
      <Button mode="contained" onPress={() => onNext({ passengers })} disabled={!isValid} style={styles.button} buttonColor={indigoBlue}>Continue</Button>
    </View>
  );
}
