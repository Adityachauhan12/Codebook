"""Adapters for the open-source coding agents that do the actual code reading.

Each adapter knows three things: how to check the tool is installed, how to
build one non-interactive command, and which environment variables the tool
needs for model access.

A note on prompt delivery. These prompts are large (schema + file list), and
Windows caps a command line at ~8,191 characters. So every adapter writes the
prompt to a file and passes either the tool's native task-file flag or a short
pointer prompt. Never pass the full prompt as an argument.

If a tool changes its flags between versions, override the command in .env
rather than editing this file - see `<TOOL>_CMD` below.
"""
from __future__ import annotations

import os
import shlex
from pathlib import Path

from ..config import settings
from .base import (
    BackendError,
    BackendStatus,
    KnowledgeBackend,
    KnowledgeTask,
    probe_version,
    resolve_executable,
)


def _litellm_env() -> dict[str, str]:
    """Credentials for tools built on LiteLLM (Aider, OpenHands).

    LiteLLM reads Azure config from AZURE_API_KEY / AZURE_API_BASE /
    AZURE_API_VERSION and addresses the deployment as `azure/<deployment>`.
    We pass these explicitly rather than relying on each tool's own config
    file, so there is exactly one place to put credentials: the project .env.
    """
    env: dict[str, str] = {}
    if settings.is_azure:
        if not settings.azure_key_is_placeholder:
            env["AZURE_API_KEY"] = settings.azure_api_key
            env["AZURE_OPENAI_API_KEY"] = settings.azure_api_key
        env["AZURE_API_BASE"] = settings.azure_endpoint
        env["AZURE_OPENAI_ENDPOINT"] = settings.azure_endpoint
        env["AZURE_API_VERSION"] = settings.azure_api_version
    elif not settings.anthropic_key_is_placeholder:
        env["ANTHROPIC_API_KEY"] = settings.anthropic_api_key
    return env


def _model_ref() -> str:
    """How the OSS agent should name the model on its command line."""
    if settings.is_azure:
        # LiteLLM/Aider convention: the deployment name, not the model name.
        return f"azure/{settings.knowledge_model}"
    return f"anthropic/{settings.knowledge_model}"


def _write_prompt_file(task: KnowledgeTask) -> Path:
    prompt_path = task.output_file.parent / f"{task.task_id}.prompt.md"
    prompt_path.parent.mkdir(parents=True, exist_ok=True)
    prompt_path.write_text(task.render_prompt(), encoding="utf-8")
    return prompt_path


def _pointer_prompt(prompt_path: Path) -> str:
    return (
        f"Read the file at {prompt_path.as_posix()} and carry out the instructions "
        f"in it exactly. It is a read-only analysis task that ends by writing one "
        f"JSON file. Do not modify any other file."
    )


def _override_command(env_key: str, **placeholders: str) -> list[str] | None:
    """Let .env replace a whole command line, e.g. when a tool changes its flags."""
    template = os.getenv(env_key, "").strip()
    if not template:
        return None
    parts = shlex.split(template, posix=(os.name != "nt"))
    return [p.format(**placeholders) for p in parts]


# --------------------------------------------------------------------------
# OpenCode - default. Lightest install, first-class non-interactive `run`.
# --------------------------------------------------------------------------


class OpenCodeBackend(KnowledgeBackend):
    name = "opencode"
    docs_url = "https://opencode.ai  (install: npm i -g opencode-ai)"

    def check(self) -> BackendStatus:
        exe = resolve_executable("opencode", "opencode.cmd", "opencode.exe")
        if not exe:
            return BackendStatus(
                False,
                "OpenCode not found on PATH. Install with:  npm install -g opencode-ai",
            )
        return BackendStatus(True, "ready", exe, probe_version(exe, "--version"))

    def build_command(self, repo_path: Path, task: KnowledgeTask) -> list[str]:
        status = self.check()
        if not status.available:
            raise BackendError(status.detail)
        prompt_file = _write_prompt_file(task)
        model = _model_ref()

        override = _override_command(
            "OPENCODE_CMD",
            exe=status.executable,
            model=model,
            prompt=_pointer_prompt(prompt_file),
            prompt_file=prompt_file.as_posix(),
            repo=repo_path.as_posix(),
            out=task.output_file.as_posix(),
        )
        if override:
            return override

        return [status.executable, "run", "--model", model, _pointer_prompt(prompt_file)]

    def environment(self) -> dict[str, str]:
        env = _litellm_env()
        if settings.is_azure:
            # OpenCode's Azure provider is configured through its own config
            # file rather than env vars alone. These help, but if OpenCode
            # cannot reach Azure on your machine, switch to KNOWLEDGE_BACKEND=aider
            # - LiteLLM's Azure support is purely env-var driven and needs no setup.
            env["AZURE_RESOURCE_NAME"] = _azure_resource_name()
            env["OPENAI_API_KEY"] = settings.azure_api_key
            env["OPENAI_BASE_URL"] = settings.azure_endpoint
        return env


def _azure_resource_name() -> str:
    """Extract `myresource` from https://myresource.openai.azure.com"""
    endpoint = settings.azure_endpoint.replace("https://", "").replace("http://", "")
    return endpoint.split(".")[0] if endpoint else ""


# --------------------------------------------------------------------------
# Aider - strongest repo map (tree-sitter); pip install, very Windows friendly.
# --------------------------------------------------------------------------


class AiderBackend(KnowledgeBackend):
    name = "aider"
    docs_url = "https://aider.chat  (install: pip install aider-install && aider-install)"

    def check(self) -> BackendStatus:
        exe = resolve_executable("aider", "aider.exe", "aider.cmd")
        if not exe:
            return BackendStatus(
                False, "Aider not found on PATH. Install with:  python -m pip install aider-chat"
            )
        return BackendStatus(True, "ready", exe, probe_version(exe, "--version"))

    def build_command(self, repo_path: Path, task: KnowledgeTask) -> list[str]:
        status = self.check()
        if not status.available:
            raise BackendError(status.detail)
        prompt_file = _write_prompt_file(task)
        model = _model_ref()

        override = _override_command(
            "AIDER_CMD",
            exe=status.executable,
            model=model,
            prompt=_pointer_prompt(prompt_file),
            prompt_file=prompt_file.as_posix(),
            repo=repo_path.as_posix(),
            out=task.output_file.as_posix(),
        )
        if override:
            return override

        return [
            status.executable,
            "--model", model,
            "--message-file", str(prompt_file),
            "--yes-always",          # non-interactive
            "--no-auto-commits",     # never touch the target repo's history
            "--no-gitignore",
            "--no-check-update",
            "--no-show-model-warnings",
            "--map-tokens", "4096",  # aider's repo map is the reason to use it
        ]

    def environment(self) -> dict[str, str]:
        env = _litellm_env()
        env["AIDER_ANALYTICS"] = "false"
        return env


# --------------------------------------------------------------------------
# OpenHands - heaviest, most autonomous. Needs Docker Desktop on Windows.
# --------------------------------------------------------------------------


class OpenHandsBackend(KnowledgeBackend):
    name = "openhands"
    docs_url = "https://docs.all-hands.dev  (install: pip install openhands-ai)"

    def check(self) -> BackendStatus:
        exe = resolve_executable("openhands", "openhands.exe", "openhands.cmd")
        if exe:
            return BackendStatus(True, "ready", exe, probe_version(exe, "--version"))
        # Fall back to the module entry point if the console script is absent.
        import importlib.util

        if importlib.util.find_spec("openhands") is not None:
            import sys

            return BackendStatus(True, "ready (module entry point)", sys.executable, None)
        return BackendStatus(
            False,
            "OpenHands not found. Install with:  python -m pip install openhands-ai "
            "(Docker Desktop must also be running for its sandbox).",
        )

    def build_command(self, repo_path: Path, task: KnowledgeTask) -> list[str]:
        status = self.check()
        if not status.available:
            raise BackendError(status.detail)
        prompt_file = _write_prompt_file(task)

        override = _override_command(
            "OPENHANDS_CMD",
            exe=status.executable,
            model=settings.knowledge_model,
            prompt=_pointer_prompt(prompt_file),
            prompt_file=prompt_file.as_posix(),
            repo=repo_path.as_posix(),
            out=task.output_file.as_posix(),
        )
        if override:
            return override

        if status.detail.startswith("ready (module"):
            return [status.executable, "-m", "openhands.core.main", "-f", str(prompt_file)]
        return [status.executable, "headless", "-f", str(prompt_file)]

    def environment(self) -> dict[str, str]:
        env = _litellm_env()
        env["LLM_MODEL"] = _model_ref()
        env["LLM_API_KEY"] = settings.api_key
        if settings.is_azure:
            env["LLM_BASE_URL"] = settings.azure_endpoint
            env["LLM_API_VERSION"] = settings.azure_api_version
        env["WORKSPACE_BASE"] = str(settings.target_dir)
        env["LOG_ALL_EVENTS"] = "false"
        return env

            return BackendStatus(
                False, "Continue CLI not found. Install with:  npm install -g @continuedev/cli"
            )
        return BackendStatus(True, "ready", exe, probe_version(exe, "--version"))

    def build_command(self, repo_path: Path, task: KnowledgeTask) -> list[str]:
        status = self.check()
        if not status.available:
            raise BackendError(status.detail)
        prompt_file = _write_prompt_file(task)

        override = _override_command(
            "CONTINUE_CMD",
            exe=status.executable,
            model=settings.knowledge_model,

            prompt_file=prompt_file.as_posix(),
            repo=repo_path.as_posix(),

# --------------------------------------------------------------------------
# Registry
# --------------------------------------------------------------------------

BACKENDS: dict[str, type[KnowledgeBackend]] = {
    "opencode": OpenCodeBackend,
    "aider": AiderBackend,
    "openhands": OpenHandsBackend,
    "continue": ContinueBackend,
}


def get_backend(name: str | None = None) -> KnowledgeBackend:
    key = (name or settings.knowledge_backend).strip().lower()
    if key not in BACKENDS:
        raise BackendError(
            f"Unknown KNOWLEDGE_BACKEND '{key}'. Choose one of: "
            f"{', '.join(sorted(BACKENDS))}, or 'builtin'."
        )
    return BACKENDS[key]()


def probe_all() -> dict[str, BackendStatus]:
    """What is actually installed on this machine? Surfaced in the UI."""
    return {name: cls().check() for name, cls in BACKENDS.items()}
