@app.get("/v1/individual-analysis")
async def individual_analysis(
    iga_code: str = Query(..., description="IGA code e.g., IGA6781"),
    crew_name: str = Query(..., description="Crew name e.g., Lavanya Singh"),
    date_from: str = Query(..., description="Start date in YYYY-MM-DD format"),
    date_to: str = Query(..., description="End date in YYYY-MM-DD format"),
):
    """
    UPDATED: Count category violations from ALL records (COMPLIANT + NON-COMPLIANT)
    if they have violations in that specific category.
    """
    try:
        start_date = datetime.strptime(date_from, "%Y-%m-%d").date()
        end_date = datetime.strptime(date_to, "%Y-%m-%d").date()

        # Load fresh records from GCS
        filters = DashboardFilters(date_from=start_date, date_to=end_date)
        all_records = load_records(filters)
        print(f"[INDIVIDUAL-ANALYSIS] Loaded {len(all_records)} total records from GCS")

        # Filter by crew
        iga_search = iga_code.strip().upper()
        crew_search = crew_name.strip().upper()

        crew_records = []
        for record in all_records:
            record_iga = record.get("iga_code", "").strip().upper()
            record_crew = record.get("crew_name", "").strip().upper()
            if record_iga == iga_search and record_crew == crew_search:
                crew_records.append(record)

        print(f"[INDIVIDUAL-ANALYSIS] Filtered to {len(crew_records)} crew records")

        if not crew_records:
            return JSONResponse(
                error="No assessments found for this crew in given date range",
                debug={
                    "searched_for": {"iga_code": iga_code, "crew_name": crew_name},
                    "date_range": f"{date_from} to {date_to}",
                    "total_records_in_range": len(all_records),
                    "matching_records": 0,
                },
                status_code=404,
            )

        # === UPDATED: Count violations from ALL records, not just NON-COMPLIANT ===
        category_violation_count = defaultdict(int)
        
        for record in crew_records:
            # Get issues from EVERY record (compliant or non-compliant)
            issues = record.get("issues") or []
            
            print(f"[INDIVIDUAL-ANALYSIS] Processing record: assessment={record.get('assessment')}, issues_count={len(issues)}")
            
            for issue in issues:
                # Map issue text to category using the existing helper
                heading = issue_heading(issue)
                
                # CRITICAL: Count only if it's one of the 5 main categories
                if heading in ["uniform", "hairstyle", "makeup", "nails", "accessories"]:
                    category_violation_count[heading] += 1
                    print(f"[INDIVIDUAL-ANALYSIS] Issue: '{issue}' → Category: '{heading}' (count: {category_violation_count[heading]})")

        # Calculate summary statistics
        total = len(crew_records)
        compliant_count = sum(1 for r in crew_records if r.get("assessment") == "COMPLIANT")
        non_compliant_count = total - compliant_count
        pass_rate = (compliant_count / total * 100) if total > 0 else 0.0

        print(f"[INDIVIDUAL-ANALYSIS] Summary: Total={total}, Compliant={compliant_count}, NC={non_compliant_count}, PassRate={pass_rate:.2f}%")

        # Build category breakdown using ALL records as denominator
        category_breakdown = {}
        for cat in ["uniform", "hairstyle", "makeup", "nails", "accessories"]:
            violation_count = category_violation_count.get(cat, 0)
            # Percentage = violations / total assessments (not just non-compliant)
            percentage = (violation_count / total * 100) if total > 0 else 0.0
            
            category_breakdown[cat] = {
                "violations": violation_count,
                "percentage": round(percentage, 2),
            }
            
            print(f"[INDIVIDUAL-ANALYSIS] {cat.upper()}: violations={violation_count}, total={total}, percentage={percentage:.2f}%")

        # Build trend (daily breakdown)
        trend_list = []
        for day in date_range(start_date, end_date):
            day_records = [r for r in crew_records if r.get("date") == day.isoformat()]
            day_compliant = sum(1 for r in day_records if r.get("assessment") == "COMPLIANT")
            day_non_compliant = len(day_records) - day_compliant
            
            trend_list.append({
                "date": day.isoformat(),
                "compliant": day_compliant,
                "non_compliant": day_non_compliant,
            })

        # Build response
        response = {
            "crew": {
                "iga_code": iga_code,
                "name": crew_name,
            },
            "date_range": {
                "from": date_from,
                "to": date_to,
                "days_analyzed": (end_date - start_date).days + 1,
            },
            "summary": {
                "total_assessments": total,
                "compliant": compliant_count,
                "non_compliant": non_compliant_count,
                "pass_rate": f"{pass_rate:.2f}",
                "pass_rate_percentage": f"{pass_rate:.2f}%",
            },
            "non_compliance_by_category": category_breakdown,
            "trend": trend_list,
            "recent_assessments": [
                {
                    "timestamp": r.get("timestamp").isoformat() if r.get("timestamp") else None,
                    "date": r.get("date").isoformat() if r.get("date") else None,
                    "score": r.get("score"),
                    "assessment": r.get("assessment"),
                    "issues": r.get("issues", []),
                }
                for r in crew_records[-10:]  # Last 10 assessments
            ],
        }

        print(f"[INDIVIDUAL-ANALYSIS] Returning response with {len(crew_records)} records")
        return response

    except Exception as e:
        import traceback
        print(f"[ERROR] individual_analysis: {str(e)}")
        print(traceback.format_exc())
        return JSONResponse(
            error="Internal server error",
            details=str(e),
            type=type(e).__name__,
            status_code=500,
        )
