"""The duration planner — the answer to "Veo only makes 8 second clips".

Veo returns at most one short clip per generation: 4, 6 or 8 seconds. A 90
second video therefore does not exist as a single API call. It is a plan of N
segments that get generated independently and accumulated by ffmpeg at render
time.

This module owns that plan. It is the single source of truth for "how many
clips" and "how long is each one" — the storyboard writer, the clip queue and
the renderer all read from it, so the three can never disagree.
"""

from __future__ import annotations

from ..config import settings
from ..errors import BadRequest
from .models import Beat, DurationPlan, Pacing, PlannedSegment, ShotFunction

#: Durations Veo accepts for a single generation.
VEO_ALLOWED_SECONDS = (4, 6, 8)

#: Layer 1 beat map, as fractions of total runtime so it scales from a 15s
#: cutdown to a three minute film. Derived from the 0-90s table in dramaturgy.md.
BEAT_MAP: list[tuple[Beat, float, float, str]] = [
    ("hook", 0.0, 5 / 90, "Hero already in tension. No setup. Problem on screen."),
    ("context", 5 / 90, 15 / 90, "Where we are. Who is near. What is at stake."),
    ("pressure", 15 / 90, 30 / 90, "Hero tries to hold control."),
    ("crack", 30 / 90, 45 / 90, "A detail appears that breaks the hero's position."),
    ("acceleration", 45 / 90, 60 / 90, "Cuts shorten. Breath tightens."),
    ("impact", 60 / 90, 75 / 90, "Decision, break, confession, or action."),
    ("aftermath", 75 / 90, 1.0, "Brief silence or visual residue."),
]

#: How long a shot in each beat wants to be. The rhythm ladder is
#: long -> shorter -> shorter -> pause -> impact, so acceleration gets the
#: shortest segments and context/aftermath get the longest.
BEAT_LENGTH_WEIGHT: dict[Beat, float] = {
    "hook": 0.9,
    "context": 1.0,
    "pressure": 0.95,
    "crack": 0.8,
    "acceleration": 0.4,
    "impact": 0.65,
    "aftermath": 1.0,
}

BEAT_SHOT_FUNCTIONS: dict[Beat, list[ShotFunction]] = {
    "hook": ["detail", "pressure"],
    "context": ["establish", "power"],
    "pressure": ["pressure", "reaction"],
    "crack": ["detail", "shift"],
    "acceleration": ["detail", "reaction", "power"],
    "impact": ["impact", "reaction"],
    "aftermath": ["aftermath", "exit"],
}

#: Preferred segment length per pacing, before exact-fit adjustment.
PACING_PREFERRED_SECONDS: dict[Pacing, int] = {"slow": 8, "standard": 6, "fast": 4}


def beat_label(beat: Beat) -> str:
    for name, _from, _to, label in BEAT_MAP:
        if name == beat:
            return label
    return ""


def _exact_combination(total: int, count: int, allowed: list[int]) -> list[int] | None:
    """Find ``count`` values from ``allowed`` summing to exactly ``total``.

    The search space is tiny — three denominations, at most ~45 segments — so a
    pruned enumeration is both exact and instant.
    """
    ordered = sorted(allowed, reverse=True)
    smallest, largest = ordered[-1], ordered[0]

    def search(remaining_total: int, remaining_count: int, start: int) -> list[int] | None:
        if remaining_count == 0:
            return [] if remaining_total == 0 else None
        # Whatever is left must still be reachable with the segments we have.
        if remaining_total < smallest * remaining_count:
            return None
        if remaining_total > largest * remaining_count:
            return None
        for index in range(start, len(ordered)):
            value = ordered[index]
            rest = search(remaining_total - value, remaining_count - 1, index)
            if rest is not None:
                return [value, *rest]
        return None

    return search(total, count, 0)


def _fit_durations(target_seconds: int, pacing: Pacing) -> tuple[list[int], int, bool]:
    """Pick the multiset of segment durations.

    Prefers an exact fit at the segment count the pacing implies, widens the
    count outward, and only then allows a small overshoot that the renderer
    trims off the tail.

    Returns ``(durations, planned_total, exact)``.
    """
    allowed = settings.allowed_clip_seconds
    shortest = min(allowed)
    preferred = min(PACING_PREFERRED_SECONDS[pacing], settings.veo_max_clip_seconds)

    if target_seconds <= shortest:
        return [shortest], shortest, target_seconds == shortest

    desired_count = max(1, round(target_seconds / preferred))
    max_count = min(settings.max_segments, -(-target_seconds // shortest))

    # Try the exact target first, walking the segment count outward from the
    # one the pacing asked for.
    for delta in range(0, max_count + 1):
        candidates = [desired_count] if delta == 0 else [desired_count + delta, desired_count - delta]
        for count in candidates:
            if not 1 <= count <= max_count:
                continue
            combo = _exact_combination(target_seconds, count, allowed)
            if combo:
                return combo, target_seconds, True

    # No exact fit. Only happens for odd targets, since 4/6/8 are all even.
    # Overshoot by the smallest amount and trim the tail during render.
    for over in range(1, max(allowed) + 1):
        total = target_seconds + over
        base = max(1, min(max_count, round(total / preferred)))
        for delta in range(0, max_count + 1):
            candidates = [base] if delta == 0 else [base + delta, base - delta]
            for count in candidates:
                if not 1 <= count <= max_count:
                    continue
                combo = _exact_combination(total, count, allowed)
                if combo:
                    return combo, total, False

    # Unreachable for any sane target, but keep the function total.
    count = -(-target_seconds // preferred)
    return [preferred] * count, count * preferred, count * preferred == target_seconds


def _beat_at(fraction: float) -> Beat:
    clamped = min(0.9999, max(0.0, fraction))
    for beat, start, end, _label in BEAT_MAP:
        if start <= clamped < end:
            return beat
    return "aftermath"


def _place_durations(durations: list[int]) -> list[tuple[int, Beat]]:
    """Order durations so the rhythm ladder reads correctly.

    Held shots land on context and aftermath; the shortest cuts land on
    acceleration.
    """
    count = len(durations)

    # Provisional beats from slot position, used only to rank the slots.
    slots = []
    for index in range(count):
        beat = _beat_at((index + 0.5) / count)
        slots.append((index, beat, BEAT_LENGTH_WEIGHT[beat]))

    longest_first = sorted(durations, reverse=True)
    by_weight = sorted(slots, key=lambda slot: (-slot[2], slot[0]))

    placed: list[int] = [0] * count
    for rank, (index, _beat, _weight) in enumerate(by_weight):
        placed[index] = longest_first[rank]

    # Now that real durations exist, re-derive the beat from actual screen time.
    total = sum(placed)
    result: list[tuple[int, Beat]] = []
    cursor = 0
    for seconds in placed:
        midpoint = cursor + seconds / 2
        result.append((seconds, _beat_at(0.0 if total == 0 else midpoint / total)))
        cursor += seconds

    return _enforce_essential_beats(result)


def _enforce_essential_beats(placed: list[tuple[int, Beat]]) -> list[tuple[int, Beat]]:
    """Pin the opening and the payoff, which the fraction map cannot guarantee.

    ``BEAT_MAP`` gives `hook` the first 5/90 of runtime — 5.6%. On a 30s film
    that is the first 1.67 seconds, shorter than the shortest clip Veo will
    render, so no segment midpoint can ever land inside it. Every film under
    roughly 72s therefore opened on `context` (exposition) instead of a hook.

    The same sparse sampling drops `impact`: a 4-segment plan lands midpoints at
    roughly 13/40/63/87% of runtime, and the impact window is 66.7-83.3%, so it
    falls straight through the gap. That plan has no climax, and a film written
    to it cannot pay off — the ad just stops.

    The map still shapes the middle. Only the first beat, the last beat, and the
    presence of a payoff are forced.
    """
    count = len(placed)
    if count < 3:
        # One or two shots have no room for an arc; leave the map's answer alone.
        return placed

    beats: list[Beat] = [beat for _seconds, beat in placed]
    beats[0] = "hook"
    beats[-1] = "aftermath"

    if "impact" not in beats:
        # Give it the last slot before the closing aftermath, never slot 0.
        index = count - 2
        while index > 0 and beats[index] == "aftermath":
            index -= 1
        if index > 0:
            beats[index] = "impact"

    return [(seconds, beat) for (seconds, _old), beat in zip(placed, beats)]


def _summarise(durations: list[int]) -> str:
    counts: dict[int, int] = {}
    for value in durations:
        counts[value] = counts.get(value, 0) + 1
    return " + ".join(f"{count}x{seconds}s" for seconds, count in sorted(counts.items(), reverse=True))


def plan_duration(target_seconds: float, pacing: Pacing = "standard") -> DurationPlan:
    if target_seconds is None or target_seconds <= 0:
        raise BadRequest("Video duration must be a positive number of seconds.")
    if target_seconds > settings.max_video_seconds:
        raise BadRequest(
            f"Video duration is capped at {settings.max_video_seconds}s. "
            "Raise MAX_VIDEO_SECONDS to go longer."
        )

    target = round(target_seconds)
    durations, planned, exact = _fit_durations(target, pacing)

    if len(durations) > settings.max_segments:
        raise BadRequest(
            f"That brief needs {len(durations)} Veo generations, above the "
            f"MAX_SEGMENTS limit of {settings.max_segments}."
        )

    placed = _place_durations(durations)

    used: dict[Beat, int] = {}
    segments: list[PlannedSegment] = []
    starts_at = 0
    for index, (seconds, beat) in enumerate(placed):
        options = BEAT_SHOT_FUNCTIONS[beat]
        seen = used.get(beat, 0)
        used[beat] = seen + 1
        segments.append(
            PlannedSegment(
                index=index,
                seconds=seconds,
                startsAt=starts_at,
                beat=beat,
                shotFunction=options[seen % len(options)],
            )
        )
        starts_at += seconds

    notes = [
        f"{len(segments)} Veo generations of {_summarise(durations)} accumulate to {planned}s."
    ]
    if not exact:
        notes.append(
            f"Veo only renders 4/6/8s clips, so the plan overshoots by {planned - target}s "
            f"and the renderer trims the tail to land on exactly {target}s."
        )
    notes.append(
        f'Pacing "{pacing}" biases toward {PACING_PREFERRED_SECONDS[pacing]}s shots; '
        "the shortest segments are placed on the acceleration beat."
    )

    return DurationPlan(
        targetSeconds=target,
        plannedSeconds=planned,
        trimSeconds=max(0, planned - target),
        exact=exact,
        strategy="stitch",
        segments=segments,
        notes=notes,
    )


def default_pacing(style: str) -> Pacing:
    """Pacing implied by a style preset, when the client does not send one."""
    if style == "ugc":
        return "fast"
    if style in ("animated", "product"):
        return "standard"
    return "slow"
