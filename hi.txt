function getAlignmentTone(status: string | undefined) {
  if (status === 'aligned') return 'aligned'
  if (status === 'warnings') return 'warnings'
  return 'pending'
}

function formatMetricKey(key: string) {
  const cleaned = key.replace(/^target_/, '').replace(/_/g, ' ')
  if (cleaned.toLowerCase() === 'roas') return 'ROAS'
  if (cleaned.toLowerCase() === 'cpa') return 'CPA'
  if (cleaned.toLowerCase() === 'ctr') return 'CTR'
  if (cleaned.toLowerCase() === 'vtr') return 'VTR'
  if (cleaned.toLowerCase() === 'reach') return 'Reach'
  return cleaned.replace(/\b\w/g, (character) => character.toUpperCase())
}

function formatMetricValue(value: unknown): string {
  if (typeof value === 'number') {
    return Number.isInteger(value) ? String(value) : value.toFixed(2)
  }

  if (typeof value === 'boolean') {
    return value ? 'Yes' : 'No'
  }

  if (Array.isArray(value)) {
    return value.map((item) => formatMetricValue(item)).join(', ')
  }

  if (value && typeof value === 'object') {
    return Object.entries(value)
      .map(([entryKey, entryValue]) => `${formatMetricKey(entryKey)}: ${formatMetricValue(entryValue)}`)
      .join(' · ')
  }

  return String(value ?? '—')
}

function formatChannelWeight(weight: unknown) {
  if (typeof weight !== 'number') return 'Allocation set'
  return `${weight}%`
}

function getFinanceChannelName(channel: unknown, index: number): string {
  if (typeof channel === 'string') return channel

  if (channel && typeof channel === 'object') {
    const item = channel as Record<string, unknown>
    return String(item.channel ?? item.name ?? `Channel ${index + 1}`)
  }

  return `Channel ${index + 1}`
}

function formatFinanceChannel(channel: unknown, index: number): string {
  if (typeof channel === 'string') return channel

  if (channel && typeof channel === 'object') {
    const item = channel as Record<string, unknown>
    const channelName = String(item.channel ?? item.name ?? `Channel ${index + 1}`)
    const allocation =
      typeof item.total_allocation === 'number'
        ? ` · ${formatCurrency(item.total_allocation)}`
        : ''

    return `${channelName}${allocation}`
  }

  return `Channel ${index + 1}`
}
export function CampaignDetail() {
  const { campaignId } = useParams()
  const [resolvedImageUrls, setResolvedImageUrls] = useState<Record<string, string | null>>({})
  const [selectedOutput, setSelectedOutput] = useState<CreativeOutput | null>(null)

  const localCampaign = useMemo(() => {
    if (!campaignId) return null
    const storedCampaigns = loadCampaigns()
    return storedCampaigns.find((item) => item.id === campaignId) ?? null
  }, [campaignId])

  // A campaign generated on another machine is not in this browser's storage,
  // so fall back to the server before deciding it does not exist.
  const [remoteCampaign, setRemoteCampaign] = useState<CampaignRecord | null>(null)
  const [isFetching, setIsFetching] = useState(!localCampaign)

  useEffect(() => {
    if (!campaignId || localCampaign) {
      setIsFetching(false)
      return
    }

    const controller = new AbortController()
    setIsFetching(true)

    fetchCampaign(campaignId, controller.signal)
      .then((found) => setRemoteCampaign(found))
      .catch((error: unknown) => {
        if (error instanceof Error && error.name === 'AbortError') return
        setRemoteCampaign(null)
      })
      .finally(() => setIsFetching(false))

    return () => controller.abort()
  }, [campaignId, localCampaign])

  const campaign = localCampaign ?? remoteCampaign

  useEffect(() => {
    const outputs = campaign?.summary.creative?.outputs ?? []
    const refs = outputs
      .flatMap((output) => [output.image_url, ...(output.image_urls ?? [])])
      .filter((value): value is string => typeof value === 'string' && isIndexedImageRef(value))

    if (refs.length === 0) return

    let cancelled = false
    void Promise.all(refs.map(async (ref) => ({ ref, url: await resolveIndexedImageRef(ref) }))).then((entries) => {
      if (cancelled) return
      setResolvedImageUrls((current) => {
        const next = { ...current }
        for (const entry of entries) {
          next[entry.ref] = entry.url
        }
        return next
      })
    })

    return () => {
      cancelled = true
    }
  }, [campaign?.id, campaign?.summary.creative?.outputs])

  if (!campaign && isFetching) {
    return (
      <section className="empty-route">
        <h2>Loading campaign…</h2>
        <p>Fetching this plan from the server.</p>
      </section>
    )
  }

  if (!campaign) {
    return (
      <section className="empty-route">
        <h2>Campaign not found</h2>
        <p>It is not on this device and the server has no record of it.</p>
        <Link className="button button--secondary" to="/campaigns">
          Back to campaigns
        </Link>
      </section>
    )
  }

  const channels = [...(campaign.summary.channels ?? [])].sort((left, right) => {
    const leftWeight = typeof left.weight === 'number' ? left.weight : -1
    const rightWeight = typeof right.weight === 'number' ? right.weight : -1
    return rightWeight - leftWeight
  })
  const channelWeightTotal = channels.reduce((sum, channel) => sum + (typeof channel.weight === 'number' ? channel.weight : 0), 0)
  const channelTargets = Object.entries(
    campaign.summary.channel_targets && typeof campaign.summary.channel_targets === 'object'
      ? campaign.summary.channel_targets
      : {}
  )
  const finance = campaign.summary.finance
  const financeChannels = Array.isArray(finance?.channels) ? finance.channels : []
  const reallocationRules = Array.isArray(finance?.reallocation_rules) ? finance.reallocation_rules : []
  const financeValidation = finance?.validation
  const financeValidationTone = financeValidation?.is_valid === false ? 'warnings' : financeValidation?.is_valid ? 'aligned' : 'pending'
  const creativeOutputs: CreativeOutput[] = (campaign.summary.creative?.outputs ?? []).flatMap((output) => {
    const urls = output.image_urls?.length
      ? output.image_urls
      : output.image_url
        ? [output.image_url]
        : []
    if (!urls.length) return [output]
    return urls.map((url, index) => ({
      ...output,
      image_url: url,
      image_urls: [url],
      // Each image comes from its own prompt variant.
      prompt: output.used_prompts?.[index] ?? output.prompt,
      variant_index: index + 1,
    }))
  })

  const getRenderableImageUrl = (value: string | null | undefined) => {
    if (!value) return null
    if (!isIndexedImageRef(value)) return value
    return resolvedImageUrls[value] ?? null
  }
  const selectedImageUrl = getRenderableImageUrl(selectedOutput?.image_url ?? null)

  useEffect(() => {
    if (!selectedOutput) return

    const handleKeyDown = (event: KeyboardEvent) => {
      if (event.key === 'Escape') {
        setSelectedOutput(null)
      }
    }

    window.addEventListener('keydown', handleKeyDown)
    return () => window.removeEventListener('keydown', handleKeyDown)
  }, [selectedOutput])

  return (
    <section className="page-stack">
      <header className="page-header">
        <div>
          <span className="eyebrow">Campaign detail</span>
          <h1>{campaign.summary.product ?? campaign.prompt}</h1>
          <p>{campaign.summary.objective ?? campaign.prompt}</p>
        </div>
        <Link className="button button--secondary" to="/campaigns">
          Back to campaigns
        </Link>
      </header>

      <div className="detail-hero">
        <div className="detail-hero__main">
          <div className={`campaign-card__badge campaign-card__badge--${getAlignmentTone(campaign.summary.alignment_status)}`}>
            {campaign.summary.alignment_status ?? 'pending'}
          </div>
          <span>{campaign.summary.primary_kpi ?? 'KPI pending'}</span>
          <strong>{formatCurrency(campaign.summary.total_budget)}</strong>
          <p>{campaign.summary.objective ?? campaign.prompt}</p>
          <div className="detail-hero__chips">
            <span>{campaign.summary.archetype ?? 'unknown'}</span>
            <span>Revisions {campaign.summary.revision_count ?? 0}</span>
            <span>Updated {formatDate(campaign.updatedAt)}</span>
          </div>
        </div>

        <div className="detail-hero__meta">
          <div>
            <span>Primary KPI</span>
            <strong>{campaign.summary.primary_kpi ?? 'pending'}</strong>
          </div>
          <div>
            <span>Status</span>
            <strong>{campaign.summary.alignment_status ?? 'pending'}</strong>
          </div>
          <div>
            <span>Archetype</span>
            <strong>{campaign.summary.archetype ?? 'unknown'}</strong>
          </div>
          <div>
            <span>Budget</span>
            <strong>{formatCurrency(campaign.summary.total_budget)}</strong>
          </div>
          <div>
            <span>Updated</span>
            <strong>{formatDate(campaign.updatedAt)}</strong>
          </div>
        </div>
      </div>

      <div className="detail-grid">
        <section className="detail-card">
          <span className="eyebrow">Channel mix</span>
          <div className="detail-list channel-mix">
            {channels.length === 0 ? (
              <p className="muted">No channel allocation was captured.</p>
            ) : (
              channels.map((channel, index) => {
                const weight = typeof channel.weight === 'number' ? channel.weight : 0
                const channelName = String(channel.name ?? `Channel ${index + 1}`)

                return (
                  <div key={`${campaign.id}-channel-${index}`} className="detail-row channel-mix__row">
                    <div className="channel-mix__header">
                      <strong>{channelName}</strong>
                      <span>{formatChannelWeight(channel.weight)}</span>
                    </div>
                    <div className="channel-mix__bar-track" aria-hidden="true">
                      <div className="channel-mix__bar" style={{ width: `${Math.max(0, Math.min(weight, 100))}%` }} />
                    </div>
                    <div className="channel-mix__footer">
                      <span>{weight}% of mix</span>
                      <span>{channelWeightTotal > 0 ? `${Math.round((weight / channelWeightTotal) * 100)}% of allocated budget` : 'No allocation total'}</span>
                    </div>
                  </div>
                )
              })
            )}
          </div>
        </section>

        <section className="detail-card">
          <span className="eyebrow">Finance</span>
          <div className="detail-list finance-stack">
            <div className="detail-row finance-stack__summary">
              <strong>Reserve buffer</strong>
              <span>{formatCurrency(finance?.reserve_buffer)}</span>
            </div>

            <div className="detail-row">
              <strong>Coverage channels</strong>
              <div className="finance-chips">
                {financeChannels.length === 0 ? (
                  <span className="muted">No coverage channels were captured.</span>
                ) : (
                  financeChannels.map((channel, index) => (
                    <span
                      key={`${campaign.id}-finance-channel-${getFinanceChannelName(channel, index)}-${index}`}
                      className="finance-chip"
                    >
                      {formatFinanceChannel(channel, index)}
                    </span>
                  ))
                )}
              </div>
            </div>

            <div className="detail-row">
              <strong>Reallocation rules</strong>
              {reallocationRules.length === 0 ? (
                <span className="muted">No reallocation rules were captured.</span>
              ) : (
                <ul className="finance-list">
                  {reallocationRules.map((rule, index) => (
                    <li key={`${campaign.id}-reallocation-${index}`}>{formatMetricValue(rule)}</li>
                  ))}
                </ul>
              )}
            </div>

            <div className={`detail-row validation-pill validation-pill--${financeValidationTone}`}>
              <strong>Validation</strong>
              <span>{financeValidation?.is_valid === false ? 'Needs review' : financeValidation?.is_valid ? 'Validated' : 'Pending validation'}</span>
              {financeValidation ? <small>{formatMetricValue(financeValidation)}</small> : null}
            </div>
          </div>
        </section>

        <section className="detail-card detail-card--wide">
          <span className="eyebrow">Channel KPIs</span>
          <div className="detail-list kpi-stack kpi-stack--grid">
            {channelTargets.length === 0 ? (
              <p className="muted">No per-channel KPI targets were captured.</p>
            ) : (
              channelTargets.map(([channel, target]) => (
                <div key={channel} className="detail-row kpi-card">
                  <div className="kpi-card__header">
                    <strong>{channel}</strong>
                    <span>{Object.keys(target).length} target{Object.keys(target).length === 1 ? '' : 's'}</span>
                  </div>
                  <dl className="kpi-table">
                    {Object.entries(target).map(([metric, value]) => (
                      <div key={`${channel}-${metric}`} className="kpi-table__row">
                        <dt>{formatMetricKey(metric)}</dt>
                        <dd>{formatMetricValue(value)}</dd>
                      </div>
                    ))}
                  </dl>
                </div>
              ))
            )}
          </div>
        </section>

        <section className="detail-card detail-card--wide">
          <span className="eyebrow">Creatives</span>
          <div className="detail-list creative-grid">
            {creativeOutputs.length === 0 ? (
              <p className="muted">No creative assets were captured for this campaign.</p>
            ) : (
              creativeOutputs.map((output, index) => {
                const imageUrl = getRenderableImageUrl(output.image_url ?? null)
                return (
                  <article
                    key={`${campaign.id}-${output.channel}-${output.variant_index ?? index}`}
                    className="creative-card detail-row"
                  >
                    {imageUrl ? (
                      <button type="button" className="creative-card__preview" onClick={() => setSelectedOutput(output)}>
                        <img
                          src={imageUrl}
                          alt={`${output.channel} ${output.format ?? ''} creative`}
                          className="creative-card__image"
                        />
                      </button>
                    ) : (
                      <div className="creative-card__media creative-card__placeholder">
                        <p className="muted" style={{ margin: 0 }}>Preview unavailable</p>
                      </div>
                    )}
                    <div className="creative-card__meta">
                      <div>
                        <div className="creative-card__title">
                          {output.channel}
                        </div>

                        {output.format && (
                          <div className="creative-card__format">
                            {output.format}
                          </div>
                        )}

                        {output.variant_index && (
                          <div className="creative-card__format">
                            Image {output.variant_index}
                          </div>
                        )}
                      </div>
                      <span>{output.status}</span>
                    </div>
                    {output.error_message ? <p className="creative-card__error">{output.error_message}</p> : null}
                  </article>
                )
              })
            )}
          </div>
        </section>
      </div>

      {selectedOutput && selectedImageUrl && typeof document !== 'undefined' && createPortal(
        <div className="creative-modal" role="dialog" aria-modal="true" aria-label="Creative preview" onClick={() => setSelectedOutput(null)}>
          <div className="creative-modal__surface" onClick={(event) => event.stopPropagation()}>
            <button
              type="button"
              className="creative-modal__close"
              onClick={() => setSelectedOutput(null)}
              aria-label="Close creative preview"
            >
              <X size={16} />
            </button>
            <div className="creative-modal__media">
              <img src={selectedImageUrl} alt={`${selectedOutput.channel} ${selectedOutput.format ?? ''} creative`} />
            </div>
            <div className="creative-modal__body">
              <div className="creative-modal__header">
                <div>
                  <span className="eyebrow">Creative preview</span>
                  <h3>
                    {selectedOutput.channel}
                    {selectedOutput.format ? ` · ${selectedOutput.format}` : ''}
                    {selectedOutput.variant_index ? ` · Image ${selectedOutput.variant_index}` : ''}
                  </h3>
                </div>
                <span>{selectedOutput.status}</span>
              </div>
              {selectedOutput.headline && (
                <div className="creative-modal__prompt">
                  <strong>Headline</strong>
                  <p>{selectedOutput.headline}</p>
                </div>
              )}

              {selectedOutput.concept && (
                <div className="creative-modal__prompt">
                  <strong>Concept</strong>
                  <p>{selectedOutput.concept}</p>
                </div>
              )}

              <div className="creative-modal__prompt">
                <strong>Prompt</strong>
                <p>{selectedOutput.prompt ?? 'Prompt not available for this asset.'}</p>
              </div>
            </div>
          </div>
        </div>,
        document.body,
      )}
    </section>
  )
}

function StatusBadge({ status }: { status: CreativeOutput['status'] }) {
  const styles: Record<string, { bg: string; color: string; label: string }> = {
    success: { bg: '#dcfce7', color: '#16a34a', label: 'Generated' },
    error:   { bg: '#fee2e2', color: '#dc2626', label: 'Failed' },
    pending: { bg: '#f3f4f6', color: '#6b7280', label: 'Pending' },
  }
  const s = styles[status] ?? styles.pending
  return (
    <span style={{ fontSize: '0.65rem', fontWeight: 600, padding: '2px 6px', borderRadius: '4px', background: s.bg, color: s.color, textTransform: 'uppercase', letterSpacing: '0.05em' }}>
      {s.label}
    </span>
  )
}

function CreativeSection({ planData, activeAgent }: { planData: PlanData; activeAgent: string | null }) {
  const creative = planData.creative
  const isGenerating = activeAgent === 'creative'
  const filled = !!(creative?.outputs?.length || creative?.summary)
  const [selectedOutput, setSelectedOutput] = useState<CreativeOutput | null>(null)
  const [resolvedImageUrls, setResolvedImageUrls] = useState<Record<string, string | null>>({})

  useEffect(() => {
    const outputs = creative?.outputs ?? []
    const refs = outputs
    .flatMap((output) => [
      output.image_url,
      ...(output.image_urls ?? []),
    ])
    .filter(
      (value): value is string =>
        typeof value === 'string' && isIndexedImageRef(value),
    )

    if (refs.length === 0) return

    let cancelled = false
    void Promise.all(refs.map(async (ref) => ({ ref, url: await resolveIndexedImageRef(ref) }))).then((entries) => {
      if (cancelled) return
      setResolvedImageUrls((current) => {
        const next = { ...current }
        for (const entry of entries) {
          next[entry.ref] = entry.url
        }
        return next
      })
    })

    return () => {
      cancelled = true
    }
  }, [creative?.outputs])

  const getRenderableImageUrl = (value: string | null | undefined) => {
    if (!value) return null
    if (!isIndexedImageRef(value)) return value
    return resolvedImageUrls[value] ?? null
  }
  const displayOutputs: CreativeOutput[] = (
    creative?.outputs ?? []
  ).flatMap((output) => {
    const imageUrls = output.image_urls?.length
      ? output.image_urls
      : output.image_url
        ? [output.image_url]
        : []

    if (!imageUrls.length) return [output]

    return imageUrls.map((imageUrl, index) => ({
      ...output,
      image_url: imageUrl,
      image_urls: [imageUrl],
      // Each image comes from its own prompt variant, so attribute the real one
      // rather than repeating the channel's first prompt on every card.
      prompt: output.used_prompts?.[index] ?? output.prompt,
      variant_index: index + 1,
    }))
  })

  const selectedImageUrl = getRenderableImageUrl(selectedOutput?.image_url)
  const visibleOutputs = displayOutputs.filter(
    (output) => output.status !== 'pending',
  )
  const isLoadingGrid = isGenerating && (!creative?.outputs?.length || creative.outputs.some((output) => output.status === 'pending'))
  const skeletonCount = Math.max(3, creative?.planning?.length ?? 0)
  const creativeSummary = creative?.summary
  const creativePlanning = creative?.planning ?? []

  useEffect(() => {
    if (!selectedOutput) return

    const handleKeyDown = (event: KeyboardEvent) => {
      if (event.key === 'Escape') {
        setSelectedOutput(null)
      }
    }

    window.addEventListener('keydown', handleKeyDown)
    return () => window.removeEventListener('keydown', handleKeyDown)
  }, [selectedOutput])

  return (
    <div className="panel-section">
      <SectionHeader title="Creative Assets" filled={filled} />
      {!filled && !isGenerating && !creative?.planning && (
        <p className="muted" style={{ fontSize: '0.8rem' }}>Waiting for Creative agent...</p>
      )}
      {!filled && isGenerating && creativePlanning.length > 0 && (
        <div style={{ display: 'flex', flexDirection: 'column', gap: '6px' }}>
          <p style={{ fontSize: '0.78rem', color: '#6b7280', margin: 0 }}>Generating banners for:</p>
          <div style={{ display: 'flex', flexWrap: 'wrap', gap: '6px' }}>
            {creativePlanning.map((ch) => (
              <span key={ch} style={{ fontSize: '0.72rem', fontWeight: 600, padding: '2px 8px', borderRadius: '4px', background: '#eff6ff', color: '#1d4ed8' }}>{ch}</span>
            ))}
          </div>
        </div>
      )}
      {(filled || isLoadingGrid) && (
        <div style={{ display: 'grid', gap: '10px' }}>
          {creativeSummary && (
            <div style={{ fontSize: '0.72rem', color: '#6b7280', padding: '5px 10px', background: '#fafafa', borderRadius: '6px', border: '1px solid #f3f4f6' }}>
              {creativeSummary.total_images ??
                visibleOutputs.filter(
                  (output) => output.status === 'success',
                ).length}{' '}
              images generated

              {(creativeSummary.failed_channels ??
                creativeSummary.failed ??
                0) > 0 && (
                <span style={{ color: '#dc2626' }}>
                  {' '}·{' '}
                  {creativeSummary.failed_channels ??
                    creativeSummary.failed}{' '}
                  channel(s) failed
                </span>
              )}
            </div>
          )}
          <div className="creative-grid">
            {visibleOutputs.map((out, i) => (
              <div
                key={`${out.channel}-${out.variant_index ?? i}`}
                className="creative-card"
              >
                {getRenderableImageUrl(out.image_url) ? (
                  <button
                    type="button"
                    onClick={() => setSelectedOutput(out)}
                    className="creative-card__preview"
                  >
                    <img
                      src={getRenderableImageUrl(out.image_url) ?? undefined}
                      alt={`${out.channel} ${out.format ?? ''} banner`}
                      className="creative-card__image"
                      onError={(e) => { (e.currentTarget as HTMLImageElement).style.display = 'none' }}
                    />
                  </button>
                ) : out.status === 'success' ? (
                  <div className="creative-card__media creative-card__placeholder creative-card__placeholder--success">
                    <BadgeCheck size={14} />
                    Asset ready — no preview URL
                  </div>
                ) : (
                  <div className="creative-card__media creative-card__placeholder">
                    <ImageOff size={20} />
                  </div>
                )}
                <div className="creative-card__meta">
                 <div>
                    <div className="creative-card__title">
                      {out.channel}
                    </div>

                    {out.format && (
                      <div className="creative-card__format">
                        {out.format}
                      </div>
                    )}

                    {out.variant_index && (
                      <div className="creative-card__format">
                        Image {out.variant_index}
                      </div>
                    )}
                  </div>
                  <div className="creative-card__actions">
                    <StatusBadge status={out.status} />
                    {getRenderableImageUrl(out.image_url) && (
                      <button
                        type="button"
                        onClick={() => setSelectedOutput(out)}
                        className="creative-card__view"
                        aria-label={`Open ${out.channel} creative preview`}
                      >
                        <span>View</span>
                      </button>
                    )}
                  </div>
                </div>
                {out.status === 'error' && out.error_message && (
                  <p className="creative-card__error">{out.error_message}</p>
                )}
              </div>
            ))}
            {isLoadingGrid && Array.from({ length: skeletonCount }).map((_, index) => (
              <CreativeSkeletonCard key={`creative-skeleton-${index}`} />
            ))}
          </div>
        </div>
      )}

      {selectedOutput && selectedImageUrl && typeof document !== 'undefined' && createPortal(
        <div className="creative-modal" role="dialog" aria-modal="true" aria-label="Creative preview" onClick={() => setSelectedOutput(null)}>
          <div className="creative-modal__surface" onClick={(event) => event.stopPropagation()}>
            <button
              type="button"
              className="creative-modal__close"
              onClick={() => setSelectedOutput(null)}
              aria-label="Close creative preview"
            >
              <X size={16} />
            </button>

            <div className="creative-modal__media">
              <img
                src={selectedImageUrl}
                alt={`${selectedOutput.channel} ${selectedOutput.format ?? ''} banner`}
              />
            </div>

            <div className="creative-modal__body">
              <div className="creative-modal__header">
                <div>
                  <span className="eyebrow">Creative preview</span>
                  <h3>
                    {selectedOutput.channel}
                    {selectedOutput.format
                      ? ` · ${selectedOutput.format}`
                      : ''}
                    {selectedOutput.variant_index
                      ? ` · Image ${selectedOutput.variant_index}`
                      : ''}
                  </h3>
                </div>
                <StatusBadge status={selectedOutput.status} />
              </div>

              {selectedOutput.headline && (
                <div className="creative-modal__prompt">
                  <strong>Headline</strong>
                  <p>{selectedOutput.headline}</p>
                </div>
              )}

              {selectedOutput.concept && (
                <div className="creative-modal__prompt">
                  <strong>Concept</strong>
                  <p>{selectedOutput.concept}</p>
                </div>
              )}

              <div className="creative-modal__prompt">
                <strong>Prompt</strong>
                <p>{selectedOutput.prompt ?? 'Prompt not available for this asset.'}</p>
              </div>
            </div>
          </div>
        </div>,
        document.body,
      )}
    </div>
  )
}

import { useState } from 'react'
import { ArrowRight } from 'lucide-react'

type Props = {
  onSubmit: (value: string) => void
  busy?: boolean
}

// Models all five fields the intake agent needs (product, offer, objective, budget,
// visual direction) so the user's first message answers most of them up front instead
// of being drawn out one question at a time. The visual direction deliberately shows
// the brand constraint by example -- no airport or aircraft imagery.
const EXAMPLE_BRIEF =
  'Domestic flight sale for IndiGo, fares from ₹1,499, ₹8 lakh budget, ' +
  'aiming to maximise bookings over a 48-hour window. ' +
  'Visuals: the moment a trip stops being hypothetical — candid and uncluttered, no posed smiles.'

export function CenteredComposer({ onSubmit, busy }: Props) {
  const [value, setValue] = useState(EXAMPLE_BRIEF)

  return (
    <section className="hero-panel pt-4">
      <div className="hero-panel__copy">
        {/* <span className="eyebrow">Autonomous marketing platform</span> */}
        <h1>Start with one prompt. The system turns it into an operating plan.</h1>
        <p>
          Enter a campaign brief and the platform will stream strategy, metrics, budget allocation,
          and banner creative in real time.
        </p>
      </div>

      <form
        className="composer composer--centered"
        onSubmit={(event) => {
          event.preventDefault()
          onSubmit(value.trim())
        }}
      >
        <textarea
          value={value}
          onChange={(event) => setValue(event.target.value)}
          rows={4}
          placeholder="What are you selling, what's the offer, what does success look like, what's the budget, and how should the banners feel?"
          onKeyDown={(event) => {
            // Enter sends, Shift+Enter makes a new line.
            // isComposing guards IME input, where Enter confirms a candidate
            // and must not submit the message.
            if (event.key !== 'Enter' || event.shiftKey) return
            if (event.nativeEvent.isComposing) return
            event.preventDefault()
            if (busy || !value.trim()) return
            event.currentTarget.form?.requestSubmit()
          }}
        />
        <div className="composer__actions">
          <span>Tip: include the product, the offer, your objective, the budget, and how the banners should feel.</span>
          <button className="button button--primary" type="submit" disabled={busy || !value.trim()}>
            <span>Start campaign</span>
            <ArrowRight size={16} />
          </button>
        </div>
      </form>
    </section>
  )
}
