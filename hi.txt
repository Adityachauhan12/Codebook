"""Show how a brief gets split, without generating a single frame.

    python test_brief.py                 # the Japan brief, as a regression test
    python test_brief.py my_brief.txt    # your own

One cheap Azure call. Proves the fix that matters most — that on-screen copy is
pulled out of the brief and never reaches a prompt — before any Veo spend.
"""

from __future__ import annotations

import asyncio
import sys
from pathlib import Path

sys.path.insert(0, str(Path(__file__).parent))

from app.domain.models import ProjectBrief  # noqa: E402
from app.services.brief import analyse_brief  # noqa: E402

# The brief that produced "A new horizon awasts." — unchanged, on purpose.
JAPAN = """\
IndiGo x Japan - A New Journey Begins
Duration: 45-60 seconds
Style: Ultra-cinematic, luxury travel commercial, emotionally uplifting
Aspect Ratio: 16:9
Resolution: 4K HDR
Lens: Anamorphic, shallow depth of field, natural cinematic motion
Colour Grade: Cool blues with warm sunrise highlights, premium airline aesthetic

Create a breathtaking cinematic airline launch film announcing IndiGo's new
destination: Japan. Open with a sunrise above the clouds. An IndiGo aircraft
emerges through golden morning light, soaring over a sea of clouds.

Transition to aerial views of Japan: Mount Fuji at dawn, Tokyo's skyline,
Shibuya Crossing at blue hour, Kyoto temples surrounded by cherry blossoms,
bamboo forests, bullet trains, lakes reflecting autumn colours.

Show elegant travel moments: a traveller gazing out of the aircraft window as
Japan appears beneath the clouds; cabin crew welcoming passengers; close-ups of
smiling travellers and children looking excited.

Feature the aircraft landing at a modern Japanese airport during golden hour.

Music
Epic orchestral score blended with modern ambient textures and subtle Japanese
instruments such as koto and shakuhachi. Build toward an emotional crescendo.

On-Screen Text
0-10 sec: "A new horizon awaits."
10-30 sec: "From India to the heart of Japan."
30-50 sec: "Discover timeless traditions. Experience tomorrow."

Final Frame
INDIGO NOW FLIES TO JAPAN
Your Next Journey Begins
Logo reveal with elegant fade over clouds.

Negative Prompt
Low quality, cartoon, CGI look, oversaturated colours, distorted aircraft,
incorrect airline branding, extra aircraft, text artifacts, watermark, blurry
faces, warped architecture, incorrect Japanese landmarks, shaky camera.
"""


def rule(title: str) -> None:
    print(f"\n{'=' * 68}\n{title}\n{'=' * 68}")


async def main() -> None:
    raw = Path(sys.argv[1]).read_text(encoding="utf-8") if len(sys.argv) > 1 else JAPAN
    seconds = 50

    from app.config import settings

    smart = bool(settings.azure_openai_key and settings.azure_openai_base_url)
    print(f"Raw brief: {len(raw):,} characters")
    print(f"Path     : {'Azure model' if smart else 'DETERMINISTIC FALLBACK (no Azure configured)'}")
    if not smart:
        print("           The fallback cuts at the copy heading. It keeps copy away from")
        print("           Veo — the thing that matters — but cannot recover timings, so")
        print("           expect 0 supers. Configure Azure for the full split.")

    analysis = await analyse_brief(raw, seconds=seconds)

    rule("INTENT — the ONLY part the scene writer sees")
    print(analysis.intent)

    rule(f"SUPERS — {len(analysis.supers)} line(s), composited in post, never prompted")
    for item in analysis.supers:
        print(f"  {item.start:5.1f}-{item.end:5.1f}s  [{item.position}]  {item.text!r}")
    if not analysis.supers:
        print("  (none)")

    rule("END CARD")
    card = analysis.end_card
    if card.enabled:
        print(f"  headline : {card.headline!r}")
        print(f"  subline  : {card.subline!r}")
        print(f"  mark     : {card.mark}   last {card.seconds}s")
    else:
        print("  disabled")

    rule("ROUTED ELSEWHERE")
    print(f"  music -> composer : {(analysis.music or '(none)')[:150]}")
    print(f"  avoid -> negative : {len(analysis.avoid)} item(s)")
    for item in analysis.avoid[:6]:
        print(f"      - {item}")

    rule("REFUSED OUT LOUD instead of silently ignored")
    for item in analysis.unsupported:
        print(f"  ! {item}")
    if not analysis.unsupported:
        print("  (nothing unsupported)")

    # --- The actual regression assertions ---------------------------------
    rule("VERDICT")
    brief = ProjectBrief(brief=raw, seconds=seconds, intent=analysis.intent)
    seen = brief.creative().lower()

    # The load-bearing check is the first one. The rest describe quality of the
    # split, not safety of it.
    critical = [
        ("no super copy leaked into intent",
         not any(item.text.lower()[:24] in seen for item in analysis.supers if len(item.text) > 8)),
        ("the literal end-card copy is not in intent",
         "now flies to japan" not in seen and "new horizon awaits" not in seen),
        ("intent is shorter than the raw brief", len(analysis.intent) < len(raw)),
    ]
    quality = [
        ("copy was extracted with timings", bool(analysis.supers)),
        ("end card recognised", card.enabled),
        ("music routed to the composer", bool(analysis.music)),
        ("4K/HDR refused out loud",
         any("4K" in u or "HDR" in u for u in analysis.unsupported)),
    ]

    for label, ok in critical:
        print(f"  [{'PASS' if ok else 'FAIL'}] {label}   (critical)")
    for label, ok in quality:
        print(f"  [{'PASS' if ok else '  - '}] {label}")

    if all(ok for _, ok in critical):
        print("\nSAFE: nothing a viewer reads can reach Veo.")
        if not all(ok for _, ok in quality) and not smart:
            print("Quality items are blank because the fallback ran. Expected without Azure.")
    else:
        print("\nUNSAFE: copy survived into the intent. Read the INTENT block above.")


asyncio.run(main())
