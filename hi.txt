import { create } from 'zustand'
import { persist } from 'zustand/middleware'
import type { ScheduleFrequency, Priority } from '@/features/eventScheduler/types'
import type { SurveyQuestion } from '@/features/eventScheduler/types/survey'

export interface MessageDraft {
  id: string
  type: 'message'
  message: string
  documentName?: string
  priority: Priority
  frequency: ScheduleFrequency
  fromDate: string
  toDate: string
  customDays: string[]
  topLevel: 'cabin_crew' | 'backend_ops' | null
  ccRoles: string[]
  ccBases: string[]
  ccIndividuals: string[]
  boTeams: string[]
  boRoles: string[]
  boIndividuals: string[]
  updatedAt: string
}

export interface SurveyDraft {
  id: string
  type: 'survey'
  surveyTitle: string
  surveyDescription: string
  questions: SurveyQuestion[]
  allowDocUpload: boolean
  priority: Priority
  frequency: ScheduleFrequency
  fromDate: string
  toDate: string
  topLevel: 'cabin_crew' | 'backend_ops' | null
  ccRoles: string[]
  ccBases: string[]
  ccIndividuals: string[]
  boTeams: string[]
  boRoles: string[]
  boIndividuals: string[]
  updatedAt: string
}

export type Draft = MessageDraft | SurveyDraft

interface DraftStore {
  drafts: Draft[]
  saveDraft: (draft: Draft) => void
  removeDraft: (id: string) => void
  getDraft: (id: string) => Draft | undefined
}

export const useDraftStore = create<DraftStore>()(
  persist(
    (set, get) => ({
      drafts: [],
      saveDraft: (draft) =>
        set((state) => {
          const existing = state.drafts.findIndex((d) => d.id === draft.id)
          if (existing >= 0) {
            const updated = [...state.drafts]
            updated[existing] = draft
            return { drafts: updated }
          }
          return { drafts: [draft, ...state.drafts] }
        }),
      removeDraft: (id) =>
        set((state) => ({ drafts: state.drafts.filter((d) => d.id !== id) })),
      getDraft: (id) => get().drafts.find((d) => d.id === id),
    }),
    { name: 'event-scheduler-drafts' },
  ),
)
