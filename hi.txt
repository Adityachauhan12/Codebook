"""Brand guardrails.

Brand is a separate axis from craft. ``app/skills`` teaches the model how to
direct a film; this package tells it whose film it is, and refuses the result
when it is off-brand.

Everything routes through ``BRAND`` in the environment, so the pipeline is not
hardwired to one client — set ``BRAND=none`` and every hook below turns into a
no-op, leaving the craft layer untouched.
"""

from __future__ import annotations

from ..config import get_settings
from ..domain.models import AuditIssue, StoryOption
from .checks import brand_issues as _indigo_issues
from .policy import (
    INDIGO_BRAND_POLICY,
    INDIGO_CONTINUITY_REQUIREMENTS,
    INDIGO_NEGATIVE_PROMPT,
    INDIGO_VIDEO_BRAND_POLICY,
    INDIGO_VIDEO_SAFETY_POLICY,
)

__all__ = [
    "active_brand",
    "brand_name",
    "brand_system_block",
    "brand_brief_block",
    "brand_negative_prompt",
    "brand_issues",
]


def active_brand() -> str | None:
    """The configured brand id, or ``None`` when brand enforcement is off."""
    value = (get_settings().brand or "").strip().lower()
    return value if value and value not in {"none", "off", "false", "generic"} else None


def brand_name() -> str:
    return "IndiGo" if active_brand() == "indigo" else ""


def brand_system_block(*, safety_only: bool = False) -> str:
    """The policy text to append to a task's system prompt.

    ``safety_only`` is for paths driven by a raw user instruction, where the full
    styling document would fight the thing the user just asked for, but the
    content prohibitions still hold absolutely.
    """
    if active_brand() != "indigo":
        return ""

    if safety_only:
        return "\n".join(
            ["---", "", "# Brand safety — IndiGo", "", INDIGO_VIDEO_SAFETY_POLICY.strip()]
        )

    return "\n".join(
        [
            "---",
            "",
            "# Brand guardrails — IndiGo",
            "",
            "This film is for IndiGo. The rules below are not style preferences; they are the "
            "brand book, and they override any conflicting instinct from the craft references "
            "above. A technically excellent film that breaks them is a failed film.",
            "",
            INDIGO_BRAND_POLICY.strip(),
            "",
            INDIGO_VIDEO_BRAND_POLICY.strip(),
        ]
    )


def brand_brief_block() -> str:
    """Concrete brand requirements for the *user* message.

    The system prompt carries the policy; this carries the values the model has
    to physically write into the continuity block and every scene prompt.
    """
    if active_brand() != "indigo":
        return ""

    return "\n".join(
        [
            "## Brand — IndiGo (non-negotiable)",
            "",
            INDIGO_CONTINUITY_REQUIREMENTS.strip(),
            "",
            "Two failures are automatically rejected before this reaches Veo:",
            "1. Any request to render the logo, wordmark, brand name, or ANY on-screen text. "
            "Veo cannot spell or draw a mark — it comes back wrong every time. The mark is "
            "composited from the master file after the render. Keep the plate clean.",
            "2. Any scene whose palette is not blue-dominant, or that uses red or purple.",
        ]
    )


def brand_negative_prompt() -> str:
    """Brand terms appended to every scene's negative prompt."""
    return INDIGO_NEGATIVE_PROMPT if active_brand() == "indigo" else ""


def brand_issues(story: StoryOption) -> list[AuditIssue]:
    """Brand compliance issues for one story, or none when brand is off."""
    return _indigo_issues(story) if active_brand() == "indigo" else []

"""Brand compliance, enforced in code.

Same reasoning as ``app/skills/audit.py``: a model asked whether it followed the
brand book will say yes. These rules run against the structured storyboard the
model actually returned, and anything at ``error`` severity is fed back as a
repair brief before a single second of Veo time is spent.

The rules here encode ``policy.INDIGO_VIDEO_BRAND_POLICY``. When that document
changes, this file changes with it — a guardrail nobody checks is decoration.
"""

from __future__ import annotations

import re

from ..domain.models import AuditIssue, StoryOption

# ---------------------------------------------------------------------------
# Brand name
# ---------------------------------------------------------------------------

_INDIGO_TOKEN = re.compile(r"\bindigo\b", re.IGNORECASE)

#: "indigo" is also a legitimate colour word, and the approved palette language
#: ("deep indigo navy-blue") depends on it. Colour usage is allowed; brand usage
#: must be the exact IndiGo casing.
_COLOUR_BEFORE = re.compile(
    r"\b(?:deep|dark|pale|light|rich|soft|muted|bright|colour|color|shade|tint|tone|hue)\s*[-,]?\s*$",
    re.IGNORECASE,
)
_COLOUR_AFTER = re.compile(
    r"^\s*[-,]?\s*(?:blue|navy|violet|dye|tone|hue|tint|shade|grade|palette|wash|cast)\b",
    re.IGNORECASE,
)

#: Name forms the brand book explicitly forbids.
_BANNED_NAME_FORMS = re.compile(
    r"\b(?:air\s+indigo|indigo\s+airways|indigo\s+airlines|goindigo(?!\.in\b)|"
    r"indigoing|indigoes|indigone|indiwent)\b",
    re.IGNORECASE,
)

#: The URL must carry both capitals.
_URL_TOKEN = re.compile(r"\bgo\s?indigo\s?\.\s?in\b", re.IGNORECASE)


def _brand_name_issues(text: str) -> list[str]:
    """Return the misspelt brand-name occurrences in ``text``."""
    problems: list[str] = []

    for match in _INDIGO_TOKEN.finditer(text):
        token = match.group(0)
        if token == "IndiGo":
            continue
        before = text[max(0, match.start() - 24) : match.start()]
        after = text[match.end() : match.end() + 24]
        if _COLOUR_BEFORE.search(before) or _COLOUR_AFTER.match(after):
            continue  # the colour, not the brand
        problems.append(token)

    return problems


# ---------------------------------------------------------------------------
# Frame content
# ---------------------------------------------------------------------------

#: Asking Veo for any of these produces a mangled mark or unreadable type.
#: Policy section 2 — all marks and type are composited after the render.
_MARK_OR_TYPE = re.compile(
    r"\b(?:logo|logotype|wordmark|brand\s?mark|arrow[- ]plane|20[- ]dot|twenty[- ]dot|"
    # "supers" plural only — "Super 35" and "super wide" are legitimate craft terms.
    r"subtitle|caption|supers\b|lower[- ]third|end\s?card|title\s?card|on[- ]screen\s+text|"
    r"text\s+overlay|overlay\s+text|tail\s+letter|fuselage\s+letter|livery\s+letter|"
    r"signage\s+read|readable\s+(?:text|signage|lettering))\b",
    re.IGNORECASE,
)

#: Banned dominant palettes. Policy section 3.
_BANNED_COLOURS = re.compile(
    r"\b(?:red|crimson|scarlet|maroon|purple|violet|magenta|fuchsia)\b", re.IGNORECASE
)

#: A prompt that correctly restates the palette *names* the banned colours in
#: order to forbid them ("no red, no purple") — the guardrails ask for exactly
#: that. Negation is therefore scoped to the clause, so "no red, crimson,
#: magenta" clears the whole list while "crimson sunset" in its own clause does
#: not. Getting this wrong would flag the very compliance we asked for.
_NEGATION_CUE = re.compile(
    r"\b(?:no|not|never|avoid|avoiding|without|banned?|ban|excluding|exclude|free\s+of|"
    r"absent|zero|minus|prohibited|forbidden)\b",
    re.IGNORECASE,
)
_CLAUSE_SPLIT = re.compile(r"[.;\n]")


#: How far back a negation can reach. Long enough to cover "banned: red,
#: crimson, purple, violet", short enough that an unrelated "not" earlier in a
#: sentence does not silently excuse a colour genuinely in frame.
_NEGATION_WINDOW = 60


def _unnegated(pattern: re.Pattern[str], text: str) -> list[str]:
    """Matches of ``pattern`` that are asserted, not forbidden.

    Compliant prompts name the banned things in order to ban them — the
    continuity block says "no logo, wordmark or on-screen text in frame", and
    that block is restated in every scene. Matching naively would flag the exact
    compliance the guardrails demand, on every single scene. A negation only
    covers what follows it, so "crimson and gold, no blue" is still crimson.
    """
    found: list[str] = []
    for clause in _CLAUSE_SPLIT.split(text):
        for match in pattern.finditer(clause):
            lookback = clause[max(0, match.start() - _NEGATION_WINDOW) : match.start()]
            if not _NEGATION_CUE.search(lookback):
                found.append(match.group(0).lower())
    return found


def _banned_colour_hits(text: str) -> set[str]:
    return set(_unnegated(_BANNED_COLOURS, text))

#: At least one of these must appear so the frame reads as IndiGo blue.
_REQUIRED_BLUE = re.compile(r"\b(?:blue|navy|indigo|azure|cyan|cerulean)\b", re.IGNORECASE)

#: Camera behaviour that contradicts "effortless". Policy section 4.
_FRANTIC_CAMERA = re.compile(
    r"\b(?:whip\s?pan|snap\s?zoom|crash\s?zoom|speed\s?ramp|shaky|frantic|chaotic|"
    r"jerky|violent\s+(?:shake|movement))\b",
    re.IGNORECASE,
)

# ---------------------------------------------------------------------------
# Safety. Policy section 9 — zero tolerance.
# ---------------------------------------------------------------------------

_SAFETY_TERMS = re.compile(
    r"\b(?:crash(?:es|ing|ed)?|collision|colliding|mid[- ]air|emergency\s+landing|"
    r"crash[- ]land|aircraft\s+(?:on\s+fire|in\s+distress)|uncontrolled\s+descent|"
    r"nose[- ]dive|nosedive|runway\s+(?:incident|overrun)|oxygen\s+mask|evacuation\s+slide|"
    r"emergency\s+slide|evacuat(?:e|ing|ion)|severe\s+turbulence|explosion|exploding|"
    r"detonat|terroris|sabotage|hijack|wreckage|debris\s+field|casualt|fatalit)\b",
    re.IGNORECASE,
)

# ---------------------------------------------------------------------------
# Tone. Policy sections 1 and 6.
# ---------------------------------------------------------------------------

_CHEAP_TONE = re.compile(
    r"\b(?:cheap(?:est)?|dirt[- ]cheap|bargain|discount|rock[- ]bottom|slash(?:ed|ing)?\s+(?:price|fare)|"
    r"unbeatable\s+(?:price|fare|deal)|lowest\s+(?:price|fare)\s+ever)\b",
    re.IGNORECASE,
)

#: "routes"/"sectors" are trade words; the brand book says "flights".
_TRADE_WORDS = re.compile(r"\b(?:routes|sectors)\b", re.IGNORECASE)

_ALL_CAPS_RUN = re.compile(r"\b[A-Z]{4,}\b")

#: Acronyms and codes that are legitimately upper case.
_CAPS_ALLOWLIST = {
    "FPV", "SFX", "LED", "HDR", "IMAX", "JSON", "MMKG", "USB", "IFE", "ATC", "VFX",
}

#: A hero shot of hardware with nobody in it. Policy section 5 — a warning, not
#: a block: aviation settings are legitimate in a brand film, the cliché is not.
_AVIATION_SUBJECT = re.compile(
    r"\b(?:aircraft|airplane|aeroplane|plane|jet|airbus|a320|a321|runway|tarmac|"
    r"airport|terminal|apron|taxiway)\b",
    re.IGNORECASE,
)
_HUMAN_PRESENCE = re.compile(
    r"\b(?:passenger|traveller|traveler|child|kid|boy|girl|man|woman|mother|father|"
    r"crew|attendant|pilot|hand|face|eyes|shoulder|family|couple|person|people|"
    r"grandmother|grandfather|daughter|son|figure|silhouette\s+of\s+a)\b",
    re.IGNORECASE,
)


def _all_caps_hits(text: str) -> list[str]:
    return [
        word
        for word in _ALL_CAPS_RUN.findall(text)
        if word not in _CAPS_ALLOWLIST and not word.isdigit()
    ]


def brand_issues(story: StoryOption) -> list[AuditIssue]:
    """Audit one story option against the IndiGo video guardrails."""
    issues: list[AuditIssue] = []

    def push(rule: str, severity: str, message: str) -> None:
        issues.append(AuditIssue(rule=rule, severity=severity, message=message))  # type: ignore[arg-type]

    def at(scene, rule: str, severity: str, message: str) -> None:
        issues.append(
            AuditIssue(
                sceneId=scene.id,
                sceneOrder=scene.order,
                rule=rule,
                severity=severity,  # type: ignore[arg-type]
                message=message,
            )
        )

    # --- Story level --------------------------------------------------------
    story_text = " ".join(
        [story.title, story.summary, story.treatment, story.angle, *story.continuity.rules]
    )

    misspelt = _brand_name_issues(story_text)
    if misspelt:
        push(
            "brand/name",
            "error",
            f"The brand name is written as {', '.join(sorted(set(misspelt)))}. "
            "It is always IndiGo — capital I, capital G.",
        )

    if _BANNED_NAME_FORMS.search(story_text):
        push(
            "brand/name-form",
            "error",
            "Forbidden name form. Never Air IndiGo, IndiGo Airways, IndiGo Airlines, or "
            "goindigo, and never the name as a verb.",
        )

    palette = story.continuity.palette
    if palette:
        if not _REQUIRED_BLUE.search(palette):
            push(
                "brand/palette-blue",
                "error",
                "The locked palette does not name a blue. Deep indigo navy-blue "
                "(Pantone 2738 C) must dominate, with pale sky blue (Pantone 291 C) secondary.",
            )
        banned = _banned_colour_hits(palette)
        if banned:
            push(
                "brand/palette-banned",
                "error",
                f"The locked palette contains prohibited colours: {', '.join(sorted(banned))}. "
                "Red and purple palettes are banned.",
            )

    if not any(
        _MARK_OR_TYPE.search(rule) or "text" in rule.lower() for rule in story.continuity.rules
    ):
        push(
            "brand/no-mark-rule",
            "warning",
            "The continuity block does not state that no logo, wordmark, or on-screen text "
            "appears in frame. Lock it there so every scene inherits it.",
        )

    caps = _all_caps_hits(f"{story.title} {story.summary}")
    if caps:
        push(
            "brand/sentence-case",
            "warning",
            f"ALL CAPS in story copy: {', '.join(sorted(set(caps))[:4])}. "
            "IndiGo is always sentence case.",
        )

    if _CHEAP_TONE.search(story_text):
        push(
            "brand/tone",
            "error",
            "Discount-led language. Low-cost must never read as cheap — sell effortlessness, "
            "not price.",
        )

    if _TRADE_WORDS.search(story_text):
        push("brand/vocabulary", "warning", 'Use "flights", never "routes" or "sectors".')

    # --- Scene level --------------------------------------------------------
    for scene in story.scenes:
        prompt = scene.prompt or ""
        card = scene.shot_card
        haystack = "\n".join(
            [
                prompt,
                scene.summary,
                *(value for value in card.model_dump().values() if isinstance(value, str)),
            ]
        )

        marks = _unnegated(_MARK_OR_TYPE, haystack)
        if marks:
            at(
                scene,
                "brand/generated-mark",
                "error",
                f'This scene asks the model to render "{marks[0]}". Veo cannot render a '
                "wordmark or type legibly — it will come back misspelt and off-colour. Remove it; "
                "the mark is composited from the master file after the render.",
            )

        scene_misspelt = _brand_name_issues(haystack)
        if scene_misspelt:
            at(
                scene,
                "brand/name",
                "error",
                f"Brand name written as {', '.join(sorted(set(scene_misspelt)))}. It is always "
                "IndiGo. Better still: describe the colour as deep indigo navy-blue and keep the "
                "brand name out of the prompt entirely.",
            )

        if not _REQUIRED_BLUE.search(f"{card.palette} {prompt}"):
            at(
                scene,
                "brand/palette-blue",
                "error",
                "No blue in this scene. Blue must dominate every frame, not just the average of "
                "the film — Veo has no memory between clips, so the palette must be restated here.",
            )

        banned = _banned_colour_hits(f"{card.palette}. {prompt}")
        if banned:
            at(
                scene,
                "brand/palette-banned",
                "error",
                f"Prohibited colours in frame: {', '.join(sorted(banned))}. "
                "Red and purple palettes are banned.",
            )

        if _SAFETY_TERMS.search(haystack):
            at(
                scene,
                "brand/safety",
                "error",
                "This scene depicts an aviation incident, emergency, or destruction. Absolutely "
                "prohibited — rewrite the beat entirely rather than softening it.",
            )

        if _FRANTIC_CAMERA.search(haystack):
            at(
                scene,
                "brand/camera-calm",
                "warning",
                "Frantic camera work contradicts effortless. Prefer a calm, motivated move — "
                "slow push-in, gentle track, or locked-off.",
            )

        if _CHEAP_TONE.search(haystack):
            at(scene, "brand/tone", "error", "Discount-led language. Never frame IndiGo as cheap.")

        caps = _all_caps_hits(prompt)
        if caps:
            at(
                scene,
                "brand/sentence-case",
                "warning",
                f"ALL CAPS in the prompt: {', '.join(sorted(set(caps))[:4])}.",
            )

        if _AVIATION_SUBJECT.search(haystack) and not _HUMAN_PRESENCE.search(haystack):
            at(
                scene,
                "brand/aviation-cliche",
                "warning",
                "An aircraft, runway, or terminal with no human presence is a stock aviation "
                "shot, not a scene. Put a person and a feeling in the frame.",
            )

    return issues

"""IndiGo brand guardrails for the video pipeline.

Single source of truth for brand policy in this codebase. The image and text
policies live in the sibling marketing repo (``prompts/platform_*_prompt.py``);
this file carries the shared rules that survive the jump to motion, plus
``INDIGO_VIDEO_BRAND_POLICY`` — the video-specific document, which has no
counterpart there because video breaks brand in ways stills cannot.

Three things make video different, and they are why this file exists:

1. **Every clip is an independent generation.** Veo has no memory between the
   4/6/8-second segments the duration planner produces, so a palette stated
   once drifts off-brand by the middle of the film. Brand has to be restated in
   every single prompt, exactly like the continuity block.
2. **Generative video cannot render a wordmark.** Asking Veo for the IndiGo
   logo produces a misspelt, wrong-coloured mark every time. The brand book
   already forbids this by another route — "wordmark should always be used from
   the master file, and should not be typed" — so the mark is banned from the
   prompt and composited from the master asset at render time.
3. **Motion carries brand meaning that a still cannot.** "Effortless" is a
   camera behaviour before it is a colour.
"""

from __future__ import annotations

# ---------------------------------------------------------------------------
# Colour
# ---------------------------------------------------------------------------
# Named colours are what a diffusion model actually responds to; hex is what the
# post-production overlay needs. Both are recorded, but the prompt layer leads
# with language, not numbers.
#
# NOTE: the two source policy files disagreed on the light blue (291 C in the
# top-level rules, 284 C in the image policy). Confirmed as **Pantone 291 C**.
#
# The hex values below are approximate sRGB conversions for the ffmpeg overlay
# and UI swatches. Confirm them against the brand book before anything ships to
# air — a Pantone-to-sRGB conversion is never exact, and "close enough" blue is
# the exact failure this file exists to prevent.

INDIGO_DARK_BLUE_PANTONE = "Pantone 2738 C"
INDIGO_DARK_BLUE_HEX = "#2E1A87"  # approximate — deep indigo-navy, violet lean

INDIGO_LIGHT_BLUE_PANTONE = "Pantone 291 C"
INDIGO_LIGHT_BLUE_HEX = "#9BCBEB"  # approximate — pale sky blue

#: How the palette should be *described* to a generative model. Colour names
#: steer a diffusion model; hex codes do not.
INDIGO_PALETTE_LANGUAGE = (
    "deep indigo navy-blue with a violet lean (Pantone 2738 C) as the dominant colour, "
    "pale sky blue (Pantone 291 C) as the secondary, and clean white as the breathing space"
)


# ---------------------------------------------------------------------------
# Shared rules — the brand book, as it applies to any medium
# ---------------------------------------------------------------------------

INDIGO_BRAND_POLICY = """
HARD RULES:
- Brand name is always written as: IndiGo (capital I, capital G). Never Indigo, INDIGO,
  indigo, Air IndiGo, IndiGo Airways, IndiGo Airlines, or goindigo
- Never turn the name into a verb: no IndiGoing, IndiGoes, IndiGone, IndiWent
- URL is always written as goIndiGo.in (I and G upper case)
- Logo has 20 dots and must always appear in correct proportion and direction
- Wordmark always comes from the master file and is never typed or recreated
- Typeface: Bauhaus. Light weight for titles and headlines — never bold, semibold,
  medium, or regular
- Always sentence case. Never ALL CAPS. Title case only for names, properties, and assets
- Never say anything negative about IndiGo
- Always polite and professional
- Never place the logo on an aircraft tail; place it in a corner that suits the composition

SOFT RULES:
- Black, dark blue, or light blue background -> white logo
- Arrow-plane may be used creatively (patterns, circles) but must always imply
  forward/upward motion, never pure decoration
- Where a logo meets a building, start it from the building rather than overlapping it
- Primary colour: Dark Blue (Pantone 2738 C). Secondary: Light Blue (Pantone 291 C)
- Simple, eye-catching, non-controversial subject that adapts to many sizes
- Soft colours, balanced composition, simple design
- Integrate pop culture and modern trends where they genuinely fit
"""


# ---------------------------------------------------------------------------
# The video policy — the document this module exists for
# ---------------------------------------------------------------------------

INDIGO_VIDEO_BRAND_POLICY = """
========================
1. BRAND FOUNDATION IN MOTION
========================

IndiGo believes low-cost should never feel cheap. In film this is carried by how
the camera behaves, not by what is said.

Every generated film must feel:
- Effortless — nothing strains, nothing rushes
- On-time and precise — clean, deliberate framing
- Hassle-free — uncluttered frames, room to breathe
- Affordable without ever looking cheap

Brand personality on screen: fun, fresh, smart, simple. Confident, never arrogant.
Warm, never cheesy. Witty, never forced. Straightforward, never boring.

Core principle: say simple truths in interesting ways.


========================
2. THE MARK IS NEVER GENERATED  (ABSOLUTE)
========================

This is the single most-violated rule and the most damaging.

A generative video model cannot render a wordmark, a logotype, or a 20-dot logo
legibly. It will produce a misspelt, wrong-coloured, wrong-proportioned mark in
every clip. This also violates the brand book directly: the wordmark must come
from the master artwork and must never be typed.

Therefore, in every scene prompt:
- NEVER ask for the IndiGo logo, wordmark, logotype, or the arrow-plane device
- NEVER ask for the word "IndiGo" to appear anywhere in frame
- NEVER ask for the 20-dot pattern
- NEVER ask for goIndiGo.in or any URL on screen
- NEVER ask for aircraft tail lettering, fuselage lettering, livery text, or
  branded signage that would need readable brand type
- NEVER ask for subtitles, captions, supers, lower-thirds, end cards, or any
  on-screen text of any kind

All brand marks and all on-screen type are composited after the render, from the
master asset, in Bauhaus light, in sentence case. The generated footage is the
plate underneath, and nothing more.

When an aircraft appears, frame it so unreadable lettering is not the subject:
prefer partial fuselage, wing, underside, shadow, silhouette, or motion blur over
a clean broadside of a tail.


========================
3. COLOUR — AND WHY IT DRIFTS
========================

Palette:
- Dominant: deep indigo navy-blue with a violet lean (Pantone 2738 C)
- Secondary: pale sky blue (Pantone 291 C)
- Breathing space: clean white
- Any further colours must be soft and pastel

Blue must dominate the frame in EVERY scene, not on average across the film.

Strictly prohibited:
- Red-dominant palettes
- Purple-dominant palettes
- High-contrast or harsh palettes
- Sodium-orange or teal-and-orange grades

THE DRIFT PROBLEM. Each clip is a separate Veo generation with no memory of the
last one. A palette stated only in the treatment will be honoured in scene 1 and
gone by scene 6, and the film will change colour halfway through. The palette
therefore belongs in the continuity block AND restated inside every single scene
prompt, in colour words, not hex codes — a diffusion model responds to
"deep indigo navy-blue", not to "#2E1A87".


========================
4. CAMERA AND MOTION
========================

"Effortless" is a camera behaviour first.

Preferred:
- Calm, precise, motivated moves — slow push-in, gentle lateral track, locked-off
- Smooth glides that echo the arrow-plane's forward/upward direction
- Generous negative space; let the frame breathe
- One primary move per shot, with a reason

Avoid:
- Frantic whip pans, snap zooms, aggressive speed ramps
- Shaky or chaotic handheld
- Restless drifting with no motivation
- Over-designed, busy, or cluttered frames

Direction of travel carries meaning: movement should read as forward or upward
wherever the story allows it.


========================
5. PEOPLE AND PHOTOGRAPHY
========================

Prohibited:
- Stock-photography look
- Posed or smiling stock faces
- Glamourised or cliché travel imagery
- Luxury-airline styling, and equally, discount-airline styling

Preferred:
- Candid, natural, unposed people
- Conceptual and insight-driven moments
- Evocative, atmospheric imagery
- Thoughtful or unexpected perspectives
- Destination-led storytelling

Aviation settings (tarmac, terminal, cabin, boarding) are permitted in a brand
film — but the aircraft is never the story. A hero beauty-shot of an aircraft, a
runway, or an airport with no human presence and no emotional beat is a cliché,
not a scene. Put a person and a feeling in the frame.


========================
6. SOUND AND ANY SPOKEN LINE
========================

- Room tone, texture, and the story's motif. Never ask Veo for music; a score is
