"""
Azure Application Insights telemetry via azure-monitor-opentelemetry.

configure_appinsights() — call once at startup. Wires Azure Monitor into
FastAPI via OpenTelemetry auto-instrumentation (HTTP requests → requests table)
and sets up structured log export (logger.info/warning/error → traces table).

track_event()        — emit a named structured log to the traces table.
claims_props()       — extract standard user fields from a JWT claims dict.
log_blocked_prompt() — DB audit + telemetry for safety-blocked prompts.
"""

import logging
import os
from contextvars import ContextVar
from typing import Any, Dict, Optional

logger = logging.getLogger(__name__)

# Dedicated logger whose records are exported to App Insights as traces.
_ai_logger = logging.getLogger("marketing-platform.events")

# Set once per request in middleware; inherited by background tasks spawned
# within the same request context. Falls back to APP_BASE_URL env var for
# startup events or scheduled jobs that have no request context.
_request_url: ContextVar[str] = ContextVar("_request_url", default="")


def set_request_url(url: str) -> None:
    """Called from HTTP middleware to record the current request base URL."""
    _request_url.set(url)


def _url_prop() -> Dict[str, str]:
    url = _request_url.get() or os.getenv("APP_BASE_URL", "unknown")
    return {"environment_url": url}


def configure_appinsights() -> None:
    """Wire up Azure Monitor + FastAPI auto-instrumentation at startup."""
    conn_str = os.getenv("APPLICATIONINSIGHTS_CONNECTION_STRING", "").strip()
    if not conn_str:
        logger.warning("App Insights disabled: APPLICATIONINSIGHTS_CONNECTION_STRING not set")
        return
    if conn_str.startswith("@Microsoft.KeyVault("):
        logger.warning("App Insights disabled: APPLICATIONINSIGHTS_CONNECTION_STRING is a Key Vault reference — replace with the actual value in your local .env")
        return
    try:
        from azure.monitor.opentelemetry import configure_azure_monitor
        configure_azure_monitor(connection_string=conn_str)
        logger.info("App Insights initialised via azure-monitor-opentelemetry")
    except Exception as exc:
        logger.error("App Insights setup failed | error=%s", exc)


def claims_props(claims: Dict[str, Any]) -> Dict[str, str]:
    """Extract standard user fields from a JWT claims dict."""
    return {
        "user_oid":   claims.get("oid", ""),
        "user_name":  claims.get("name", ""),
        "user_email": claims.get("preferred_username") or claims.get("upn") or claims.get("email", ""),
        "user_roles": str(claims.get("roles", [])),
    }


def track_event(
    name: str,
    properties: Optional[Dict[str, str]] = None,
    measurements: Optional[Dict[str, float]] = None,
) -> None:
    """Emit a structured log entry to the App Insights traces table.

    All fields land in customDimensions, queryable via KQL:
        traces | where customDimensions["event_name"] == "<name>"
    Fire-and-forget — never raises.
    """
    try:
        dimensions: Dict[str, object] = {"event_name": name, **_url_prop()}
        if properties:
            dimensions.update(properties)
        if measurements:
            dimensions.update(measurements)
        _ai_logger.info(name, extra={"custom_dimensions": dimensions})
    except Exception as exc:
        logger.error("track_event failed | event=%s | error=%s", name, exc)


def log_blocked_prompt(
    user_id: str,
    prompt: str,
    category: str,
    real_reason: str,
    endpoint: str = "",
) -> None:
    """DB audit log + App Insights telemetry for a safety-blocked prompt.

    Single call replaces the previous pattern of calling db.connection.log_blocked_prompt
    and track_event separately at every block site.
    """
    try:
        from db.connection import log_blocked_prompt as _db_log
        _db_log(user_id, prompt, category, real_reason)
    except Exception as exc:
        logger.error("log_blocked_prompt DB write failed | user=%s | error=%s", user_id, exc)

    track_event("prompt.blocked", properties={
        "user_id":        user_id,
        "category":       category,
        "blocked_reason": real_reason,
        "endpoint":       endpoint,
        "prompt_snippet": prompt[:200],
    })
