const pageKeyAliases: Record<string, string> = {
  mailer_reports: "mailer-reports",
  mailerReports: "mailer-reports",
  "base-ops": "base-osp",
  "base-operations": "base-osp",
};

const normalizePageKey = (pageKey: string): string => {
  return pageKeyAliases[pageKey] || pageKey;
};

const mapPageIdsToPermissions = (
  pageIds: string[],
  dynamicPageMapping: Record<string, Permission>
): Permission[] => {
  const seenPermissionIds = new Set<string>();

  return pageIds
    .filter((pageId: string) => !pageId.endsWith("_agent"))
    .map((pageId: string) => {
      const normalizedPageKey = normalizePageKey(pageId);
      return dynamicPageMapping[normalizedPageKey] || dynamicPageMapping[pageId];
    })
    .filter((permission): permission is Permission => {
      if (!permission) {
        return false;
      }
      if (seenPermissionIds.has(permission.id)) {
        return false;
      }
      seenPermissionIds.add(permission.id);
      return true;
    });
};

// Fetch page mapping from backend
const fetchPageMapping = async (getRbacPages: () => Promise<any[]>): Promise<Record<string, Permission>> => {
  try {
    const data = await getRbacPages();
    const mapping: Record<string, Permission> = {};
    data?.forEach((page: any) => {
      const normalizedPageKey = normalizePageKey(page.key);
      const permission: Permission = {
        id: normalizedPageKey,
        label: page.label,
        path: getPathForPage(normalizedPageKey),
        icon: getIconForPage(normalizedPageKey),
      };

      mapping[normalizedPageKey] = permission;
      mapping[page.key] = permission;
    });

    // Keep backward-compatible aliases for renamed keys.
    Object.entries(pageKeyAliases).forEach(([aliasKey, canonicalKey]) => {
      if (!mapping[canonicalKey] && mapping[aliasKey]) {
        mapping[canonicalKey] = mapping[aliasKey];
      }
      if (!mapping[aliasKey] && mapping[canonicalKey]) {
        mapping[aliasKey] = mapping[canonicalKey];
      }
    });

    return mapping;
  } catch (err) {
    console.error("Failed to fetch page mapping:", err);
    return pageMapping;
  }
};

const reorderPermissionsWithAISearchInMiddle = (
permissions: Permission[]
): Permission[] => {
const aiSearchIndex = permissions.findIndex((p) => p.id === "ai-search");
if (aiSearchIndex === -1) return permissions;
const otherPermissions = permissions.filter((p) => p.id !== "ai-search");
const aiSearch = permissions[aiSearchIndex];
const middleIndex = Math.floor(otherPermissions.length / 2);
const result = [...otherPermissions];
result.splice(middleIndex, 0, aiSearch);
return result;
};

const reorderPermissionsWithHomeFirst = (
  permissions: Permission[]
): Permission[] => {
  const homeIndex = permissions.findIndex((permission) => permission.id === "home");
  if (homeIndex <= 0) {
    return permissions;
  }

  const [homePermission] = permissions.splice(homeIndex, 1);
  return [homePermission, ...permissions];
};

const withHomeFallback = (
  mappedPermissions: Permission[],
  dynamicPageMapping: Record<string, Permission>
): Permission[] => {
  if (mappedPermissions.length > 0) {
    return mappedPermissions;
  }
  return [dynamicPageMapping.home || pageMapping.home];
};

// Statically-available pages not (yet) driven by the backend RBAC list.
// Appended so they surface in the sidebar and pass RouteGuard.
const STATIC_PERMISSIONS: Permission[] = [
  {
    id: "event-scheduler",
    label: "Event Scheduler",
    path: "/admin/event-scheduler",
    icon: "Calendar",
  },
];

const withStaticPermissions = (permissions: Permission[]): Permission[] => {
  const existingPaths = new Set(permissions.map((p) => p.path));
  const extras = STATIC_PERMISSIONS.filter((p) => !existingPaths.has(p.path));
  return [...permissions, ...extras];
};

export const usePermissions = (): UserPermissions => {
const [permissions, setPermissions] = useState<Permission[]>([]);
const [loading, setLoading] = useState(true);
const [error, setError] = useState<string | null>(null);
const { testRole, testPages } = useTestRole();
const { userInfo, userRole, setUserInfo, getRbacPages, getUserData } = useUser();
const hasFetchedRef = useRef(false);

const isAuthorizedUser = () => {
  return userInfo?.testRolePermission === true;
};
useEffect(() => {
  const fetchPermissions = async () => {
    // Prevent multiple fetches
    if (hasFetchedRef.current && !testRole) return;
    
    try {
     // console.log('📋 usePermissions - Starting fetch, testRole:', testRole, 'testPages:', testPages);
      // Get dynamic page mapping from backend
      const dynamicPageMapping = await fetchPageMapping(getRbacPages);
      
      // PRIORITY 1: If test role is active AND user is authorized, use test pages
      if (isAuthorizedUser() && testRole && testPages.length > 0) {
        //console.log('📋 usePermissions - Using TEST ROLE pages:', testPages);
        const mappedPermissions = mapPageIdsToPermissions(testPages, dynamicPageMapping);
        const safePermissions = withHomeFallback(mappedPermissions, dynamicPageMapping);
        const reorderedPermissions = reorderPermissionsWithHomeFirst(
          reorderPermissionsWithAISearchInMiddle(withStaticPermissions(safePermissions))
        );
        //console.log('📋 usePermissions - Mapped test role permissions:', reorderedPermissions);
        setPermissions(reorderedPermissions);
        setError(null);
        setLoading(false);
        return;
      }
      
      // PRIORITY 2: Development mode - use user's actual pages from context
      const isDevelopment = window.location.hostname === "localhost";
      if (isDevelopment && userRole && userInfo) {
       // console.log('📋 usePermissions - Using DEVELOPMENT MODE pages:', userInfo.pages);
        const userPages = userInfo.pages || [];
        const mappedPermissions = mapPageIdsToPermissions(userPages, dynamicPageMapping);
        const safePermissions = withHomeFallback(mappedPermissions, dynamicPageMapping);
        const reorderedPermissions = reorderPermissionsWithHomeFirst(
          reorderPermissionsWithAISearchInMiddle(withStaticPermissions(safePermissions))
        );
        setPermissions(reorderedPermissions);
        setError(null);
        setLoading(false);
        return;
      }
      
      // PRIORITY 3: Otherwise, try to get from backend API
     // console.log('📋 usePermissions - Fetching from backend API');
      const userData = await getUserData();
      const userPages = userData.pages || [];
      // Map backend page IDs to navigation permissions
      const mappedPermissions = mapPageIdsToPermissions(userPages, dynamicPageMapping);
      const safePermissions = withHomeFallback(mappedPermissions, dynamicPageMapping);
      const reorderedPermissions = reorderPermissionsWithHomeFirst(
        reorderPermissionsWithAISearchInMiddle(safePermissions)
      );
      setPermissions(reorderedPermissions);
      setError(null);
    } catch (err) {
      console.error("Error fetching permissions:", err);
      setError("Failed to load permissions");
      setPermissions([]);
    } finally {
      setLoading(false);
      if (!testRole) {
        hasFetchedRef.current = true;
      }
    }
  };
  fetchPermissions();
  // eslint-disable-next-line
}, [testRole, testPages]);
return { permissions, loading, error };
};
