#!/usr/bin/env python3
"""
Standalone helper -- run this BEFORE trying run_scan.py if you have an Azure
OpenAI base URL + key but don't know the deployment name or API version.

Azure OpenAI resources expose a "list my deployments" endpoint that works
with just the key you already have (no extra Azure login needed). This
script tries a handful of known API-version strings against that endpoint
until one works, then prints every deployment you have access to.

Usage:
    python3 discover_azure.py https://your-resource-name.openai.azure.com your-api-key
"""

import json
import sys
import urllib.error
import urllib.request

# Tried newest-first. Azure OpenAI's list-deployments endpoint has accepted
# any of these at various points -- we just need ONE that this resource accepts.
CANDIDATE_API_VERSIONS = [
    "2024-10-21",
    "2024-08-01-preview",
    "2024-06-01",
    "2024-02-01",
    "2023-05-15",
    "2023-03-15-preview",
]


def try_list_deployments(base_url: str, api_key: str, api_version: str):
    base_url = base_url.rstrip("/")
    url = f"{base_url}/openai/deployments?api-version={api_version}"
    req = urllib.request.Request(url, headers={"api-key": api_key})
    with urllib.request.urlopen(req, timeout=15) as resp:
        return json.loads(resp.read().decode())


def main():
    if len(sys.argv) != 3:
        print("Usage: python3 discover_azure.py <base_url> <api_key>")
        sys.exit(1)

    base_url, api_key = sys.argv[1], sys.argv[2]

    for version in CANDIDATE_API_VERSIONS:
        print(f"Trying API version {version} ...", end=" ")
        try:
            data = try_list_deployments(base_url, api_key, version)
        except urllib.error.HTTPError as e:
            print(f"failed ({e.code})")
            continue
        except Exception as e:  # noqa: BLE001
            print(f"failed ({e})")
            continue

        deployments = data.get("data", [])
        print("SUCCESS")
        print()
        print(f"This API version works: {version}")
        print(f"Put this in your .env: AZURE_OPENAI_API_VERSION={version}")
        print()
        if not deployments:
            print("No deployments found on this resource -- ask whoever gave you "
                  "the key whether any model is actually deployed here.")
            return

        print(f"Found {len(deployments)} deployment(s) on this resource:")
        for d in deployments:
            print(f"  - deployment name: {d.get('id')}   (underlying model: {d.get('model')})")
        print()
        print("Use one of the 'deployment name' values above as AZURE_OPENAI_DEPLOYMENT.")
        return

    print()
    print("None of the common API versions worked. This usually means either:")
    print("  - the key or base URL is wrong, or")
    print("  - this Azure resource has disabled the plain API-key access this")
    print("    script relies on, and needs Azure AD login instead.")
    print("In that case, go back to whoever gave you the key and ask for the")
    print("deployment name and API version directly.")


if __name__ == "__main__":
    main()
