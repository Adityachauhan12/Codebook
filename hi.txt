name: 'Sandbox CI Pipeline'
on:
  push:
    branches:
      - sandbox 

jobs:
  SANDBOX_CI:
    name: 'Sandbox CI Pipeline'
    runs-on: ubuntu-latest   
    environment: sandbox
    defaults:
      run:
        shell: bash
    steps:
      - name: 'Checkout GitHub Action'
        uses: actions/checkout@v4
      - name: 'Docker Login'
        uses: azure/docker-login@v1
        with:
          login-server: ${{ secrets.REGISTRY_LOGIN_SERVER }}
          username: ${{ secrets.REGISTRY_USERNAME }}
          password: ${{ secrets.REGISTRY_PASSWORD }}
      - name: 'Build and Push Image'
        uses: docker/build-push-action@v5
        with:
          push: true
          tags: ${{ secrets.REGISTRY_LOGIN_SERVER }}/marketing-platform:sandbox  # Use the tag directly
          file: Dockerfile
          build-args: |
            REGISTRY=${{ secrets.REGISTRY_LOGIN_SERVER }}



name: 'Dev CI Pipeline'
on:
  push:
    branches:
      - dev 

jobs:
  DEV_CI:
    name: 'Dev CI Pipeline'
    runs-on: [self-hosted, self-hosted-nonprod]
    environment: non-prod
    defaults:
      run:
        shell: bash
    steps:
      - name: 'Checkout GitHub Action'
        uses: actions/checkout@v4
      - name: 'Free Disk Space'
        run: |
          docker system prune -af --volumes
          df -h
      - name: 'Docker Login'
        uses: azure/docker-login@v1
        with:
          login-server: ${{ secrets.REGISTRY_LOGIN_SERVER }}
          username: ${{ secrets.REGISTRY_USERNAME }}
          password: ${{ secrets.REGISTRY_PASSWORD }}
      - name: 'Build and Push Image'
        uses: docker/build-push-action@v5
        with:
          push: true
          tags: ${{ secrets.REGISTRY_LOGIN_SERVER }}/marketing-platform:dev  # Use the tag directly
          file: Dockerfile
          build-args: |
            REGISTRY=${{ secrets.REGISTRY_LOGIN_SERVER }}
