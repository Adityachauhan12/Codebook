import React, { useState, useCallback, useEffect, useRef } from 'react';
import { useIdleTimer } from './hooks/useIdleTimer';
import { useIsAuthenticated, useMsal } from '@azure/msal-react';
import { Routes, Route, Navigate, NavLink, useLocation } from 'react-router-dom';
import { LoginPage } from './components/LoginPage';
import { ALLOWED_DOMAIN } from './auth/authConfig';
import { ProjectsPage } from './components/ProjectsPage';
import { StudioPage } from './components/StudioPage';
import { ExplorePage } from './components/ExplorePage';
import { SixECreativesPage } from './components/SixECreativesPage';
import { AnalyticsPage } from './components/AnalyticsPage';
import { FeedbackPage } from './components/FeedbackPage';
import { MyCreativesPage } from './components/MyCreativesPage';
import { ApproverCreativesPage } from './components/ApproverCreativesPage';
import EditorPage from './components/EditorPage';
import { FolderOpen, Compass, LogOut, MessageSquare, TrendingUp, Images, Image,Crop, ClipboardCheck, Clapperboard } from 'lucide-react';
import { setAuthToken, setUnauthorizedHandler, apiFetch } from './services/apiClient';
import { getPublicLogoUrl } from './utils/logoAsset';
import { CONFIG } from './config';
import { VideoPage } from './features/video/VideoPage';

// ─── Root: handles auth gate ──────────────────────────────────────────────────

export default function App() {
  const isAuthenticated = useIsAuthenticated();
  const { instance, accounts } = useMsal();
  const [loggedOut, setLoggedOut] = useState(false);
  const [sessionExpired, setSessionExpired] = useState(false);
  const [idleWarning, setIdleWarning] = useState(false);
  const [countdown, setCountdown] = useState(120); // seconds
  const countdownRef = useRef<ReturnType<typeof setInterval> | null>(null);

  const handleLogout = useCallback(async () => {
    // Revoke the current token on the backend before wiping local session state.
    // Errors are swallowed so a network failure never blocks the client-side logout.
    try {
      await apiFetch('/api/v1/auth/logout', { method: 'POST' });
    } catch {
      // proceed regardless
    }
    setAuthToken('');
    sessionStorage.clear();
    localStorage.clear();
    setLoggedOut(true);
    setIdleWarning(false);
    if (countdownRef.current) clearInterval(countdownRef.current);
    window.history.pushState(null, '', '/');
  }, []);

  // Start countdown when idle warning appears
  useEffect(() => {
    if (idleWarning) {
      setCountdown(120);
      countdownRef.current = setInterval(() => {
        setCountdown(prev => {
          if (prev <= 1) {
            clearInterval(countdownRef.current!);
            return 0;
          }
          return prev - 1;
        });
      }, 1000);
    } else {
      if (countdownRef.current) clearInterval(countdownRef.current);
    }
    return () => { if (countdownRef.current) clearInterval(countdownRef.current); };
  }, [idleWarning]);

  const handleIdleWarn   = useCallback(() => { setIdleWarning(true); }, []);
  const handleIdleExpire = useCallback(() => {
    setIdleWarning(false);
    setSessionExpired(true);
  }, []);
  const handleIdleReset  = useCallback(() => {
    setIdleWarning(false);
    // Do NOT clear sessionExpired here — once the session has expired it must
    // stay expired until the user explicitly clicks Re-login.
  }, []);

  // On 401: try a silent renewal before declaring the session dead.
  //
  // The token is acquired once on mount and MSAL caches it for about an hour.
  // Anything that outlives that — a video render runs for fifteen minutes and
  // can start late in the hour — comes back to a stale token, and the first
  // request afterwards is the one that fetches the finished film. Returning
  // false unconditionally made every such 401 permanent and left the retry
  // branch in apiFetch unreachable, so a completed render presented as a dead
  // session. `acquireTokenSilent` renews from the refresh token without any
  // user interaction; only a genuine failure is a real expiry.
  useEffect(() => {
    const handle401 = async (): Promise<boolean> => {
      const account = accounts[0];
      if (account) {
        try {
          const result = await instance.acquireTokenSilent({
            scopes: ['openid', 'profile', 'email'],
            account,
          });
          setAuthToken(result.idToken);
          return true;
        } catch {
          // Silent renewal failed — the session really is over.
        }
      }
      setSessionExpired(true);
      return false;
    };
    setUnauthorizedHandler(handle401);
    return () => setUnauthorizedHandler(() => {});
  }, [accounts, instance]);

  //console.log('Accounts:', accounts[0]);
  const userEmail = accounts[0]?.username ?? '';
  const wrongDomain = isAuthenticated && userEmail && !userEmail.toLowerCase().endsWith(ALLOWED_DOMAIN);

  if (loggedOut || !isAuthenticated || wrongDomain) {
    if (wrongDomain && !loggedOut) handleLogout();
    return <LoginPage />;
  }

  return (
    <>
      {idleWarning && !sessionExpired && (
        <div className="fixed inset-0 z-[9998] flex items-center justify-center bg-black/50 backdrop-blur-sm">
          <div className="bg-white rounded-2xl shadow-2xl p-8 max-w-sm w-full mx-4 flex flex-col items-center gap-5">
            <div className="w-14 h-14 rounded-full bg-blue-100 flex items-center justify-center">
              <svg className="w-7 h-7 text-[#000099]" fill="none" viewBox="0 0 24 24" stroke="currentColor" strokeWidth={2}>
                <path strokeLinecap="round" strokeLinejoin="round" d="M12 8v4l3 3m6-3a9 9 0 11-18 0 9 9 0 0118 0z" />
              </svg>
            </div>
            <div className="text-center">
              <h2 className="text-lg font-bold text-gray-900 mb-1">Still there?</h2>
              <p className="text-sm text-gray-500">
                You've been inactive. Your current session will expire in{' '}
                <span className="font-semibold text-[#000099]">
                  {Math.floor(countdown / 60)}:{String(countdown % 60).padStart(2, '0')}
                </span>
                .
              </p>
            </div>
            <button
              onClick={() => setIdleWarning(false)}
              className="w-full cursor-pointer px-4 py-2 rounded-lg bg-[#000099] text-white text-sm font-medium hover:bg-[#0000cc] transition-colors"
            >
              I'm still here
            </button>
          </div>
        </div>
      )}
      {sessionExpired && (
        <div className="fixed inset-0 z-[9999] flex items-center justify-center bg-black/50 backdrop-blur-sm">
          <div className="bg-white rounded-2xl shadow-2xl p-8 max-w-sm w-full mx-4 flex flex-col items-center gap-5">
            <div className="w-14 h-14 rounded-full bg-amber-100 flex items-center justify-center">
              <svg className="w-7 h-7 text-amber-500" fill="none" viewBox="0 0 24 24" stroke="currentColor" strokeWidth={2}>
                <path strokeLinecap="round" strokeLinejoin="round" d="M12 9v4m0 4h.01M10.29 3.86L1.82 18a2 2 0 001.71 3h16.94a2 2 0 001.71-3L13.71 3.86a2 2 0 00-3.42 0z" />
              </svg>
            </div>
            <div className="text-center">
              <h2 className="text-lg font-bold text-gray-900 mb-1">Session Expired</h2>
              <p className="text-sm text-gray-500">Your session has timed out. Please log in again to continue.</p>
            </div>
            <button
              onClick={handleLogout}
              className="w-full cursor-pointer px-4 py-2 rounded-lg bg-[#000099] text-white text-sm font-medium hover:bg-[#0000cc] transition-colors"
            >
              Re-login
            </button>
          </div>
        </div>
      )}
      <AuthenticatedApp
        onLogout={handleLogout}
        onIdleWarn={handleIdleWarn}
        onIdleExpire={handleIdleExpire}
        onIdleReset={handleIdleReset}
      />
    </>
  );
}

// ─── Shell: header + React Router routes ─────────────────────────────────────

function AuthenticatedApp({
  onLogout,
  onIdleWarn,
  onIdleExpire,
  onIdleReset,
}: {
  onLogout: () => void;
  onIdleWarn: () => void;
  onIdleExpire: () => void;
  onIdleReset: () => void;
}) {
  const { accounts, instance } = useMsal();
  const location = useLocation();
  const currentUser = accounts[0];
  const userName = currentUser?.name ?? currentUser?.username ?? '';
  const userEmail = currentUser?.username ?? '';
  const [tokenReady, setTokenReady] = useState(false);

  // Roles are injected by Azure AD into the ID token — no hardcoded lists needed.
  const idTokenClaims = currentUser?.idTokenClaims as Record<string, unknown> | undefined;
  const roles: string[] = (idTokenClaims?.roles as string[]) ?? [];
  const canViewAnalytics = roles.includes('Analytics.View') || roles.includes('view.analytics');
  const canHideInExplore = roles.some((role) => role.toLowerCase() === 'analytics.view');
  const groupClaims = Array.isArray(idTokenClaims?.groups) ? (idTokenClaims?.groups as string[]) : [];
  const [userSubDepartment, setUserSubDepartment] = useState<string | null>(null);

  const canViewApprover = Boolean(
    CONFIG.PNM_APPROVERS_GROUP_ID && groupClaims.includes(CONFIG.PNM_APPROVERS_GROUP_ID),
  );
  const canViewMyCreatives = userSubDepartment === CONFIG.PNM_SUB_DEPARTMENT;
  const isEditorRoute = location.pathname === '/editor';

  useIdleTimer({
    onWarn:   onIdleWarn,
    onExpire: onIdleExpire,
    onReset:  onIdleReset,
  });

  // Acquire / refresh ID token for all API calls.
  // We gate rendering until the token is available so that child components
  // never fire an API request with an empty Bearer token (which causes 401s
  // on page refresh before acquireTokenSilent has finished).
  useEffect(() => {
    if (!currentUser) return;
    const supplyToken = async () => {
      const tokenRequest = { scopes: ['openid', 'profile', 'email'], account: currentUser };
      try {
        const result = await instance.acquireTokenSilent(tokenRequest);
        setAuthToken(result.idToken);
        setTokenReady(true);
      } catch {
        try {
          const result = await instance.acquireTokenPopup(tokenRequest);
          setAuthToken(result.idToken);
          setTokenReady(true);
        } catch (err) {
          console.error('Failed to acquire auth token:', err);
        }
      }
    };
    void supplyToken();

    // MSAL hands back the *cached* ID token until it expires and nothing was
    // ever asking for a fresh one, so a token minted early in a browser session
    // could lapse an hour later while a job was still running. A video render
    // takes twenty minutes, which made the request that fetched the finished
    // film the one that wore the failure — the render had succeeded, and the UI
    // showed a spinner. Renew well inside the lifetime so it is never stale.
    //
    // Silent only: `acquireTokenPopup` fired from a background timer would
    // ambush someone mid-work. A genuine silent failure still falls through to
    // the 401 handler, which can prompt properly.
    const renew = setInterval(
      () => {
        instance
          .acquireTokenSilent({ scopes: ['openid', 'profile', 'email'], account: currentUser })
          .then((result) => setAuthToken(result.idToken))
          .catch(() => {
            /* Handled on the next 401. */
          });
      },
      10 * 60 * 1000,
    );
    return () => clearInterval(renew);
  }, [currentUser, instance]);

  useEffect(() => {
    if (!tokenReady) return;

    const loadCurrentUser = async () => {
      try {
        const res = await apiFetch(`${CONFIG.API_BASE_URL}/api/v1/users/me`);
        if (!res.ok) {
          setUserSubDepartment(null);
          return;
        }
        const user = (await res.json()) as { sub_department?: string | null };
        setUserSubDepartment(user.sub_department ?? null);
      } catch {
        setUserSubDepartment(null);
      }
    };

    void loadCurrentUser();
  }, [tokenReady]);

  const navLinkClass = ({ isActive }: { isActive: boolean }) =>
    `flex items-center gap-1.5 px-3 py-1.5 rounded-lg text-sm font-medium transition-colors cursor-pointer ${
      isActive
        ? 'bg-[#e6e6ff] text-[#000099]'
        : 'text-gray-600 hover:bg-gray-100 hover:text-gray-900'
    }`;

  // Wait until the access token is ready before rendering any authenticated
  // content — prevents child components from firing API calls with an empty
  // Bearer token on page refresh.
  if (!tokenReady) {
    return (
      <div className="min-h-screen flex items-center justify-center bg-gray-50">
        <div className="w-8 h-8 border-4 border-[#000099] border-t-transparent rounded-full animate-spin" />
      </div>
    );
  }

  return (
    <div className={isEditorRoute ? 'h-screen overflow-hidden' : 'min-h-screen bg-gray-50 flex flex-col font-sans'}>
      {!isEditorRoute && (
        <>
          {/* ── Approval disclaimer banner ──────────────────────────────────── */}
          <div className="w-full bg-amber-50 border-b border-amber-300 px-4 py-2 flex items-center justify-center gap-2 text-amber-800 text-xs font-medium z-20 sticky top-0">
            <svg xmlns="http://www.w3.org/2000/svg" className="w-4 h-4 shrink-0 text-amber-500" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="2" strokeLinecap="round" strokeLinejoin="round"><path d="M10.29 3.86L1.82 18a2 2 0 0 0 1.71 3h16.94a2 2 0 0 0 1.71-3L13.71 3.86a2 2 0 0 0-3.42 0z"/><line x1="12" y1="9" x2="12" y2="13"/><line x1="12" y1="17" x2="12.01" y2="17"/></svg>
            All creatives generated by this app must be reviewed and approved by the Marketing team before being used in any campaign or published externally.
          </div>
          {/* ── Global header ───────────────────────────────────────────────── */}
          <header className="bg-white border-b border-gray-200 px-6 py-4 flex items-center gap-3 shadow-sm z-10">
            <img
              src={getPublicLogoUrl()}
              alt=""
              className="h-6 w-auto max-w-[140px] object-contain rounded"
              onClick={() => window.location.href = '/projects'}
            />
            <h1 className="text-xl font-bold text-gray-900 tracking-tight cursor-pointer" onClick={() => window.location.href = '/projects'}>IndiGoBluPrint</h1>

            {/* Nav links */}
            <nav className="hidden sm:flex items-center gap-1 ml-6">
              <NavLink to="/projects" className={navLinkClass} end>
                <FolderOpen className="w-4 h-4" />
                Projects
              </NavLink>
              <NavLink to="/explore" className={navLinkClass}>
                <Compass className="w-4 h-4" />
                Explore
              </NavLink>
              <NavLink to="/6e-creatives" className={navLinkClass}>
                <Images className="w-4 h-4" />
                6ECreatives
              </NavLink>
              <NavLink to="/video" className={navLinkClass}>
                <Clapperboard className="w-4 h-4" />
                AI Video
              </NavLink>
              {canViewMyCreatives && (
                <NavLink to="/my-creatives" className={navLinkClass}>
                  <Image className="w-4 h-4" />
                  My Creatives
                </NavLink>
              )}
              <NavLink to="/editor?tab=uploads" className={navLinkClass}>
                <Crop className="w-4 h-4" />
                Editor
              </NavLink>
              {canViewApprover && (
                <NavLink to="/approver" className={navLinkClass}>
                  <ClipboardCheck className="w-4 h-4" />
                  My Approvals
                </NavLink>
              )}
              {canViewAnalytics && (
                <NavLink to="/analytics" className={navLinkClass}>
                  <TrendingUp className="w-4 h-4" />
                  Analytics
                </NavLink>
              )}
              {canViewAnalytics && (
                <NavLink to="/feedback" className={navLinkClass}>
                  <MessageSquare className="w-4 h-4" />
                  Feedbacks
                </NavLink>
              )}
            </nav>

            {/* User info + sign out */}
            <div className="ml-auto flex items-center gap-3">
              <div className="hidden sm:flex flex-col items-end leading-tight">
                <span className="text-sm font-semibold text-gray-800">{userName}</span>
                <span className="text-xs text-gray-500">{userEmail} ({userSubDepartment})</span>
              </div>
              <button
                onClick={onLogout}
                title="Sign out"
                className="flex cursor-pointer items-center gap-1.5 px-3 py-1.5 rounded-lg text-sm font-medium text-gray-600 border border-gray-200 hover:border-red-300 hover:text-red-600 hover:bg-red-50 transition-colors"
              >
                <LogOut className="w-4 h-4" />
                <span className="hidden sm:inline">Sign out</span>
              </button>
            </div>
          </header>
        </>
      )}

      {/* ── Page routes ─────────────────────────────────────────────────── */}
      <Routes>
        <Route index element={<Navigate to="/projects" replace />} />
        <Route
          path="/editor"
          element={<EditorPage />}
        />
        <Route
          path="/projects"
          element={
            <main className="flex-1 flex flex-col bg-gray-50">
              <ProjectsPage />
            </main>
          }
        />
        <Route
          path="/projects/:projectId"
          element={
            <main className="bg-gray-50">
              <StudioPage />
            </main>
          }
        />
        <Route
          path="/explore"
          element={
            <main className="flex-1 flex flex-col overflow-hidden">
              <ExplorePage canHideCreatives={canHideInExplore} />
            </main>
          }
        />
        <Route
          path="/6e-creatives"
          element={
            <main className="flex-1 flex flex-col overflow-hidden">
              <SixECreativesPage />
            </main>
          }
        />
        <Route
          path="/video"
          element={
            <main className="flex-1 flex flex-col bg-gray-50">
              <VideoPage />
            </main>
          }
        />
        <Route
          path="/my-creatives"
          element={
            canViewMyCreatives ? (
              <main className="flex-1 flex flex-col bg-gray-50">
                <MyCreativesPage />
              </main>
            ) : (
              <Navigate to="/projects" replace />
            )
          }
        />
        <Route
          path="/approver"
          element={
            canViewApprover ? (
              <main className="flex-1 flex flex-col bg-gray-50">
                <ApproverCreativesPage />
              </main>
            ) : (
              <Navigate to="/projects" replace />
            )
          }
        />
        <Route
          path="/analytics"
          element={
            canViewAnalytics ? (
              <main className="flex-1 flex flex-col bg-gray-50">
                <AnalyticsPage />
              </main>
            ) : (
              <Navigate to="/projects" replace />
            )
          }
        />
        <Route
          path="/feedback"
          element={
            canViewAnalytics ? (
              <main className="flex-1 flex flex-col bg-gray-50">
                <FeedbackPage />
              </main>
            ) : (
              <Navigate to="/projects" replace />
            )
          }
        />
        <Route path="*" element={<Navigate to="/projects" replace />} />
      </Routes>
    </div>
  );
}
