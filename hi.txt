import pyodbc
from dataclasses import dataclass, field
from typing import Optional
from datetime import date

DB_CONN_STR = (
    "DRIVER={ODBC Driver 18 for SQL Server};"
    "SERVER=zpmi1-ea-pdb01.4166029f8cdd.database.windows.net,1433;"
    "DATABASE=CLMS;"
    "Authentication=ActiveDirectoryInteractive;"  # Use Azure AD / service principal in prod
    "Encrypt=yes;"
)

@dataclass
class CrewURTIContext:
    crew_id: str
    crew_name: str
    base: str
    email: str
    sl_balance: float
    urti_balance: float
    lwp_balance: float
    leave_year_id: int
    leave_year_name: str
    is_prev_bucket: bool
    prev_bucket_urti_balance: Optional[float]
    existing_sl_entries: list = field(default_factory=list)
    is_lwp_case: bool = False


def read_crew_by_iga_code(
    iga_code: str,
    from_date: date,
    to_date: date
) -> CrewURTIContext:
    """
    MCP Tool: Fetch crew leave context from CLMS by IGA code.
    Returns all info needed for URTI leave update decision.
    """
    conn = pyodbc.connect(DB_CONN_STR)
    cursor = conn.cursor()

    # 1. Fetch crew master info
    cursor.execute("""
        SELECT CrewID, CrewName, Base, Email, Active, Status
        FROM M_CLMS_Crew
        WHERE CrewID = ?  -- IGA code IS the CrewID in CLMS
        AND Active = 1
    """, (iga_code,))
    crew = cursor.fetchone()
    if not crew:
        raise ValueError(f"No active crew found for IGA code: {iga_code}")

    crew_id, crew_name, base, email = crew.CrewID, crew.CrewName, crew.Base, crew.Email

    # 2. Identify current leave year
    cursor.execute("""
        SELECT LeaveYearId, LeaveYearName, LeaveYearStartDate, LeaveYearEndDate
        FROM M_LeaveYear
        WHERE IsActive = 1 AND IsVisible = 1
    """)
    curr_year = cursor.fetchone()
    leave_year_id = curr_year.LeaveYearId

    # 3. Resolve LeaveTypeIDs for SL, URTI, LWP
    cursor.execute("""
        SELECT LeaveTypeID, LeaveType FROM M_LMS_LeaveType
        WHERE LeaveType IN ('SL', 'URTI', 'LWP')
    """)
    leave_type_map = {row.LeaveType: row.LeaveTypeID for row in cursor.fetchall()}
    sl_type_id   = leave_type_map.get('SL')
    urti_type_id = leave_type_map.get('URTI')
    lwp_type_id  = leave_type_map.get('LWP')

    # 4. Fetch current leave balances (SL, URTI, LWP)
    def get_balance(leave_type_id, year_id):
        cursor.execute("""
            SELECT Balance FROM M_CLMS_LeaveBalance
            WHERE CrewID = ? AND LeaveTypeID = ? AND LeaveYearId = ?
        """, (crew_id, leave_type_id, year_id))
        row = cursor.fetchone()
        return row.Balance if row else 0.0

    sl_balance   = get_balance(sl_type_id, leave_year_id)
    urti_balance = get_balance(urti_type_id, leave_year_id)
    lwp_balance  = get_balance(lwp_type_id, leave_year_id)

    # 5. Check if URTI dates fall in a PREVIOUS leave bucket
    is_prev_bucket = from_date < curr_year.LeaveYearStartDate.date()
    prev_bucket_urti_balance = None
    if is_prev_bucket:
        cursor.execute("""
            SELECT LeaveYearId FROM M_LeaveYear
            WHERE LeaveYearStartDate <= ? AND LeaveYearEndDate >= ?
        """, (from_date, from_date))
        prev_year = cursor.fetchone()
        if prev_year:
            prev_bucket_urti_balance = get_balance(urti_type_id, prev_year.LeaveYearId)

    # 6. Fetch existing SL/LWP leave requests for the URTI date range
    cursor.execute("""
        SELECT lrm.LeaveDetailID, lrm.LeaveTypeid, lrm.StatusTypeID,
               lrm.FromDT, lrm.ToDt, lrm.GndCode,
               lrd.NoOfLeaves
        FROM T_CLMS_LeaveReqMaster lrm
        JOIN T_CLMS_LeaveReqDetail lrd ON lrm.LeaveDetailID = lrd.LeaveDetailID
        WHERE lrm.CrewID = ?
          AND lrm.LeaveTypeid IN (?, ?)
          AND lrm.FromDT <= ? AND lrm.ToDt >= ?
          AND lrm.StatusTypeID NOT IN (
              SELECT StatusTypeID FROM M_LMS_StatusType WHERE StatusType = 'Revoked'
          )
        ORDER BY lrm.FromDT
    """, (crew_id, sl_type_id, lwp_type_id, to_date, from_date))
    existing_sl_entries = cursor.fetchall()

    is_lwp_case = any(row.LeaveTypeid == lwp_type_id for row in existing_sl_entries)

    conn.close()

    return CrewURTIContext(
        crew_id=crew_id,
        crew_name=crew_name,
        base=base,
        email=email,
        sl_balance=sl_balance,
        urti_balance=urti_balance,
        lwp_balance=lwp_balance,
        leave_year_id=leave_year_id,
        leave_year_name=curr_year.LeaveYearName,
        is_prev_bucket=is_prev_bucket,
        prev_bucket_urti_balance=prev_bucket_urti_balance,
        existing_sl_entries=existing_sl_entries,
        is_lwp_case=is_lwp_case,
    )
DB Update Method — How It Works
The DB is on Azure SQL Server, so updates go via pyodbc with parameterized SQL transactions. The key principle: wrap all changes in a single atomic transaction so a mid-way failure doesn't corrupt balances.

def update_urti_in_clms(ctx: CrewURTIContext, from_date: date, to_date: date, no_of_days: int, approved_by: str):
    """
    Executes the URTI update in CLMS DB inside a single transaction.
    ONLY called after human approval is confirmed.
    """
    conn = pyodbc.connect(DB_CONN_STR)
    cursor = conn.cursor()

    try:
        conn.autocommit = False  # Begin transaction

        # Resolve IDs
        cursor.execute("SELECT LeaveTypeID FROM M_LMS_LeaveType WHERE LeaveType='SL'")
        sl_type_id = cursor.fetchone().LeaveTypeID
        cursor.execute("SELECT LeaveTypeID FROM M_LMS_LeaveType WHERE LeaveType='URTI'")
        urti_type_id = cursor.fetchone().LeaveTypeID
        cursor.execute("SELECT StatusTypeID FROM M_LMS_StatusType WHERE StatusType='Revoked'")
        revoked_status_id = cursor.fetchone().StatusTypeID
        cursor.execute("SELECT StatusTypeID FROM M_LMS_StatusType WHERE StatusType='Approved'")
        approved_status_id = cursor.fetchone().StatusTypeID

        # STEP 1: Revoke existing SL/LWP entries for the date range
        for entry in ctx.existing_sl_entries:
            cursor.execute("""
                UPDATE T_CLMS_LeaveReqMaster
                SET StatusTypeID = ?, ActionBy = ?, ActionDate = GETDATE(), Remarks = 'Revoked - URTI Applied'
                WHERE LeaveDetailID = ?
            """, (revoked_status_id, approved_by, entry.LeaveDetailID))

        # STEP 2: Credit SL balance back (only SL entries, not LWP)
        sl_days_to_credit = sum(
            e.NoOfLeaves for e in ctx.existing_sl_entries
            if e.LeaveTypeid == sl_type_id
        )
        if sl_days_to_credit > 0:
            cursor.execute("""
                UPDATE M_CLMS_LeaveBalance
                SET Balance = Balance + ?
                WHERE CrewID = ? AND LeaveTypeID = ? AND LeaveYearId = ?
            """, (sl_days_to_credit, ctx.crew_id, sl_type_id, ctx.leave_year_id))

        # STEP 3: Insert new URTI leave request (Master + Detail)
        cursor.execute("""
            INSERT INTO T_CLMS_LeaveReqMaster_OtherRocadeCodes
            (CrewID, RequestedDate, LeaveTypeid, StatusTypeID, FromDT, ToDt, Remarks, ActionBy, ActionDate, CreatedDtm, GndCode)
            VALUES (?, GETDATE(), ?, ?, ?, ?, 'URTI Applied via Agentic Flow', ?, GETDATE(), GETDATE(), 'URTI')
        """, (ctx.crew_id, urti_type_id, approved_status_id, from_date, to_date, approved_by))

        cursor.execute("SELECT SCOPE_IDENTITY() AS new_id")
        new_leave_detail_id = cursor.fetchone().new_id

        cursor.execute("""
            INSERT INTO T_CLMS_LeaveReqDetail_OtherRocadeCodes
            (LeaveDetailID, LeaveTypeID, NoOfLeaves, Balance, CurrentLeave, FromDt, ToDt)
            VALUES (?, ?, ?, ?, ?, ?, ?)
        """, (new_leave_detail_id, urti_type_id, no_of_days,
              ctx.urti_balance - no_of_days, no_of_days, from_date, to_date))

        # STEP 4: Deduct URTI balance
        # If prev bucket → deduct from prev bucket and REVERSE any current-bucket deduction
        target_year_id = ctx.leave_year_id
        if ctx.is_prev_bucket:
            cursor.execute("""
                SELECT LeaveYearId FROM M_LeaveYear
                WHERE LeaveYearStartDate <= ? AND LeaveYearEndDate >= ?
            """, (from_date, from_date))
            prev_year = cursor.fetchone()
            if prev_year:
                target_year_id = prev_year.LeaveYearId
                # Reverse any current bucket deduction if already done
                cursor.execute("""
                    UPDATE M_CLMS_LeaveBalance
                    SET Balance = Balance + ?
                    WHERE CrewID = ? AND LeaveTypeID = ? AND LeaveYearId = ?
                """, (no_of_days, ctx.crew_id, urti_type_id, ctx.leave_year_id))

        cursor.execute("""
            UPDATE M_CLMS_LeaveBalance
            SET Balance = Balance - ?
            WHERE CrewID = ? AND LeaveTypeID = ? AND LeaveYearId = ?
        """, (no_of_days, ctx.crew_id, urti_type_id, target_year_id))

        # STEP 5: Log the balance change in audit table
        cursor.execute("""
            INSERT INTO CrewBalanceUpdateByScheduler
            (crewID, OldSLBalance, updatedSLBalance, OldURTIBalance, updatedURTIBalance,
             IsSchedulerUpdate, LeaveYearID, UpdatedTime, IsRevoked, CreatedDtm)
            VALUES (?, ?, ?, ?, ?, 0, ?, GETDATE(), 0, GETDATE())
        """, (
            ctx.crew_id,
            ctx.sl_balance, ctx.sl_balance + sl_days_to_credit,
            ctx.urti_balance, ctx.urti_balance - no_of_days,
            target_year_id
        ))

        # STEP 6: LWP Special Case → Insert bonus date flag for Payroll
        if ctx.is_lwp_case:
            cursor.execute("""
                INSERT INTO M_CLMS_BonusDate (crewid, bonusdate)
                VALUES (?, GETDATE())
            """, (ctx.crew_id,))
            # Payroll notification handled separately by Agent 2

        conn.commit()
        print(f"[✓] URTI update committed for crew: {ctx.crew_name} ({ctx.crew_id})")
        return {"status": "success", "lwp_case": ctx.is_lwp_case}

    except Exception as e:
        conn.rollback()
        print(f"[✗] Transaction rolled back: {e}")
        raise
    finally:
        conn.close()
Human-in-the-Loop Approval Gate ⚠️
This is the critical checkpoint between read and write. The agent pauses and requires explicit confirmation:

def human_approval_gate(ctx: CrewURTIContext, from_date: date, to_date: date, no_of_days: int) -> bool:
    """
    Presents a warning summary to the human operator.
    Returns True only on explicit 'APPROVE' input.
    """
    lwp_warning = ""
    if ctx.is_lwp_case:
        lwp_warning = "\n  ⚠️  LWP CASE DETECTED — Crew will become eligible for ARREARS & EFFICIENCY BONUS. Payroll team will be notified."

    bucket_warning = ""
    if ctx.is_prev_bucket:
        bucket_warning = f"\n  ⚠️  PREV BUCKET DATES — URTI balance in prior bucket: {ctx.prev_bucket_urti_balance} days. Current bucket deduction will be REVERSED."

    warning_msg = f"""
╔══════════════════════════════════════════════════════════════╗
║              ⚠️  URTI LEAVE UPDATE — HUMAN APPROVAL REQUIRED  ║
╚══════════════════════════════════════════════════════════════╝

  Crew       : {ctx.crew_name}  (IGA: {ctx.crew_id})
  Base       : {ctx.base}
  URTI Dates : {from_date} → {to_date}  ({no_of_days} days)

  Current Balances:
    → SL    : {ctx.sl_balance} days   (will be credited back: +{sum(e.NoOfLeaves for e in ctx.existing_sl_entries)} days)
    → URTI  : {ctx.urti_balance} days  (will be deducted: -{no_of_days} days)
    → LWP   : {ctx.lwp_balance} days
{bucket_warning}{lwp_warning}

  Type APPROVE to confirm or REJECT to cancel:
    """
    print(warning_msg)
    user_input = input("Your Decision: ").strip().upper()
    return user_input == "APPROVE"

import smtplib
from email.mime.text import MIMEText

def notify_crew_and_base(ctx: CrewURTIContext, from_date: date, to_date: date, no_of_days: int):
    """Agent 2: Send email notification to crew + base after successful URTI update."""
    conn = pyodbc.connect(DB_CONN_STR)
    cursor = conn.cursor()

    cursor.execute("SELECT Email FROM M_CLMS_BaseEmail WHERE Base = ? AND Active = 1", (ctx.base,))
    base_email = cursor.fetchone()
    cursor.execute("SELECT Email FROM M_CLMS_BaseMgrEmail WHERE Base = ? AND Active = 1", (ctx.base,))
    base_mgr_email = cursor.fetchone()
    conn.close()

    recipients = [ctx.email]
    cc = []
    if base_email: cc.append(base_email.Email)
    if base_mgr_email: cc.append(base_mgr_email.Email)

    lwp_note = ""
    if ctx.is_lwp_case:
        lwp_note = "\n\nNote: Your LWP marking has been reversed. You are now eligible for arrears and efficiency bonus. The Payroll team has been informed."

    body = f"""Dear {ctx.crew_name},

Your URTI leave has been successfully updated in CLMS.

Details:
  IGA Code   : {ctx.crew_id}
  From Date  : {from_date}
  To Date    : {to_date}
  No of Days : {no_of_days}
  
Your Sick Leave (SL) for the above period has been credited back and replaced with URTI code.
Please follow up on CLMS portal after 7 working days to confirm the update.{lwp_note}

Regards,
Crew Leave Management System (Automated)
"""
    msg = MIMEText(body)
    msg['Subject'] = f"URTI Leave Updated — {ctx.crew_name} ({ctx.crew_id})"
    msg['From'] = "clms-noreply@goindigo.in"
    msg['To'] = ", ".join(recipients)
    msg['Cc'] = ", ".join(cc)

    with smtplib.SMTP("smtp.goindigo.in", 587) as server:
        server.starttls()
        server.login("clms-service-account", "***")
        server.sendmail(msg['From'], recipients + cc, msg.as_string())

    print(f"[✓] Notification sent to {ctx.crew_name} ({ctx.email}) and base {ctx.base}")

def urti_agentic_flow(mail_payload: dict):
    """
    Full flow: Agent1 (read + approve + update) → Agent2 (notify)
    mail_payload = { "iga": "12345", "crew_name": "...", "from_date": date(...), "to_date": date(...), "no_of_days": 3 }
    """
    iga       = mail_payload["iga"]
    from_date = mail_payload["from_date"]
    to_date   = mail_payload["to_date"]
    no_of_days= mail_payload["no_of_days"]

    # AGENT 1 — Read context
    ctx = read_crew_by_iga_code(iga, from_date, to_date)

    # Validation checks
    if ctx.urti_balance < no_of_days:
        print(f"[!] Insufficient URTI balance ({ctx.urti_balance}) for {no_of_days} days. Halting.")
        return

    # Human Approval Gate ← MANDATORY
    approved = human_approval_gate(ctx, from_date, to_date, no_of_days)
    if not approved:
        print("[✗] Update REJECTED by operator. No changes made.")
        return

    # DB Update
    result = update_urti_in_clms(ctx, from_date, to_date, no_of_days, approved_by="operator@goindigo.in")

    # AGENT 2 — Notify crew + base
    if result["status"] == "success":
        notify_crew_and_base(ctx, from_date, to_date, no_of_days)
Key Design Decisions Summary
DB Update Method: Parameterized SQL via pyodbc inside an atomic transaction (autocommit=False + commit/rollback) — this is the safest approach for Azure SQL and ensures no partial balance corruption
​

Human Gate: Blocks the transaction until explicit APPROVE — satisfies the HITL requirement from your doc
​

Bucket Logic: Automatically detects if URTI dates are in a prior leave year and handles both bucket deduction + current-bucket reversal via M_LeaveYear

LWP Case: Detected from existing leave request entries, triggers M_CLMS_BonusDate insert and flags payroll in notification
​

Audit Trail: Every balance change is logged in CrewBalanceUpdateByScheduler with old/new values
