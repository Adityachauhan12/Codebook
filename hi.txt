"""Brand guardrails.

Brand is a separate axis from craft. ``app/skills`` teaches the model how to
direct a film; this package tells it whose film it is, and refuses the result
when it is off-brand.

Everything routes through ``BRAND`` in the environment, so the pipeline is not
hardwired to one client — set ``BRAND=none`` and every hook below turns into a
no-op, leaving the craft layer untouched.
"""

from __future__ import annotations

from ..config import get_settings
from ..domain.models import AuditIssue, StoryOption
from .checks import brand_issues as _indigo_issues
from .policy import (
    INDIGO_BEAT_LABELS,
    INDIGO_BRAND_POLICY,
    INDIGO_BRAND_SURFACE,
    INDIGO_CONTINUITY_REQUIREMENTS,
    INDIGO_NEGATIVE_PROMPT,
    INDIGO_VIDEO_BRAND_POLICY,
    INDIGO_VIDEO_SAFETY_POLICY,
)

__all__ = [
    "active_brand",
    "brand_name",
    "brand_system_block",
    "brand_brief_block",
    "brand_negative_prompt",
    "brand_beat_label",
    "brand_issues",
]


def active_brand() -> str | None:
    """The configured brand id, or ``None`` when brand enforcement is off."""
    value = (get_settings().brand or "").strip().lower()
    return value if value and value not in {"none", "off", "false", "generic"} else None


def brand_name() -> str:
    return "IndiGo" if active_brand() == "indigo" else ""


def brand_system_block(*, safety_only: bool = False) -> str:
    """The policy text to append to a task's system prompt.

    ``safety_only`` is for paths driven by a raw user instruction, where the full
    styling document would fight the thing the user just asked for, but the
    content prohibitions still hold absolutely.
    """
    if active_brand() != "indigo":
        return ""

    if safety_only:
        return "\n".join(
            ["---", "", "# Brand safety — IndiGo", "", INDIGO_VIDEO_SAFETY_POLICY.strip()]
        )

    return "\n".join(
        [
            "---",
            "",
            "# Brand guardrails — IndiGo",
            "",
            "This film is for IndiGo. The rules below are not style preferences; they are the "
            "brand book, and they override any conflicting instinct from the craft references "
            "above. A technically excellent film that breaks them is a failed film.",
            "",
            INDIGO_BRAND_POLICY.strip(),
            "",
            INDIGO_VIDEO_BRAND_POLICY.strip(),
        ]
    )


def brand_brief_block() -> str:
    """Concrete brand requirements for the *user* message.

    The system prompt carries the policy; this carries the values the model has
    to physically write into the continuity block and every scene prompt.
    """
    if active_brand() != "indigo":
        return ""

    return "\n".join(
        [
            "## Brand — IndiGo (non-negotiable)",
            "",
            "This is an advertisement for IndiGo. Its job is to make the viewer feel good about "
            "flying IndiGo. A beautiful film that could be for any airline has failed the brief.",
            "",
            INDIGO_CONTINUITY_REQUIREMENTS.strip(),
            "",
            "### Making IndiGo visible without the logo",
            INDIGO_BRAND_SURFACE.strip(),
            "",
            "### Emotional register",
            "The beat map below is written for drama, but this is a brand film. Keep the dramatic "
            "shape and change the colour: the obstacle is never IndiGo and never flying — flying "
            "is what RESOLVES the story. 'Pressure' means anticipation, not anxiety. The 'crack' "
            "is a moment of warmth or recognition, not a breakdown. Nobody is frightened, "
            "panicked, stressed, or overwhelmed. Longing, anticipation, relief and delight: yes.",
            "",
            "Four failures are automatically rejected before this reaches Veo:",
            "1. Any request to render the logo, wordmark, brand name, or ANY on-screen text. "
            "Veo cannot spell or draw a mark — it comes back wrong every time. The mark is "
            "composited from the master file after the render. Keep the plate clean.",
            "2. Any scene whose palette is not blue-dominant, or that uses red or purple.",
            "3. Any scene carrying no IndiGo surface at all — no cabin, crew, aircraft form, "
            "or blue brand object.",
            "4. Any scene that plays as fear, stress, or anxiety, or that blames the journey "
            "(delays, queues, lost bags, turbulence, cramped cabins).",
        ]
    )


def brand_negative_prompt() -> str:
    """Brand terms appended to every scene's negative prompt."""
    return INDIGO_NEGATIVE_PROMPT if active_brand() == "indigo" else ""


def brand_beat_label(beat: str) -> str | None:
    """The brand's reading of a dramatic beat, or ``None`` to use the craft default.

    The craft beat map is written for short drama — its own `hook` label is
    "Hero already in tension. Problem on screen." Fed to the writer scene by
    scene, that is what puts a frightened woman in an airport. The beat
    *structure* is kept; only its emotional colour is rebranded.
    """
    return INDIGO_BEAT_LABELS.get(beat) if active_brand() == "indigo" else None


def brand_issues(story: StoryOption) -> list[AuditIssue]:
    """Brand compliance issues for one story, or none when brand is off."""
    return _indigo_issues(story) if active_brand() == "indigo" else []
