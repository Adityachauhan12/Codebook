#!/usr/bin/env python3
"""
Parent orchestrator graph: integrates the GTM and Metrics agents as SUBGRAPHS
over one shared CampaignState blackboard, with a real cross-agent feedback loop.

Outer flow:
    START -> intake -> gtm -> metrics -> route
        route: "revise" -> revise -> gtm   (Metrics bounced strategy back to GTM)
               "done"   -> END

Two levels of loops:
    * INNER  (inside each subgraph): the agent fixes ITSELF
        - GTM:     strategize <-> validate
        - Metrics: gtm_alignment <-> negotiation
    * OUTER  (this graph): Metrics tells GTM to re-strategize when the channel
        mix doesn't match the economics (bounded by MAX_REVISIONS).

Future agents (Finance, Creative) drop in as new nodes after "metrics" with no
rewrite — they just read/write their own keys on the blackboard.

Run:
    python3 campaign_graph.py --input indigo_flymore_input.json
"""
from __future__ import annotations

import argparse
import json
import os
import sys
from pathlib import Path

from dotenv import load_dotenv
load_dotenv()

from langgraph.graph import StateGraph, START, END

from campaign_state import CampaignState, objective_to_key
from campaign_memory import InMemoryStore, SQLiteMemory, DEFAULT_DB_PATH
from gtm_agent import build_gtm_subgraph, load_skill_prompt, SKILL_FILE
from metrics_agent import build_metrics_subgraph, build_performance_feedback_graph

MAX_REVISIONS = 1  # how many times Metrics may bounce the plan back to GTM


# --------------------------------------------------------------------------- #
# Parent nodes
# --------------------------------------------------------------------------- #
def make_intake(memory):
    """Orchestrator intake: normalise the brief and seed the blackboard.
    Returns a node (closure) so the chosen MemoryStore rides along in state."""

    def intake(state: CampaignState) -> dict:
        brief = state["brief"]
        grounded = brief.get("grounded", brief)
        assumed = brief.get("assumed", {})

        obj_raw = grounded.get("objective") or assumed.get("objective") or ""
        budget = (assumed.get("budget_inr") or grounded.get("budget_inr")
                  or grounded.get("budget") or 0)
        product = grounded.get("product") or "campaign"

        return {
            "objective_raw": obj_raw,
            "objective": objective_to_key(obj_raw),
            "total_budget": float(budget),
            "product": product,
            "system_prompt": load_skill_prompt(state.get("skill_path", SKILL_FILE)),
            "memory": memory,
            # init counters
            "gtm_attempts": 0,
            "gtm_errors": [],
            "negotiation_round": 0,
            "revision_count": 0,
            "feedback": [],
        }

    return intake


def revise(state: CampaignState) -> dict:
    """Cross-agent feedback node: bump the revision counter and reset GTM's
    internal loop so it gets a fresh attempt budget. feedback[] is left in place
    for strategize() to read."""
    return {
        "revision_count": state.get("revision_count", 0) + 1,
        "gtm_attempts": 0,
        "gtm_errors": [],
        "negotiation_round": 0,
    }


def route_after_metrics(state: CampaignState) -> str:
    if state.get("feedback") and state.get("revision_count", 0) < MAX_REVISIONS:
        return "revise"
    return "done"


# --------------------------------------------------------------------------- #
# Build
# --------------------------------------------------------------------------- #
def build_campaign_graph(memory, gtm_subgraph=None):
    # gtm_subgraph is injectable so tests can swap in a deterministic fake GTM
    # (no LLM). Production passes nothing and gets the real LLM-backed subgraph.
    g = StateGraph(CampaignState)
    g.add_node("intake", make_intake(memory))
    g.add_node("gtm", gtm_subgraph or build_gtm_subgraph())  # subgraph as a node
    g.add_node("metrics", build_metrics_subgraph())          # subgraph as a node
    g.add_node("revise", revise)

    g.add_edge(START, "intake")
    g.add_edge("intake", "gtm")
    g.add_edge("gtm", "metrics")
    g.add_conditional_edges("metrics", route_after_metrics,
                            {"revise": "revise", "done": END})
    g.add_edge("revise", "gtm")
    return g.compile()


def run(brief: dict, memory=None, skill_path: str = SKILL_FILE,
        gtm_subgraph=None) -> CampaignState:
    memory = memory or InMemoryStore()
    app = build_campaign_graph(memory, gtm_subgraph=gtm_subgraph)
    return app.invoke({"brief": brief, "skill_path": skill_path})


def studio_graph():
    """Factory for LangGraph Studio (`langgraph dev`). Uses a DURABLE SQLite
    store seeded with priors, so live runs in Studio both demonstrate the
    prior->learned path AND persist what they learn across restarts.
    Referenced by langgraph.json."""
    import sys
    sys.path.insert(0, str(Path(__file__).resolve().parent / "tests"))
    from seed_data import seeded_sqlite_store
    return build_campaign_graph(seeded_sqlite_store())


# --------------------------------------------------------------------------- #
# Flow logging (via LangGraph streaming — no changes to node code)
# --------------------------------------------------------------------------- #
def _trace_label(namespace: tuple, node: str) -> str:
    """Turn the stream namespace into 'gtm/strategize', 'metrics/kpi_selector'."""
    if namespace:
        parent = namespace[0].split(":")[0]  # 'gtm:<uuid>' -> 'gtm'
        return f"{parent}/{node}"
    return node


def _trace_summary(node: str, d: dict) -> str:
    """One-line summary of what a node produced (reads its state delta)."""
    if not d:
        return "(no change — skipped)"
    if node == "intake":
        return f"objective={d.get('objective')}  budget=Rs {d.get('total_budget'):,.0f}"
    if node == "classify":
        return f"archetype={d.get('archetype')!r}  ({d.get('archetype_reason','')[:40]})"
    if node == "strategize":
        return f"strategy YAML generated  (gtm_attempt={d.get('gtm_attempts')})"
    if node == "validate":
        ok = d.get("gtm_valid")
        errs = d.get("gtm_errors") or []
        return f"valid={ok}  errors={len(errs)}" + (f" {errs}" if errs else "")
    if node == "parse":
        chs = d.get("channels") or []
        return "channels=[" + ", ".join(f"{c['name']}:{c['weight']}" for c in chs) + "]"
    if node == "kpi_selector":
        return f"primary_kpi={d.get('primary_kpi')}"
    if node == "manual_target_setter":
        return "manual (benchmark) targets set"
    if node == "historical_data_loader":
        return f"matched {len(d.get('historical_performance') or [])} history rows"
    if node == "learning_target":
        learned = d.get("learned_targets") or {}
        return "learned global targets " + (str(learned) if learned else "(none — cold start)")
    if node == "benchmark_checker":
        return f"benchmarks loaded for {len(d.get('benchmark_data') or {})} channels"
    if node == "channel_target":
        ct = d.get("channel_targets") or {}
        confs = {t.get("confidence") for t in ct.values()}
        return f"{len(ct)} channel targets set  confidence={confs}"
    if node == "gtm_alignment":
        iss = d.get("metrics_issues") or []
        strat = sum(1 for i in iss if i["level"] == "strategy")
        targ = sum(1 for i in iss if i["level"] == "target")
        return f"issues: {strat} strategy, {targ} target"
    if node == "negotiation":
        return f"negotiated target issues (round {d.get('negotiation_round')})"
    if node == "final_decision":
        return f"alignment={d.get('alignment_status')}  feedback={len(d.get('feedback') or [])}"
    if node == "revise":
        return f">>> BOUNCE back to GTM (revision {d.get('revision_count')})"
    # fallback
    return f"updated: {list(d.keys())}"


def run_logged(brief: dict, memory=None, skill_path: str = SKILL_FILE,
               gtm_subgraph=None, log=print) -> CampaignState:
    """Run the full graph while logging every node as it fires (the live flow)."""
    memory = memory or InMemoryStore()
    app = build_campaign_graph(memory, gtm_subgraph=gtm_subgraph)
    final: dict = {}
    step = 0
    for namespace, update in app.stream(
        {"brief": brief, "skill_path": skill_path},
        stream_mode="updates", subgraphs=True,
    ):
        for node, delta in update.items():
            # Skip the top-level aggregate emission of each subgraph node
            # (namespace empty + node 'gtm'/'metrics'); we already log the
            # subgraph's INTERNAL nodes via their namespace.
            if not namespace and node in ("gtm", "metrics"):
                if delta:
                    final.update(delta)
                continue
            step += 1
            log(f"  {step:>2}. {_trace_label(namespace, node):<26} {_trace_summary(node, delta)}")
            if delta:
                final.update(delta)
    return final


# --------------------------------------------------------------------------- #
# Pretty printer
# --------------------------------------------------------------------------- #
def _report(final: CampaignState) -> None:
    print("=" * 64)
    print("CAMPAIGN PLAN")
    print("=" * 64)
    print(f"Product   : {final.get('product')}")
    print(f"Objective : {final.get('objective_raw')}  ->  [{final.get('objective')}]")
    print(f"Budget    : Rs {final.get('total_budget'):,.0f}")
    print(f"Archetype : {final.get('archetype')}  ({final.get('archetype_reason')})")
    print(f"Revisions : {final.get('revision_count')}  (GTM<->Metrics bounces)")

    print("\n-- GTM: channels --")
    for c in final.get("channels", []):
        print(f"  {c['name']:<16} weight={c['weight']:>3}  priority={c['priority']}")

    print(f"\n-- Metrics: primary KPI = {final.get('primary_kpi')} --")
    for ch, t in final.get("channel_targets", {}).items():
        print(f"  {ch:<16} ROAS={t.get('target_roas')}  CPA={t.get('target_cpa')}  "
              f"CTR={t.get('target_ctr')}  [{t.get('confidence')}]")

    print(f"\n-- Alignment: {final.get('alignment_status')} --")
    print(f"  {final.get('alignment_message')}")
    if final.get("feedback"):
        print("  Outstanding feedback to GTM:")
        for f in final["feedback"]:
            print(f"    - {f}")


# --------------------------------------------------------------------------- #
# CLI
# --------------------------------------------------------------------------- #
def main() -> None:
    ap = argparse.ArgumentParser()
    ap.add_argument("--input", required=True, help="Campaign brief JSON")
    ap.add_argument("--skill", default=SKILL_FILE, help="Path to gtm-agent.md")
    ap.add_argument("--demo-feedback", action="store_true",
                    help="Also run the post-launch performance loop with sample data")
    ap.add_argument("--db", default=DEFAULT_DB_PATH,
                    help="SQLite file for DURABLE memory (persists learnings "
                         "across runs). Default: %(default)s")
    ap.add_argument("--seed", action="store_true",
                    help="Seed the DB with cold-start KPI priors if it is empty")
    ap.add_argument("--cold", action="store_true",
                    help="Use ephemeral in-RAM memory instead of SQLite "
                         "(nothing persists — the old default)")
    args = ap.parse_args()

    if not os.environ.get("OPENAI_API_KEY"):
        sys.exit("Set OPENAI_API_KEY (the GTM subgraph needs the LLM).")

    if args.cold:
        memory = InMemoryStore()  # ephemeral: nothing survives the process
    elif args.seed:
        sys.path.insert(0, str(Path(__file__).resolve().parent / "tests"))
        from seed_data import seeded_sqlite_store
        memory = seeded_sqlite_store(args.db)
    else:
        memory = SQLiteMemory(args.db)  # durable; starts empty unless --seed
    print(f"[memory] {type(memory).__name__}"
          + (f"  ->  {args.db}" if not args.cold else "  (ephemeral)")
          + f"  | {len(memory.history)} rows at start")

    brief = json.loads(Path(args.input).read_text(encoding="utf-8"))
    final = run(brief, memory=memory, skill_path=args.skill)
    _report(final)

    if args.demo_feedback:
        print("\n" + "=" * 64)
        print("POST-LAUNCH FEEDBACK LOOP (sample live data)")
        print("=" * 64)
        # Fabricate live numbers for whatever channels GTM picked.
        live = {}
        for c in final.get("channels", []):
            t = final["channel_targets"].get(c["name"], {})
            live[c["name"]] = {
                "spend": 10000,
                "roas": (t.get("target_roas") or 3) * 1.1,   # beat ROAS
                "cpa": (t.get("target_cpa") or 500) * 0.9,   # beat CPA
                "ctr": (t.get("target_ctr") or 3) * 1.1,
                "conversion_rate": (t.get("target_conversion_rate") or 3) * 1.1,
                "conversions": 50,
            }
        fb_graph = build_performance_feedback_graph()
        fb = fb_graph.invoke({**final, "live_performance": live, "memory": memory})
        for r in fb["channel_ranking"]:
            print(f"  {r['channel']:<16} score={r['score']:>3}  {r['status']}")
        print(f"\n  Memory now holds {len(memory.history)} learned rows "
              f"(next campaign will use them).")


if __name__ == "__main__":
    main()

#!/usr/bin/env python3
"""
GTM agent as a LangGraph SUBGRAPH operating on the shared CampaignState.

Self-contained: the LLM + prompt helpers (previously in gtm_graph.py) are inlined
here so the package has no external dependency on that module.

Internal loop (self-correction):   classify -> strategize <-> validate -> parse
Cross-agent loop:  strategize ALSO reads state["feedback"] (recommendations sent
                   back by the Metrics agent), so a Metrics->GTM bounce makes GTM
                   re-strategize with that feedback in the prompt.
"""
from __future__ import annotations

import json
import os
import re
from pathlib import Path

import yaml
from langchain_openai import ChatOpenAI
from langchain_core.messages import HumanMessage, SystemMessage
from langgraph.graph import StateGraph, START, END

from campaign_state import (
    CampaignState,
    normalize_channel_name,
    weight_to_priority,
)

MODEL = os.environ.get("OPENAI_MODEL", "gpt-4o-mini")
SKILL_FILE = "gtm-agent.md"
MAX_ATTEMPTS = 2  # GTM's internal retries on structural validation failure


def _llm() -> ChatOpenAI:
    # ChatOpenAI reads OPENAI_API_KEY from the environment automatically.
    # max_retries lets the OpenAI SDK honor 429 `retry-after` headers and pause
    # automatically — this paces us under the rate limit so a burst of
    # classify+strategize+revision calls doesn't hard-fail.
    return ChatOpenAI(model=MODEL, max_tokens=2000, max_retries=6)


def _strip_fences(text: str) -> str:
    """Strip a leading/trailing ```yaml ... ``` markdown fence from LLM output."""
    text = text.strip()
    if text.startswith("```"):
        text = re.sub(r"^```[a-zA-Z]*\n", "", text)
        text = re.sub(r"\n```$", "", text)
    return text.strip()


def _brief_block(brief: dict) -> str:
    """Format the campaign brief into GROUNDED / ASSUMED sections for the prompt."""
    g = brief.get("grounded", brief)
    a = brief.get("assumed", {})
    return (
        f"GROUNDED (real, use as-is):\n{json.dumps(g, indent=2)}\n\n"
        f"ASSUMED (placeholders — if used, list under `assumptions`):\n"
        f"{json.dumps(a, indent=2)}"
    )


def load_skill_prompt(path: str = SKILL_FILE) -> str:
    """Read gtm-agent.md and strip the YAML frontmatter."""
    body = Path(path).read_text(encoding="utf-8")
    return re.sub(r"^---\n.*?\n---\n", "", body, count=1, flags=re.DOTALL).strip()


# --------------------------------------------------------------------------- #
# Nodes
# --------------------------------------------------------------------------- #
def classify(state: CampaignState) -> dict:
    """Pick the archetype. Skips itself on cross-agent re-entry (already known)."""
    if state.get("archetype"):
        return {}  # don't waste an LLM call re-classifying on a revision pass

    user = (
        f"{_brief_block(state['brief'])}\n\n"
        "Do ONLY Step 0: classify this product into archetype A, B, C, or D. "
        'Respond with strict JSON only: {"archetype": "<A|B|C|D>", "reason": "<one line>"}'
    )
    resp = _llm().invoke(
        [SystemMessage(content=state["system_prompt"]), HumanMessage(content=user)]
    )
    try:
        data = json.loads(_strip_fences(resp.content))
        return {
            "archetype": data["archetype"].strip()[:1].upper(),
            "archetype_reason": data.get("reason", ""),
        }
    except Exception:
        return {"archetype": "", "archetype_reason": ""}


def strategize(state: CampaignState) -> dict:
    """Produce the full strategy_pack YAML, folding in BOTH error types."""
    extra = ""

    # (1) GTM's own structural errors (weights != 100, missing field, ...)
    if state.get("gtm_errors"):
        extra += (
            "\n\nYour previous output FAILED validation:\n- "
            + "\n- ".join(state["gtm_errors"])
            + "\nFix these and re-output the complete strategy_pack."
        )

    # (2) Cross-agent feedback from the Metrics agent (strategy-level).
    if state.get("feedback"):
        extra += (
            "\n\nThe Metrics agent reviewed your plan and recommends:\n- "
            + "\n- ".join(state["feedback"])
            + "\nRevise the channel mix / weights accordingly and re-output the "
            "complete strategy_pack."
        )

    user = (
        f"{_brief_block(state['brief'])}\n\n"
        f"Chosen archetype: {state.get('archetype') or '(decide it)'} "
        f"- {state.get('archetype_reason', '')}\n\n"
        "Produce the COMPLETE strategy_pack YAML exactly per your Output Format. "
        "Return only the YAML." + extra
    )
    resp = _llm().invoke(
        [SystemMessage(content=state["system_prompt"]), HumanMessage(content=user)]
    )
    return {
        "strategy_pack": _strip_fences(resp.content),
        "gtm_attempts": state.get("gtm_attempts", 0) + 1,
    }


def validate(state: CampaignState) -> dict:
    """Deterministic structural self-check (no LLM)."""
    errors: list[str] = []
    try:
        pack = yaml.safe_load(state["strategy_pack"])["strategy_pack"]
    except Exception as exc:
        return {"gtm_valid": False, "gtm_errors": [f"YAML did not parse: {exc}"]}

    if pack.get("archetype", "").strip()[:1].upper() not in {"A", "B", "C", "D"}:
        errors.append("archetype must be one of A, B, C, D")

    channels = pack.get("channels") or []
    total = sum(int(c.get("weight", 0)) for c in channels)
    if total != 100:
        errors.append(f"channel weights must sum to 100 (got {total})")

    pricing = pack.get("pricing") or {}
    for key in ("mechanic", "offer", "anchor", "urgency"):
        if not pricing.get(key):
            errors.append(f"pricing.{key} is missing")

    anxiety = ((pack.get("positioning") or {}).get("forces") or {}).get("anxiety")
    if not anxiety:
        errors.append("positioning.forces.anxiety is required and must be specific")

    return {"gtm_valid": not errors, "gtm_errors": errors}


def route_after_validate(state: CampaignState) -> str:
    if state["gtm_valid"] or state.get("gtm_attempts", 0) >= MAX_ATTEMPTS:
        return "parse"
    return "strategize"


def parse_strategy(state: CampaignState) -> dict:
    """
    Turn the LLM's YAML string into STRUCTURED blackboard fields exactly once.
    Channel names are normalised to canonical, and channels that collapse to the
    same canonical name (website + app + whatsapp -> Owned) are merged.
    """
    pack = yaml.safe_load(state["strategy_pack"])["strategy_pack"]

    merged: dict[str, int] = {}
    rationale: dict[str, str] = {}
    for c in pack.get("channels") or []:
        canon = normalize_channel_name(c.get("name", ""))
        merged[canon] = merged.get(canon, 0) + int(c.get("weight", 0))
        rationale.setdefault(canon, c.get("rationale", ""))

    channels = [
        {
            "name": name,
            "weight": weight,
            "priority": weight_to_priority(weight),
            "rationale": rationale.get(name, ""),
        }
        for name, weight in sorted(merged.items(), key=lambda kv: -kv[1])
    ]

    return {
        "archetype": pack.get("archetype", state.get("archetype", "")).strip()[:1].upper(),
        "archetype_reason": pack.get("archetype_reason", state.get("archetype_reason", "")),
        "positioning": pack.get("positioning", {}),
        "channels": channels,
        "pricing": pack.get("pricing", {}),
    }


# --------------------------------------------------------------------------- #
# Subgraph
# --------------------------------------------------------------------------- #
def build_gtm_subgraph():
    g = StateGraph(CampaignState)
    g.add_node("classify", classify)
    g.add_node("strategize", strategize)
    g.add_node("validate", validate)
    g.add_node("parse", parse_strategy)

    g.add_edge(START, "classify")
    g.add_edge("classify", "strategize")
    g.add_edge("strategize", "validate")
    g.add_conditional_edges("validate", route_after_validate, ["strategize", "parse"])
    g.add_edge("parse", END)

# --------------------------------------------------------------------------- #
# Tee: mirror everything printed to BOTH the console and a saved log file, so
# every run leaves a shareable artifact under logs/ without changing any print.
# --------------------------------------------------------------------------- #
class _Tee:
    def __init__(self, *streams):
        self.streams = streams

    def write(self, data):
        for s in self.streams:
            s.write(data)
            s.flush()

    def flush(self):
        for s in self.streams:
            s.flush()

        tag = t.get("confidence", "?").upper()
        print(f"    {ch:<16} ROAS={str(t.get('target_roas')):<6} "
              f"CPA={str(t.get('target_cpa')):<6} [{tag}]")

    if final.get("feedback"):
        print("\n  Outstanding feedback to GTM:")
        for f in final["feedback"]:

def confidence_summary(final: dict) -> str:
    tags = [t.get("confidence") for t in final.get("channel_targets", {}).values()]
    learned = sum(1 for t in tags if t == "learned")
    return f"{learned}/{len(tags)} channels LEARNED from history"

def _run() -> None:
    # ---- PART 1: seed (uses tests/seed_data.py: flights, gift card, shop) ----
    memory = seeded_store()
    print(f"PART 1 — Seeded memory with {len(memory.history)} rows from "
          f"tests/seed_data.py (flights, gift card, shop x objectives x channels).")
    print("         NOTE: 'hotels' is deliberately NOT seeded -> stays cold start.")

    # ---- PART 2: generate 3 campaigns on the SAME memory ----
    print("\nPART 2 — Generating 3 campaigns through the full LLM pipeline...")

    print("\n>>> FLOW TRACE: CAMPAIGN A — FLIGHTS")
    flights = run_campaign(FLIGHTS, memory=memory)
    show("CAMPAIGN A — FLIGHTS (seeded -> should LEARN)", flights)
    print(f"\n  >> {confidence_summary(flights)}")

    print("\n>>> FLOW TRACE: CAMPAIGN B — GIFT CARDS")
    giftcards = run_campaign(GIFTCARDS, memory=memory)
    show("CAMPAIGN B — GIFT CARDS (seeded -> should LEARN)", giftcards)
    print(f"\n  >> {confidence_summary(giftcards)}")

    print("\n>>> FLOW TRACE: CAMPAIGN C — HOTELS")
    hotels = run_campaign(HOTELS, memory=memory)
    show("CAMPAIGN C — HOTELS (NOT seeded -> cold start, all PRIOR)", hotels)
    print(f"\n  >> {confidence_summary(hotels)}")

    # ---- PART 3: close the learning loop on Hotels ----
    print("\n" + "=" * 70)
    print("  PART 3 — CLOSING THE LOOP: feed Hotels live results back, re-run")
    print("=" * 70)

    # Fabricate live performance for whatever channels Hotels used.
    live = {}
    for c in hotels.get("channels", []):
        t = hotels["channel_targets"].get(c["name"], {})
        live[c["name"]] = {
            "roas": (t.get("target_roas") or 3) * 1.15,   # beat ROAS
            "cpa": (t.get("target_cpa") or 500) * 0.85,   # beat CPA
            "ctr": (t.get("target_ctr") or 3) * 1.1,
            "conversion_rate": (t.get("target_conversion_rate") or 3) * 1.1,
            "spend": 20000, "conversions": 60,
        }

    before = len(memory.history)
    fb = build_performance_feedback_graph().invoke(
        {**hotels, "live_performance": live, "memory": memory})
    print(f"\n  Live results scored, {len(memory.history) - before} rows saved to memory.")
    for r in fb["channel_ranking"]:
        print(f"    {r['channel']:<16} score={r['score']:>3}  {r['status']}")

    # Re-run Hotels — now memory has hotel history, so targets should be LEARNED.
    print("\n>>> FLOW TRACE: CAMPAIGN C (RE-RUN) — HOTELS after the loop")
    hotels_v2 = run_campaign(HOTELS, memory=memory)
    show("CAMPAIGN B (RE-RUN) — HOTELS after the loop (should LEARN now)", hotels_v2)
    print(f"\n  >> BEFORE: {confidence_summary(hotels)}")
    print(f"  >> AFTER : {confidence_summary(hotels_v2)}   <- learning loop closed")

    if not os.environ.get("OPENAI_API_KEY"):

    # Mirror all console output into a timestamped, shareable log file.
    logdir = Path(__file__).resolve().parent / "logs"
    logdir.mkdir(exist_ok=True)

    original_stdout = sys.stdout
    with open(logpath, "w", encoding="utf-8") as logfile:
        sys.stdout = _Tee(original_stdout, logfile)
        try:
            _run()
        finally:
            sys.stdout = original_stdout
    print(f"\n[Full flow log saved to: {logpath}]")
