from __future__ import annotations

import mimetypes

from fastapi import APIRouter, Query, Request
from fastapi.responses import FileResponse, Response
from fastapi import HTTPException

from ..errors import NotFound
from ..media_token import verify as verify_media_token
from ..services.store import require_project
from ..storage import resolve_in_storage

router = APIRouter(prefix="/api/v1/video", tags=["video"])


@router.get("/media/{path:path}")
async def serve_media(
    path: str,
    request: Request,
    token: str | None = Query(default=None),
) -> Response:
    """Serve generated clips, audio and finals out of the storage directory.

    Accepts either:
    - A short-lived signed ?token= query param issued by GET /render/token, OR
    - A valid Azure AD Bearer token in the Authorization header.

    FileResponse handles Range requests so the browser can scrub without
    downloading the whole file first.
    """
    parts = path.replace("\\", "/").split("/")
    if len(parts) < 3 or parts[0] != "projects":
        raise NotFound(f"Media not found: {path}")

    project_id = parts[1]

    if token:
        if not verify_media_token(token, project_id):
            raise HTTPException(status_code=401, detail="Media token is invalid or expired")
    else:
        # Fall back to Bearer token validation.
        auth_header = request.headers.get("Authorization", "")
        if not auth_header.startswith("Bearer "):
            raise HTTPException(status_code=401, detail="Not authenticated")
        raw_token = auth_header[len("Bearer "):]
        from auth.azure_auth import validate_token
        validate_token(raw_token)  # raises 401/403 on failure

    await require_project(project_id)
    target = resolve_in_storage(path)

    if not target.is_file():
        raise NotFound(f"Media not found: {path}")

    media_type = mimetypes.guess_type(target.name)[0] or "application/octet-stream"
    return FileResponse(
        target,
        media_type=media_type,
        headers={"Accept-Ranges": "bytes", "Cache-Control": "private, max-age=3600"},
    )
