import { create } from 'zustand'
import { persist } from 'zustand/middleware'
import type { SurveyRecord } from '@/features/eventScheduler/types/survey'
import { MOCK_SURVEY_RECORDS } from '@/features/eventScheduler/mocks/survey.mock'

interface SurveyStore {
  surveys: SurveyRecord[]
  addSurvey: (survey: SurveyRecord) => void
  submitResponse: (surveyId: string, userId: string, userName: string, answers: Record<string, string | string[]>) => void
}

export const useSurveyStore = create<SurveyStore>()(
  persist(
    (set) => ({
      surveys: MOCK_SURVEY_RECORDS,
      addSurvey: (survey) => set((state) => ({ surveys: [survey, ...state.surveys] })),
      submitResponse: (surveyId, userId, userName, answers) =>
        set((state) => ({
          surveys: state.surveys.map((s) => {
            if (s.id !== surveyId) return s
            const existingIdx = s.responses.findIndex((r) => r.userId === userId)
            const entry = {
              userId,
              userName,
              iga: '',
              base: '-',
              status: 'filled' as const,
              submittedAt: new Date().toISOString(),
              answers,
            }
            const responses =
              existingIdx >= 0
                ? s.responses.map((r, i) => (i === existingIdx ? { ...r, ...entry } : r))
                : [...s.responses, entry]
            const filledCount = responses.filter((r) => r.status === 'filled').length
            return { ...s, responses, filledCount }
          }),
        })),
    }),
    { name: 'survey-store' },
  ),
)
