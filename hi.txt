"""Environment configuration.

Two providers, two jobs:

    Azure OpenAI  -> the writing brain. Storyboards, scene prompts, repairs,
                     audio direction. Anything that produces text.
    Google Gemini -> the pixels. Veo for the video segments, and the Gemini
                     image model for optional scene keyframes used as Veo first
                     frames.

Nothing is required at boot. The keys arrive later, and a server that refuses to
start without them is useless for wiring up the UI. Each capability reports
itself through ``capabilities()``, and routes fail loudly, per request, with an
actionable message.
"""

from __future__ import annotations

from functools import lru_cache
from pathlib import Path
import tempfile
from typing import Literal

from pydantic import Field, field_validator
from pydantic_settings import BaseSettings, SettingsConfigDict

SERVER_ROOT = Path(__file__).resolve().parent.parent
REPO_ROOT = SERVER_ROOT.parent

# server/.env wins; the repo root .env is the fallback so one file can feed both apps.
_ENV_FILES = (REPO_ROOT / ".env", SERVER_ROOT / ".env")


class Settings(BaseSettings):
    model_config = SettingsConfigDict(
        env_file=[str(path) for path in _ENV_FILES],
        env_file_encoding="utf-8",
        extra="ignore",
        case_sensitive=False,
    )

    # --- Server -----------------------------------------------------------
    host: str = "0.0.0.0"
    port: int = 8000
    public_base_url: str = ""
    cors_origins: str = "http://localhost:5173,http://localhost:8080"
    log_level: str = "INFO"

    # --- Brand -------------------------------------------------------------
    #: Which brand guardrails to enforce. "indigo" applies app/brand/policy.py to
    #: every storyboard, scene prompt and audit. "none" turns enforcement off and
    #: leaves the craft layer alone.
    brand: str = "indigo"
    #: TrueType/OpenType file used for burnt-in supers. Veo is never asked to
    #: render type, so this font is the only one a viewer ever sees. Left unset,
    #: `brand.assets.font_path()` picks the best available geometric sans for the
    #: platform — Futura on macOS, Century Gothic on Windows.
    brand_font_path: str | None = None
    #: Refuse to render when the storyboard audit still has `error`-severity
    #: issues. The audit exists to stop broken prompts reaching Veo; leaving this
    #: off makes it advisory, which is how misspelt supers shipped.
    brand_enforce_audit: bool = True

    # --- Azure OpenAI: the writing brain ----------------------------------
    azure_openai_base_url: str | None = None
    azure_openai_key: str | None = None
    #: The deployment name, which is what actually goes in the URL path.
    azure_openai_deployment_name: str | None = None
    #: Underlying model behind the deployment. Decides the parameter dialect.
    azure_openai_model: str = "gpt-4.1"
    azure_openai_version: str = "2024-10-21"
    #: Optional second deployment for cheap passes. Falls back to the main one.
    azure_openai_fast_deployment_name: str | None = None
    #: Optional TTS deployment for voiceover. Voiceover is disabled without it.
    azure_openai_tts_deployment_name: str | None = None
    azure_openai_tts_voice: str = "alloy"
    azure_openai_max_output_tokens: int = 16000
    azure_openai_timeout_seconds: float = 180.0

    # --- Google: Veo + image ----------------------------------------------
    #: The Google AI Studio key. Named for the image model in this product's
    #: existing config; the same key authenticates Veo. GEMINI_API_KEY is an alias.
    gemini_image_api_key: str | None = None
    gemini_api_key: str | None = None
    gemini_api_base: str = "https://generativelanguage.googleapis.com/v1beta"
    #: Image model, used for optional scene keyframes fed to Veo as first frames.
    gemini_image_model_name: str = "gemini-2.5-flash-image"
    veo_model: str = "veo-3.1-generate-preview"
    veo_fast_model: str = "veo-3.1-fast-generate-preview"

    # --- Veo behaviour ------------------------------------------------------
    #: Longest single Veo generation, in seconds. Veo accepts 4, 6 or 8.
    veo_max_clip_seconds: int = Field(default=8, ge=4, le=8)
    #: Seconds added by one ``extend`` call on a Veo-produced video.
    veo_extend_seconds: int = 7
    #: Hard ceiling on extends per video, per the Veo API.
    veo_max_extensions: int = 20
    veo_resolution: Literal["720p", "1080p"] = "1080p"
    veo_person_generation: Literal["allow_all", "allow_adult", "dont_allow"] = "allow_all"
    veo_poll_interval_seconds: float = 10.0
    veo_poll_timeout_seconds: float = 900.0
    #: How many Veo generations may be in flight at once.
    veo_concurrency: int = 3
    #: Attempts per clip before it is marked failed.
    veo_max_attempts: int = 2
    #: Render a still keyframe per scene with the Gemini image model and pass it
    #: to Veo as the first frame. One image call per clip, large continuity win.
    #:
    #: On by default because the keyframe is also where the real IndiGo livery
    #: enters the pipeline: it is conditioned on the photographs in
    #: ``brand/assets/references/livery/``, which is the only way the aircraft
    #: comes back as an IndiGo aircraft rather than an anonymous white jet.
    #: Turning this off costs one image call per clip and the accurate livery.
    veo_use_keyframes: bool = True
    #: Seed each clip with the LAST FRAME of the previous clip, so the face,
    #: wardrobe and grade carry across a cut instead of being re-guessed. Costs
    #: nothing extra per clip, but forces generation to run in story order rather
    #: than `veo_concurrency` at a time — a 7-clip film goes from ~3 rounds to 7.
    #: Worth it when characters recur; pointless for a montage of unrelated shots.
    veo_chain_keyframes: bool = False

    # --- Guard rails --------------------------------------------------------
    max_video_seconds: int = 180
    max_segments: int = 30

    # --- Media / storage ----------------------------------------------------
    video_storage_dir: str = str(Path(tempfile.gettempdir()) / "marketing-platform-video")
    ffmpeg_path: str | None = None
    ffprobe_path: str | None = None
    render_fps: int = 24
    #: Gain applied to the music bed under the clip audio.
    music_gain: float = 0.28
    voiceover_gain: float = 1.0

    # --- Optional music providers ------------------------------------------
    music_provider: Literal["none", "elevenlabs", "vertex-lyria"] = "none"
    elevenlabs_api_key: str | None = None
    elevenlabs_music_model: str = "music_v1"
    google_cloud_project: str | None = None
    google_cloud_location: str = "us-central1"
    google_access_token: str | None = None
    vertex_lyria_model: str = "lyria-002"

    @field_validator(
        "azure_openai_base_url",
        "azure_openai_key",
        "azure_openai_deployment_name",
        "azure_openai_fast_deployment_name",
        "azure_openai_tts_deployment_name",
        "gemini_image_api_key",
        "gemini_api_key",
        "elevenlabs_api_key",
        "google_cloud_project",
        "google_access_token",
        "ffmpeg_path",
        "ffprobe_path",
        mode="before",
    )
    @classmethod
    def _blank_to_none(cls, value: object) -> object:
        """Treat empty strings and Azure Key Vault references as "not set".

        Key Vault references (``@Microsoft.KeyVault(...)``) are only resolved
        by Azure App Service at runtime. Locally they arrive as literal strings
        and must be ignored so the capability check reports "not configured"
        rather than sending the reference text as an API key.
        """
        if isinstance(value, str):
            stripped = value.strip()
            if not stripped or stripped.startswith("@Microsoft.KeyVault("):
                return None
        return value

    # --- Derived ------------------------------------------------------------
    @property
    def cors_origin_list(self) -> list[str]:
        return [origin.strip() for origin in self.cors_origins.split(",") if origin.strip()]

    @property
    def storage_path(self) -> Path:
        path = Path(self.video_storage_dir)
        return path if path.is_absolute() else (SERVER_ROOT / path).resolve()

    @property
    def google_key(self) -> str | None:
        """The Google key, under either accepted name."""
        return self.gemini_image_api_key or self.gemini_api_key

    @property
    def azure_configured(self) -> bool:
        return bool(
            self.azure_openai_base_url
            and self.azure_openai_key
            and self.azure_openai_deployment_name
        )

    @property
    def allowed_clip_seconds(self) -> list[int]:
        """Durations Veo accepts for a single generation, capped by config."""
        allowed = [s for s in (4, 6, 8) if s <= self.veo_max_clip_seconds]
        return allowed or [4]


@lru_cache
def get_settings() -> Settings:
    return Settings()


settings = get_settings()


def capabilities() -> list[dict[str, object]]:
    """What this server can actually do right now, given the keys it has."""
    google = settings.google_key
    return [
        {
            "id": "storyboard",
            "ready": settings.azure_configured,
            "detail": (
                f"Azure OpenAI {settings.azure_openai_model} "
                f"(deployment {settings.azure_openai_deployment_name})"
                if settings.azure_configured
                else "Set AZURE_OPENAI_BASE_URL, AZURE_OPENAI_KEY and "
                "AZURE_OPENAI_DEPLOYMENT_NAME to plan storyboards."
            ),
        },
        {
            "id": "clips",
            "ready": bool(google),
            "detail": (
                f"{settings.veo_model}, max {settings.veo_max_clip_seconds}s per generation"
                if google
                else "Set GEMINI_IMAGE_API_KEY to generate video clips with Veo."
            ),
        },
        {
            "id": "keyframes",
            "ready": bool(google) and settings.veo_use_keyframes,
            "detail": (
                "Set GEMINI_IMAGE_API_KEY to generate scene keyframes."
                if not google
                else f"{settings.gemini_image_model_name} first-frame keyframes are on"
                if settings.veo_use_keyframes
                else "Set VEO_USE_KEYFRAMES=true to lock continuity with generated first frames."
            ),
        },
        {
            "id": "voiceover",
            "ready": settings.azure_configured and bool(settings.azure_openai_tts_deployment_name),
            "detail": (
                f"Azure OpenAI TTS deployment {settings.azure_openai_tts_deployment_name}"
                if settings.azure_openai_tts_deployment_name
                else "Set AZURE_OPENAI_TTS_DEPLOYMENT_NAME for voiceover. Veo still "
                "generates in-scene dialogue without it."
            ),
        },
        {
            "id": "music",
            "ready": (
                settings.music_provider == "elevenlabs" and bool(settings.elevenlabs_api_key)
            )
            or (
                settings.music_provider == "vertex-lyria"
                and bool(settings.google_cloud_project and settings.google_access_token)
            ),
            "detail": (
                "No music generator configured — upload a track instead, or set MUSIC_PROVIDER."
                if settings.music_provider == "none"
                else f"Provider {settings.music_provider}"
            ),
        },
    ]
