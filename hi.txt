"""Short-lived HMAC tokens for unauthenticated media downloads.

A render can take 20+ minutes. By the time it finishes the Azure AD ID token
the browser holds may have expired, causing the media fetch to 401. Rather
than forcing a re-login for a completed render, the frontend requests a
short-lived signed token immediately after the render job succeeds (while the
session is still valid) and uses that token as a query-param on the media URL.

Token format (URL-safe base64):  <expiry_unix_int>.<project_id>.<hmac_sha256>
"""

from __future__ import annotations

import hashlib
import hmac
import os
import time
from base64 import urlsafe_b64decode, urlsafe_b64encode

_TOKEN_TTL = 600  # 10 minutes


def _secret() -> bytes:
    key = os.getenv("MEDIA_TOKEN_SECRET") or os.getenv("AZURE_CLIENT_SECRET") or ""
    if not key:
        raise RuntimeError("No secret available for media token signing")
    return key.encode()


def _sign(payload: str) -> str:
    return hmac.new(_secret(), payload.encode(), hashlib.sha256).hexdigest()


def issue(project_id: str) -> str:
    """Return a signed token valid for _TOKEN_TTL seconds."""
    expiry = int(time.time()) + _TOKEN_TTL
    payload = f"{expiry}.{project_id}"
    sig = _sign(payload)
    raw = f"{payload}.{sig}"
    return urlsafe_b64encode(raw.encode()).decode()


def verify(token: str, project_id: str) -> bool:
    """Return True iff the token is valid, unexpired, and bound to project_id."""
    try:
        raw = urlsafe_b64decode(token.encode() + b"==").decode()
        expiry_str, pid, sig = raw.split(".", 2)
        if pid != project_id:
            return False
        if int(expiry_str) < int(time.time()):
            return False
        payload = f"{expiry_str}.{pid}"
        return hmac.compare_digest(_sign(payload), sig)
    except Exception:
        return False
