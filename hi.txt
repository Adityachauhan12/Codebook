import { create } from 'zustand'

/**
 * Minimal auth store vendored for the Event Scheduler feature.
 *
 * The Event Scheduler only reads the current user's display fields
 * (name, role, crewIga, initials) to stamp the "sender" on published
 * notifications, plus the ROLE_LABELS lookup for the targeting UI.
 * The host app manages real auth via UserContext; this store simply
 * provides a stable admin identity so the feature is self-contained.
 */

export type UserRole =
  | 'crew'
  | 'lead'
  | 'mentor'
  | 'manager'
  | 'admin'
  | 'clms_verifier'
  | 'crew_ops'

/** Human-readable role labels shown across the UI. */
export const ROLE_LABELS: Record<UserRole, string> = {
  crew: 'Cabin Crew',
  lead: 'Lead Crew',
  mentor: 'PEP Mentor',
  manager: 'Manager',
  admin: 'Admin',
  clms_verifier: 'CLMS Verifier',
  crew_ops: 'Crew Ops Base Manager',
}

export interface AppUser {
  id: string
  name: string
  crewIga: string
  role: UserRole
  email: string
  base: string
  initials: string
}

interface AuthState {
  user: AppUser | null
}

const DEFAULT_ADMIN: AppUser = {
  id: 'admin',
  name: 'Hemant Admin',
  crewIga: '10001',
  role: 'admin',
  email: 'hemant.admin@indigo.in',
  base: 'DEL',
  initials: 'HA',
}

export const useAuthStore = create<AuthState>(() => ({
  user: DEFAULT_ADMIN,
}))
