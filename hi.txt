"""Veo, via the Gemini API.

Shape of the world:

  - One generation returns one clip of 4, 6 or 8 seconds. That is the ceiling,
    and it is the reason the whole segment plan exists.
  - Generation is a long-running operation: POST returns an operation name, you
    poll it until ``done``, then download the result from a URI that still needs
    the API key on the request.
  - ``extend`` continues a Veo-produced video by ~7s, up to 20 times. Not used
    by the main pipeline — see ``extend_clip`` for why.
"""

from __future__ import annotations

import asyncio
import base64
from dataclasses import dataclass
from pathlib import Path
from typing import Any, Awaitable, Callable

import httpx

from ..config import settings
from ..domain.models import VeoAspectRatio
from ..errors import Unavailable, UpstreamError
from ..logging_setup import get_logger
from ..storage import ensure_dir

log = get_logger("veo")

ProgressFn = Callable[[float, str], Awaitable[None] | None]


@dataclass(slots=True)
class InlineAsset:
    mime_type: str
    #: base64, without a ``data:`` prefix.
    data: str

    def to_payload(self) -> dict[str, str]:
        return {"mimeType": self.mime_type, "data": self.data}


@dataclass(slots=True)
class GeneratedClip:
    file: Path
    bytes: int
    operation: str
    model: str


def _require_key() -> str:
    key = settings.google_key
    if not key:
        raise Unavailable(
            "GEMINI_IMAGE_API_KEY is not set. Add it to server/.env "
            "(see server/.env.example) and restart the server."
        )
    return key


def _api_url(path: str) -> str:
    return f"{settings.gemini_api_base.rstrip('/')}/{path.lstrip('/')}"


def normalise_seconds(seconds: int) -> int:
    """Veo accepts 4, 6 or 8. Anything else snaps down to a legal value."""
    allowed = settings.allowed_clip_seconds
    if seconds in allowed:
        return seconds
    below = [value for value in allowed if value <= seconds]
    return max(below) if below else min(allowed)


async def _report(callback: ProgressFn | None, fraction: float, message: str) -> None:
    if callback is None:
        return
    result = callback(fraction, message)
    if asyncio.iscoroutine(result):
        await result


# ---------------------------------------------------------------------------
# Start
# ---------------------------------------------------------------------------


async def start_generation(
    *,
    prompt: str,
    seconds: int,
    aspect_ratio: VeoAspectRatio,
    negative_prompt: str | None = None,
    model: str | None = None,
    resolution: str | None = None,
    first_frame: InlineAsset | None = None,
    last_frame: InlineAsset | None = None,
    reference_images: list[InlineAsset] | None = None,
    video: InlineAsset | None = None,
) -> tuple[str, str]:
    """Submit a generation. Returns ``(operation_name, model)``."""
    key = _require_key()
    target_model = model or settings.veo_model
    duration = normalise_seconds(seconds)

    instance: dict[str, Any] = {"prompt": prompt}
    if first_frame:
        instance["image"] = {"inlineData": first_frame.to_payload()}
    if last_frame:
        instance["lastFrame"] = {"inlineData": last_frame.to_payload()}
    if video:
        instance["video"] = {"inlineData": video.to_payload()}
    if reference_images:
        instance["referenceImages"] = [
            {"image": {"inlineData": image.to_payload()}, "referenceType": "asset"}
            for image in reference_images
        ]

    # Veo only supports 1080p for 8-second clips; shorter durations require 720p.
    effective_resolution = resolution or settings.veo_resolution
    if duration < 8 and effective_resolution == "1080p":
        effective_resolution = "720p"

    parameters: dict[str, Any] = {
        "aspectRatio": aspect_ratio,
        "resolution": effective_resolution,
        "durationSeconds": duration,
        "personGeneration": settings.veo_person_generation,
    }
    if negative_prompt and negative_prompt.strip():
        parameters["negativePrompt"] = negative_prompt.strip()

    async def _submit() -> httpx.Response:
        async with httpx.AsyncClient(timeout=120.0) as client:
            return await client.post(
                _api_url(f"models/{target_model}:predictLongRunning"),
                headers={"x-goog-api-key": key, "content-type": "application/json"},
                json={"instances": [instance], "parameters": parameters},
            )

    response = await _submit()

    # `referenceImages` is a newer Veo capability and not every model or API
    # version accepts it. Dropping the brand reference costs one less accurate
    # frame; failing the request costs the whole film. Retry once without.
    if response.status_code >= 400 and "referenceImages" in instance:
        log.warning(
            "Veo rejected the request (%s) with %d reference image(s) attached; retrying "
            "without them. The aircraft may not read as IndiGo in this clip. %s",
            response.status_code,
            len(instance["referenceImages"]),
            response.text[:300],
        )
        instance.pop("referenceImages")
        response = await _submit()

    if response.status_code >= 400:
        log.error("Veo %s error body: %s", response.status_code, response.text[:1200])
        raise UpstreamError(
            f"Veo rejected the generation request ({response.status_code}).",
            details=response.text[:800],
        )

    operation = response.json()
    name = operation.get("name")
    if not name:
        raise UpstreamError("Veo accepted the request but returned no operation name.", details=operation)

    log.info("Started %s generation (%ds, %s): %s", target_model, duration, aspect_ratio, name)
    return name, target_model


# ---------------------------------------------------------------------------
# Poll
# ---------------------------------------------------------------------------


async def poll_operation(name: str, on_progress: ProgressFn | None = None) -> dict[str, Any]:
    key = _require_key()
    loop = asyncio.get_running_loop()
    deadline = loop.time() + settings.veo_poll_timeout_seconds

    async with httpx.AsyncClient(timeout=60.0) as client:
        while True:
            if loop.time() > deadline:
                raise UpstreamError(
                    f"Veo did not finish within {int(settings.veo_poll_timeout_seconds)}s. "
                    f"Operation {name} is still running — raise VEO_POLL_TIMEOUT_SECONDS if this "
                    "is normal for your quota."
                )

            await asyncio.sleep(settings.veo_poll_interval_seconds)

            response = await client.get(_api_url(name), headers={"x-goog-api-key": key})

            if response.status_code >= 400:
                # A transient 5xx while polling should not kill a generation
                # that is probably fine; only 4xx is fatal.
                if response.status_code >= 500:
                    log.warning("Polling %s returned %s; retrying.", name, response.status_code)
                    continue
                raise UpstreamError(
                    f"Veo polling failed ({response.status_code}).", details=response.text[:600]
                )

            operation = response.json()

            if operation.get("error"):
                error = operation["error"]
                raise UpstreamError(
                    f"Veo generation failed: {error.get('message', 'unknown error')}", details=error
                )

            metadata = operation.get("metadata") or {}
            percent = metadata.get("progressPercent")
            if isinstance(percent, (int, float)):
                await _report(on_progress, min(0.95, percent / 100), f"Rendering… {round(percent)}%")
            else:
                state = metadata.get("state")
                await _report(
                    on_progress, 0.5, f"Veo {state.lower()}…" if state else "Rendering…"
                )

            if operation.get("done"):
                return operation


def _extract_video_uri(operation: dict[str, Any]) -> str:
    result = (operation.get("response") or {}).get("generateVideoResponse") or {}
    samples = result.get("generatedSamples") or []
    uri = (samples[0].get("video") or {}).get("uri") if samples else None

    if not uri:
        filtered = result.get("raiMediaFilteredCount") or 0
        if filtered:
            reasons = "; ".join(result.get("raiMediaFilteredReasons") or []) or "no reason given"
            raise UpstreamError(
                f"Veo filtered the generation before it produced a clip ({reasons}). Rewrite the "
                "scene prompt — usually it names a real person, a brand, or content the safety "
                "filter blocks.",
                details={"filtered": filtered, "reasons": reasons},
            )
        raise UpstreamError("Veo finished but returned no video.", details=operation.get("response"))

    return uri


# ---------------------------------------------------------------------------
# Download
# ---------------------------------------------------------------------------


async def download_video(uri: str, destination: Path) -> int:
    key = _require_key()
    ensure_dir(destination.parent)

    # The URI is on a Google host and still requires the API key.
    async with httpx.AsyncClient(timeout=600.0, follow_redirects=True) as client:
        response = await client.get(uri, headers={"x-goog-api-key": key})

    if response.status_code >= 400:
        raise UpstreamError(
            f"Downloading the generated clip failed ({response.status_code}).",
            details=response.text[:400],
        )

    payload = response.content
    if not payload:
        raise UpstreamError("The generated clip downloaded as an empty file.")

    destination.write_bytes(payload)
    return len(payload)


# ---------------------------------------------------------------------------
# The whole round trip
# ---------------------------------------------------------------------------


async def generate_clip(
    destination: Path,
    *,
    prompt: str,
    seconds: int,
    aspect_ratio: VeoAspectRatio,
    negative_prompt: str | None = None,
    model: str | None = None,
    first_frame: InlineAsset | None = None,
    reference_images: list[InlineAsset] | None = None,
    on_progress: ProgressFn | None = None,
) -> GeneratedClip:
    await _report(on_progress, 0.05, "Submitting to Veo…")

    operation, used_model = await start_generation(
        prompt=prompt,
        seconds=seconds,
        aspect_ratio=aspect_ratio,
        negative_prompt=negative_prompt,
        model=model,
        first_frame=first_frame,
        reference_images=reference_images,
    )

    async def relay(fraction: float, message: str) -> None:
        await _report(on_progress, 0.05 + fraction * 0.8, message)

    finished = await poll_operation(operation, relay)

    await _report(on_progress, 0.9, "Downloading clip…")
    size = await download_video(_extract_video_uri(finished), destination)

    await _report(on_progress, 1.0, "Clip ready.")
    log.info("Downloaded %.1fMB to %s", size / 1_048_576, destination.name)

    return GeneratedClip(file=destination, bytes=size, operation=operation, model=used_model)


# ---------------------------------------------------------------------------
# Extend
# ---------------------------------------------------------------------------


async def extend_clip(
    source_mp4: Path,
    prompt: str,
    destination: Path,
    *,
    aspect_ratio: VeoAspectRatio = "16:9",
    model: str | None = None,
) -> GeneratedClip:
    """Continue a Veo-produced clip by roughly ``VEO_EXTEND_SECONDS``.

    Not used by the main pipeline: extension carries the previous shot's motion
    forward, which is what you want for one continuous take and precisely what
    you do not want for a cut sequence with distinct framings per beat. It also
    serialises — twelve extensions is twelve round trips end to end, where
    twelve independent segments run three at a time.

    Kept because it is the right tool for a future "one continuous take" mode,
    and because it is the only way past 8 seconds without a cut.
    """
    encoded = base64.b64encode(source_mp4.read_bytes()).decode("ascii")

    operation, used_model = await start_generation(
        prompt=prompt,
        seconds=settings.veo_max_clip_seconds,
        aspect_ratio=aspect_ratio,
        model=model,
        video=InlineAsset(mime_type="video/mp4", data=encoded),
    )

    finished = await poll_operation(operation)
    size = await download_video(_extract_video_uri(finished), destination)
    return GeneratedClip(file=destination, bytes=size, operation=operation, model=used_model)
