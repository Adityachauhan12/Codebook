import { create } from 'zustand'

/**
 * Minimal notification store vendored for the Event Scheduler feature.
 *
 * The Event Scheduler publishes notifications via `addNotification`. The host
 * app has its own notification delivery; here we simply keep published items
 * in memory so the feature's publish/toast flow works end-to-end. The type
 * shapes mirror the source app so the feature compiles unchanged.
 */

export interface AppNotification {
  id: string
  title: string
  message: string
  timestamp: string
  read: boolean
  link?: string
  meta?: Record<string, string>
  targets?: NotificationTarget | null
  source?: 'system' | 'event_scheduler'
  priority?: 'normal' | 'urgent'
  sender?: {
    name: string
    role: string
    crewIga: string
    initials: string
    avatarUrl?: string
  }
  attachmentName?: string
}

/** Targeting payload — mirrors Event Scheduler selection. */
export interface NotificationTarget {
  roles?: string[]
  bases?: string[]
  organisations?: string[]
  teams?: string[]
  userIds?: string[]
}

interface NotificationState {
  notifications: AppNotification[]
  addNotification: (n: AppNotification) => void
}

export const useNotificationStore = create<NotificationState>((set) => ({
  notifications: [],
  addNotification: (n) =>
    set((state) => ({ notifications: [n, ...state.notifications] })),
}))
