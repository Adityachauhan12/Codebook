# ---------------------------------------------------------------------------
# Veo behaviour
#
# THE CORE CONSTRAINT: Veo renders 4, 6 or 8 seconds per generation and nothing
# longer. A 90 second video is 12 generations that ffmpeg accumulates into one
# file. The planner in app/domain/duration.py owns that arithmetic — see
# GET /api/video/plan?seconds=90 to inspect it without spending a token.
# ---------------------------------------------------------------------------
VEO_MAX_CLIP_SECONDS=8
VEO_RESOLUTION=1080p
VEO_PERSON_GENERATION=allow_all

# How many generations run at once. Raise it if your quota allows; a 12-clip
# film at concurrency 3 is roughly 4 sequential rounds.
VEO_CONCURRENCY=3
# Attempts per clip before it is marked failed in the UI.
VEO_MAX_ATTEMPTS=2
VEO_POLL_INTERVAL_SECONDS=10
VEO_POLL_TIMEOUT_SECONDS=900

# Generate a still first frame per scene with the image model and feed it to
# Veo. Costs one image call per clip; buys a large jump in cross-clip
# continuity, because identity is then fixed by pixels rather than by hoping
# twelve independent prompts converge.
VEO_USE_KEYFRAMES=false

# Seed each clip with the LAST FRAME of the previous clip, so a face, a costume
# and a grade survive a cut instead of being re-guessed. Costs no extra API
# calls, but forces clips to generate in story order rather than
# VEO_CONCURRENCY at a time — a 7-clip film takes 7 rounds instead of 3.
# Turn on when characters recur across shots; leave off for a montage.
VEO_CHAIN_KEYFRAMES=false

# Font for burnt-in supers. Veo is never asked to render type, so this is the
# only font a viewer sees. Left blank, the best available geometric sans is
# picked per platform: Futura on macOS, Century Gothic on Windows. Drop a .ttf
# into server/app/brand/assets/fonts/ to override without editing config.
# BRAND_FONT_PATH=C:\Windows\Fonts\GOTHIC.TTF

# Refuse to render when the storyboard audit still has error-severity issues.
# With this off the audit is advisory, which is how misspelt supers and a
# mangled wordmark reached a finished film.
BRAND_ENFORCE_AUDIT=true

# Extension is implemented (app/services/veo.py extend_clip) but not used by the
# stitch pipeline. These describe the API's limits: +7s per extend, 20 max.
VEO_EXTEND_SECONDS=7
VEO_MAX_EXTENSIONS=20


# --- Guard rails -----------------------------------------------------------
# Cost ceilings. 30 segments at 8s is 240 seconds of Veo per render.
MAX_VIDEO_SECONDS=180
MAX_SEGMENTS=30
