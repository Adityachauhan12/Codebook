import type { Brief } from '../types'

function extractBudget(prompt: string) {
  const inrMatch = prompt.match(/(?:₹|rs\.?|inr)\s?([0-9][0-9,]*)/i)
  if (inrMatch?.[1]) return Number(inrMatch[1].replace(/,/g, ''))
  const plainMatch = prompt.match(/\b([0-9][0-9,]{2,})\b/)
  if (plainMatch?.[1]) return Number(plainMatch[1].replace(/,/g, ''))
  return undefined
}

function inferObjective(prompt: string) {
  const lower = prompt.toLowerCase()
  if (lower.includes('awareness')) return 'awareness'
  if (lower.includes('reach')) return 'reach'
  if (lower.includes('lead')) return 'leads'
  if (lower.includes('signup') || lower.includes('sign up') || lower.includes('enrol')) return 'signups'
  if (lower.includes('trial')) return 'trials'
  if (lower.includes('booking')) return 'bookings'
  if (lower.includes('revenue') || lower.includes('sales') || lower.includes('sell')) return 'sales'
  return 'bookings'
}

/**
 * UNUSED. The brief is built server-side by the intake agent, which streams back a
 * `chat_ready` event containing a properly split grounded/assumed brief.
 *
 * Do not wire this up as-is: it puts the user's entire sentence into `grounded.product`,
 * so `product` becomes something like "Launch a spring campaign with a ₹8 lakh budget".
 * That string is interpolated into the GTM and Creative system prompts and ends up inside
 * the image generation prompt — meaning internal budget language can be rendered as text
 * into a customer-facing banner. It also marks an inferred objective as `grounded`, which
 * the GTM agent treats as a client-stated fact it must not contradict.
 */
export function promptToBrief(prompt: string): Brief {
  const text = prompt.trim()
  const budget = extractBudget(text)

  return {
    grounded: {
      product: text,
      budget_inr: budget,
    },
    assumed: {
      objective: inferObjective(text),
      original_prompt: text,
    },
  }
}
