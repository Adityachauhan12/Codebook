import type { Brief, CampaignRecord, ChatSSEEvent, SSEEvent } from '../types'

const API_BASE = import.meta.env.VITE_API_BASE ?? 'https://autonomous-marketing-application.azurewebsites.net'

async function readSSEStream<T>(
  response: Response,
  onEvent: (event: T) => void,
) {
  if (!response.ok || !response.body) {
    throw new Error(`Request failed (${response.status})`)
  }

  const reader = response.body.getReader()
  const decoder = new TextDecoder()
  let buffer = ''

  const flush = (raw: string) => {
    const payload = raw
      .split(/\r?\n/)
      .map((l) => l.trim())
      .filter((l) => l.startsWith('data:'))
      .map((l) => l.slice(5).trim())
      .join('')
    if (!payload) return
    onEvent(JSON.parse(payload) as T)
  }

  while (true) {
    const { value, done } = await reader.read()
    if (done) break
    buffer += decoder.decode(value, { stream: true })
    let idx = buffer.indexOf('\n\n')
    while (idx >= 0) {
      flush(buffer.slice(0, idx))
      buffer = buffer.slice(idx + 2)
      idx = buffer.indexOf('\n\n')
    }
  }
  if (buffer.trim()) flush(buffer)
}

export async function sendChatMessage(
  sessionId: string,
  message: string,
  options: {
    signal?: AbortSignal
    onEvent: (event: ChatSSEEvent) => void
  },
) {
  const response = await fetch(`${API_BASE}/chat/message`, {
    method: 'POST',
    headers: { 'Content-Type': 'application/json', Accept: 'text/event-stream' },
    body: JSON.stringify({ session_id: sessionId, message }),
    signal: options.signal,
  })
  await readSSEStream<ChatSSEEvent>(response, options.onEvent)
}

export async function streamCampaign(
  brief: Brief,
  options: {
    signal?: AbortSignal
    onEvent: (event: SSEEvent) => void
  },
) {
  const response = await fetch(`${API_BASE}/campaign/run`, {
    method: 'POST',
    headers: { 'Content-Type': 'application/json', Accept: 'text/event-stream' },
    body: JSON.stringify(brief),
    signal: options.signal,
  })
  await readSSEStream<SSEEvent>(response, options.onEvent)
}

export async function refineCampaign(
  sessionId: string,
  message: string,
  currentBrief: Brief,
  options: {
    signal?: AbortSignal
    onEvent: (event: SSEEvent) => void
  },
) {
  const response = await fetch(`${API_BASE}/campaign/refine`, {
    method: 'POST',
    headers: { 'Content-Type': 'application/json', Accept: 'text/event-stream' },
    body: JSON.stringify({ session_id: sessionId, message, current_brief: currentBrief }),
    signal: options.signal,
  })
  await readSSEStream<SSEEvent>(response, options.onEvent)
}

/* -------------------------------------------------------------------------- */
/* Campaign history — server-backed, so every machine sees the same list.      */
/* The backend has always exposed these; the app previously read only          */
/* localStorage, which is why each browser showed a different set.             */
/* -------------------------------------------------------------------------- */

export async function fetchCampaigns(signal?: AbortSignal): Promise<CampaignRecord[]> {
  const response = await fetch(`${API_BASE}/campaigns`, {
    headers: { Accept: 'application/json' },
    signal,
  })
  if (!response.ok) throw new Error(`Could not load campaigns (${response.status})`)
  const data = await response.json()
  return Array.isArray(data) ? (data as CampaignRecord[]) : []
}

export async function fetchCampaign(
  campaignId: string,
  signal?: AbortSignal,
): Promise<CampaignRecord | null> {
  const response = await fetch(`${API_BASE}/campaigns/${encodeURIComponent(campaignId)}`, {
    headers: { Accept: 'application/json' },
    signal,
  })
  if (response.status === 404) return null
  if (!response.ok) throw new Error(`Could not load campaign (${response.status})`)
  return (await response.json()) as CampaignRecord
}

export async function saveCampaign(campaign: CampaignRecord): Promise<void> {
  const response = await fetch(`${API_BASE}/campaigns/save`, {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({ campaign }),
  })
  if (!response.ok) throw new Error(`Could not save campaign (${response.status})`)
}

export async function deleteCampaign(campaignId: string): Promise<void> {
  const response = await fetch(`${API_BASE}/campaigns/${encodeURIComponent(campaignId)}`, {
    method: 'DELETE',
  })
  if (!response.ok && response.status !== 404) {
    throw new Error(`Could not delete campaign (${response.status})`)
  }
}
