INTAKE_SYSTEM_PROMPT = """\
You are Aria, a friendly marketing planning assistant for IndiGo airlines.
Your job is to gather campaign details from the user through natural conversation before generating a plan.

REQUIRED fields (must collect all five before proceeding):
1. product        -- what is being marketed (e.g. flights to Singapore, hotel getaway sale, IndiGo gift cards)
2. offer          -- the specific promotion (e.g. "12% off base fare, promo code FLY12", "30% off hotel stays")
3. objective      -- what success looks like (bookings, signups, awareness, revenue, sales)
4. budget_inr     -- total campaign budget as a number in Indian Rupees
5. creative_brief -- visual direction for the banner ads: what imagery to show, the mood, and any must-have or must-avoid elements (e.g. "show a family arriving at a beach house, not posing", "quiet and understated, close-up textures", "avoid stock-photo smiles")
   When asking for this, ask about the FEELING and the SUBJECT, not the styling. Do not
   suggest aircraft, runways, airports or terminals -- IndiGo banners sell the destination
   and the ease of getting there, never the hardware, and the brand guardrails reject
   aviation imagery. Do not ask about logos, fonts or exact colours either: those are
   applied automatically downstream, so collecting them just creates conflicting
   instructions.

OPTIONAL fields (make sensible assumptions if not mentioned):
- available_channels (Google Search, Meta, Email, YouTube, SMS, goindigo.in)
- booking_window (how long the sale runs, e.g. "7-day sale window")
- bluchip_eligible (whether to target BluChip loyalty members)

RULES:
- Acknowledge what the user has already shared and extract every detail from their first message
- Ask exactly ONE focused question at a time, for the most important missing required field
- Be warm, brief, and professional -- like a smart colleague, not a form
- Do not list all questions at once
- When you have all 5 required fields, write a 2--3 sentence confirmation in plain conversational prose -- no bullet points, no bold labels, no markdown. Weave together what the user told you and what you're assuming into a single natural paragraph, like you're confirming a plan with a colleague. For example: "Got it -- I'll build a bookings campaign for IndiGo flights to Singapore with a Rs8 lakh budget, running a 12% off fare promo across Google Search, Meta, and Email over a 7-day window. The banners will show families at the destination in a vibrant, aspirational style. Generating your plan now." Then on the very last line output:

BRIEF_READY:{"grounded":{...},"assumed":{...}}

"grounded" = only what the user explicitly told you (always include creative_brief here)
"assumed"  = sensible defaults for an Indian airline for anything not mentioned

Example assumed values: {"budget_inr":1500000,"objective":"bookings","existing_list":"BluChip base and opted-in email/SMS list exist","bluchip_eligible":true}
Do NOT output BRIEF_READY until you truly have all 5 required fields confirmed by the user.\
"""

REFINE_SYSTEM_PROMPT = """\
You are a JSON editor for a marketing campaign brief.
The user wants to refine their existing campaign plan.
Given the current brief and the user's request, return ONLY an updated JSON object with keys "grounded" and "assumed".
Make minimal targeted changes -- only update what the user asked to change.
Return valid JSON only, no markdown, no explanation.\
"""


def _llm() -> AsyncAzureOpenAI:
    return AsyncAzureOpenAI(
        api_key=os.getenv("AZURE_OPENAI_API_KEY"),
        azure_endpoint=os.getenv("AZURE_OPENAI_ENDPOINT"),
        api_version=os.getenv("AZURE_OPENAI_API_VERSION", "2024-05-01-preview"),
    )

    grounded: dict
    assumed: dict = {}


class ChatMessageRequest(BaseModel):

    message: str


class RefineRequest(BaseModel):
    session_id: str
    message: str
    current_brief: dict

    campaign: dict

@app.get("/health")
async def health():
    return {"status": "ok"}


@app.post("/chat/message")
async def chat_message(req: ChatMessageRequest):
    """
    Intake chat endpoint. Streams the assistant reply word-by-word as SSE tokens.
    When the LLM has collected all required fields it emits chat_ready with the
    extracted brief so the frontend can immediately kick off /campaign/run.

    SSE events:
      {"event": "token",      "content": "..."}
      {"event": "chat_done"}
      {"event": "chat_ready", "brief": {"grounded": {...}, "assumed": {...}}}
      {"event": "error",      "message": "..."}
    """
    if req.session_id not in _sessions:
        _sessions[req.session_id] = []

    history = _sessions[req.session_id]
    history.append({"role": "user", "content": req.message})
    messages = [{"role": "system", "content": INTAKE_SYSTEM_PROMPT}] + history

    async def generate():
        try:
            client = _llm()
            stream = await client.chat.completions.create(
                model=os.getenv("AZURE_OPENAI_DEPLOYMENT", "gpt-4.1"),
                messages=messages,
                temperature=0.4,
                max_completion_tokens=500,
                stream=True,
            )

            collected = ""
            async for chunk in stream:
                delta = chunk.choices[0].delta.content if chunk.choices else None
                if delta:
                    collected += delta

            # Strip BRIEF_READY marker from display text
            brief = None
            display_text = collected.strip()
            marker_match = re.search(r'\nBRIEF_READY:(\{.+\})\s*$', collected, re.DOTALL)
            if marker_match:
                try:
                    brief = json.loads(marker_match.group(1))
                    display_text = collected[:marker_match.start()].strip()
                except json.JSONDecodeError:
                    pass

            history.append({"role": "assistant", "content": display_text})

            # Re-stream display text word-by-word for natural typing feel
            words = display_text.split(' ')
            for i, word in enumerate(words):
                chunk_text = word if i == len(words) - 1 else word + ' '
                yield f"data: {json.dumps({'event': 'token', 'content': chunk_text})}\n\n"
                await asyncio.sleep(0.025)

            if brief:
                yield f"data: {json.dumps({'event': 'chat_ready', 'brief': brief})}\n\n"
            else:
                yield f"data: {json.dumps({'event': 'chat_done'})}\n\n"

        except Exception as exc:
            yield f"data: {json.dumps({'event': 'error', 'message': str(exc)})}\n\n"

    return StreamingResponse(
        generate(),
        media_type="text/event-stream",
        headers={"Cache-Control": "no-cache", "X-Accel-Buffering": "no", "Connection": "keep-alive"},
    )


@app.post("/campaign/run")
async def run_campaign(brief: CampaignBrief):
    """
    Run the full GTM -> Metrics -> Finance agent pipeline and stream SSE events.
    """
    brief_dict = {"grounded": brief.grounded, "assumed": brief.assumed}

    def event_generator():
        try:
            for event in stream_campaign_events(brief_dict, memory=_memory):
                yield f"data: {json.dumps(event)}\n\n"
        except Exception as exc:
            yield f"data: {json.dumps({'event': 'error', 'message': str(exc)})}\n\n"
        finally:
            yield f"data: {json.dumps({'event': 'stream_end'})}\n\n"

    return StreamingResponse(
        event_generator(),
        media_type="text/event-stream",
        headers={"Cache-Control": "no-cache", "X-Accel-Buffering": "no", "Connection": "keep-alive"},
    )


@app.post("/campaign/refine")
async def refine_campaign(req: RefineRequest):
    """
    Merge user refinement feedback into the existing brief and re-run the pipeline.
    Streams the same SSE events as /campaign/run.
    """
    client = _llm()
    try:
        resp = await client.chat.completions.create(
            model=os.getenv("AZURE_OPENAI_DEPLOYMENT", "gpt-4.1"),
            messages=[
                {"role": "system", "content": REFINE_SYSTEM_PROMPT},
                {"role": "user", "content": (
                    f"Current brief:\n{json.dumps(req.current_brief, indent=2)}\n\n"
                    f"User refinement request: {req.message}"
                )},
            ],
            temperature=0.1,
            max_completion_tokens=800,
        )
        raw = resp.choices[0].message.content or ""
        json_match = re.search(r'\{.*\}', raw, re.DOTALL)
        updated_brief = json.loads(json_match.group()) if json_match else req.current_brief
    except Exception:
        updated_brief = req.current_brief

    def event_generator():
        try:
            for event in stream_campaign_events(updated_brief, memory=_memory):
                yield f"data: {json.dumps(event)}\n\n"
        except Exception as exc:
            yield f"data: {json.dumps({'event': 'error', 'message': str(exc)})}\n\n"
        finally:
            yield f"data: {json.dumps({'event': 'stream_end'})}\n\n"

    return StreamingResponse(
        event_generator(),
        media_type="text/event-stream",
        headers={"Cache-Control": "no-cache", "X-Accel-Buffering": "no", "Connection": "keep-alive"},
    )

@app.get("/campaigns")
async def list_campaigns():
    try:
        return db.list_campaign_snapshots()
    except Exception as exc:
        logger.error("[campaigns/list] %s", exc)
        raise HTTPException(status_code=503, detail="Campaign database is unavailable") from exc


@app.get("/campaigns/{campaign_id}")
async def get_campaign(campaign_id: str):
    try:
        campaign = db.get_campaign_snapshot(campaign_id)
    except Exception as exc:
        logger.error("[campaigns/detail] campaign_id=%s %s", campaign_id, exc)
        raise HTTPException(status_code=503, detail="Campaign database is unavailable") from exc
    if campaign is None:
        raise HTTPException(status_code=404, detail="Campaign not found")
    return campaign


@app.post("/campaigns/save")
async def save_campaign(req: CampaignSnapshotRequest):
    try:
        return db.save_campaign_snapshot(req.campaign)
    except ValueError as exc:
        raise HTTPException(status_code=400, detail=str(exc)) from exc
    except Exception as exc:
        logger.error("[campaigns/save] %s", exc)
        raise HTTPException(status_code=503, detail="Campaign could not be saved") from exc


@app.delete("/campaigns/{campaign_id}")
async def delete_campaign(campaign_id: str):
    try:
        deleted = db.delete_campaign_snapshot(campaign_id)
    except Exception as exc:
        logger.error("[campaigns/delete] campaign_id=%s %s", campaign_id, exc)
        raise HTTPException(status_code=503, detail="Campaign database is unavailable") from exc
    if not deleted:
        raise HTTPException(status_code=404, detail="Campaign not found or already deleted")
    return {"deleted": True, "campaign_id": campaign_id}
