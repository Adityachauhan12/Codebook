// Update baselinesRef initialization to include lastSetTime for debounce
const baselinesRef = React.useRef<{
  ear: number | null;
  noseX: number | null;
  faceWidth: number | null;
  faceHeight: number | null;
  lastSetTime: number | null;
}>({ ear: null, noseX: null, faceWidth: null, faceHeight: null, lastSetTime: null });

// Updated detection loop with video improvements
const runLiveDetectionLoop = useCallback(() => {
  const video = videoRef.current;

  const loop = () => {
    if (!video || video.paused || !landmarkerRef.current) {
      detectionLoopRef.current = requestAnimationFrame(loop);
      return;
    }

    if (status !== 'detecting' && status !== 'recording') {
      return;
    }

    try {
      const results = landmarkerRef.current.detectForVideo(video, performance.now());
      if (results.faceLandmarks && results.faceLandmarks.length > 0) {
        const landmarks = results.faceLandmarks[0];
        if (!landmarks || landmarks.length < 470) {
          detectionLoopRef.current = requestAnimationFrame(loop);
          return;
        }

        const LEYE = [33, 160, 158, 133, 153, 144];
        const REYE = [362, 385, 387, 263, 373, 380];
        const NOSE = 1;
        const LEFT_CHEEK = 234;
        const RIGHT_CHEEK = 454;
        const CHIN = 152;
        const FOREHEAD = 10;

        const dist = (p1: NormalizedLandmark, p2: NormalizedLandmark): number =>
          Math.hypot(p1.x - p2.x, p1.y - p2.y, (p1.z ?? 0) - (p2.z ?? 0));

        const getEAR = (idx: number[]): number => {
          const p1 = landmarks[idx[0]],
            p2 = landmarks[idx[1]],
            p3 = landmarks[idx[2]];
          const p4 = landmarks[idx[3]],
            p5 = landmarks[idx[4]],
            p6 = landmarks[idx[5]];
          return (dist(p2, p6) + dist(p3, p5)) / (2 * dist(p1, p4));
        };

        const ear = (getEAR(LEYE) + getEAR(REYE)) / 2;
        const noseX = landmarks[NOSE].x;

        const leftEyeCenter = {
          x: (landmarks[33].x + landmarks[133].x) / 2,
          y: (landmarks[33].y + landmarks[133].y) / 2,
          z: ((landmarks[33].z ?? 0) + (landmarks[133].z ?? 0)) / 2,
        };
        const rightEyeCenter = {
          x: (landmarks[362].x + landmarks[263].x) / 2,
          y: (landmarks[362].y + landmarks[263].y) / 2,
          z: ((landmarks[362].z ?? 0) + (landmarks[263].z ?? 0)) / 2,
        };

        const faceWidth = dist(leftEyeCenter as NormalizedLandmark, rightEyeCenter as NormalizedLandmark);
        const faceHeight = dist(landmarks[FOREHEAD], landmarks[CHIN]);

        const leftFaceX = (landmarks[LEFT_CHEEK].x + leftEyeCenter.x + landmarks[234].x) / 3;
        const rightFaceX = (landmarks[RIGHT_CHEEK].x + rightEyeCenter.x + landmarks[454].x) / 3;
        const avgFaceX = (leftFaceX + rightFaceX + noseX) / 3;

        const now = performance.now();
        // baseline reset rules:
        // - If no baseline, set it
        // - Only reset if last set > 3000 ms ago or in photo flow (to tolerate video flow stabilization)
        if (
          baselinesRef.current.ear === null ||
          baselinesRef.current.lastSetTime === null ||
          (now - baselinesRef.current.lastSetTime > 3000 || flowRef.current === 'photo')
        ) {
          baselinesRef.current.ear = ear;
          baselinesRef.current.noseX = avgFaceX;
          baselinesRef.current.faceWidth = faceWidth;
          baselinesRef.current.faceHeight = faceHeight;
          baselinesRef.current.lastSetTime = now;
          if (flowRef.current === 'video') console.debug('Video flow baseline reset');
        }

        const widthDeviation = Math.abs(faceWidth - (baselinesRef.current.faceWidth ?? faceWidth));
        const heightDeviation = Math.abs(faceHeight - (baselinesRef.current.faceHeight ?? faceHeight));
        const isFaceStable = widthDeviation < 0.02 && heightDeviation < 0.02;

        if (!isFaceStable) {
          detectionLoopRef.current = requestAnimationFrame(loop);
          return;
        }

        const leftEAR = getEAR(LEYE);
        const rightEAR = getEAR(REYE);
        const eyesVisible = leftEAR > 0.1 && rightEAR > 0.1 && Math.abs(leftEAR - rightEAR) < 0.15;

        if (!eyesVisible) {
          detectionLoopRef.current = requestAnimationFrame(loop);
          return;
        }

        // Relax thresholds for video flow
        const blinkThreshold = flowRef.current === 'video' ? 0.85 : 0.8;
        const leftTurnThreshold = flowRef.current === 'video' ? -0.02 : -0.03;
        const rightTurnThreshold = flowRef.current === 'video' ? 0.03 : 0.04;
        const sideMovementLeftThreshold = flowRef.current === 'video' ? -0.06 : -0.03;
        const sideMovementRightThreshold = flowRef.current === 'video' ? 0.06 : 0.03;

        if (baselinesRef.current.ear !== null && ear < baselinesRef.current.ear * blinkThreshold) {
          if (Math.abs(leftEAR - rightEAR) < 0.08) {
            if (!actionsDoneRef.current.blink) {
              actionsDoneRef.current.blink = true;
              if (flowRef.current === 'video') console.debug('Blink detected');
            }
          }
        }

        if (baselinesRef.current.noseX !== null && isFaceStable) {
          const deviation = avgFaceX - baselinesRef.current.noseX;

          const leftSideMovement = leftFaceX - (baselinesRef.current.noseX - 0.05);
          const rightSideMovement = rightFaceX - (baselinesRef.current.noseX + 0.05);

          if (
            deviation < leftTurnThreshold &&
            leftSideMovement < sideMovementLeftThreshold
          ) {
            const eyeCenterX = (leftEyeCenter.x + rightEyeCenter.x) / 2;
            if (eyeCenterX < baselinesRef.current.noseX + sideMovementLeftThreshold) {
              if (!actionsDoneRef.current.left) {
                actionsDoneRef.current.left = true;
                if (flowRef.current === 'video') console.debug('Left turn detected');
              }
            }
          }

          if (
            deviation > rightTurnThreshold &&
            rightSideMovement > sideMovementRightThreshold
          ) {
            const eyeCenterX = (leftEyeCenter.x + rightEyeCenter.x) / 2;
            if (eyeCenterX > baselinesRef.current.noseX + sideMovementRightThreshold) {
              if (!actionsDoneRef.current.right) {
                actionsDoneRef.current.right = true;
                if (flowRef.current === 'video') console.debug('Right turn detected');
              }
            }
          }
        }

        const { blink, left, right } = actionsDoneRef.current;

        setFeedback(
          `${blink ? 'Blink ✓' : 'Blink'}  ${left ? 'Left ✓' : 'Left'}  ${right ? 'Right ✓' : 'Right'}`
        );

        if (blink && left && right) {
          livenessPassedRef.current = true;

          if (flowRef.current === 'photo' && status === 'detecting') {
            handlePhotoPassAndUpload();
            return;
          }
        }
      } else {
        setFeedback('No face detected - please position yourself in the camera');
      }
    } catch (err) {
      console.error('Detection loop error:', err);
    }

    detectionLoopRef.current = requestAnimationFrame(loop);
  };

  detectionLoopRef.current = requestAnimationFrame(loop);
}, [status]);
