"""Central configuration, loaded from the .env file at the project root."""
from __future__ import annotations

import os
from dataclasses import dataclass, field
from pathlib import Path

from dotenv import load_dotenv

# backend/app/config.py -> backend/app -> backend -> <project root>
PROJECT_ROOT = Path(__file__).resolve().parents[2]

load_dotenv(PROJECT_ROOT / ".env")

PLACEHOLDER_MARKERS = ("paste-your", "your-key-here", "<your", "changeme", "xxxx")


def _str(name: str, default: str = "") -> str:
    return os.getenv(name, default).strip()


def _int(name: str, default: int) -> int:
    try:
        return int(os.getenv(name, "").strip() or default)
    except ValueError:
        return default


def _bool(name: str, default: bool) -> bool:
    raw = os.getenv(name, "").strip().lower()
    if raw in ("1", "true", "yes", "on"):
        return True
    if raw in ("0", "false", "no", "off"):
        return False
    return default


def _is_placeholder(value: str) -> bool:
    if not value:
        return True
    lowered = value.lower()
    return any(marker in lowered for marker in PLACEHOLDER_MARKERS)


@dataclass
class Settings:
    # --- which vendor serves the models -----------------------------------
    llm_provider: str = field(default_factory=lambda: _str("LLM_PROVIDER", "azure").lower())

    # --- Azure OpenAI ------------------------------------------------------
    azure_endpoint: str = field(
        default_factory=lambda: _str("AZURE_OPENAI_ENDPOINT").rstrip("/")
    )
    azure_api_key: str = field(default_factory=lambda: _str("AZURE_OPENAI_API_KEY"))
    azure_deployment: str = field(
        default_factory=lambda: _str("AZURE_OPENAI_DEPLOYMENT", "gpt-5.3-codex")
    )
    azure_api_version: str = field(
        default_factory=lambda: _str("AZURE_OPENAI_API_VERSION", "2025-04-01")
    )
    azure_use_responses_api: bool = field(
        default_factory=lambda: _bool("AZURE_USE_RESPONSES_API", True)
    )
    # Optional second deployment for the cheap high-volume reading pass. Most
    # Azure resources have one deployment, so this defaults to the same one.
    azure_worker_deployment: str = field(
        default_factory=lambda: _str("AZURE_OPENAI_WORKER_DEPLOYMENT")
    )

    # --- Anthropic (alternative provider) ---------------------------------
    anthropic_api_key: str = field(default_factory=lambda: _str("ANTHROPIC_API_KEY"))
    anthropic_deep_model: str = field(
        default_factory=lambda: _str("DEEP_MODEL", "claude-opus-5")
    )
    anthropic_worker_model: str = field(
        default_factory=lambda: _str("WORKER_MODEL", "claude-sonnet-5")
    )

    # --- reasoning effort --------------------------------------------------
    deep_effort: str = field(default_factory=lambda: _str("DEEP_EFFORT", "high"))
    worker_effort: str = field(default_factory=lambda: _str("WORKER_EFFORT", "medium"))

    # --- knowledge stage: which open-source coding agent reads the code ---
    knowledge_backend: str = field(
        default_factory=lambda: _str("KNOWLEDGE_BACKEND", "aider").lower()
    )
    knowledge_task_timeout: int = field(
        default_factory=lambda: _int("KNOWLEDGE_TASK_TIMEOUT", 1800)
    )
    max_modules: int = field(default_factory=lambda: _int("MAX_MODULES", 0))

    # --- pipeline tuning ---------------------------------------------------
    max_concurrency: int = field(default_factory=lambda: _int("MAX_CONCURRENCY", 4))
    module_char_budget: int = field(default_factory=lambda: _int("MODULE_CHAR_BUDGET", 110_000))
    max_files_per_module: int = field(default_factory=lambda: _int("MAX_FILES_PER_MODULE", 40))
    llm_cache: bool = field(default_factory=lambda: _bool("LLM_CACHE", True))
    dry_run: bool = field(default_factory=lambda: _bool("DRY_RUN", False))

    target_dir: Path = field(
        default_factory=lambda: PROJECT_ROOT / _str("TARGET_DIR", "target")
    )
    reports_dir: Path = field(
        default_factory=lambda: PROJECT_ROOT / _str("REPORTS_DIR", "reports")
    )

    # -- derived -----------------------------------------------------------

    @property
    def cache_dir(self) -> Path:
        return self.reports_dir / "_llm_cache"

    @property
    def is_azure(self) -> bool:
        return self.llm_provider in ("azure", "azure_openai")

    @property
    def azure_key_is_placeholder(self) -> bool:
        return _is_placeholder(self.azure_api_key)

    @property
    def anthropic_key_is_placeholder(self) -> bool:
        return _is_placeholder(self.anthropic_api_key)

    @property
    def key_is_placeholder(self) -> bool:
        """Is the key for the *selected* provider missing?"""
        if self.is_azure:
            return self.azure_key_is_placeholder
        return self.anthropic_key_is_placeholder

    @property
    def deep_model(self) -> str:
        """Model id for synthesis / BRD / convertibility."""
        return self.azure_deployment if self.is_azure else self.anthropic_deep_model

    @property
    def worker_model(self) -> str:
        """Model id for the high-volume pass."""
        if self.is_azure:
            return self.azure_worker_deployment or self.azure_deployment
        return self.anthropic_worker_model

    @property
    def knowledge_model(self) -> str:
        """What the external OSS coding agent should use."""
        return self.worker_model

    @property
    def api_key(self) -> str:
        return self.azure_api_key if self.is_azure else self.anthropic_api_key

    def ready(self) -> tuple[bool, str]:
        """Can the pipeline actually call a model right now?"""
        if self.dry_run:
            return True, "DRY_RUN=1 - running with the offline stub model, no API calls."

        from .llm.providers import get_provider  # local import: avoids a cycle

        try:
            return get_provider().check()
        except Exception as exc:
            return False, str(exc)


settings = Settings()
