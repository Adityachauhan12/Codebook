"""Optional scene keyframes, via the Gemini image model.

Veo has no memory between generations, so the strongest continuity lever
available is image-to-video: render the scene's opening frame as a still, then
hand that still to Veo as the first frame. The character, wardrobe, location and
grade are then fixed by pixels rather than by hoping twelve independent prompts
converge.

Off by default (``VEO_USE_KEYFRAMES``) because it adds one image call per clip.
"""

from __future__ import annotations

import base64
import re
from functools import lru_cache
from pathlib import Path

import httpx

from ..brand import active_brand
from ..brand.assets import livery_references
from ..config import settings
from ..domain.models import Scene, StoryOption, VeoAspectRatio
from ..logging_setup import get_logger
from .veo import InlineAsset

log = get_logger("imagegen")

_MIME_BY_SUFFIX = {
    ".png": "image/png",
    ".jpg": "image/jpeg",
    ".jpeg": "image/jpeg",
    ".webp": "image/webp",
}

#: Sent alongside the reference photographs. Colour words alone get the blues
#: approximately right and the tail form entirely wrong; naming the airline
#: without pixels to copy makes the model invent a wordmark it cannot spell.
#: The references fix both, but only if the prompt says to follow them.
_LIVERY_DIRECTION = (
    "Reference photographs of the real IndiGo aircraft and cabin are attached. "
    "Brand accuracy is the point of this frame — match them exactly: pearl-white "
    "fuselage, deep indigo navy-blue tail with its upward-sweeping form, the blue "
    "sweep along the lower fuselage, deep blue engine nacelles, and the IndiGo "
    "wordmark reproduced exactly as it appears in the reference, correctly spelt "
    "with a capital I and a capital G. Cabin: blue fabric seats, pale blue "
    "headrest covers, white panels. Crew: deep navy uniform. Do not invent a "
    "different airline, a different livery, or a plain unbranded white aircraft."
)


def asset_from_image(path: Path) -> InlineAsset | None:
    """Read an image file as a Veo inline asset. ``None`` if unusable.

    Used for tail-frame chaining: the last frame of clip N becomes the first
    frame of clip N+1, which is the only continuity mechanism that costs no
    extra generation and carries a face across a cut exactly.
    """
    try:
        data = path.read_bytes()
    except OSError as error:
        log.warning("Could not read keyframe %s: %s", path.name, error)
        return None
    if not data:
        return None
    return InlineAsset(
        mime_type=_MIME_BY_SUFFIX.get(path.suffix.lower(), "image/png"),
        data=base64.b64encode(data).decode("ascii"),
    )

#: Strip camera movement out of the shot card's camera field — a still has none.
_MOVEMENT = re.compile(
    r"\b(push[- ]in|pull[- ]back|tracking|pan|tilt|orbit|dolly|handheld|whip|crane|glide)\b.*",
    re.IGNORECASE,
)


def _keyframe_prompt(scene: Scene, story: StoryOption, aspect: VeoAspectRatio) -> str:
    """Direct a still, not a shot.

    The image model must not be told about camera movement or audio, and it must
    be told to leave the frame clean.
    """
    card = scene.shot_card
    continuity = story.continuity
    lens = _MOVEMENT.sub("", card.camera).strip() or "50mm"

    return "\n".join(
        [
            f"A single film frame, {aspect} aspect ratio. This is the opening frame of a shot.",
            "",
            f"Framing: {card.frame}. {card.composition}",
            f"Lens: {lens}.",
            f"Subject and action at the start of the shot: {card.action}",
            f"On the body: {card.micro_action}",
            f"In the space: {card.environmental_pressure}",
            "",
            f"Characters: {' | '.join(continuity.characters) or '—'}",
            f"Wardrobe: {' | '.join(continuity.wardrobe) or '—'}",
            f"Location: {continuity.location}",
            f"Lighting: {card.lighting or continuity.lighting_constant}",
            f"Palette: {card.palette or continuity.palette}",
            "",
            # "no logos" used to sit in this line, which stripped IndiGo out of
            # the very frame the film is an advertisement for. Only *overlay*
            # graphics are unwanted here; branding inside the scene is the point.
            "Photographic, natural skin texture, motivated lighting. No captions, no "
            "subtitles, no overlay graphics or text, no watermark, no borders, no collage, "
            "no split frames. Branding that belongs in the world of the shot — aircraft "
            "livery, crew uniform, gate signage — is wanted in frame.",
            *(("", _LIVERY_DIRECTION) if livery_assets() else ()),
        ]
    )


@lru_cache(maxsize=1)
def livery_assets() -> tuple[InlineAsset, ...]:
    """Real IndiGo photography, read once and reused for every keyframe."""
    if active_brand() != "indigo":
        return ()
    assets = tuple(
        asset for path in livery_references() if (asset := asset_from_image(path))
    )
    if not assets:
        log.warning(
            "No livery references found; keyframes will be text-only and the aircraft "
            "will not reliably read as IndiGo. Add photographs to brand/assets/references/livery/."
        )
    return assets


async def generate_keyframe(
    scene: Scene, story: StoryOption, aspect: VeoAspectRatio
) -> InlineAsset | None:
    """Render the scene's first frame. Returns ``None`` on any failure.

    A keyframe is an optimisation, never a reason to fail a clip.
    """
    key = settings.google_key
    if not key:
        return None

    url = f"{settings.gemini_api_base.rstrip('/')}/models/{settings.gemini_image_model_name}:generateContent"

    # Text first, then the reference photographs. The livery is the one thing a
    # prompt cannot specify accurately — this is what makes the aircraft read as
    # IndiGo rather than as a generic white jet.
    request_parts: list[dict[str, object]] = [
        {"text": _keyframe_prompt(scene, story, aspect)}
    ]
    request_parts.extend({"inlineData": asset.to_payload()} for asset in livery_assets())

    try:
        async with httpx.AsyncClient(timeout=180.0) as client:
            response = await client.post(
                url,
                headers={"x-goog-api-key": key, "content-type": "application/json"},
                json={
                    "contents": [{"role": "user", "parts": request_parts}],
                    "generationConfig": {
                        "responseModalities": ["IMAGE"],
                        "imageConfig": {"aspectRatio": aspect},
                    },
                },
            )

        if response.status_code >= 400:
            log.warning(
                "Keyframe for scene %d failed (%s): %s",
                scene.order, response.status_code, response.text[:200],
            )
            return None

        payload = response.json()

        block_reason = (payload.get("promptFeedback") or {}).get("blockReason")
        if block_reason:
            log.warning("Keyframe for scene %d was blocked (%s).", scene.order, block_reason)
            return None

        candidates = payload.get("candidates") or []
        parts = ((candidates[0].get("content") or {}).get("parts") or []) if candidates else []
        inline = next((part.get("inlineData") for part in parts if part.get("inlineData")), None)

        if not inline or not inline.get("data"):
            log.warning(
                "Keyframe for scene %d returned no image; continuing text-to-video.", scene.order
            )
            return None

        return InlineAsset(mime_type=inline.get("mimeType") or "image/png", data=inline["data"])

    except Exception as error:  # noqa: BLE001 — never fail a clip over a keyframe
        log.warning(
            "Keyframe for scene %d failed; falling back to text-to-video: %s", scene.order, error
        )
        return None
