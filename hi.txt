export default function HealthBar({ health }) {
  if (!health) return null

  const backends = Object.entries(health.backends || {})
  const installed = backends.filter(([, s]) => s.available).map(([name]) => name)
  const active = health.knowledge_backend
  const activeInstalled = installed.includes(active)

  return (
    <div className={`banner ${health.ok ? 'ok' : 'warn'}`}>
      <div className="health-row">
        <span className="pill">
          {health.dry_run
            ? 'DRY RUN — offline stub, no API calls'
            : health.api_key_configured
              ? 'API key loaded'
              : 'API key not set'}
        </span>

        <span className="pill">
          Code reader: <strong>{active}</strong>
          {!activeInstalled && active !== 'builtin' && ' (not installed → builtin fallback)'}
        </span>

        <span className="pill dim" title={health.azure_endpoint || ''}>
          {health.provider_detail}
        </span>

        <span className="pill dim">
          OSS agents found: {installed.length ? installed.join(', ') : 'none'}
        </span>
      </div>

      {!health.ok && <p className="health-detail">{health.detail}</p>}

      {!activeInstalled && active !== 'builtin' && (
        <p className="health-detail">
          {backends.find(([n]) => n === active)?.[1]?.detail}
        </p>
      )}
    </div>
  )
}
