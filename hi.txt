"""Brand master files and the exact colours sampled from them.

The composite-in-post design in ``policy.py`` only works if there is a master
file to composite. This module is that file's address, plus the colour values
taken from the artwork itself rather than from a brand deck nobody can find.

Marks are straight (unpremultiplied) RGBA PNGs, trimmed to their alpha bounding
box, so ``overlay`` in ffmpeg places them without a halo and without guesswork
about padding.
"""

from __future__ import annotations

import sys
from pathlib import Path

from ..config import settings

__all__ = [
    "INDIGO_BLUE",
    "INDIGO_BLUE_HEX",
    "INDIGO_SKY",
    "INDIGO_SKY_HEX",
    "INDIGO_MID_BLUE_HEX",
    "MARKS_DIR",
    "REFERENCES_DIR",
    "LIVERY_DIR",
    "FONTS_DIR",
    "wordmark",
    "icon",
    "reference_images",
    "livery_references",
    "font_path",
]

ASSETS_DIR = Path(__file__).parent / "assets"
MARKS_DIR = ASSETS_DIR / "marks"
REFERENCES_DIR = ASSETS_DIR / "references"
LIVERY_DIR = REFERENCES_DIR / "livery"
FONTS_DIR = ASSETS_DIR / "fonts"

#: Sampled from ``marks/wordmark-blue.png`` and ``marks/icon.svg`` — both give
#: exactly this value. This is the brand's primary deep blue.
INDIGO_BLUE = (0, 0, 153)
INDIGO_BLUE_HEX = "#000099"

#: The secondary light blue, sampled from the BluChip campaign headline.
INDIGO_SKY = (151, 201, 234)
INDIGO_SKY_HEX = "#97C9EA"

#: The mid blue used in mailer headers. Useful for gradients, never for type.
INDIGO_MID_BLUE_HEX = "#0058AC"


def wordmark(*, on_dark: bool = True) -> Path:
    """The full IndiGo lockup — wordmark plus 20-dot device. 1543x466.

    ``on_dark`` picks the white cut, which is what an end card over sky or
    cloud wants. The blue cut is for light plates only.
    """
    name = "wordmark-white.png" if on_dark else "wordmark-blue.png"
    return MARKS_DIR / name


def icon(*, on_dark: bool = True) -> Path:
    """The 20-dot arrow-plane device on its own. 835x832."""
    return MARKS_DIR / ("icon-white.png" if on_dark else "icon-blue.png")


#: Ordered best-first. IndiGo's own typeface is not redistributable and is not in
#: the asset drop, so these are the closest widely installed geometric sans
#: faces. Anything dropped into ``assets/fonts`` wins over all of them.
_FONT_CANDIDATES: dict[str, tuple[str, ...]] = {
    "darwin": (
        "/System/Library/Fonts/Supplemental/Futura.ttc",
        "/System/Library/Fonts/Avenir Next.ttc",
        "/System/Library/Fonts/HelveticaNeue.ttc",
        "/System/Library/Fonts/Supplemental/Arial.ttf",
    ),
    "win32": (
        r"C:\Windows\Fonts\GOTHIC.TTF",
        r"C:\Windows\Fonts\Futura.ttf",
        r"C:\Windows\Fonts\segoeui.ttf",
        r"C:\Windows\Fonts\arial.ttf",
    ),
    "linux": (
        "/usr/share/fonts/truetype/dejavu/DejaVuSans.ttf",
        "/usr/share/fonts/truetype/liberation/LiberationSans-Regular.ttf",
    ),
}


def font_path() -> Path | None:
    """The font file for burnt-in supers, or ``None`` if nothing usable exists.

    Order: the configured override, then anything bundled in ``assets/fonts``,
    then the platform's best geometric sans. Returning ``None`` rather than
    raising lets the caller decide — a film with no supers is still a film.
    """
    configured = (settings.brand_font_path or "").strip()
    if configured:
        candidate = Path(configured)
        return candidate if candidate.is_file() else None

    if FONTS_DIR.is_dir():
        bundled = sorted(
            path
            for path in FONTS_DIR.iterdir()
            if path.suffix.lower() in {".ttf", ".otf", ".ttc"}
        )
        if bundled:
            return bundled[0]

    platform = "win32" if sys.platform.startswith("win") else sys.platform
    for name in _FONT_CANDIDATES.get(platform, _FONT_CANDIDATES["linux"]):
        path = Path(name)
        if path.is_file():
            return path
    return None


def reference_images() -> list[Path]:
    """Campaign reference art — type, palette and product-photography tone.

    Deliberately excludes ``livery/``: these are food creatives and mailers, and
    conditioning an aircraft shot on them biases the frame toward product
    photography. For aircraft and cabin, use `livery_references`.
    """
    if not REFERENCES_DIR.is_dir():
        return []
    return sorted(
        path
        for path in REFERENCES_DIR.iterdir()
        if path.suffix.lower() in {".png", ".jpg", ".jpeg"}
    )


def livery_references() -> list[Path]:
    """Real IndiGo aircraft and cabin photography, for keyframe conditioning.

    The livery is the one thing a text prompt cannot specify accurately — colour
    words get the blues roughly right and the tail form entirely wrong, and
    naming the airline alone makes the model invent a wordmark it cannot spell.
    Conditioning on real pixels is what fixes both. See
    ``imagegen.generate_keyframe``.
    """
    if not LIVERY_DIR.is_dir():
        return []
    return sorted(
        path
        for path in LIVERY_DIR.iterdir()
        if path.suffix.lower() in {".png", ".jpg", ".jpeg"}
    )
